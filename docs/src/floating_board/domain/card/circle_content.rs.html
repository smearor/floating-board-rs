1use std::f64::consts::SQRT_2;
2
3/// How a [`CardShape::Circle`] card allocates its content inside the
4/// circle's bounding square.
5///
6/// [`CardShape::Circle`]: crate::CardShape
7#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
8#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize), serde(rename_all = "kebab-case"))]
9pub enum CircleContent {
10    /// The content is allocated the square inscribed in the inner
11    /// circle: `side = (d - 2·ring) / √2`. By construction the content
12    /// can never paint or catch picks outside the circle - the card
13    /// keeps the full ring as its own surface.
14    #[default]
15    Inscribed,
16    /// The content is allocated the full inner square (`side = d -
17    /// 2·ring`) and is expected to draw its own circular shape with
18    /// transparent crescents (a clock face, a pie menu, a chart).
19    /// Crescent picks inside the circle hit the content itself.
20    Fill,
21    /// Like [`CircleContent::Fill`], plus a circular clip on the
22    /// content: rendering and picking stop at the inner circle, so
23    /// crescent picks fall back onto the card - the ring stays a drag
24    /// affordance even over a content that fills its square.
25    Clipped,
26}
27
28impl CircleContent {
29    /// The card diameter for `content_natural` - the larger of the
30    /// content's natural width and height - given a `ring` band width
31    /// (px) around the inner circle.
32    ///
33    /// For `Inscribed` the inner circle circumscribes the content
34    /// square (`d_inner = side · √2`); for `Fill`/`Clipped` the inner
35    /// circle is inscribed in it (`d_inner = side`).
36    pub fn diameter_for(&self, content_natural: f64, ring: f64) -> f64 {
37        let inner_diameter = match self {
38            Self::Inscribed => content_natural * SQRT_2,
39            Self::Fill | Self::Clipped => content_natural,
40        };
41        inner_diameter + 2.0 * ring
42    }
43
44    /// The content square's side length for a card of `diameter` with a
45    /// ring band of `ring` (px). Inverse of [`Self::diameter_for`]:
46    /// `content_square_for(diameter_for(nat)) == nat`.
47    pub fn content_square_for(&self, diameter: f64, ring: f64) -> f64 {
48        let inner = (diameter - 2.0 * ring).max(0.0);
49        match self {
50            Self::Inscribed => inner / SQRT_2,
51            Self::Fill | Self::Clipped => inner,
52        }
53    }
54}

56#[cfg(test)]
57mod tests {
58    use super::*;
59
60    #[test]
61    fn default_content_is_inscribed() {
62        assert_eq!(CircleContent::default(), CircleContent::Inscribed);
63    }
64
65    #[test]
66    fn inscribed_diameter_circumscribes_the_content_square() {
67        // A 100px content square needs an inner circle of 100·√2 plus
68        // a 4px ring on both sides.
69        let diameter = CircleContent::Inscribed.diameter_for(100.0, 4.0);
70        assert!((diameter - (100.0 * SQRT_2 + 8.0)).abs() < 1e-9);
71    }
72
73    #[test]
74    fn fill_diameter_is_content_plus_ring() {
75        let diameter = CircleContent::Fill.diameter_for(100.0, 4.0);
76        assert!((diameter - 108.0).abs() < 1e-9);
77        assert!((CircleContent::Clipped.diameter_for(100.0, 4.0) - 108.0).abs() < 1e-9);
78    }
79
80    #[test]
81    fn diameter_and_square_are_inverses() {
82        for content in [CircleContent::Inscribed, CircleContent::Fill, CircleContent::Clipped] {
83            let diameter = content.diameter_for(120.0, 6.0);
84            assert!((content.content_square_for(diameter, 6.0) - 120.0).abs() < 1e-9);
85        }
86    }
87
88    #[test]
89    fn degenerate_diameter_clamps_to_zero() {
90        assert_eq!(CircleContent::Fill.content_square_for(4.0, 6.0), 0.0);
91    }
