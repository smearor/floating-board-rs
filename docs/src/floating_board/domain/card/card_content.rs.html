7/// How a shaped card allocates its content inside the shape's
8/// bounding box, in ascending extent: `Inscribed` < `Fill` <
9/// `Clipped` < `Cover`.
10///
11/// Picking: GTK descends into children unconditionally and offers no
12/// non-rectangular pick masks, so on non-rectangular shapes the
13/// content subtree of `Fill`/`Clipped`/`Cover` is made non-targetable:
14/// content in those modes is display-only and clicks in the cut
15/// corners pass through to cards underneath. `Inscribed` and
16/// `Rectangle` content stays interactive because its allocation never
17/// covers the cut corners.
18///
19/// [`CardShape::Circle`]: crate::CardShape
20#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
21#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize), serde(rename_all = "kebab-case"))]
22pub enum CardContent {
23    /// The content is allocated the square inscribed in the inner
24    /// shape (`side = (d - 2·ring) / √2` on a `Circle`; the
25    /// analogous inscribed square on polygons). By construction the
26    /// content can never paint or catch picks outside the shape -
27    /// the card keeps the full ring as its own surface.
28    #[default]
29    Inscribed,
30    /// The content is allocated the full inner region (`side = d -
31    /// 2·ring` on a `Circle`; the rim-shrunk silhouette's bounding
32    /// box on polygons) and is expected to draw its own shape with
33    /// transparent corners (a clock face, a pie menu, a chart).
34    Fill,
35    /// Like [`CardContent::Fill`], plus a silhouette clip on the
36    /// content: rendering stops at the inner shape, so the cut
37    /// regions show the card - the ring stays a visible drag
38    /// affordance even over a content that fills its square.
39    Clipped,
40    /// The content is allocated the whole bounding box - the card's
41    /// silhouette is the only bound, no ring stays card surface.
42    /// The card's own silhouette clip is the bound on every shape,
43    /// so a content that paints its square still shows the
44    /// silhouette. Ignored for `Rectangle` (its bounding box is the
45    /// silhouette).
46    Cover,
47}
48
49impl CardContent {
50    /// The card diameter for `content_natural` - the larger of the
51    /// content's natural width and height - given a `ring` band width
52    /// (px) around the inner circle.
53    ///
54    /// For `Inscribed` the inner circle circumscribes the content
55    /// square (`d_inner = side · √2`); for `Fill`/`Clipped` the inner
56    /// circle is inscribed in it (`d_inner = side`).
57    pub fn diameter_for(&self, content_natural: Distance<Pixels>, ring: Distance<Pixels>) -> Distance<Pixels> {
58        match self {
59            // `with_inscribed_square` gives `d_inner = side * sqrt(2)`.
60            Self::Inscribed => Distance::new(Circle::<()>::radius_for_inscribed_square(content_natural.as_f64()) * 2.0) + ring * 2.0,
61            Self::Fill | Self::Clipped => content_natural + ring * 2.0,
62            // `Cover` fills the bounding box - no ring applies.
63            Self::Cover => content_natural,
64        }
65    }
66
67    /// The content square's side length for a card of `diameter` with a
68    /// ring band of `ring` (px). Inverse of [`Self::diameter_for`]:
69    /// `content_square_for(diameter_for(nat)) == nat`.
70    pub fn content_square_for(&self, diameter: Distance<Pixels>, ring: Distance<Pixels>) -> Distance<Pixels> {
71        let inner = (diameter - ring * 2.0).max(Distance::ZERO);
72        match self {
73            Self::Inscribed => Distance::new(Circle::<()>::inscribed_square_side_for_radius(inner.as_f64() / 2.0)),
74            Self::Fill | Self::Clipped => inner,
75            Self::Cover => diameter,
76        }
77    }
78
79    /// The dome's radius for a `Semicircle` card, given the content's
80    /// `natural` size, a `ring` band width inside the dome's edge and
81    /// a `bottom_inset` that lifts the content off the flat edge (the
82    /// straight bar's band width, `Distance::ZERO` without one).
83    ///
84    /// For `Inscribed` the inner dome circumscribes a square that is
85    /// bottom-aligned `bottom_inset` above the flat edge: the binding
86    /// corners are the square's top two, so
87    /// `r_inner² = (side/2)² + (side + bottom_inset)²`. For
88    /// `Fill`/`Clipped` the inner dome bounds a `2·r_inner ×
89    /// (r_inner − bottom_inset)` content rect, so
90    /// `r_inner = max(width/2, height + bottom_inset)`. For `Cover`
91    /// the dome's bounding box `2r × r` holds the content, so
92    /// `r = max(width/2, height)` and no ring applies.
93    pub fn dome_radius_for(&self, natural: Vec2<Pixels>, insets: DomeInsets) -> Distance<Pixels> {
94        match self {
95            Self::Inscribed => {
96                let side = natural.x.max(natural.y);
97                Distance::new(((side / 2.0).powi(2) + (side + insets.bottom_inset.as_f64()).powi(2)).sqrt()) + insets.ring
98            }
99            Self::Fill | Self::Clipped => Distance::new((natural.x / 2.0).max(natural.y + insets.bottom_inset.as_f64())) + insets.ring,
100            Self::Cover => Distance::new((natural.x / 2.0).max(natural.y)),
101        }
102    }
103
104    /// The content extent inside a `Semicircle` card of `radius`
105    /// with a ring band of `ring` and a bottom inset of
106    /// `bottom_inset`. Inverse of [`Self::dome_radius_for`].
107    /// The content rect is bottom-aligned `bottom_inset` above the
108    /// dome's flat edge: `Inscribed` solves the circumscribed-square
109    /// quadratic for its side, `Fill`/`Clipped` gives the inner
110    /// dome's bounding box (`2·r_inner × (r_inner − bottom_inset)`),
111    /// `Cover` the dome's bounding box (`2r × r`).
112    pub fn dome_content_for(&self, radius: Distance<Pixels>, insets: DomeInsets) -> Vec2<Pixels> {
113        let inner = (radius - insets.ring).max(Distance::ZERO).as_f64();
114        let inset = insets.bottom_inset.as_f64();
115        match self {
116            Self::Inscribed => {
117                // 1.25·side² + 2·bottom_inset·side + bottom_inset² = r_inner²
118                let side = ((5.0 * inner.powi(2) - inset.powi(2)).max(0.0).sqrt() - 2.0 * inset) / 2.5;
119                Vec2::new(side.max(0.0), side.max(0.0))
120            }
121            Self::Fill | Self::Clipped => Vec2::new(2.0 * inner, (inner - inset).max(0.0)),
122            Self::Cover => Vec2::new(2.0 * radius.as_f64(), radius.as_f64()),
123        }
124    }
125}

131#[cfg(test)]
132mod tests {
133    use super::*;
134    use std::f64::consts::SQRT_2;
135
136    #[test]
137    fn default_content_is_inscribed() {
138        assert_eq!(CardContent::default(), CardContent::Inscribed);
139    }
140
141    #[test]
142    fn inscribed_diameter_circumscribes_the_content_square() {
143        // A 100px content square needs an inner circle of 100·√2 plus
144        // a 4px ring on both sides.
145        let diameter = CardContent::Inscribed.diameter_for(Distance::new(100.0), Distance::new(4.0)).as_f64();
146        assert!((diameter - (100.0 * SQRT_2 + 8.0)).abs() < 1e-9);
147    }
148
149    #[test]
150    fn fill_diameter_is_content_plus_ring() {
151        let diameter = CardContent::Fill.diameter_for(Distance::new(100.0), Distance::new(4.0)).as_f64();
152        assert!((diameter - 108.0).abs() < 1e-9);
153        assert!((CardContent::Clipped.diameter_for(Distance::new(100.0), Distance::new(4.0)).as_f64() - 108.0).abs() < 1e-9);
154    }
155
156    #[test]
157    fn diameter_and_square_are_inverses() {
158        for content in [CardContent::Inscribed, CardContent::Fill, CardContent::Clipped, CardContent::Cover] {
159            let diameter = content.diameter_for(Distance::new(120.0), Distance::new(6.0));
160            assert!((content.content_square_for(diameter, Distance::new(6.0)).as_f64() - 120.0).abs() < 1e-9);
161        }
162    }
163
164    #[test]
165    fn degenerate_diameter_clamps_to_zero() {
166        assert_eq!(CardContent::Fill.content_square_for(Distance::new(4.0), Distance::new(6.0)), Distance::ZERO);
167    }
168
169    #[test]
170    fn dome_radius_and_content_are_inverses() {
171        // With no chrome the bottom inset vanishes and both helpers
172        // reduce to the flat-edge formulas.
173        for content in [CardContent::Inscribed, CardContent::Fill, CardContent::Clipped, CardContent::Cover] {
174            for bottom_inset in [0.0, 20.0, 40.0] {
175                let radius = content.dome_radius_for(Vec2::new(150.0, 120.0), DomeInsets::new(Distance::new(8.0), Distance::new(bottom_inset)));
176                let Vec2 { x: width, y: height, .. } = content.dome_content_for(radius, DomeInsets::new(Distance::new(8.0), Distance::new(bottom_inset)));
177                assert!(width >= 150.0 - 1e-9, "{content:?} width {width} < 150 (inset {bottom_inset})");
178                assert!(height >= 120.0 - 1e-9, "{content:?} height {height} < 120 (inset {bottom_inset})");
179            }
180        }
181    }
182
183    #[test]
184    fn dome_bottom_inset_grows_the_radius() {
185        // A bar under the content lifts it: the dome must grow so the
186        // same square still fits above the bar.
187        let flat = CardContent::Inscribed.dome_radius_for(Vec2::new(100.0, 100.0), DomeInsets::new(Distance::new(8.0), Distance::ZERO));
188        let lifted = CardContent::Inscribed.dome_radius_for(Vec2::new(100.0, 100.0), DomeInsets::new(Distance::new(8.0), Distance::new(36.0)));
189        assert!(lifted > flat);
190        let side = CardContent::Inscribed
191            .dome_content_for(lifted, DomeInsets::new(Distance::new(8.0), Distance::new(36.0)))
192            .x;
193        assert!((side - 100.0).abs() < 1e-9);
194    }
