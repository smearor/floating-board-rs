3/// Where in the fold choreography an animated fold
4/// (`CardConfig::fold_animation`) plays relative to the satellite
5/// set's close transitions.
6#[derive(Debug, Clone, Copy, PartialEq, Default)]
7#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize), serde(rename_all = "kebab-case"))]
8pub enum CardFoldTiming {
9    /// Satellites sink into the card first; the fold morph plays once
10    /// the last close transition lands. Reads as a causal sequence:
11    /// first the contents, then the container.
12    #[default]
13    Sequential,
14    /// The fold morph starts while the set is still closing - once the
15    /// leading satellite's close progress reaches the enclosed
16    /// fraction. `0.0` starts fully parallel with the close, values
17    /// approaching `1.0` converge on `Sequential` behavior.
18    Overlap(Fraction),
19}
