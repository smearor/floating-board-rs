1/// Where a [`crate::domain::ChromeElement`] is mounted on a card.
2#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Default)]
3#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize), serde(rename_all = "kebab-case"))]
4pub enum ChromePlacement {
5    /// Inside the card chrome at the renderer's default position -
6    /// header strip or content overlay, whichever the element's mount
7    /// semantics dictate. Action elements land on the trailing end.
8    #[default]
9    Inline,
10    /// Inside the header strip at the leading (`start`) end - the left
11    /// end in left-to-right direction, mirrored for RTL. Slotted order
12    /// is kept: the first `InlineStart` slot sits outermost.
13    InlineStart,
14    /// Inside the header strip at the trailing (`end`) end - the right
15    /// end in left-to-right direction. Slotted order is kept: the last
16    /// `InlineEnd` slot sits outermost.
17    InlineEnd,
18    /// As a satellite widget floating next to the card. Only elements
19    /// with [`crate::domain::ChromeElement::supports_satellite`] may use
20    /// this placement.
21    Satellite,
22    /// In the center region of a circle card's top arc band - the
23    /// explicit counterpart to `Inline` once arc placements are used
24    /// (see `arc_span` on `CardConfig`). Ignored on `Rectangle`.
25    ArcTop,
26    /// At the leading end of a circle card's top arc band - the card's
27    /// left rim. `InlineStart` maps to the same zone. Ignored on
28    /// `Rectangle`.
29    ArcTopStart,
30    /// At the trailing end of a circle card's top arc band - the
31    /// card's right rim. `InlineEnd` maps to the same zone. Ignored
32    /// on `Rectangle`.
33    ArcTopEnd,
34    /// In the center region of a curved card's bottom header bar - a
35    /// second band mirrored at the bottom: an arc band on a `Circle`,
36    /// a straight strip on the flat edge of a `Semicircle`. Ignored on
37    /// `Rectangle`.
38    Bar,
39    /// At the leading end of a curved card's bottom bar - the bar is
40    /// mirrored, so the leading end sits on the screen's right side:
41    /// a user facing the card's bottom reads start-to-end
42    /// left-to-right. Ignored on `Rectangle`.
43    BarStart,
44    /// At the trailing end of a curved card's bottom bar - the
45    /// mirrored counterpart of `ArcTopEnd`. Ignored on `Rectangle`.
46    BarEnd,
47}
48
49impl ChromePlacement {
50    /// Whether the placement mounts on a circle card's top arc band
51    /// (`Inline*` and `ArcTop*` placements).
52    pub fn is_top_arc(&self) -> bool {
53        matches!(self, Self::Inline | Self::InlineStart | Self::InlineEnd | Self::ArcTop | Self::ArcTopStart | Self::ArcTopEnd)
54    }
55
56    /// Whether the placement mounts on a curved card's bottom bar
57    /// (`Bar*` placements) - the geometry follows the shape.
58    pub fn is_bar(&self) -> bool {
59        matches!(self, Self::Bar | Self::BarStart | Self::BarEnd)
60    }
61}

67    #[test]

69        assert_eq!(ChromePlacement::default(), ChromePlacement::Inline);

72    #[cfg(feature = "serde")]
73    #[test]
74    fn inline_ends_serialize_kebab_case() {
75        let start = serde_json::to_string(&ChromePlacement::InlineStart).unwrap_or_default();
76        let end = serde_json::to_string(&ChromePlacement::InlineEnd).unwrap_or_default();
77        assert_eq!(start, r#""inline-start""#);
78        assert_eq!(end, r#""inline-end""#);
