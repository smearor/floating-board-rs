22/// The geometric outline a card renders and hit-tests with.
23///
24/// The widget hierarchy always stays rectangular - the shape decides how
25/// the content is allocated inside the bounding box, how the card's
26/// background is drawn and which points are pickable. On non-rectangular
27/// shapes there is no header strip and no corner grips; chrome elements
28/// mount as satellites instead.
29#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
30#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize), serde(rename_all = "kebab-case"))]
31pub enum CardShape {
32    /// The classic rectangular card: header strip plus rectangular content.
33    #[default]
34    Rectangle,
35    /// A circular card: the bounding box is the circle's square, the
36    /// content is allocated inside the circle (see [`CardContent`]) and
37    /// the ring between content and edge is the card's own rounded
38    /// background. Taps in the bounding-box corners pass through to
39    /// whatever is underneath.
40    ///
41    /// [`CardContent`]: crate::CircleContent
42    Circle,
43    /// A dome: the upper half of a circle - an arc on top, a flat
44    /// diameter edge at the bottom. The bounding box is 2r wide and r
45    /// tall; the content is allocated inside the dome (see
46    /// [`CardContent`], which also drives the dome's content
47    /// region). Taps outside the dome pass through to whatever is
48    /// underneath.
49    ///
50    /// Arc chrome mounts only on the top band (`Inline*`/`ArcTop*`
51    /// placements); `Bar*` placements mount a straight bar along
52    /// the flat bottom edge instead of an arc.
53    ///
54    /// [`CardContent`]: crate::CircleContent
55    Semicircle,
56    /// A flat-top hexagon: horizontal top and bottom edges, slanted
57    /// left/right edges meeting in vertices. The bounding box is
58    /// `2R` wide and `sqrt(3) * R` tall (`R` = circumradius); the
59    /// content lives in the inscribed square (see
60    /// `hexagon_content_for`). Taps in the bounding-box corners pass
61    /// through to whatever is underneath.
62    ///
63    /// There is no header strip and no corner grips; chrome elements
64    /// mount as a straight bar along the flat bottom edge.
65    Hexagon,
66    /// A flat-top octagon: horizontal/vertical flat edges, diagonal
67    /// edges cutting the four corners at 45 degrees. The bounding
68    /// box is the square `2a x 2a` (`a` = inradius, center to flat
69    /// edge); the content lives in the inscribed square (see
70    /// `octagon_content_for`). Taps in the bounding-box corners pass
71    /// through to whatever is underneath.
72    ///
73    /// There is no header strip and no corner grips; chrome elements
74    /// mount as straight bars along the flat top/bottom edges.
75    Octagon,
76    /// A ring / annulus: the area between the outer rim and the
77    /// see-through hole (`CardConfig::hole_ratio` of the outer
78    /// radius). The bounding box is the outer circle's square; the
79    /// hole shows whatever is underneath and taps inside it pass
80    /// through - the card surface lives on the band only.
81    ///
82    /// There is no header strip and no corner grips; chrome elements
83    /// mount along the ring's arc bands like on a `Circle`.
84    Annulus,
85}
86
87/// `sqrt(3) / 2` - a flat-top hexagon's height is `sqrt(3) * R`
88/// for circumradius `R`, so the folded pin height is
89/// `diameter * sqrt(3) / 2`.
90const SQRT_3_OVER_2: f64 = 0.866_025_403_784_438_6;
91
92impl CardShape {
93    /// How far the shape's outline extends along `direction`, measured
94    /// from the bounding box's center: the disk radius for a `Circle`
95    /// card, the axis-aligned box support `(|cos| * w + |sin| * h) / 2`
96    /// for a `Rectangle` card (exact for the bbox - the visible
97    /// corners of a rotated card slightly overshoot, which is fine
98    /// for an emerge seed). A `Semicircle` card orbits like its full
99    /// circle - the bbox's inscribed radius, a uniform gap all
100    /// around. A `Hexagon` card reports the exact support of the
101    /// flat-top hexagon (the largest vertex projection along the
102    /// direction); an `Octagon` card reports its exact support
103    /// likewise.
104    pub fn contour_support(&self, direction: RotationRad, footprint: Vec2<Pixels>) -> Distance<Pixels> {
105        match self {
106            Self::Circle | Self::Semicircle | Self::Annulus => Distance::new(footprint.x / 2.0),
107            Self::Rectangle => {
108                let (sin, cos) = direction.sin_cos();
109                Distance::new((cos.abs() * footprint.x + sin.abs() * footprint.y) / 2.0)
110            }
111            Self::Hexagon => Distance::new(Hexagon::flat_top(Vec2::<Pixels>::new(footprint.x / 2.0, footprint.y / 2.0), footprint.x / 2.0).support(direction)),
112            Self::Octagon => Distance::new(
113                Octagon::with_inradius(Vec2::<Pixels>::new(footprint.x / 2.0, footprint.y / 2.0), footprint.x / 2.0, PolygonOrientation::FlatTop)
114                    .support(direction),
115            ),
116        }
117    }
118
119    /// The radius of the centered circle the shape's outline fits
120    /// into: the disk radius for `Circle`, the bounding box's
121    /// circumscribed circle (`magnitude / 2`) for `Rectangle`, and
122    /// the dome's virtual-circle radius for `Semicircle` (the flat
123    /// edge's corners overshoot it by ~12% - the same uniform
124    /// approximation [`Self::contour_support`] documents). For a
125    /// `Hexagon` card the circumradius is exact - every vertex sits
126    /// on it. For an `Octagon` the vertices reach `a / cos(22.5°)`,
127    /// slightly past the bounding square's half extent.
128    pub fn bounding_radius(&self, footprint: Vec2<Pixels>) -> Distance<Pixels> {
129        match self {
130            Self::Circle | Self::Semicircle | Self::Hexagon | Self::Annulus => Distance::new(footprint.x / 2.0),
131            Self::Octagon => Distance::new(
132                Octagon::with_inradius(Vec2::<Pixels>::new(footprint.x / 2.0, footprint.y / 2.0), footprint.x / 2.0, PolygonOrientation::FlatTop).circumradius,
133            ),
134            Self::Rectangle => Distance::new(footprint.magnitude_f64() / 2.0),
135        }
136    }
137
138    /// The rotation snap target one step past `current` in
139    /// `direction`: a `Hexagon` steps in `SixthTurn`s - 60 degrees,
140    /// its sixfold symmetry - an `Octagon` in `EighthTurn`s (45
141    /// degrees), every other shape in `QuarterTurn`s. The typed
142    /// `Rotation` variant carries the shape's symmetry.
143    pub fn rotation_snap_target(&self, current: RotationRad, direction: RotationDirection) -> Rotation {
144        match self {
145            CardShape::Hexagon => SixthTurn::snap_target(current, direction).into(),
146            CardShape::Octagon => EighthTurn::snap_target(current, direction).into(),
147            _ => QuarterTurn::snap_target(current, direction).into(),
148        }
149    }
150
151    /// The angular distance between the shape's rotation snap
152    /// boundaries: 60 degrees for a `Hexagon`, 45 degrees for an
153    /// `Octagon`, 90 degrees for every other shape - the
154    /// drag-rotation counterpart of [`Self::rotation_snap_target`].
155    pub fn rotation_step(&self) -> Degrees {
156        match self {
157            CardShape::Hexagon => Degrees::SIXTH_TURN,
158            CardShape::Octagon => Degrees::EIGHTH_TURN,
159            _ => Degrees::QUARTER_TURN,
160        }
161    }
162
163    /// The shape's geometry inscribed in `bounds`: the disc for
164    /// `Circle`, the dome (center at the bounds' bottom middle) for
165    /// `Semicircle`, the flat-top polygons for `Hexagon`/`Octagon` and
166    /// the ring with `hole` as its inner-radius share for `Annulus`.
167    /// `Rectangle` has no inscribed outline - its contour is the
168    /// bounding box itself - so callers fall back to rectangular
169    /// defaults (`parent_contains` for hit-tests, no silhouette clip
170    /// path for snapshots).
171    pub fn outline(&self, bounds: &Rect<()>, hole: Fraction) -> Option<CardOutline> {
172        let outline = match self {
173            Self::Circle => CardOutline::Circle(Circle::inscribed_in(bounds)),
174            Self::Semicircle => CardOutline::Semicircle(Semicircle::inscribed_in(bounds)),
175            Self::Hexagon => CardOutline::Hexagon(Hexagon::inscribed_in(bounds, PolygonOrientation::FlatTop)),
176            Self::Octagon => CardOutline::Octagon(Octagon::inscribed_in(bounds, PolygonOrientation::FlatTop)),
177            Self::Annulus => CardOutline::Annulus(Annulus::inscribed_in(bounds, hole)),
178            Self::Rectangle => return None,
179        };
180        Some(outline)
181    }
182
183    /// Whether `point` hits the shape's contour inside `bounds` -
184    /// `None` for `Rectangle`, whose callers keep the rectangular
185    /// fallback test.
186    pub fn contains(&self, bounds: &Rect<()>, hole: Fraction, point: &Vec2<()>) -> Option<bool> {
187        self.outline(bounds, hole).map(|outline| outline.contains(bounds, point))
188    }
189
190    /// The radius the arc chrome rides at: an `Annulus` ring floats
191    /// its chrome on the band's middle (`(inner + outer) / 2`); every
192    /// other shape rides the inscribed circle's rim minus half the
193    /// `band` thickness.
194    pub fn arc_chrome_radius(&self, bounds: &Rect<()>, hole: Fraction, band: f64) -> f64 {
195        match self.outline(bounds, hole) {
196            Some(CardOutline::Annulus(annulus)) => (annulus.inner + annulus.outer) / 2.0,
197            _ => bounds.size.x.min(bounds.size.y) / 2.0 - band / 2.0,
198        }
199    }
200
201    /// The extent a folded card of this shape pins to, derived from
202    /// the content's `natural` size: a `Circle` collapses onto its
203    /// ring, so the diameter (the larger natural axis) stays pinned
204    /// as a square; a `Semicircle` dome keeps its `2r × r`
205    /// proportion; a `Hexagon` keeps its `2R × sqrt(3)·R`
206    /// proportion; a `Rectangle` pins the width only. A `-1` height
207    /// means the folded measure decides it.
208    pub fn collapsed_extent(&self, natural: Size<i32>) -> Size<i32> {
209        match self {
210            Self::Circle | Self::Annulus => {
211                let diameter = natural.x.max(natural.y);
212                Size::new(diameter, diameter)
213            }
214            Self::Semicircle => {
215                let diameter = natural.x.max(natural.y);
216                Size::new(diameter, diameter / 2)
217            }
218            Self::Hexagon => {
219                let diameter = natural.x.max(natural.y);
220                Size::new(diameter, (f64::from(diameter) * SQRT_3_OVER_2).ceil() as i32)
221            }
222            Self::Octagon => {
223                let diameter = natural.x.max(natural.y);
224                Size::new(diameter, diameter)
225            }
226            Self::Rectangle => Size::new(natural.x, -1),
227        }
228    }
229}

231#[cfg(test)]
232mod tests {
233    use super::*;
234
235    #[test]
236    fn default_shape_is_rectangle() {
237        assert_eq!(CardShape::default(), CardShape::Rectangle);
238    }
239
240    #[test]
241    fn circle_support_is_the_disk_radius_everywhere() {
242        let footprint = Vec2::<Pixels>::new(200.0, 200.0);
243        for direction in [RotationRad::ZERO, RotationRad::QUARTER_TURN, RotationRad::new(1.1)] {
244            assert_eq!(CardShape::Circle.contour_support(direction, footprint).as_f64(), 100.0);
245        }
246    }
247
248    #[test]
249    fn rectangle_support_follows_the_box_contour() {
250        let footprint = Vec2::<Pixels>::new(200.0, 100.0);
251        // Straight down: the support reaches the bottom edge (h/2).
252        assert!((CardShape::Rectangle.contour_support(RotationRad::QUARTER_TURN, footprint).as_f64() - 50.0).abs() < 1e-9);
253        // Diagonal (cos=0.8, sin=0.6): (0.8*200 + 0.6*100) / 2 = 110.
254        let direction = RotationRad::new(0.6435);
255        assert!((CardShape::Rectangle.contour_support(direction, footprint).as_f64() - 110.0).abs() < 1.0);
256    }
257
258    #[test]
259    fn bounding_radius_circumscribes_the_shape() {
260        // Circle/dome: the disk's own radius. Rectangle: the
261        // circumscribed circle sqrt(200^2 + 100^2) / 2.
262        let square = Vec2::<Pixels>::new(200.0, 200.0);
263        let rect = Vec2::<Pixels>::new(200.0, 100.0);
264        assert_eq!(CardShape::Circle.bounding_radius(square).as_f64(), 100.0);
265        assert_eq!(CardShape::Semicircle.bounding_radius(rect).as_f64(), 100.0);
266        assert!((CardShape::Rectangle.bounding_radius(rect).as_f64() - 111.803).abs() < 0.001);
267    }
268
269    #[test]
270    fn semicircle_support_matches_the_full_circle() {
271        let footprint = Vec2::<Pixels>::new(200.0, 100.0);
272        assert_eq!(CardShape::Semicircle.contour_support(RotationRad::QUARTER_TURN, footprint).as_f64(), 100.0);
273    }
274
275    #[test]
276    fn collapsed_extent_keeps_each_shape_silhouette() {
277        let natural = Size::new(200, 120);
278        // Circle: square pin on the larger axis; dome: 2r x r;
279        // rectangle: width only (-1 = the folded measure decides).
280        assert_eq!(CardShape::Circle.collapsed_extent(natural), Size::new(200, 200));
281        assert_eq!(CardShape::Semicircle.collapsed_extent(natural), Size::new(200, 100));
282        assert_eq!(CardShape::Rectangle.collapsed_extent(natural), Size::new(200, -1));
283        // Hexagon: 2R x sqrt(3)R pin on the larger axis.
284        assert_eq!(CardShape::Hexagon.collapsed_extent(natural), Size::new(200, 174));
285    }
286
287    #[test]
288    fn hexagon_support_reaches_the_vertices_exactly() {
289        let footprint = Vec2::<Pixels>::new(200.0, 174.0);
290        // Along +x the support is the circumradius (a vertex sits
291        // there); straight up it is the inradius R * sqrt(3)/2.
292        assert!((CardShape::Hexagon.contour_support(RotationRad::ZERO, footprint).as_f64() - 100.0).abs() < 1e-9);
293        assert!((CardShape::Hexagon.contour_support(RotationRad::QUARTER_TURN, footprint).as_f64() - 86.6).abs() < 0.01);
294        assert_eq!(CardShape::Hexagon.bounding_radius(footprint).as_f64(), 100.0);
295    }
