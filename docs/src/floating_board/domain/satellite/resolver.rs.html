22/// Pure placement resolution: card geometry + footprint + layout +
23/// measured items -> one [`SatellitePlacement`] per item, in workspace
24/// coordinates.
25///
26/// The sidebar path lets `EdgeStripLayout` arrange items in the
27/// card-local frame (card center at the origin, natural footprint),
28/// then maps each placement through the card transform: scale, rotate
29/// about the card center, translate. Widget orientation follows
30/// `mode` (`CoRotate` -> card rotation, the upright modes -> no
31/// rotation); positions are always card-local for a sidebar -
32/// `CounterRotate` orbit semantics are radial matter (Phase 3).
33/// Satellites inherit the card's scale factor (rigid docking).
34///
35/// The `Radial` path lets `OrbitLayout` place items on an arc around
36/// the card-local center (`radius` = distance outward from the card's
37/// bounding circle via `orbit_radius_for`), then maps through the same
38/// card transform. `CounterRotate` is radial-only semantics: orbit
39/// angles stay screen-fixed - positions scale and translate but do not
40/// rotate with the card, so rotating the card orbits nothing while
41/// dragging carries the whole ring along.
42pub fn resolve(
43    geometry: &CardGeometry,
44    card_footprint: Vec2<Pixels>,
45    card_shape: CardShape,
46    layout: &SatelliteLayout,
47    mode: SatelliteRotationMode,
48    items: &[SatelliteItem],
49) -> Result<Vec<SatellitePlacement>, LayoutError> {
50    match layout {
51        SatelliteLayout::Sidebar { sides } => resolve_sidebar(geometry, card_footprint, *sides, mode, items),
52        SatelliteLayout::Radial(config) => resolve_radial(geometry, card_footprint, card_shape, config, mode, items),
53    }
54}

56/// The `Sidebar` path: `EdgeStripLayout` in the card-local frame, then
57/// mapped through the card transform.
58fn resolve_sidebar(
59    geometry: &CardGeometry,
60    card_footprint: Vec2<Pixels>,
61    sides: SideSet,
62    mode: SatelliteRotationMode,
63    items: &[SatelliteItem],
64) -> Result<Vec<SatellitePlacement>, LayoutError> {
65    let bounds = Rect {
66        min: card_footprint * -0.5,
67        size: card_footprint,
68    };
69    let layout = EdgeStripLayout::<Pixels> { sides, ..Default::default() };
70    let layout_items: Vec<LayoutItem<Pixels>> = items.iter().map(|item| LayoutItem::new(item.size, Rotation::ZERO)).collect();
71    let placements = layout.arrange(bounds, &layout_items)?;
72    let rotation = match mode {
73        SatelliteRotationMode::CoRotate => geometry.rotation,
74        SatelliteRotationMode::KeepCardAngle | SatelliteRotationMode::CounterRotate => Rotation::ZERO,
75    };
76    let card_rotation = geometry.rotation.to_rotation_rad();
77    let scale = geometry.scale.value();
78    let center = Vec2::<Pixels>::from(geometry.position);
79    Ok(items
80        .iter()
81        .zip(placements.iter())
82        .map(|(item, placement)| SatellitePlacement {
83            id: item.id.clone(),
84            geometry: CardGeometry {
85                position: CardPosition::new(center + (placement.center * scale).rotated(card_rotation)),
86                rotation,
87                scale: geometry.scale,
88            },
89        })
90        .collect())
91}

93/// The `Radial` path: `OrbitLayout` in the card-local frame, then the
94/// card transform - [`SatelliteDistribution::Anchored`] forwards each
95/// item's anchor verbatim, `Even` drops them. `OrbitLayout` never
96/// fails (degenerate inputs yield deterministic placements), so this
97/// path always resolves.
98fn resolve_radial(
99    geometry: &CardGeometry,
100    card_footprint: Vec2<Pixels>,
101    card_shape: CardShape,
102    config: &RadialConfig,
103    mode: SatelliteRotationMode,
104    items: &[SatelliteItem],
105) -> Result<Vec<SatellitePlacement>, LayoutError> {
106    let bounds = Rect {
107        min: card_footprint * -0.5,
108        size: card_footprint,
109    };
110    // `radius` reads as the gap outward from the card's outline,
111    // added onto the shape's bounding radius (`CardShape::bounding_radius`
112    // - the bounding circle for a rectangle, the disk radius for
113    // round shapes).
114    let orbit_radius = card_shape.bounding_radius(card_footprint) + config.radius;
115    let layout = OrbitLayout::new(orbit_radius, config.span);
116    let anchored = config.distribution == SatelliteDistribution::Anchored;
117    let orbit_items: Vec<OrbitItem<Pixels>> = items
118        .iter()
119        .map(|item| {
120            let mut orbit_item = OrbitItem::new(item.size, Rotation::ZERO);
121            if anchored && let Some(anchor) = item.anchor {
122                orbit_item = orbit_item.with_anchor(anchor.to_rotation_rad());
123            }
124            orbit_item
125        })
126        .collect();
127    let placements = layout.arrange(bounds, &orbit_items);
128    let rotation = match mode {
129        SatelliteRotationMode::CoRotate => geometry.rotation,
130        SatelliteRotationMode::KeepCardAngle | SatelliteRotationMode::CounterRotate => Rotation::ZERO,
131    };
132    let card_rotation = geometry.rotation.to_rotation_rad();
133    let scale = geometry.scale.value();
134    let center = Vec2::<Pixels>::from(geometry.position);
135    Ok(items
136        .iter()
137        .zip(placements.iter())
138        .map(|(item, placement)| {
139            let offset = placement.center * scale;
140            // `CounterRotate` keeps orbit angles screen-fixed: the
141            // offset does not rotate with the card.
142            let offset = match mode {
143                SatelliteRotationMode::CounterRotate => offset,
144                _ => offset.rotated(card_rotation),
145            };
146            SatellitePlacement {
147                id: item.id.clone(),
148                geometry: CardGeometry {
149                    position: CardPosition::new(center + offset),
150                    rotation,
151                    scale: geometry.scale,
152                },
153            }
154        })
155        .collect())
156}

446    /// Rotation modes on the orbit: `CoRotate` rotates positions and
447    /// widgets with the card, `KeepCardAngle` rotates positions but
448    /// keeps widgets upright, `CounterRotate` keeps orbit angles
449    /// screen-fixed (position stays, widget upright).
450    #[test]
451    fn radial_rotation_modes() {
452        let items = vec![item("a", 30.0, 30.0)];
453        let radius = 200.0_f64.hypot(100.0) / 2.0 + 48.0;
454        for mode in [
455            SatelliteRotationMode::CoRotate,
456            SatelliteRotationMode::KeepCardAngle,
457            SatelliteRotationMode::CounterRotate,
458        ] {
459            for degrees in [0.0, 45.0, 90.0, 180.0] {
460                let placements = resolve(
461                    &card(0.0, 0.0, degrees, 1.0),
462                    Vec2::new(200.0, 100.0),
463                    CardShape::Rectangle,
464                    &radial(SatelliteDistribution::Even),
465                    mode,
466                    &items,
467                )
468                .expect("resolves");
469                let position = Vec2::<Pixels>::from(placements[0].geometry.position);
470                let widget_rotation = placements[0].geometry.rotation;
471                // A single item on a full circle sits at the span's
472                // midpoint - 180 degrees, opposite the arc seam.
473                let base = 180.0_f64;
474                match mode {
475                    SatelliteRotationMode::CoRotate => {
476                        // The card's rotation carries the angle along.
477                        let radians = (base + degrees).to_radians();
478                        assert!(
479                            (position.x - radius * radians.cos()).abs() < 1e-6 && (position.y - radius * radians.sin()).abs() < 1e-6,
480                            "{mode:?} {degrees}deg at {position:?}"
481                        );
482                        assert_eq!(widget_rotation, Rotation::from_degrees(degrees));
483                    }
484                    SatelliteRotationMode::KeepCardAngle => {
485                        let radians = (base + degrees).to_radians();
486                        assert!((position.x - radius * radians.cos()).abs() < 1e-6, "{mode:?} {degrees}deg at {position:?}");
487                        assert_eq!(widget_rotation, Rotation::ZERO);
488                    }
489                    SatelliteRotationMode::CounterRotate => {
490                        // Screen-fixed: 180 degrees in workspace
491                        // regardless of the card's rotation.
492                        assert!((position.x + radius).abs() < 1e-9 && (position.y).abs() < 1e-9, "{mode:?} {degrees}deg at {position:?}");
493                        assert_eq!(widget_rotation, Rotation::ZERO);
494                    }
495                }
496            }
497        }
498    }
