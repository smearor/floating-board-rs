8/// The "collapsed" satellite placement a satellite emerges from on
9/// open and sinks back into on close - the seed endpoint of the
10/// open/close choreography (Phase 5).
11///
12/// Per layout: `Radial` satellites fly out of the card center along
13/// their orbit angle (the radial pop); `Sidebar` satellites slide out
14/// of the card edge - the seed sits where the ray from the card center
15/// to the resolved position exits the card's contour. Both seeds carry
16/// the resolved rotation at minimal scale, so the lerp grows the
17/// satellite while it travels.
18pub(crate) fn collapsed_seed(
19    resolved: &CardGeometry,
20    card_geometry: &CardGeometry,
21    card_footprint: Vec2<Pixels>,
22    card_shape: CardShape,
23    layout: &SatelliteLayout,
24) -> CardGeometry {
25    let center = Vec2::<Pixels>::from(card_geometry.position);
26    let offset = Vec2::<Pixels>::from(resolved.position) - center;
27    let distance = offset.magnitude_f64();
28    let edge = match (layout, distance > 1e-9) {
29        (SatelliteLayout::Radial(_), _) | (_, false) => 0.0,
30        (SatelliteLayout::Sidebar { .. }, true) => contour_support(offset / distance, card_footprint, card_shape),
31    };
32    CardGeometry {
33        position: CardPosition::new(center + offset.normalized() * distance.min(edge)),
34        rotation: resolved.rotation,
35        scale: crate::domain::ScaleFactor::new(0.01),
36    }
37}
38
39/// How far the card's outline extends along unit direction `dir`,
40/// measured from its center: the disk radius for a `Circle` card, the
41/// axis-aligned box support `(|dx| * w + |dy| * h) / 2` for a
42/// `Rectangle` card (exact for the bbox - the visible corners of a
43/// rotated card slightly overshoot, which is fine for an emerge seed).
44fn contour_support(dir: Vec2<Pixels>, card_footprint: Vec2<Pixels>, card_shape: CardShape) -> f64 {
45    match card_shape {
46        CardShape::Circle => card_footprint.x / 2.0,
47        CardShape::Rectangle => (dir.x.abs() * card_footprint.x + dir.y.abs() * card_footprint.y) / 2.0,
48    }
49}

80    /// Radial seeds collapse to the card center so satellites pop
81    /// outward along their orbit angle.
82    #[test]
83    fn radial_seed_is_the_card_center() {
84        let seed = collapsed_seed(&placed(500.0, 100.0), &card(500.0, 300.0), Vec2::new(200.0, 100.0), CardShape::Rectangle, &radial());
85        let position = Vec2::<Pixels>::from(seed.position);
86        assert_eq!(position, Vec2::new(500.0, 300.0));
87        assert!(seed.scale.value() < 0.1);
88    }
89
90    /// A bottom-docked satellite's seed sits on the card's bottom
91    /// edge, directly under the dock slot - the slide-out start.
92    #[test]
93    fn sidebar_seed_sits_on_the_card_edge() {
94        let seed = collapsed_seed(&placed(500.0, 400.0), &card(500.0, 300.0), Vec2::new(200.0, 100.0), CardShape::Rectangle, &sidebar());
95        let position = Vec2::<Pixels>::from(seed.position);
96        assert!((position.x - 500.0).abs() < 1e-9);
97        assert!((position.y - 350.0).abs() < 1e-9, "seed at {position:?}");
98    }
99
100    /// Diagonal targets exit through the box support point
101    /// `(|dx| w + |dy| h) / 2` along the direction.
102    #[test]
103    fn sidebar_seed_follows_the_box_contour_diagonally() {
104        let seed = collapsed_seed(&placed(600.0, 450.0), &card(500.0, 300.0), Vec2::new(200.0, 100.0), CardShape::Rectangle, &sidebar());
105        let position = Vec2::<Pixels>::from(seed.position);
106        // dir = (100,150)/180.28, support = (0.555*200 + 0.832*100)/2
107        let dir = Vec2::<f64>::new(100.0, 150.0).normalized();
108        let support = (dir.x.abs() * 200.0 + dir.y.abs() * 100.0) / 2.0;
109        assert!((position.x - (500.0 + dir.x * support)).abs() < 1e-6, "x {}", position.x);
110        assert!((position.y - (300.0 + dir.y * support)).abs() < 1e-6, "y {}", position.y);
111    }
112
113    /// A circle card's seed exits at the disk radius, not the bbox
114    /// support.
115    #[test]
116    fn circle_seed_exits_at_the_disk_radius() {
117        let seed = collapsed_seed(&placed(500.0, 480.0), &card(500.0, 300.0), Vec2::new(200.0, 200.0), CardShape::Circle, &sidebar());
118        let position = Vec2::<Pixels>::from(seed.position);
119        assert!((position.y - 400.0).abs() < 1e-9, "seed at {position:?}");
120    }
121
122    /// The seed keeps the resolved rotation - emerge twists nothing.
123    #[test]
124    fn seed_keeps_the_resolved_rotation() {
125        let mut resolved = placed(500.0, 400.0);
126        resolved.rotation = smearor_commons_geometry::Rotation::from_degrees(45.0);
127        let seed = collapsed_seed(&resolved, &card(500.0, 300.0), Vec2::new(200.0, 100.0), CardShape::Rectangle, &radial());
128        assert!((seed.rotation.degrees() - 45.0).abs() < 1e-9);
129    }
