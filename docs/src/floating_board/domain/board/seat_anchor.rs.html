1/// Where the pass-to-seat operations anchor a `Seat`'s table edge.
2///
3/// Configured per board via `FloatingBoard::set_seat_anchor` and
4/// persisted in `BoardLayout`. The anchor affects only the seat
5/// placement of `rotate_to_next_user`/`rotate_to_next_occupied_seat`
6/// and the `Seat::nearest` derivation for unassigned cards - generic
7/// `LayoutStrategy` runs keep operating in workspace coordinates.
8#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
9#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize), serde(rename_all = "kebab-case"))]
10pub enum SeatAnchor {
11    /// Seats sit at the screen edge - where the users physically
12    /// sit. The whole placement (edge position, fan-out lane, margin,
13    /// target rotation) is computed in widget space and mapped back
14    /// through `to_workspace`, so a passed card lands at the visible
15    /// edge facing its user even after the view was panned, zoomed or
16    /// rotated. `Seat::nearest` derives from the rendered
17    /// (card + view) rotation. The clamp domain widens to the shown
18    /// region (`clamp_bounds`): cards may rest anywhere on screen,
19    /// including seat positions beyond the workspace extent.
20    #[default]
21    View,
22    /// Seats sit at the workspace extent - fixed positions on the
23    /// virtual table, independent of the camera. Passing a card lands
24    /// at the workspace edge wherever the view currently shows it,
25    /// and `Seat::nearest` derives from the workspace rotation.
26    Workspace,
27}
