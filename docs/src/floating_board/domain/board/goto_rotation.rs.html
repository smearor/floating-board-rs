1/// How `FloatingBoard::goto_card` resolves the camera rotation at the
2/// destination.
3#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
4#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize), serde(rename_all = "kebab-case"))]
5pub enum GotoRotation {
6    /// Rotate the camera so the card reads upright to the viewer
7    /// (its top edge towards the top of the screen).
8    #[default]
9    Upright,
10    /// Keep the current view rotation - the flight only pans and
11    /// zooms, the card keeps its on-screen orientation.
12    Keep,
13}
