13glib::wrapper! {
14    /// Single-child container that clips its content to the card
15    /// shape inscribed in its allocation - both in rendering and
16    /// picking.
17    ///
18    /// Mounted for `CardContent::Clipped` (cut-region picks fall
19    /// through to the card below, keeping the ring a drag
20    /// affordance) and `CardContent::Cover` (the full-box content is
21    /// cut to the silhouette).
22    pub struct ShapeClip(ObjectSubclass<ShapeClipImpl>)
23        @extends Widget,
24        @implements Accessible, Buildable, ConstraintTarget;
25}
26
27impl ShapeClip {
28    /// Creates an empty clip container cutting to `shape` - a
29    /// band of `inner_ratio` of its outer radius on `Annulus`,
30    /// ignored on every other shape.
31    pub(crate) fn new(shape: CardShape, inner_ratio: Fraction) -> Self {
32        let clip: Self = glib::Object::builder().build();
33        clip.imp().shape.set(shape);
34        clip.imp().inner_ratio.set(inner_ratio);
35        clip
36    }
37
38    /// Mounts `child` as the clipped content.
39    pub(crate) fn set_child(&self, child: &impl IsA<Widget>) {
40        child.set_parent(self);
41    }
42}
