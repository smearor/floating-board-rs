6/// Configuration of one satellite in `CardSatellites::satellites`.
7///
8/// Widget-layer type: a `Custom` satellite carries its consumer-provided
9/// widget, serialized like `CardConfig::folded_content` - skipped.
10#[derive(Debug, Clone, PartialEq)]
11#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
12pub struct SatelliteConfig {
13    /// Stable identifier - keys the resolver's placements back to this
14    /// satellite.
15    pub id: SatelliteId,
16    /// What the satellite renders.
17    pub kind: SatelliteKind,
18    /// Fixed anchor angle honored by `SatelliteDistribution::Anchored`
19    /// (a `QuarterTurn` anchor is `Rotation::QuarterTurn(..)`); `None`
20    /// = flexible satellite distributed by the layout.
21    #[cfg_attr(feature = "serde", serde(default))]
22    pub anchor: Option<Rotation>,
23    /// Consumer-provided widget - only meaningful for
24    /// `SatelliteKind::Custom`; `None` there skips the satellite with a
25    /// warning. Not serialized (like `folded_content`).
26    #[cfg_attr(feature = "serde", serde(skip))]
27    pub widget: Option<Widget>,
28}
29
30impl SatelliteConfig {
31    /// A chrome-element satellite (board-built widget, e.g. `Minimize`).
32    pub fn chrome(id: impl Into<SatelliteId>, element: crate::domain::ChromeElement) -> Self {
33        Self {
34            id: id.into(),
35            kind: SatelliteKind::Chrome(element),
36            anchor: None,
37            widget: None,
38        }
39    }
40
41    /// A custom satellite around a consumer-provided `widget`.
42    pub fn custom(id: impl Into<SatelliteId>, widget: impl Into<Widget>) -> Self {
43        Self {
44            id: id.into(),
45            kind: SatelliteKind::Custom,
46            anchor: None,
47            widget: Some(widget.into()),
48        }
49    }
50
51    /// Pins the satellite to `anchor` on the orbit - honored by
52    /// `SatelliteDistribution::Anchored` (a `QuarterTurn` anchor is
53    /// `Rotation::QuarterTurn(..)`); ignored by `Even`.
54    pub fn with_anchor(mut self, anchor: Rotation) -> Self {
55        self.anchor = Some(anchor);
56        self
57    }
58}
