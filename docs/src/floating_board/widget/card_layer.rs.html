1//! `CardLayer` - the transparent `gtk4::Fixed` subclass holding the
2//! card transform stacks.
3
4use crate::widget::imp::card_layer::CardLayerImpl;
5use gtk4::EventControllerLegacy;
6use gtk4::GestureDrag;
7use gtk4::gdk::Event;
8use gtk4::glib;
9use gtk4::graphene::Point;
10use gtk4::prelude::EventControllerExt;
11use gtk4::prelude::GestureDragExt;
12use gtk4::prelude::WidgetExt;
13use smearor_commons_geometry::Pixels;
14use smearor_commons_geometry::Vec2;
15
16glib::wrapper! {
17    /// Card layer: a `gtk4::Fixed` that is never itself an event
18    /// target - picks pass through to its card children or to the
19    /// board's content below.
20    pub struct CardLayer(ObjectSubclass<CardLayerImpl>)
21        @extends gtk4::Fixed, gtk4::Widget,
22        @implements gtk4::Accessible, gtk4::Buildable, gtk4::ConstraintTarget;
23}
24
25impl CardLayer {
26    /// Creates the empty card layer.
27    pub fn new() -> Self {
28        glib::Object::builder().build()
29    }
30
31    /// Pointer position of a drag gesture in layer coordinates, taken
32    /// from the gesture's own point tracking (`start_point + offset`
33    /// in its widget space, mapped through `compute_point`). The round
34    /// trip through the transform stack yields the true layer-space
35    /// position: GTK delivers the widget-space point via the inverse
36    /// transform, so applying the forward transform returns the surface
37    /// point - independent of the card's own motion, and always fresh
38    /// at signal emission time (unlike `current_event`, which can lag
39    /// one touch event behind).
40    pub(crate) fn gesture_pointer(&self, gesture: &GestureDrag) -> Option<Vec2<Pixels>> {
41        let start = Vec2::<Pixels>::from(gesture.start_point()?);
42        let offset = Vec2::<Pixels>::from(gesture.offset()?);
43        let widget = gesture.widget()?;
44        let point = widget.compute_point(self, &Point::from((start + offset).to_f32()))?;
45        Some(Vec2::<Pixels, f32>::from(point).to_f64())
46    }
47
48    /// Touch position of a raw event in layer coordinates: the event
49    /// reports its position in surface space; it is mapped into the
50    /// controller widget's coordinate space through the native's
51    /// `compute_point` and then down the transform stack into layer
52    /// (board) space.
53    pub(crate) fn event_pointer(&self, controller: &EventControllerLegacy, event: &Event) -> Option<Vec2<Pixels>> {
54        let position = Vec2::<Pixels>::from(event.position()?);
55        let widget = controller.widget()?;
56        let native = widget.native()?;
57        let surface_point = Point::from(position.to_f32());
58        let widget_point = native.compute_point(&widget, &surface_point)?;
59        let point = widget.compute_point(self, &widget_point)?;
60        Some(Vec2::<Pixels, f32>::from(point).to_f64())
61    }
62}
63
64impl Default for CardLayer {
65    fn default() -> Self {
66        Self::new()
67    }
