1//! Satellite mounting and geometry sync, implemented on [`FloatingBoard`].
2//!
3//! Satellites are `CardLayer` children next to their card, never inside
4//! it: each gets the same mini transform stack (`ScalingWidget` ->
5//! `RotationWidget` -> satellite widget) a card gets, stacked directly
6//! above `entry.scaling`. Placements come from the pure domain resolver
7//! ([`resolve`]) and are re-applied on every card geometry change via
8//! [`FloatingBoard::apply_position`], so satellites follow drag, rotate,
9//! pinch and layout animations without their own tracking code.

64impl FloatingBoard {
65    /// Builds and mounts one [`SatelliteEntry`] per effective satellite
66    /// config (chrome satellite slots plus `CardSatellites::satellites`),
67    /// each stacked directly above the card's transform stack in config
68    /// order. Satellites whose widget cannot be built are warned about
69    /// and skipped.
70    pub(crate) fn build_satellites(&self, layer: &CardLayer, card: &Card, card_scaling: &ScalingWidget, config: &CardConfig) -> Vec<SatelliteEntry> {
71        let (settings, configs) = config.effective_satellites();
72        let radial = matches!(settings.layout, SatelliteLayout::Radial(_));
73        let chrome_style = config.chrome_style.resolved();
74        let mut entries = Vec::with_capacity(configs.len());
75        for satellite in configs {
76            let Some(widget) = satellite_widget(card, &satellite, chrome_style) else {
77                continue;
78            };
79            if radial {
80                widget.add_css_class(CLASS_SATELLITE_RADIAL);
81            }
82            // Same builder options as the card stack: no built-in rotate
83            // gesture, no input-region clamp, natural child size.
84            let rotation_widget = RotationWidget::new(SmearorRotation::DEG_0)
85                .with_gesture_rotation_enabled(false)
86                .with_input_region_enabled(false)
87                .with_natural_child_size_enabled(true);
88            rotation_widget.set_child(Some(&widget));
89            let scaling = ScalingWidget::new(&rotation_widget);
90            // Group z-order: the first satellite sits directly above the
91            // card, each next directly above the previous satellite -
92            // raising the card re-puts the whole group (see `raise_card`).
93            let sibling = entries
94                .last()
95                .map_or_else(|| card_scaling.upcast_ref::<Widget>(), |entry: &SatelliteEntry| entry.scaling.upcast_ref::<Widget>());
96            scaling.insert_after(layer, Some(sibling));
97            entries.push(SatelliteEntry {
98                id: satellite.id,
99                kind: satellite.kind,
100                anchor: satellite.anchor,
101                widget,
102                scaling,
103                rotation_widget,
104                last_geometry: Cell::new(None),
105                open: Cell::new(settings.open),
106                transition: RefCell::new(None),
107                last_local_geometry: Cell::new(None),
108            });
109        }
110        entries
111    }

113    /// Re-resolves every satellite placement from the card's current
114    /// geometry and applies it to the satellite transform stacks -
115    /// through the card's own view mapping, so a sticky card pins its
116    /// satellites to the glass with it. Called from `apply_position`,
117    /// the tail of every card geometry mutation.
118    ///
119    /// Visibility: satellites hide while their card is folded or hidden
120    /// (v1 rule); a layout that does not fit hides them all with a
121    /// warning rather than dropping entries.
122    pub(crate) fn sync_satellites(&self, entry: &CardEntry) {
123        let satellites = entry.satellites.borrow();
124        if satellites.is_empty() {
125            return;
126        }
127        let Some(layer) = self.card_layer() else {
128            return;
129        };
130        let (settings, _) = entry.config.effective_satellites();
131        let items = satellite_items(&satellites);
132        let geometry = self.entry_geometry(entry);
133        let (_, natural) = entry.card.measure_size();
134        let footprint = natural.to_f64().with_unit::<Pixels>();
135        // Folded or hidden cards hide their satellite set entirely.
136        let folded = entry.card.folded();
137        let shown = !folded && entry.scaling.is_visible();
138        // Exception: a circle card has no header strip - its `Minimize`
139        // satellite is the only fold/unfold control and stays visible
140        // while folded (otherwise the fold would be a dead end).
141        let fold_exception =
142            folded && entry.scaling.is_visible() && matches!(entry.config.shape, CardShape::Circle | CardShape::Annulus) && !entry.config.unfold_on_click;
143        match resolve(&geometry, footprint, entry.config.shape, &settings.layout, settings.rotation_mode, &items) {
144            Ok(placements) => {
145                for satellite in satellites.iter() {
146                    match placements.iter().find(|placement| placement.id == satellite.id) {
147                        Some(placement) => {
148                            // Retarget detection (P5.2): a placement that
149                            // moved in card-local space (footprint or
150                            // layout change, not a card gesture) tweens
151                            // instead of jumping.
152                            self.maybe_begin_move_transition(entry, satellite, placement, &geometry);
153                            let effective =
154                                self.effective_satellite_geometry(satellite, &placement.geometry, &geometry, footprint, entry.config.shape, &settings.layout);
155                            // The circle `Minimize` fold exception also
156                            // covers the choreography window: it never
157                            // closes, so it must stay rendered while the
158                            // rest of the set animates into the seed.
159                            let fold_control = matches!(satellite.kind, SatelliteKind::Chrome(ChromeElement::Minimize))
160                                && matches!(entry.config.shape, CardShape::Circle | CardShape::Annulus)
161                                && !entry.config.unfold_on_click
162                                && (fold_exception || (entry.fold_choreo_closing.get() && entry.scaling.is_visible()));
163                            // A running close transition keeps its
164                            // satellite visible until it lands in the seed.
165                            // During the fold choreography `open` only
166                            // carries the reveal intent for the unfold -
167                            // without this carve-out a satellite whose
168                            // transition already retired would re-render
169                            // at its resolved placement (the close-end
170                            // flash).
171                            let transitioning = satellite.transition.borrow().is_some();
172                            let vis = fold_control || (shown && (transitioning || (satellite.open.get() && !entry.fold_choreo_closing.get())));
173                            satellite.scaling.set_visible(vis);
174                            // Transitions also fade: open fades in, close
175                            // fades out, retargets stay opaque (P5.1).
176                            satellite.scaling.set_opacity(transition_opacity(satellite));
177                            self.apply_satellite_placement(entry, satellite, &layer, &effective);
178                        }
179                        None => {
180                            satellite.scaling.set_visible(false);
181                            satellite.last_geometry.set(None);
182                        }
183                    }
184                }
185            }
186            Err(error) => {
187                warn!(card = %entry.card.id(), %error, "satellite layout does not fit - hiding satellites");
188                for satellite in satellites.iter() {
189                    satellite.scaling.set_visible(false);
190                    satellite.last_geometry.set(None);
191                }
192            }
193        }
194    }

196    /// Re-stacks a card's satellites directly above it - same put-on-top
197    /// semantics `raise_card`/`restack` use for the card stack itself,
198    /// so the group (card + its satellites) always raises together.
199    /// Positions are preserved; only the child order changes.
200    pub(crate) fn restack_satellites(&self, entry: &CardEntry, layer: &CardLayer) {
201        for satellite in entry.satellites.borrow().iter() {
202            let position = Vec2::<Pixels>::from(layer.child_position(&satellite.scaling));
203            layer.remove(&satellite.scaling);
204            layer.put(&satellite.scaling, position.x, position.y);
205        }
206    }

208    /// Applies one resolved workspace placement to a satellite's stack
209    /// (also recorded in `last_geometry` for the clamp union):
210    /// rendered rotation = placement rotation + view rotation, rendered
211    /// scale = placement scale x view zoom, position = view-mapped
212    /// center - the same mapping `sync_rendered_transform` applies to
213    /// the card.
214    fn apply_satellite_placement(&self, entry: &CardEntry, satellite: &SatelliteEntry, layer: &CardLayer, geometry: &CardGeometry) {
215        let view = self.card_view(entry);
216        let rendered_rotation = geometry.rotation.to_rotation_rad() + view.rotation;
217        satellite.rotation_widget.set_rotation(SmearorRotation::from(rendered_rotation));
218        let rendered_scale = geometry.scale.value() * self.card_view_zoom(entry);
219        satellite.scaling.set_scale(rendered_scale);
220        // `gtk4::Fixed` allocates (minimum width, natural height), so
221        // centering on the natural size would displace the rendered
222        // widget by (nat - min) / 2 - same contract the card stack
223        // follows via `stack_top_left`.
224        let (minimum, natural) = satellite.widget.measure_size();
225        let min_size = (minimum.to_f64().with_unit::<Pixels>() * rendered_scale).rotated_bounds(rendered_rotation);
226        let nat_size = (natural.to_f64().with_unit::<Pixels>() * rendered_scale).rotated_bounds(rendered_rotation);
227        let rendered_size = Vec2::<Pixels>::new(min_size.x, nat_size.y);
228        let widget = self.widget_bounds();
229        let center = match widget.is_degenerate() {
230            true => Vec2::<Pixels>::from(geometry.position),
231            false => view.to_widget(geometry.position.into(), widget),
232        };
233        layer.move_(&satellite.scaling, center.x - rendered_size.x / 2.0, center.y - rendered_size.y / 2.0);
234        satellite.last_geometry.set(Some(*geometry));
235    }

237    /// The placement actually applied this frame (P5.1/5.2): the live
238    /// resolved placement, or - while a [`SatelliteTransition`] is in
239    /// flight - the interpolation between the frozen start (a
240    /// mid-flight reversal or retarget spot) / collapsed seed and the
241    /// live endpoint. Because the endpoint resolves live on every sync,
242    /// card gestures keep carrying an animating satellite rigidly.
243    fn effective_satellite_geometry(
244        &self,
245        satellite: &SatelliteEntry,
246        resolved: &CardGeometry,
247        card_geometry: &CardGeometry,
248        card_footprint: Vec2<Pixels>,
249        card_shape: CardShape,
250        layout: &SatelliteLayout,
251    ) -> CardGeometry {
252        let transition = satellite.transition.borrow();
253        let Some(transition) = transition.as_ref() else {
254            return *resolved;
255        };
256        let progress = Fraction::new(transition.animation.value().unwrap_or(0.0));
257        let seed = || collapsed_seed(resolved, card_geometry, card_footprint, card_shape, layout);
258        let start = transition.from.unwrap_or_else(seed);
259        let end = if transition.opening { *resolved } else { seed() };
260        start.interpolate(end, progress)
261    }

263    /// Retarget tween (P5.2): when the resolved placement moved in
264    /// card-local space - a footprint or layout change, not a card
265    /// gesture - the satellite freezes its rendered spot and tweens to
266    /// the new target instead of jumping. Card gestures transform the
267    /// resolved placement identically, so the local placement stays
268    /// constant and drags remain rigidly coupled.
269    fn maybe_begin_move_transition(&self, entry: &CardEntry, satellite: &SatelliteEntry, placement: &SatellitePlacement, card_geometry: &CardGeometry) {
270        let local = card_local_geometry(&placement.geometry, card_geometry);
271        let previous = satellite.last_local_geometry.replace(Some(local));
272        // Hidden satellites (closed set, folded card) resolve retargets
273        // silently in the background: a move tween is `opening`-typed
274        // and would re-show the satellite mid-flight - the
275        // close-end/unfold flash. Reveals go through
276        // `begin_satellite_transitions` instead.
277        if !satellite.open.get()
278            || !satellite.scaling.is_visible()
279            || satellite.transition.borrow().is_some()
280            || entry.card.folded()
281            || entry.fold_choreo_closing.get()
282            || !entry.satellite_animations_enabled()
283        {
284            return;
285        }
286        let Some(previous) = previous else {
287            return;
288        };
289        if !previous.differs_beyond_deadband(&local) {
290            return;
291        }
292        let mut animation = TimedAnimation::new(Span::new(0.0, 1.0), entry.transition_duration(), EasingFunction::EaseInOut);
293        animation.start();
294        *satellite.transition.borrow_mut() = Some(SatelliteTransition {
295            animation,
296            from: satellite.last_geometry.get(),
297            opening: true,
298        });
299        self.start_satellite_motion(entry);
300    }

302    /// Begins a staggered open (`opening`) or close transition batch on
303    /// every satellite (P5.1): each gets a `TimedAnimation` delayed by
304    /// its config index, interpolating between the frozen mid-flight
305    /// spot - or the collapsed seed on first open - and the live
306    /// endpoint. `skip_fold_control` excludes the circle card's
307    /// `Minimize` satellite, whose fold exception keeps it visible
308    /// while folded. Reduced motion and `animation_speed_ms == 0`
309    /// resolve instantly instead.
310    pub(crate) fn begin_satellite_transitions(&self, entry: &CardEntry, opening: bool, skip_fold_control: bool) {
311        if !entry.satellite_animations_enabled() {
312            self.sync_satellites(entry);
313            return;
314        }
315        let satellites = entry.satellites.borrow();
316        if satellites.is_empty() {
317            return;
318        }
319        let (settings, _) = entry.config.effective_satellites();
320        let items = satellite_items(&satellites);
321        let geometry = self.entry_geometry(entry);
322        let (_, natural) = entry.card.measure_size();
323        let footprint = natural.to_f64().with_unit::<Pixels>();
324        if resolve(&geometry, footprint, entry.config.shape, &settings.layout, settings.rotation_mode, &items).is_err() {
325            drop(satellites);
326            self.sync_satellites(entry);
327            return;
328        }
329        let duration = entry.transition_duration();
330        // The stagger spreads the starts over the first 40% of the
331        // shared duration so the set reads as one choreographed motion.
332        let stagger = duration.mul_f64(0.4) / satellites.len().max(1) as u32;
333        let now = Instant::now();
334        let mut any = false;
335        for (index, satellite) in satellites.iter().enumerate() {
336            let fold_protected = skip_fold_control
337                && !opening
338                && matches!(entry.config.shape, CardShape::Circle | CardShape::Annulus)
339                && !entry.config.unfold_on_click
340                && matches!(satellite.kind, SatelliteKind::Chrome(ChromeElement::Minimize));
341            if fold_protected {
342                continue;
343            }
344            // A close on a satellite that is already logically closed
345            // (hidden set) would briefly re-show it mid-tween - skip.
346            // `open` is the right discriminator, not `is_visible()`:
347            // the fold choreography's revert sync legitimately hides
348            // the satellites a call earlier, but they must still get
349            // their close transitions.
350            if !opening && !satellite.open.get() && satellite.transition.borrow().is_none() {
351                continue;
352            }
353            // `from` semantics (P5.1): only a transition already in
354            // flight freezes its rendered spot (mid-flight reversal).
355            // A fresh open emerges from the collapsed seed - hidden
356            // satellites keep receiving resolved placements via
357            // background syncs, so `last_geometry` is NOT a frozen spot.
358            let from = match satellite.transition.borrow().is_some() {
359                true => satellite.last_geometry.get(),
360                false if !opening => satellite.last_geometry.get(),
361                false => None,
362            };
363            let easing = match opening {
364                true => EasingFunction::BackOut { overshoot_amount: 1.2 },
365                false => EasingFunction::EaseIn,
366            };
367            let mut animation = TimedAnimation::new(Span::new(0.0, 1.0), duration, easing);
368            animation.start_at(now + stagger.mul_f64(index as f64));
369            *satellite.transition.borrow_mut() = Some(SatelliteTransition { animation, from, opening });
370            any = true;
371        }
372        drop(satellites);
373        if any {
374            self.start_satellite_motion(entry);
375        }
376        self.sync_satellites(entry);
377    }

379    /// Starts (or restarts) the per-entry motion tick that re-syncs
380    /// satellite transitions frame by frame until none are left. A new
381    /// batch bumps `satellites_generation`, so a stale tick stops
382    /// itself - the same pattern `layout_generation` uses for card
383    /// glides. When the last transition of a fold choreography lands,
384    /// the tick applies the reverted fold through `set_folded`, whose
385    /// `card-folded` handler performs the final sync.
386    pub(crate) fn start_satellite_motion(&self, entry: &CardEntry) {
387        let generation = entry.satellites_generation.get().incremented();
388        entry.satellites_generation.set(generation);
389        let card_id = entry.card.id();
390        self.add_tick_callback(move |board, _frame_clock| {
391            enum Next {
392                Running,
393                Done,
394                ApplyFold,
395            }
396            let next = {
397                let cards = board.imp().cards.borrow();
398                let Some(entry) = cards.get(&card_id) else {
399                    return ControlFlow::Break;
400                };
401                if entry.satellites_generation.get() != generation {
402                    return ControlFlow::Break;
403                }
404                for satellite in entry.satellites.borrow().iter() {
405                    let finished = satellite
406                        .transition
407                        .borrow()
408                        .as_ref()
409                        .is_some_and(|transition| transition.animation.is_complete());
410                    if finished {
411                        satellite.transition.take();
412                    }
413                }
414                if entry.satellites.borrow().iter().any(|satellite| satellite.transition.borrow().is_some()) {
415                    // `Overlap(at)` fold timing: the morph starts once
416                    // the leading close transition passes the fraction.
417                    // `.get()` keeps the flag for `Sequential` entries
418                    // and for fractions not reached yet.
419                    if entry.fold_morph_pending.get()
420                        && let CardFoldTiming::Overlap(at) = entry.config.fold_timing
421                    {
422                        let progress = entry
423                            .satellites
424                            .borrow()
425                            .iter()
426                            .filter_map(|satellite| satellite.transition.borrow().as_ref().and_then(|transition| transition.animation.value()))
427                            .fold(0.0, f64::max);
428                        if progress >= at.as_f64().max(0.0) {
429                            entry.fold_morph_pending.set(false);
430                            let (_, natural) = entry.card.measure_size();
431                            board.begin_fold_animation(entry, true, natural.to_f64().with_unit::<Pixels>(), entry.card.folded_natural_size());
432                        }
433                    }
434                    board.sync_satellites(entry);
435                    Next::Running
436                } else if entry.fold_choreo_closing.get() && entry.fold_animation.borrow().is_some() {
437                    // An overlapping fold morph is still running - let
438                    // it finish before the real fold lands.
439                    Next::Running
440                } else if entry.fold_choreo_closing.get() {
441                    // Fold choreography finalization happens through
442                    // `set_folded`, which syncs itself - syncing here
443                    // first would pop the satellites back to their open
444                    // placements for a frame.
445                    Next::ApplyFold
446                } else {
447                    board.sync_satellites(entry);
448                    Next::Done
449                }
450            };
451            if matches!(next, Next::ApplyFold) {
452                {
453                    let cards = board.imp().cards.borrow();
454                    if let Some(entry) = cards.get(&card_id) {
455                        board.finish_fold(entry);
456                    }
457                }
458                return ControlFlow::Break;
459            }
460            match next {
461                Next::Running => ControlFlow::Continue,
462                Next::Done | Next::ApplyFold => ControlFlow::Break,
463            }
464        });
465    }

467    /// `card-folded` reaction (P5.4): a real fold first *reverts* the
468    /// instant collapse - same frame, never rendered - animates the
469    /// satellite set into the collapsed seed, and the motion tick
470    /// applies the actual fold once the last transition lands. Unfolds
471    /// replay the open transition. `fold_choreo_closing` marks the
472    /// choreography's own intermediate states so it does not retrigger
473    /// itself.
474    pub(crate) fn on_card_folded(&self, entry: &CardEntry) {
475        if entry.card.folded() {
476            if entry.fold_choreo_closing.replace(false) {
477                // The choreography's own delayed fold just landed.
478                self.sync_satellites(entry);
479                return;
480            }
481            let closeable = {
482                let satellites = entry.satellites.borrow();
483                satellites.iter().any(|satellite| {
484                    satellite.open.get()
485                        && !(matches!(entry.config.shape, CardShape::Circle | CardShape::Annulus)
486                            && !entry.config.unfold_on_click
487                            && matches!(satellite.kind, SatelliteKind::Chrome(ChromeElement::Minimize)))
488                })
489            };
490            if closeable && entry.satellite_animations_enabled() {
491                entry.fold_choreo_closing.set(true);
492                // Re-enters this handler with `folded == false` and the
493                // choreography flag set - the branch below starts the
494                // close transitions.
495                entry.card.set_folded(false);
496                return;
497            }
498            self.sync_satellites(entry);
499            return;
500        }
501        if entry.fold_choreo_closing.get() {
502            // The choreography's own revert - the card stands unfolded
503            // again while the set animates closed; the tick re-folds at
504            // the end even when no transition was actually created.
505            self.begin_satellite_transitions(entry, false, true);
506            self.start_satellite_motion(entry);
507            // `Overlap(0)` fold timing: the morph starts in lockstep
508            // with the close (later fractions fire in the motion tick).
509            // `.get()` first - `replace` must not eat a pending flag
510            // that belongs to a later trigger point.
511            if entry.fold_morph_pending.get() && matches!(entry.config.fold_timing, CardFoldTiming::Overlap(at) if at <= Fraction::ZERO) {
512                entry.fold_morph_pending.set(false);
513                let (_, natural) = entry.card.measure_size();
514                self.begin_fold_animation(entry, true, natural.to_f64().with_unit::<Pixels>(), entry.card.folded_natural_size());
515            }
516            return;
517        }
518        // The unfold morph gates the reveal (`unfold_timing`): while
519        // it runs below its trigger fraction the set stays hidden and
520        // the fold tick fires the reveal itself.
521        if entry.unfold_reveal_pending.get() {
522            self.sync_satellites(entry);
523            return;
524        }
525        // A genuine unfold reveals the satellite set again.
526        let open = entry.satellites.borrow().iter().any(|satellite| satellite.open.get());
527        if open && entry.satellite_animations_enabled() {
528            self.begin_satellite_transitions(entry, true, false);
529            return;
530        }
531        self.sync_satellites(entry);
532    }
533}

535/// Builds the satellite's widget: chrome elements come from the card's
536/// own mount-agnostic builders (P1.5), custom satellites hand over the
537/// consumer's widget (the board owns placement and lifetime only).
538/// `chrome_style` is the resolved `CardConfig::chrome_style`: `Adwaita`
539/// dresses chrome satellites as themed round buttons.
540fn satellite_widget(card: &Card, config: &SatelliteConfig, chrome_style: ChromeStyle) -> Option<Widget> {
541    let widget = match config.kind {
542        SatelliteKind::Chrome(ChromeElement::Minimize) => card.build_minimize_button(ICON_MINIMIZE_SYMBOLIC).upcast(),
543        SatelliteKind::Chrome(ChromeElement::Close) => card.build_close_button().upcast(),
544        SatelliteKind::Chrome(ChromeElement::RotateClockwise) => card.build_rotate_button(RotationDirection::Clockwise).upcast(),
545        SatelliteKind::Chrome(ChromeElement::RotateCounterClockwise) => card.build_rotate_button(RotationDirection::CounterClockwise).upcast(),
546        SatelliteKind::Chrome(ChromeElement::NextSeat) => card.build_pass_button(TurnDirection::Clockwise).upcast(),
547        SatelliteKind::Chrome(ChromeElement::PreviousSeat) => card.build_pass_button(TurnDirection::CounterClockwise).upcast(),
548        SatelliteKind::Chrome(element) => {
549            warn!(satellite = %config.id, element = ?element, "chrome element is not satellite-capable - satellite skipped");
550            return None;
551        }
552        SatelliteKind::Custom => match &config.widget {
553            Some(widget) => widget.clone(),
554            None => {
555                warn!(satellite = %config.id, "custom satellite without widget - skipped");
556                return None;
557            }
558        },
559    };
560    widget.add_css_class(CLASS_SATELLITE);
561    match config.kind {
562        SatelliteKind::Chrome(_) => {
563            widget.add_css_class(CLASS_SATELLITE_CHROME);
564            if chrome_style == ChromeStyle::Adwaita {
565                // The libadwaita look: a raised themed round button
566                // (`circular`) instead of the builtin flat icon
567                // button. `circular` is a core GTK4 class - without
568                // `adw::init()` it still renders a round button face.
569                widget.remove_css_class("flat");
570                widget.add_css_class("circular");
571            }
572        }
573        SatelliteKind::Custom => widget.add_css_class(CLASS_SATELLITE_CUSTOM),
574    }
575    Some(widget)
576}

578/// The rendered opacity during a transition: a fresh open (no frozen
579/// start - it emerges from the seed) fades in, a close fades out.
580/// Retargets and mid-flight reversals keep a frozen start and stay
581/// opaque; steady state is fully opaque.
582fn transition_opacity(satellite: &SatelliteEntry) -> f64 {
583    let transition = satellite.transition.borrow();
584    let Some(transition) = transition.as_ref() else {
585        return 1.0;
586    };
587    let progress = transition.animation.value().unwrap_or(0.0);
588    match (transition.opening, transition.from.is_some()) {
589        (true, false) => progress,
590        (true, true) => 1.0,
591        (false, _) => 1.0 - progress,
592    }
593    .clamp(0.0, 1.0)
594}

596/// Live resolver input for the mounted satellites: ids, current natural
597/// sizes and anchored angles.
598fn satellite_items(satellites: &[SatelliteEntry]) -> Vec<SatelliteItem> {
599    satellites
600        .iter()
601        .map(|satellite| SatelliteItem {
602            id: satellite.id.clone(),
603            size: satellite.widget.measure_size().1.to_f64().with_unit::<Pixels>(),
604            anchor: satellite.anchor,
605        })
606        .collect()
607}

609/// Expresses a resolved workspace placement in the card's local frame:
610/// the position offset unrotated and unscaled, rotation and scale
611/// relative to the card. Card gestures transform the resolved placement
612/// identically, so a *changed* local placement means the placement
613/// itself retargeted (footprint or layout change).
614fn card_local_geometry(placement: &CardGeometry, card_geometry: &CardGeometry) -> CardGeometry {
615    let center = Vec2::<Pixels>::from(card_geometry.position);
616    let offset = Vec2::<Pixels>::from(placement.position) - center;
617    let scale = card_geometry.scale.value().max(1e-9);
618    CardGeometry {
619        position: CardPosition::new(offset.rotated(-card_geometry.rotation.to_rotation_rad()) / scale),
620        rotation: Rotation::quantized(placement.rotation.to_rotation_rad() - card_geometry.rotation.to_rotation_rad()),
621        scale: ScaleFactor::new(placement.scale.value() / scale),
622    }
623}
