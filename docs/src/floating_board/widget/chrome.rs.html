1//! Chrome element builders and mounting, implemented on [`Card`].
2//!
3//! One [`ChromeElement`] produces one widget plus its signal wiring
4//! (P1.1). `Inline` elements mount either into the card's header strip
5//! (drag handle, title, minimize, close) or onto the content overlay
6//! (resize/rotate grips); `Satellite` placements are validated and
7//! skipped until satellite mounting lands (Phase 2/3).
8//!
9//! The `builtin` renderer assembles the stock GTK4/CSS chrome; the
10//! `adwaita` feature adds a libadwaita renderer (`adw::HeaderBar`,
11//! `adw::WindowTitle`, wired buttons in a `windowcontrols` node). The
12//! bare widget factories at the bottom of this module stay free
13//! functions - they are stateless builders for foreign types (`Label`,
14//! `DrawingArea`).
15

76impl Card {
77    /// Builds the card chrome from `config`'s effective slot list and
78    /// mounts every `Inline` element: header elements onto a header
79    /// strip prepended to the card, grip elements onto the content
80    /// overlay.
81    ///
82    /// The header widget lands in `CardImpl::header` - it stays the
83    /// drag affordance for `DragHandle::Header` and the strip that
84    /// remains visible while folded. A header without any inline
85    /// children is not mounted at all (an explicit `chrome: []` config
86    /// produces a chrome-free card; use `DragHandle::WholeCard` or
87    /// `headerless` for those).
88    pub(crate) fn mount_chrome(&self, config: &CardConfig) {
89        let title = config.title.clone().unwrap_or_else(|| l10n().message("card-untitled"));
90        *self.imp().title_text.borrow_mut() = title.clone();
91        let slots = config.effective_chrome_slots();
92        let style = config.chrome_style.resolved();
93        if style != config.chrome_style {
94            warn!("CardConfig::chrome_style is Adwaita but the `adwaita` feature is disabled - falling back to Builtin");
95        }
96        // Curved cards have no header strip: a circle's inline chrome
97        // mounts along the disc's arc bands as an "arc header", a
98        // semicircle's on the top arc plus a straight bottom bar;
99        // `Satellite` placements stay satellites either way.
100        if matches!(
101            config.shape,
102            CardShape::Circle | CardShape::Semicircle | CardShape::Hexagon | CardShape::Octagon | CardShape::Annulus
103        ) {
104            self.mount_arc_chrome(config);
105            return;
106        }
107        match style {
108            ChromeStyle::Builtin => self.mount_builtin_chrome(&title, &slots),
109            #[cfg(feature = "adwaita")]
110            ChromeStyle::Adwaita => self.mount_adwaita_chrome(config, &title, &slots),
111            // `resolved()` already fell back to Builtin - unreachable in
112            // practice, mount builtin defensively rather than panic.
113            #[cfg(not(feature = "adwaita"))]
114            ChromeStyle::Adwaita => self.mount_builtin_chrome(&title, &slots),
115        }
116        let imp = self.imp();
117        // The grip gesture wiring in `FloatingBoard` keys off the grip's
118        // existence, not the flags: a grip exists iff its slot was mounted.
119        imp.resizable.set(imp.resize_grip.get().is_some());
120        imp.rotate_grip_enabled.set(imp.rotate_grip.get().is_some());
121    }

123    /// Curved-card renderer: inline chrome slots along the disc's arc
124    /// bands (`CardConfig::arc_span` on top, `arc_bottom_span` on the
125    /// bottom - both default a quarter) as an "arc header" -
126    /// `CardImpl::circle_allocate` places them on the band's mid
127    /// radius, a `Title` pinned to the arc's center like a header
128    /// bar's. On `CardShape::Semicircle` the top slots ride the same
129    /// arc while `Bar*` slots mount the straight bottom bar
130    /// (`CardImpl::bar_items`, placed by `semicircle_allocate`). On
131    /// `CardShape::Hexagon`/`CardShape::Octagon` the flat edges replace the
132    /// arcs: `Bar*` slots ride the bottom edge's bar
133    /// (`CardImpl::bar_items`), every other placement the top edge's
134    /// (`CardImpl::top_items`).
135    /// `Satellite` slots stay satellites; the edge grips have no
136    /// mount on a curve.
137    fn mount_arc_chrome(&self, config: &CardConfig) {
138        let imp = self.imp();
139        let span = config
140            .arc_span
141            .unwrap_or(Degrees::QUARTER_TURN)
142            .as_f64()
143            .clamp(Degrees::ZERO.as_f64(), Degrees::FULL_TURN.as_f64());
144        imp.arc_span.set(Degrees::new(span).to_radians());
145        let bottom_span = config
146            .arc_bottom_span
147            .unwrap_or(Degrees::new(span))
148            .as_f64()
149            .clamp(Degrees::ZERO.as_f64(), Degrees::FULL_TURN.as_f64());
150        imp.arc_bottom_span.set(Degrees::new(bottom_span).to_radians());
151        let title = imp.title_text.borrow().clone();
152        let slots = config.effective_chrome_slots();
153        let style = config.chrome_style.resolved();
154        // `Bar*` slots route by shape: an arc zone on `Circle`,
155        // the straight bottom bar on `Semicircle`.
156        let mut items = Vec::new();
157        let mut bar_items = Vec::new();
158        let mut top_items = Vec::new();
159        for slot in &slots {
160            if slot.placement == ChromePlacement::Satellite {
161                skip_satellite(slot.element);
162                continue;
163            }
164            let widget: Widget = match slot.element {
165                ChromeElement::DragHandle => {
166                    let handle = drag_handle_label();
167                    let _ = imp.handle_label.set(handle.clone());
168                    handle.upcast()
169                }
170                ChromeElement::Title => {
171                    let label = title_label(&title);
172                    // The arc's centered item reads like a header bar's
173                    // title - it must not claim inline space.
174                    label.set_hexpand(false);
175                    label.set_halign(Align::Center);
176                    let _ = imp.title_label.set(label.clone());
177                    label.upcast()
178                }
179                ChromeElement::Minimize => {
180                    let button = self.build_minimize_button(ICON_MINIMIZE_INLINE);
181                    let _ = imp.fold_button.set(button.clone());
182                    button.upcast()
183                }
184                ChromeElement::Close => {
185                    let button = self.build_close_button();
186                    let _ = imp.close_button.set(button.clone());
187                    button.upcast()
188                }
189                ChromeElement::RotateClockwise => self.build_rotate_button(RotationDirection::Clockwise).upcast(),
190                ChromeElement::RotateCounterClockwise => self.build_rotate_button(RotationDirection::CounterClockwise).upcast(),
191                ChromeElement::NextSeat => self.build_pass_button(TurnDirection::Clockwise).upcast(),
192                ChromeElement::PreviousSeat => self.build_pass_button(TurnDirection::CounterClockwise).upcast(),
193                ChromeElement::ResizeGrip | ChromeElement::RotateGrip => {
194                    warn!(element = ?slot.element, "edge grip has no mount on a curved card - skipped");
195                    continue;
196                }
197            };
198            widget.add_css_class(CLASS_ARC_CHROME);
199            if style == ChromeStyle::Adwaita {
200                // The themed look chrome satellites get: a raised round
201                // button instead of the builtin flat icon.
202                widget.remove_css_class("flat");
203                widget.add_css_class("circular");
204            }
205            self.append(&widget);
206            // The polygon shapes have no arcs - their straight flat
207            // edges take the role of the curved bands: `Bar*` slots
208            // line up on the bottom bar, every other placement on the
209            // top bar.
210            if matches!(config.shape, CardShape::Hexagon | CardShape::Octagon) {
211                if slot.placement.is_bar() {
212                    bar_items.push((widget, *slot));
213                } else {
214                    top_items.push((widget, *slot));
215                }
216            } else if config.shape == CardShape::Semicircle && slot.placement.is_bar() {
217                bar_items.push((widget, *slot));
218            } else {
219                items.push((widget, *slot));
220            }
221        }
222        // The painted bands sit beneath the items (earlier children
223        // paint first) and are created only when items exist - an
224        // empty band would be a stray sector.
225        if !items.is_empty() {
226            let band = arc_band(self);
227            band.insert_before(self, items.first().map(|(first, _)| first));
228            let _ = imp.arc_band.set(band);
229        }
230        if !bar_items.is_empty() {
231            let band = bar_band(self, false);
232            band.insert_before(self, bar_items.first().map(|(first, _)| first));
233            let _ = imp.bar_band.set(band);
234        }
235        if !top_items.is_empty() {
236            let band = bar_band(self, true);
237            band.insert_before(self, top_items.first().map(|(first, _)| first));
238            let _ = imp.top_band.set(band);
239        }
240        *imp.arc_items.borrow_mut() = items;
241        *imp.bar_items.borrow_mut() = bar_items;
242        *imp.top_items.borrow_mut() = top_items;
243    }

245    /// `Builtin` renderer: header strip as a `gtk4::Box` of inline
246    /// chrome slots, grips on the content overlay. Reproduces the
247    /// historical card look one-to-one (P1.3 regression baseline).
248    fn mount_builtin_chrome(&self, title: &str, slots: &[ChromeSlot]) {
249        let imp = self.imp();
250        let header = SilhouetteBox::new(Orientation::Horizontal, 6);
251        header.add_css_class(CLASS_HEADER);
252        header.set_hexpand(true);
253        // Header children are grouped by placement: `InlineStart`
254        // children land before the neutral `Inline` run (which keeps
255        // slot order), `InlineEnd` children after it - so ends of the
256        // strip are controllable independent of slot order.
257        let mut start_children: Vec<Widget> = Vec::new();
258        let mut middle_children: Vec<Widget> = Vec::new();
259        let mut end_children: Vec<Widget> = Vec::new();
260        for slot in slots {
261            if slot.placement == ChromePlacement::Satellite {
262                skip_satellite(slot.element);
263                continue;
264            }
265            let target = match slot.placement {
266                ChromePlacement::InlineStart => &mut start_children,
267                ChromePlacement::InlineEnd => &mut end_children,
268                _ => &mut middle_children,
269            };
270            match slot.element {
271                ChromeElement::DragHandle => {
272                    let handle = drag_handle_label();
273                    target.push(handle.clone().upcast());
274                    let _ = imp.handle_label.set(handle);
275                }
276                ChromeElement::Title => {
277                    let label = title_label(title);
278                    target.push(label.clone().upcast());
279                    let _ = imp.title_label.set(label);
280                }
281                ChromeElement::Minimize => {
282                    let button = self.build_minimize_button(ICON_MINIMIZE_INLINE);
283                    target.push(button.clone().upcast());
284                    let _ = imp.fold_button.set(button);
285                }
286                ChromeElement::Close => {
287                    let button = self.build_close_button();
288                    target.push(button.clone().upcast());
289                    let _ = imp.close_button.set(button);
290                }
291                ChromeElement::RotateClockwise => target.push(self.build_rotate_button(RotationDirection::Clockwise).upcast()),
292                ChromeElement::RotateCounterClockwise => target.push(self.build_rotate_button(RotationDirection::CounterClockwise).upcast()),
293                ChromeElement::NextSeat => target.push(self.build_pass_button(TurnDirection::Clockwise).upcast()),
294                ChromeElement::PreviousSeat => target.push(self.build_pass_button(TurnDirection::CounterClockwise).upcast()),
295                ChromeElement::ResizeGrip => self.mount_grip(ChromeElement::ResizeGrip),
296                ChromeElement::RotateGrip => self.mount_grip(ChromeElement::RotateGrip),
297            }
298        }
299        let header_children = start_children.len() + middle_children.len() + end_children.len();
300        for child in start_children.iter().chain(&middle_children).chain(&end_children) {
301            header.append(child);
302        }
303        // A header without children is skipped entirely: no dead strip
304        // is drawn and `drag_source(DragHandle::Header)` falls back to
305        // the card widget, keeping a chrome-free card draggable.
306        if header_children > 0 {
307            imp.header_shown.set(true);
308            self.prepend(&header);
309            let _ = imp.header.set(header.upcast());
310        }
311    }

313    /// Mounts the grip widget for `element` on the card's content
314    /// overlay and records it in the matching `CardImpl` cell.
315    /// Non-grip elements are ignored.
316    fn mount_grip(&self, element: ChromeElement) {
317        let imp = self.imp();
318        let Some(overlay) = imp.content_overlay.get() else {
319            return;
320        };
321        let (grip, cell) = match element {
322            ChromeElement::ResizeGrip => (resize_grip(), &imp.resize_grip),
323            ChromeElement::RotateGrip => (rotate_grip(), &imp.rotate_grip),
324            _ => return,
325        };
326        overlay.add_overlay(&grip);
327        let _ = cell.set(grip);
328    }

409/// A `Satellite` slot never mounts inline - satellite-capable elements
410/// are picked up by the board's satellite mounting (`CardConfig::
411/// effective_satellites`); elements that can never be satellites warn.
412fn skip_satellite(element: ChromeElement) {
413    if element.supports_satellite() {
414        debug!("Chrome element {element:?} configured as satellite - mounted by the board, not inline");
415    } else {
416        warn!("Chrome element {element:?} does not support satellite placement - slot ignored");
417    }
418}

420/// The painted top-quarter annulus band behind a circle card's arc
421/// chrome (`ARC_CHROME_BAND` thick, ±45°): a `DrawingArea` stroking a
422/// round-capped arc on the band's mid radius in the card's header
423/// color - read through the CSS `color` property so palettes tint it.
424/// Input-transparent: the band is decoration, the ring stays a drag
425/// surface.
426fn arc_band(card: &Card) -> DrawingArea {
427    let band = DrawingArea::new();
428    band.add_css_class(CLASS_ARC_BAND);
429    band.set_can_target(false);
430    let card_weak = card.downgrade();
431    band.set_draw_func(move |area, cr, width, height| {
432        let Some(card) = card_weak.upgrade() else { return };
433        // The virtual circle's center and mid-band radius follow the
434        // shape: the disc's own center on `Circle`, the dome's bottom
435        // middle on `Semicircle` (the dome has no bottom arc).
436        let (center_x, center_y, radius) = match card.imp().shape.get() {
437            CardShape::Semicircle => (
438                f64::from(width) / 2.0,
439                f64::from(height),
440                (f64::from(width) / 2.0).min(f64::from(height)) - ARC_CHROME_BAND / 2.0,
441            ),
442            // The ring's chrome line is the band's mid radius, not
443            // just inside the rim like on a disc.
444            CardShape::Annulus => {
445                let outer = f64::from(width.min(height)) / 2.0;
446                (f64::from(width) / 2.0, f64::from(height) / 2.0, outer * (1.0 + card.imp().hole_ratio.get().as_f64()) / 2.0)
447            }
448            _ => (f64::from(width) / 2.0, f64::from(height) / 2.0, f64::from(width.min(height)) / 2.0 - ARC_CHROME_BAND / 2.0),
449        };
450        if radius <= 0.0 {
451            return;
452        }
453        let top_half_span = card.imp().arc_span.get() / 2.0;
454        let bottom_half_span = card.imp().arc_bottom_span.get() / 2.0;
455        let items = card.imp().arc_items.borrow();
456        let has_top = items.iter().any(|(_, slot)| slot.placement.is_top_arc());
457        let has_bottom = items.iter().any(|(_, slot)| slot.placement.is_bar());
458        drop(items);
459        let color = Color::<f64>::from(area.color());
460        color.apply_rgba(cr);
461        cr.set_line_width(ARC_CHROME_BAND);
462        cr.set_line_cap(LineCap::Round);
463        // Cairo measures angles from +x: the bands center on -90°
464        // (top) and +90° (bottom), each `arc_span`/`arc_bottom_span` wide.
465        for (center, present, half_span) in [(-FRAC_PI_2, has_top, top_half_span), (FRAC_PI_2, has_bottom, bottom_half_span)] {
466            if present {
467                cr.arc(center_x, center_y, radius, center - half_span.as_f64(), center + half_span.as_f64());
468                let _ = cr.stroke();
469            }
470        }
471    });
472    band
473}

475/// `tan(22.5°) = sqrt(2) - 1` - a flat-top octagon's flat edge is
476/// `2 * a * tan(22.5°)` wide for inradius `a`; its diagonal edges
477/// widen a cross-section at a 1:1 gain per unit height.
478const TAN_22_5: f64 = 0.414_213_562_373_095_03;
479
480/// The straight bar painted along a card's flat edge: on
481/// `Semicircle` a trapezoid strip (`ARC_CHROME_BAND` tall, the
482/// dome's full width at the flat bottom edge, the chord at its top
483/// edge), on `Hexagon`/`Octagon` a trapezoid on either flat edge
484/// (`top` mirrors it onto the top edge) in `CLASS_BAR_BAND`'s `color` -
485/// the palette's header color like the arc band. Non-interactive
486/// (`set_can_target(false)`); reads the shape's extents from the
487/// card's allocation per frame.
488fn bar_band(card: &Card, top: bool) -> DrawingArea {
489    let band = DrawingArea::new();
490    band.add_css_class(CLASS_BAR_BAND);
491    band.set_can_target(false);
492    let card_weak = card.downgrade();
493    band.set_draw_func(move |area, cr, width, height| {
494        let Some(card) = card_weak.upgrade() else { return };
495        let size = Vec2::<()>::new(f64::from(width), f64::from(height));
496        let center_x = size.x / 2.0;
497        match card.imp().shape.get() {
498            CardShape::Hexagon => {
499                // The hexagon's flat edge is `R = w/2` wide; the slanted
500                // edges widen the bar by `BAND * w / (2h)` per side at
501                // the band's far edge (the slant gains `1/sqrt(3)` per
502                // unit height), so it ends exactly on the silhouette.
503                let edge_half = size.x / 4.0;
504                let far_half = edge_half + ARC_CHROME_BAND * size.x / (2.0 * size.y);
505                let (edge_y, far_y) = if top { (0.0, ARC_CHROME_BAND) } else { (size.y, size.y - ARC_CHROME_BAND) };
506                cr.move_to(center_x - edge_half, edge_y);
507                cr.line_to(center_x - far_half, far_y);
508                cr.line_to(center_x + far_half, far_y);
509                cr.line_to(center_x + edge_half, edge_y);
510                cr.close_path();
511            }
512            CardShape::Octagon => {
513                // The octagon's flat edge is `2a * tan(22.5°)` wide
514                // (`a = w/2`); the 45° diagonal edges widen the bar by
515                // `BAND` per side at the band's far edge (the slant
516                // gains 1:1 per unit height), so it ends exactly on
517                // the silhouette.
518                let edge_half = size.x / 2.0 * TAN_22_5;
519                let far_half = edge_half + ARC_CHROME_BAND;
520                let (edge_y, far_y) = if top { (0.0, ARC_CHROME_BAND) } else { (size.y, size.y - ARC_CHROME_BAND) };
521                cr.move_to(center_x - edge_half, edge_y);
522                cr.line_to(center_x - far_half, far_y);
523                cr.line_to(center_x + far_half, far_y);
524                cr.line_to(center_x + edge_half, edge_y);
525                cr.close_path();
526            }
527            _ => {
528                let radius = (size.x / 2.0).min(size.y);
529                // The chord where the bar's top edge crosses the dome rim.
530                let chord = (radius * radius - ARC_CHROME_BAND * ARC_CHROME_BAND).max(0.0).sqrt();
531                cr.move_to(center_x - radius, size.y);
532                cr.line_to(center_x - chord, size.y - ARC_CHROME_BAND);
533                cr.line_to(center_x + chord, size.y - ARC_CHROME_BAND);
534                cr.line_to(center_x + radius, size.y);
535                cr.close_path();
536            }
537        }
538        let color = Color::<f64>::from(area.color());
539        color.apply_rgba(cr);
540        let _ = cr.fill();
541    });
542    band
543}

596/// Edge length of the rotate grip's circular arrow (px).
597const ROTATE_GRIP_SIZE: i32 = 18;
598
599/// The bottom-left rotate affordance: a circular arrow in the card's
600/// foreground color, `grab` cursor on hover.
601fn rotate_grip() -> DrawingArea {
602    let grip = DrawingArea::new();
603    grip.add_css_class(CLASS_ROTATE_GRIP);
604    // Overlay children default to `Align::Fill` - without Start/End
605    // alignment the grip would cover the whole content area.
606    grip.set_halign(Align::Start);
607    grip.set_valign(Align::End);
608    grip.set_content_width(ROTATE_GRIP_SIZE);
609    grip.set_content_height(ROTATE_GRIP_SIZE);
610    grip.set_cursor_from_name(Some("grab"));
611    grip.set_tooltip_text(Some(&l10n().message("card-rotate")));
612    grip.set_draw_func(|grip, cr, _, _| {
613        let color = Color::<f64>::from(grip.color());
614        let size = grip.allocated_size().to_f64();
615        color.apply_rgba(cr);
616        cr.set_line_width(1.5);
617        cr.set_line_cap(LineCap::Round);
618        // A three-quarter clockwise arc with an arrowhead on its tip:
619        // the circular arrow suggests turning in either direction.
620        let radius = size.x.min(size.y) / 2.0 - 3.0;
621        let center_x = size.x / 2.0;
622        let center_y = size.y / 2.0;
623        let start = -FRAC_PI_2 + 0.7;
624        let end = start + 3.0 * FRAC_PI_2;
625        cr.arc(center_x, center_y, radius, start, end);
626        let _ = cr.stroke();
627        let tip_x = center_x + radius * end.cos();
628        let tip_y = center_y + radius * end.sin();
629        let tangent = end + FRAC_PI_2;
630        for spread in [-0.7, 0.7] {
631            cr.move_to(tip_x, tip_y);
632            cr.line_to(tip_x - 4.0 * (tangent + spread).cos(), tip_y - 4.0 * (tangent + spread).sin());
633        }
634        let _ = cr.stroke();
635    });
636    grip
637}

723    impl Card {
724        /// Builds the `adw::HeaderBar` chrome for the card.
725        pub(super) fn mount_adwaita_chrome(&self, config: &CardConfig, title: &str, slots: &[ChromeSlot]) {
726            let imp = self.imp();
727            if !adw::is_initialized() {
728                debug!("ChromeStyle::Adwaita configured - call `adw::init()` (or use `adw::Application`) for themed styling");
729            }
730            let header = adw::HeaderBar::new();
731            header.set_show_start_title_buttons(false);
732            header.set_show_end_title_buttons(false);
733            // The headerbar paints the window chrome background; keep the
734            // card's header CSS class so the palette/theme keeps styling
735            // the frame around it.
736            header.add_css_class(CLASS_HEADER);
737            // Real Adwaita *window* titlebars carry `default-decoration`:
738            // the stylesheet shrinks them to 37px with 22px controls. A
739            // plain `adw::HeaderBar` would render at the 46px app-header
740            // height - noticeably taller than both our builtin header and
741            // a real window titlebar.
742            header.add_css_class("default-decoration");
743            // The HeaderBar registers internal gestures that drive the
744            // *toplevel window* (move-drag, double-click maximize). On a
745            // card they race the card's own drag - a fast drag moved the
746            // whole window instead of the card. Watch for them from the
747            // start, then strip the subtree after our children exist.
748            let watcher = header.observe_controllers();
749            watch_foreign_gestures(&header, &watcher);
750            // The watcher only covers the header's *own* controller
751            // list; libadwaita can also register gestures lazily on
752            // inner widgets (windowhandle) at map time. Strip the whole
753            // subtree once more when the header is actually shown.
754            header.connect_map(strip_foreign_gestures);
755            // Wired control buttons, collected per slot and placed by
756            // `pack_window_controls` afterwards: side and order follow
757            // the `gtk-decoration-layout` setting (themes like Rewaita
758            // move window buttons to the left).
759            // Control buttons route by placement: `InlineStart`
760            // collects into the start `windowcontrols` container,
761            // `InlineEnd` into the end container, neutral `Inline`
762            // keeps the historical behavior - `Minimize`/`Close`
763            // follow `gtk-decoration-layout`, other actions append to
764            // the end controls.
765            let mut minimize_button: Option<gtk4::Button> = None;
766            let mut close_button: Option<gtk4::Button> = None;
767            let mut start_extras: Vec<gtk4::Button> = Vec::new();
768            let mut end_extras: Vec<gtk4::Button> = Vec::new();
769            let mut header_children = 0usize;
770            for slot in slots {
771                if slot.placement == ChromePlacement::Satellite {
772                    skip_satellite(slot.element);
773                    continue;
774                }
775                match slot.element {
776                    ChromeElement::DragHandle => {
777                        let handle = drag_handle_label();
778                        match slot.placement {
779                            ChromePlacement::InlineEnd => header.pack_end(&handle),
780                            _ => header.pack_start(&handle),
781                        }
782                        let _ = imp.handle_label.set(handle);
783                        header_children += 1;
784                    }
785                    ChromeElement::Title => {
786                        let title_widget = adw::WindowTitle::new(title, "");
787                        title_widget.set_visible(!config.headerless);
788                        match slot.placement {
789                            ChromePlacement::InlineStart => header.pack_start(&title_widget),
790                            ChromePlacement::InlineEnd => header.pack_end(&title_widget),
791                            _ => header.set_title_widget(Some(&title_widget)),
792                        }
793                        header_children += 1;
794                    }
795                    _ => {
796                        let (button, class) = match slot.element {
797                            ChromeElement::Minimize => (self.build_minimize_button(ICON_MINIMIZE_SYMBOLIC), "minimize"),
798                            ChromeElement::Close => (self.build_close_button(), "close"),
799                            ChromeElement::RotateClockwise => (self.build_rotate_button(RotationDirection::Clockwise), "rotate-clockwise"),
800                            ChromeElement::RotateCounterClockwise => {
801                                (self.build_rotate_button(RotationDirection::CounterClockwise), "rotate-counter-clockwise")
802                            }
803                            ChromeElement::NextSeat => (self.build_pass_button(TurnDirection::Clockwise), "next-seat"),
804                            ChromeElement::PreviousSeat => (self.build_pass_button(TurnDirection::CounterClockwise), "previous-seat"),
805                            ChromeElement::ResizeGrip => {
806                                self.mount_grip(ChromeElement::ResizeGrip);
807                                continue;
808                            }
809                            ChromeElement::RotateGrip => {
810                                self.mount_grip(ChromeElement::RotateGrip);
811                                continue;
812                            }
813                            ChromeElement::DragHandle | ChromeElement::Title => continue,
814                        };
815                        // A real wired button: `WindowControls` buttons
816                        // are decorative (they target a real toplevel).
817                        // Let the `windowcontrols` node rules own the
818                        // look (the `flat` class would fight their
819                        // padding/hover styling), and carry the role
820                        // class themes key colors off.
821                        button.remove_css_class("flat");
822                        button.add_css_class(class);
823                        match slot.placement {
824                            ChromePlacement::InlineStart => start_extras.push(button.clone()),
825                            ChromePlacement::InlineEnd => end_extras.push(button.clone()),
826                            _ => match slot.element {
827                                ChromeElement::Minimize => minimize_button = Some(button.clone()),
828                                ChromeElement::Close => close_button = Some(button.clone()),
829                                _ => end_extras.push(button.clone()),
830                            },
831                        }
832                        if slot.element == ChromeElement::Minimize {
833                            let _ = imp.fold_button.set(button.clone());
834                        }
835                        if slot.element == ChromeElement::Close {
836                            let _ = imp.close_button.set(button);
837                        }
838                        header_children += 1;
839                    }
840                }
841            }
842            pack_window_controls(&header, minimize_button, close_button, start_extras, end_extras);
843            strip_foreign_gestures(&header);
844            // Same rule as the builtin renderer: no children, no header -
845            // `drag_source(DragHandle::Header)` falls back to the card.
846            if header_children > 0 {
847                imp.header_shown.set(true);
848                self.prepend(&header);
849                let _ = imp.header.set(header.upcast());
850            }
851        }
852    }

854    /// Fallback when `gtk-decoration-layout` is unset: all controls on
855    /// the right in GNOME order.
856    const DEFAULT_DECORATION_LAYOUT: &str = ":minimize,maximize,close";
857
858    /// Splits `gtk-decoration-layout` into the button names left and
859    /// right of the `:` separator (e.g. Rewaita's `"close,minimize:"`
860    /// moves all controls to the left side).
861    fn decoration_layout_names() -> (Vec<String>, Vec<String>) {
862        let layout = gtk4::Settings::default()
863            .and_then(|settings| settings.gtk_decoration_layout())
864            .map(|layout| layout.to_string())
865            .unwrap_or_else(|| DEFAULT_DECORATION_LAYOUT.to_string());
866        let names = |side: &str| side.split(',').map(str::trim).filter(|name| !name.is_empty()).map(str::to_string).collect();
867        let mut sides = layout.splitn(2, ':');
868        let start = names(sides.next().unwrap_or_default());
869        let end = names(sides.next().unwrap_or_default());
870        (start, end)
871    }

873    /// Packs the card's control buttons into `windowcontrols` containers,
874    /// following the decoration layout: a button goes to its named side in
875    /// layout order; slotted buttons the layout does not mention append to
876    /// the end side (a configured slot is never dropped, only reordered).
877    /// `extras` are card actions outside the window-button vocabulary
878    /// (rotate/pass steps): `start_extras` lead the start controls
879    /// (outermost first, like GNOME's app-buttons-left convention),
880    /// `end_extras` trail the end controls.
881    fn pack_window_controls(
882        header: &adw::HeaderBar,
883        minimize: Option<gtk4::Button>,
884        close: Option<gtk4::Button>,
885        start_extras: Vec<gtk4::Button>,
886        end_extras: Vec<gtk4::Button>,
887    ) {
888        let (start_names, end_names) = decoration_layout_names();
889        let start_controls = ChromeControls::new();
890        let end_controls = ChromeControls::new();
891        let mut minimize = minimize;
892        let mut close = close;
893        let mut take = |name: &str| match name {
894            "minimize" => minimize.take(),
895            "close" => close.take(),
896            _ => None,
897        };
898        for extra in &start_extras {
899            start_controls.append(extra);
900        }
901        for name in &start_names {
902            if let Some(button) = take(name) {
903                start_controls.append(&button);
904            }
905        }
906        for name in &end_names {
907            if let Some(button) = take(name) {
908                end_controls.append(&button);
909            }
910        }
911        drop(take);
912        if let Some(button) = minimize.take() {
913            end_controls.append(&button);
914        }
915        if let Some(button) = close.take() {
916            end_controls.append(&button);
917        }
918        for extra in &end_extras {
919            end_controls.append(extra);
920        }
921        if start_controls.first_child().is_some() {
922            // `start`/`end` mirror what GtkWindowControls' own boxes
923            // carry; idempotent where the headerbar already adds them.
924            start_controls.add_css_class("start");
925            header.pack_start(&start_controls);
926        }
927        if end_controls.first_child().is_some() {
928            end_controls.add_css_class("end");
929            header.pack_end(&end_controls);
930        }
931    }

933    /// A gesture controller not tagged `CARD_GESTURE_NAME` - i.e. one of
934    /// the HeaderBar's internal window gestures.
935    fn is_foreign_gesture(controller: &gtk4::EventController) -> bool {
936        controller.is::<gtk4::Gesture>() && controller.name().as_deref() != Some(CARD_GESTURE_NAME)
937    }
938
939    /// Removes every foreign gesture on `widget` and its descendants,
940    /// recursing into children but skipping `gtk4::Button` subtrees
941    /// *before* touching controllers - a button's internal `GestureClick`
942    /// is a foreign gesture by the tag rule, yet it is exactly what
943    /// makes the button work.
944    fn strip_foreign_gestures(widget: &impl IsA<gtk4::Widget>) {
945        let widget = widget.upcast_ref::<gtk4::Widget>();
946        if widget.is::<gtk4::Button>() {
947            return;
948        }
949        let controllers = widget.observe_controllers();
950        for index in (0..controllers.n_items()).rev() {
951            let Some(item) = controllers.item(index) else { continue };
952            let Ok(controller) = item.downcast::<gtk4::EventController>() else {
953                continue;
954            };
955            if is_foreign_gesture(&controller) {
956                widget.remove_controller(&controller);
957            }
958        }
959        let mut child = widget.first_child();
960        while let Some(sub) = child {
961            strip_foreign_gestures(&sub);
962            child = sub.next_sibling();
963        }
964    }
965
966    /// Watches the header's controller list for foreign gestures
967    /// registered *after* construction (e.g. on realize) and removes
968    /// them. Removal is deferred out of the `items_changed` emission -
969    /// mutating a list model inside its own change handler is fragile.
970    fn watch_foreign_gestures(header: &adw::HeaderBar, controllers: &gtk4::gio::ListModel) {
971        let header_weak = header.downgrade();
972        controllers.connect_items_changed(move |model, position, _removed, added| {
973            for index in position..position + added {
974                let Some(item) = model.item(index) else { continue };
975                let Ok(controller) = item.downcast::<gtk4::EventController>() else {
976                    continue;
977                };
978                if !is_foreign_gesture(&controller) {
979                    continue;
980                }
981                let Some(header) = header_weak.upgrade() else { continue };
982                gtk4::glib::idle_add_local_once(move || header.remove_controller(&controller));
983            }
984        });
985    }
