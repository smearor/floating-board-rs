1//! Chrome element builders and mounting, implemented on [`Card`].
2//!
3//! One [`ChromeElement`] produces one widget plus its signal wiring
4//! (P1.1). `Inline` elements mount either into the card's header strip
5//! (drag handle, title, minimize, close) or onto the content overlay
6//! (resize/rotate grips); `Satellite` placements are validated and
7//! skipped until satellite mounting lands (Phase 2/3).
8//!
9//! The `builtin` renderer assembles the stock GTK4/CSS chrome; the
10//! `adwaita` feature adds a libadwaita renderer (`adw::HeaderBar`,
11//! `adw::WindowTitle`, wired buttons in a `windowcontrols` node). The
12//! bare widget factories at the bottom of this module stay free
13//! functions - they are stateless builders for foreign types (`Label`,
14//! `DrawingArea`).
15

66impl Card {
67    /// Builds the card chrome from `config`'s effective slot list and
68    /// mounts every `Inline` element: header elements onto a header
69    /// strip prepended to the card, grip elements onto the content
70    /// overlay.
71    ///
72    /// The header widget lands in `CardImpl::header` - it stays the
73    /// drag affordance for `DragHandle::Header` and the strip that
74    /// remains visible while folded. A header without any inline
75    /// children is not mounted at all (an explicit `chrome: []` config
76    /// produces a chrome-free card; use `DragHandle::WholeCard` or
77    /// `headerless` for those).
78    pub(crate) fn mount_chrome(&self, config: &CardConfig) {
79        let title = config.title.clone().unwrap_or_else(|| l10n().message("card-untitled"));
80        *self.imp().title_text.borrow_mut() = title.clone();
81        // A circle card has no header strip and no corner grips: every
82        // chrome element mounts as a satellite (wired through
83        // `CardConfig::effective_satellites`); elements that cannot be
84        // satellites are dropped.
85        if config.shape == CardShape::Circle {
86            for slot in &config.effective_chrome_slots() {
87                if slot.placement != ChromePlacement::Satellite {
88                    if slot.element.supports_satellite() {
89                        warn!(element = ?slot.element, "inline chrome element has no mount on a circle card - mounted as satellite");
90                    } else {
91                        warn!(element = ?slot.element, "chrome element has no mount on a circle card - skipped");
92                    }
93                }
94            }
95            return;
96        }
97        let slots = config.effective_chrome_slots();
98        let style = config.chrome_style.resolved();
99        if style != config.chrome_style {
100            warn!("CardConfig::chrome_style is Adwaita but the `adwaita` feature is disabled - falling back to Builtin");
101        }
102        match style {
103            ChromeStyle::Builtin => self.mount_builtin_chrome(&title, &slots),
104            #[cfg(feature = "adwaita")]
105            ChromeStyle::Adwaita => self.mount_adwaita_chrome(config, &title, &slots),
106            // `resolved()` already fell back to Builtin - unreachable in
107            // practice, mount builtin defensively rather than panic.
108            #[cfg(not(feature = "adwaita"))]
109            ChromeStyle::Adwaita => self.mount_builtin_chrome(&title, &slots),
110        }
111        let imp = self.imp();
112        // The grip gesture wiring in `FloatingBoard` keys off the grip's
113        // existence, not the flags: a grip exists iff its slot was mounted.
114        imp.resizable.set(imp.resize_grip.get().is_some());
115        imp.rotate_grip_enabled.set(imp.rotate_grip.get().is_some());
116    }

118    /// `Builtin` renderer: header strip as a `gtk4::Box` of inline
119    /// chrome slots, grips on the content overlay. Reproduces the
120    /// historical card look one-to-one (P1.3 regression baseline).
121    fn mount_builtin_chrome(&self, title: &str, slots: &[ChromeSlot]) {
122        let imp = self.imp();
123        let header = gtk4::Box::new(Orientation::Horizontal, 6);
124        header.add_css_class(CLASS_HEADER);
125        header.set_hexpand(true);
126        // Header children are grouped by placement: `InlineStart`
127        // children land before the neutral `Inline` run (which keeps
128        // slot order), `InlineEnd` children after it - so ends of the
129        // strip are controllable independent of slot order.
130        let mut start_children: Vec<Widget> = Vec::new();
131        let mut middle_children: Vec<Widget> = Vec::new();
132        let mut end_children: Vec<Widget> = Vec::new();
133        for slot in slots {
134            if slot.placement == ChromePlacement::Satellite {
135                skip_satellite(slot.element);
136                continue;
137            }
138            let target = match slot.placement {
139                ChromePlacement::InlineStart => &mut start_children,
140                ChromePlacement::InlineEnd => &mut end_children,
141                _ => &mut middle_children,
142            };
143            match slot.element {
144                ChromeElement::DragHandle => {
145                    let handle = drag_handle_label();
146                    target.push(handle.clone().upcast());
147                    let _ = imp.handle_label.set(handle);
148                }
149                ChromeElement::Title => {
150                    let label = title_label(title);
151                    target.push(label.clone().upcast());
152                    let _ = imp.title_label.set(label);
153                }
154                ChromeElement::Minimize => {
155                    let button = self.build_minimize_button(ICON_MINIMIZE_INLINE);
156                    target.push(button.clone().upcast());
157                    let _ = imp.fold_button.set(button);
158                }
159                ChromeElement::Close => {
160                    let button = self.build_close_button();
161                    target.push(button.clone().upcast());
162                    let _ = imp.close_button.set(button);
163                }
164                ChromeElement::RotateClockwise => target.push(self.build_rotate_button(RotationDirection::Clockwise).upcast()),
165                ChromeElement::RotateCounterClockwise => target.push(self.build_rotate_button(RotationDirection::CounterClockwise).upcast()),
166                ChromeElement::NextSeat => target.push(self.build_pass_button(TurnDirection::Clockwise).upcast()),
167                ChromeElement::PreviousSeat => target.push(self.build_pass_button(TurnDirection::CounterClockwise).upcast()),
168                ChromeElement::ResizeGrip => self.mount_grip(ChromeElement::ResizeGrip),
169                ChromeElement::RotateGrip => self.mount_grip(ChromeElement::RotateGrip),
170            }
171        }
172        let header_children = start_children.len() + middle_children.len() + end_children.len();
173        for child in start_children.iter().chain(&middle_children).chain(&end_children) {
174            header.append(child);
175        }
176        // A header without children is skipped entirely: no dead strip
177        // is drawn and `drag_source(DragHandle::Header)` falls back to
178        // the card widget, keeping a chrome-free card draggable.
179        if header_children > 0 {
180            imp.header_shown.set(true);
181            self.prepend(&header);
182            let _ = imp.header.set(header.upcast());
183        }
184    }

186    /// Mounts the grip widget for `element` on the card's content
187    /// overlay and records it in the matching `CardImpl` cell.
188    /// Non-grip elements are ignored.
189    fn mount_grip(&self, element: ChromeElement) {
190        let imp = self.imp();
191        let Some(overlay) = imp.content_overlay.get() else {
192            return;
193        };
194        let (grip, cell) = match element {
195            ChromeElement::ResizeGrip => (resize_grip(), &imp.resize_grip),
196            ChromeElement::RotateGrip => (rotate_grip(), &imp.rotate_grip),
197            _ => return,
198        };
199        overlay.add_overlay(&grip);
200        let _ = cell.set(grip);
201    }

203    /// Mount-agnostic minimize button (P1.5): the same builder serves
204    /// the inline header slot and, later, the satellite mount;
205    /// `icon_name` differs per mount (`go-down-symbolic` inline,
206    /// `window-minimize-symbolic` satellite). Wired to `Card::fold`,
207    /// tooltip `card-fold`.
208    pub(crate) fn build_minimize_button(&self, icon_name: &str) -> Button {
209        let button = Button::from_icon_name(icon_name);
210        button.add_css_class("flat");
211        button.set_tooltip_text(Some(&l10n().message("card-fold")));
212        let card_weak = self.downgrade();
213        button.connect_clicked(move |_| {
214            if let Some(card) = card_weak.upgrade() {
215                card.fold();
216            }
217        });
218        button
219    }

282/// A `Satellite` slot never mounts inline - satellite-capable elements
283/// are picked up by the board's satellite mounting (`CardConfig::
284/// effective_satellites`); elements that can never be satellites warn.
285fn skip_satellite(element: ChromeElement) {
286    if element.supports_satellite() {
287        debug!("Chrome element {element:?} configured as satellite - mounted by the board, not inline");
288    } else {
289        warn!("Chrome element {element:?} does not support satellite placement - slot ignored");
290    }
291}

313/// Edge length of the resize grip's diagonal strokes (px).
314const RESIZE_GRIP_SIZE: i32 = 18;
315
316/// The bottom-right resize affordance: three diagonal strokes in the
317/// card's foreground color, `nwse-resize` cursor on hover.
318fn resize_grip() -> DrawingArea {
319    let grip = DrawingArea::new();
320    grip.add_css_class(CLASS_RESIZE_GRIP);
321    // Overlay children default to `Align::Fill` - without End
322    // alignment the grip would cover the whole content area.
323    grip.set_halign(Align::End);
324    grip.set_valign(Align::End);
325    grip.set_content_width(RESIZE_GRIP_SIZE);
326    grip.set_content_height(RESIZE_GRIP_SIZE);
327    grip.set_cursor_from_name(Some("nwse-resize"));
328    grip.set_tooltip_text(Some(&l10n().message("card-resize")));
329    grip.set_draw_func(|grip, cr, _, _| {
330        let color = grip.color();
331        let width = f64::from(grip.width());
332        let height = f64::from(grip.height());
333        cr.set_source_rgba(f64::from(color.red()), f64::from(color.green()), f64::from(color.blue()), f64::from(color.alpha()));
334        cr.set_line_width(1.5);
335        cr.set_line_cap(LineCap::Round);
336        for inset in [4.0, 8.0, 12.0] {
337            cr.move_to(width - inset, height - 2.0);
338            cr.line_to(width - 2.0, height - inset);
339            let _ = cr.stroke();
340        }
341    });
342    grip
343}

345/// Edge length of the rotate grip's circular arrow (px).
346const ROTATE_GRIP_SIZE: i32 = 18;
347
348/// The bottom-left rotate affordance: a circular arrow in the card's
349/// foreground color, `grab` cursor on hover.
350fn rotate_grip() -> DrawingArea {
351    let grip = DrawingArea::new();
352    grip.add_css_class(CLASS_ROTATE_GRIP);
353    // Overlay children default to `Align::Fill` - without Start/End
354    // alignment the grip would cover the whole content area.
355    grip.set_halign(Align::Start);
356    grip.set_valign(Align::End);
357    grip.set_content_width(ROTATE_GRIP_SIZE);
358    grip.set_content_height(ROTATE_GRIP_SIZE);
359    grip.set_cursor_from_name(Some("grab"));
360    grip.set_tooltip_text(Some(&l10n().message("card-rotate")));
361    grip.set_draw_func(|grip, cr, _, _| {
362        let color = grip.color();
363        let width = f64::from(grip.width());
364        let height = f64::from(grip.height());
365        cr.set_source_rgba(f64::from(color.red()), f64::from(color.green()), f64::from(color.blue()), f64::from(color.alpha()));
366        cr.set_line_width(1.5);
367        cr.set_line_cap(LineCap::Round);
368        // A three-quarter clockwise arc with an arrowhead on its tip:
369        // the circular arrow suggests turning in either direction.
370        let radius = width.min(height) / 2.0 - 3.0;
371        let center_x = width / 2.0;
372        let center_y = height / 2.0;
373        let start = -FRAC_PI_2 + 0.7;
374        let end = start + 3.0 * FRAC_PI_2;
375        cr.arc(center_x, center_y, radius, start, end);
376        let _ = cr.stroke();
377        let tip_x = center_x + radius * end.cos();
378        let tip_y = center_y + radius * end.sin();
379        let tangent = end + FRAC_PI_2;
380        for spread in [-0.7, 0.7] {
381            cr.move_to(tip_x, tip_y);
382            cr.line_to(tip_x - 4.0 * (tangent + spread).cos(), tip_y - 4.0 * (tangent + spread).sin());
383        }
384        let _ = cr.stroke();
385    });
386    grip
387}

473    impl Card {
474        /// Builds the `adw::HeaderBar` chrome for the card.
475        pub(super) fn mount_adwaita_chrome(&self, config: &CardConfig, title: &str, slots: &[ChromeSlot]) {
476            let imp = self.imp();
477            if !adw::is_initialized() {
478                debug!("ChromeStyle::Adwaita configured - call `adw::init()` (or use `adw::Application`) for themed styling");
479            }
480            let header = adw::HeaderBar::new();
481            header.set_show_start_title_buttons(false);
482            header.set_show_end_title_buttons(false);
483            // The headerbar paints the window chrome background; keep the
484            // card's header CSS class so the palette/theme keeps styling
485            // the frame around it.
486            header.add_css_class(CLASS_HEADER);
487            // Real Adwaita *window* titlebars carry `default-decoration`:
488            // the stylesheet shrinks them to 37px with 22px controls. A
489            // plain `adw::HeaderBar` would render at the 46px app-header
490            // height - noticeably taller than both our builtin header and
491            // a real window titlebar.
492            header.add_css_class("default-decoration");
493            // The HeaderBar registers internal gestures that drive the
494            // *toplevel window* (move-drag, double-click maximize). On a
495            // card they race the card's own drag - a fast drag moved the
496            // whole window instead of the card. Watch for them from the
497            // start, then strip the subtree after our children exist.
498            let watcher = header.observe_controllers();
499            watch_foreign_gestures(&header, &watcher);
500            // The watcher only covers the header's *own* controller
501            // list; libadwaita can also register gestures lazily on
502            // inner widgets (windowhandle) at map time. Strip the whole
503            // subtree once more when the header is actually shown.
504            header.connect_map(strip_foreign_gestures);
505            // Wired control buttons, collected per slot and placed by
506            // `pack_window_controls` afterwards: side and order follow
507            // the `gtk-decoration-layout` setting (themes like Rewaita
508            // move window buttons to the left).
509            // Control buttons route by placement: `InlineStart`
510            // collects into the start `windowcontrols` container,
511            // `InlineEnd` into the end container, neutral `Inline`
512            // keeps the historical behavior - `Minimize`/`Close`
513            // follow `gtk-decoration-layout`, other actions append to
514            // the end controls.
515            let mut minimize_button: Option<gtk4::Button> = None;
516            let mut close_button: Option<gtk4::Button> = None;
517            let mut start_extras: Vec<gtk4::Button> = Vec::new();
518            let mut end_extras: Vec<gtk4::Button> = Vec::new();
519            let mut header_children = 0usize;
520            for slot in slots {
521                if slot.placement == ChromePlacement::Satellite {
522                    skip_satellite(slot.element);
523                    continue;
524                }
525                match slot.element {
526                    ChromeElement::DragHandle => {
527                        let handle = drag_handle_label();
528                        match slot.placement {
529                            ChromePlacement::InlineEnd => header.pack_end(&handle),
530                            _ => header.pack_start(&handle),
531                        }
532                        let _ = imp.handle_label.set(handle);
533                        header_children += 1;
534                    }
535                    ChromeElement::Title => {
536                        let title_widget = adw::WindowTitle::new(title, "");
537                        title_widget.set_visible(!config.headerless);
538                        match slot.placement {
539                            ChromePlacement::InlineStart => header.pack_start(&title_widget),
540                            ChromePlacement::InlineEnd => header.pack_end(&title_widget),
541                            _ => header.set_title_widget(Some(&title_widget)),
542                        }
543                        header_children += 1;
544                    }
545                    _ => {
546                        let (button, class) = match slot.element {
547                            ChromeElement::Minimize => (self.build_minimize_button(ICON_MINIMIZE_SYMBOLIC), "minimize"),
548                            ChromeElement::Close => (self.build_close_button(), "close"),
549                            ChromeElement::RotateClockwise => (self.build_rotate_button(RotationDirection::Clockwise), "rotate-clockwise"),
550                            ChromeElement::RotateCounterClockwise => {
551                                (self.build_rotate_button(RotationDirection::CounterClockwise), "rotate-counter-clockwise")
552                            }
553                            ChromeElement::NextSeat => (self.build_pass_button(TurnDirection::Clockwise), "next-seat"),
554                            ChromeElement::PreviousSeat => (self.build_pass_button(TurnDirection::CounterClockwise), "previous-seat"),
555                            ChromeElement::ResizeGrip => {
556                                self.mount_grip(ChromeElement::ResizeGrip);
557                                continue;
558                            }
559                            ChromeElement::RotateGrip => {
560                                self.mount_grip(ChromeElement::RotateGrip);
561                                continue;
562                            }
563                            ChromeElement::DragHandle | ChromeElement::Title => continue,
564                        };
565                        // A real wired button: `WindowControls` buttons
566                        // are decorative (they target a real toplevel).
567                        // Let the `windowcontrols` node rules own the
568                        // look (the `flat` class would fight their
569                        // padding/hover styling), and carry the role
570                        // class themes key colors off.
571                        button.remove_css_class("flat");
572                        button.add_css_class(class);
573                        match slot.placement {
574                            ChromePlacement::InlineStart => start_extras.push(button.clone()),
575                            ChromePlacement::InlineEnd => end_extras.push(button.clone()),
576                            _ => match slot.element {
577                                ChromeElement::Minimize => minimize_button = Some(button.clone()),
578                                ChromeElement::Close => close_button = Some(button.clone()),
579                                _ => end_extras.push(button.clone()),
580                            },
581                        }
582                        if slot.element == ChromeElement::Minimize {
583                            let _ = imp.fold_button.set(button.clone());
584                        }
585                        if slot.element == ChromeElement::Close {
586                            let _ = imp.close_button.set(button);
587                        }
588                        header_children += 1;
589                    }
590                }
591            }
592            pack_window_controls(&header, minimize_button, close_button, start_extras, end_extras);
593            strip_foreign_gestures(&header);
594            // Same rule as the builtin renderer: no children, no header -
595            // `drag_source(DragHandle::Header)` falls back to the card.
596            if header_children > 0 {
597                imp.header_shown.set(true);
598                self.prepend(&header);
599                let _ = imp.header.set(header.upcast());
600            }
601        }
602    }

604    /// Fallback when `gtk-decoration-layout` is unset: all controls on
605    /// the right in GNOME order.
606    const DEFAULT_DECORATION_LAYOUT: &str = ":minimize,maximize,close";
607
608    /// Splits `gtk-decoration-layout` into the button names left and
609    /// right of the `:` separator (e.g. Rewaita's `"close,minimize:"`
610    /// moves all controls to the left side).
611    fn decoration_layout_names() -> (Vec<String>, Vec<String>) {
612        let layout = gtk4::Settings::default()
613            .and_then(|settings| settings.gtk_decoration_layout())
614            .map(|layout| layout.to_string())
615            .unwrap_or_else(|| DEFAULT_DECORATION_LAYOUT.to_string());
616        let names = |side: &str| side.split(',').map(str::trim).filter(|name| !name.is_empty()).map(str::to_string).collect();
617        let mut sides = layout.splitn(2, ':');
618        let start = names(sides.next().unwrap_or_default());
619        let end = names(sides.next().unwrap_or_default());
620        (start, end)
621    }
622
623    /// Packs the card's control buttons into `windowcontrols` containers,
624    /// following the decoration layout: a button goes to its named side in
625    /// layout order; slotted buttons the layout does not mention append to
626    /// the end side (a configured slot is never dropped, only reordered).
627    /// `extras` are card actions outside the window-button vocabulary
628    /// (rotate/pass steps): `start_extras` lead the start controls
629    /// (outermost first, like GNOME's app-buttons-left convention),
630    /// `end_extras` trail the end controls.
631    fn pack_window_controls(
632        header: &adw::HeaderBar,
633        minimize: Option<gtk4::Button>,
634        close: Option<gtk4::Button>,
635        start_extras: Vec<gtk4::Button>,
636        end_extras: Vec<gtk4::Button>,
637    ) {
638        let (start_names, end_names) = decoration_layout_names();
639        let start_controls = ChromeControls::new();
640        let end_controls = ChromeControls::new();
641        let mut minimize = minimize;
642        let mut close = close;
643        let mut take = |name: &str| match name {
644            "minimize" => minimize.take(),
645            "close" => close.take(),
646            _ => None,
647        };
648        for extra in &start_extras {
649            start_controls.append(extra);
650        }
651        for name in &start_names {
652            if let Some(button) = take(name) {
653                start_controls.append(&button);
654            }
655        }
656        for name in &end_names {
657            if let Some(button) = take(name) {
658                end_controls.append(&button);
659            }
660        }
661        drop(take);
662        if let Some(button) = minimize.take() {
663            end_controls.append(&button);
664        }
665        if let Some(button) = close.take() {
666            end_controls.append(&button);
667        }
668        for extra in &end_extras {
669            end_controls.append(extra);
670        }
671        if start_controls.first_child().is_some() {
672            // `start`/`end` mirror what GtkWindowControls' own boxes
673            // carry; idempotent where the headerbar already adds them.
674            start_controls.add_css_class("start");
675            header.pack_start(&start_controls);
676        }
677        if end_controls.first_child().is_some() {
678            end_controls.add_css_class("end");
679            header.pack_end(&end_controls);
680        }
681    }

683    /// A gesture controller not tagged `CARD_GESTURE_NAME` - i.e. one of
684    /// the HeaderBar's internal window gestures.
685    fn is_foreign_gesture(controller: &gtk4::EventController) -> bool {
686        controller.is::<gtk4::Gesture>() && controller.name().as_deref() != Some(CARD_GESTURE_NAME)
687    }

689    /// Removes every foreign gesture on `widget` and its descendants,
690    /// recursing into children but skipping `gtk4::Button` subtrees
691    /// *before* touching controllers - a button's internal `GestureClick`
692    /// is a foreign gesture by the tag rule, yet it is exactly what
693    /// makes the button work.
694    fn strip_foreign_gestures(widget: &impl IsA<gtk4::Widget>) {
695        let widget = widget.upcast_ref::<gtk4::Widget>();
696        if widget.is::<gtk4::Button>() {
697            return;
698        }
699        let controllers = widget.observe_controllers();
700        for index in (0..controllers.n_items()).rev() {
701            let Some(item) = controllers.item(index) else { continue };
702            let Ok(controller) = item.downcast::<gtk4::EventController>() else {
703                continue;
704            };
705            if is_foreign_gesture(&controller) {
706                widget.remove_controller(&controller);
707            }
708        }
709        let mut child = widget.first_child();
710        while let Some(sub) = child {
711            strip_foreign_gestures(&sub);
712            child = sub.next_sibling();
713        }
714    }

716    /// Watches the header's controller list for foreign gestures
717    /// registered *after* construction (e.g. on realize) and removes
718    /// them. Removal is deferred out of the `items_changed` emission -
719    /// mutating a list model inside its own change handler is fragile.
720    fn watch_foreign_gestures(header: &adw::HeaderBar, controllers: &gtk4::gio::ListModel) {
721        let header_weak = header.downgrade();
722        controllers.connect_items_changed(move |model, position, _removed, added| {
723            for index in position..position + added {
724                let Some(item) = model.item(index) else { continue };
725                let Ok(controller) = item.downcast::<gtk4::EventController>() else {
726                    continue;
727                };
728                if !is_foreign_gesture(&controller) {
729                    continue;
730                }
731                let Some(header) = header_weak.upgrade() else { continue };
732                gtk4::glib::idle_add_local_once(move || header.remove_controller(&controller));
733            }
734        });
735    }
