1//! Chrome element builders and mounting, implemented on [`Card`].
2//!
3//! One [`ChromeElement`] produces one widget plus its signal wiring
4//! (P1.1). `Inline` elements mount either into the card's header strip
5//! (drag handle, title, minimize, close) or onto the content overlay
6//! (resize/rotate grips); `Satellite` placements are validated and
7//! skipped until satellite mounting lands (Phase 2/3).
8//!
9//! The `builtin` renderer assembles the stock GTK4/CSS chrome; the
10//! `adwaita` feature adds a libadwaita renderer (`adw::HeaderBar`,
11//! `adw::WindowTitle`, wired buttons in a `windowcontrols` node). The
12//! bare widget factories at the bottom of this module stay free
13//! functions - they are stateless builders for foreign types (`Label`,
14//! `DrawingArea`).
15

70impl Card {
71    /// Builds the card chrome from `config`'s effective slot list and
72    /// mounts every `Inline` element: header elements onto a header
73    /// strip prepended to the card, grip elements onto the content
74    /// overlay.
75    ///
76    /// The header widget lands in `CardImpl::header` - it stays the
77    /// drag affordance for `DragHandle::Header` and the strip that
78    /// remains visible while folded. A header without any inline
79    /// children is not mounted at all (an explicit `chrome: []` config
80    /// produces a chrome-free card; use `DragHandle::WholeCard` or
81    /// `headerless` for those).
82    pub(crate) fn mount_chrome(&self, config: &CardConfig) {
83        let title = config.title.clone().unwrap_or_else(|| l10n().message("card-untitled"));
84        *self.imp().title_text.borrow_mut() = title.clone();
85        let slots = config.effective_chrome_slots();
86        let style = config.chrome_style.resolved();
87        if style != config.chrome_style {
88            warn!("CardConfig::chrome_style is Adwaita but the `adwaita` feature is disabled - falling back to Builtin");
89        }
90        // A circle card has no header strip: its inline chrome mounts
91        // along the disc's top arc as an "arc header" instead;
92        // `Satellite` placements stay satellites.
93        if config.shape == CardShape::Circle {
94            self.mount_arc_chrome(config);
95            return;
96        }
97        match style {
98            ChromeStyle::Builtin => self.mount_builtin_chrome(&title, &slots),
99            #[cfg(feature = "adwaita")]
100            ChromeStyle::Adwaita => self.mount_adwaita_chrome(config, &title, &slots),
101            // `resolved()` already fell back to Builtin - unreachable in
102            // practice, mount builtin defensively rather than panic.
103            #[cfg(not(feature = "adwaita"))]
104            ChromeStyle::Adwaita => self.mount_builtin_chrome(&title, &slots),
105        }
106        let imp = self.imp();
107        // The grip gesture wiring in `FloatingBoard` keys off the grip's
108        // existence, not the flags: a grip exists iff its slot was mounted.
109        imp.resizable.set(imp.resize_grip.get().is_some());
110        imp.rotate_grip_enabled.set(imp.rotate_grip.get().is_some());
111    }

113    /// `Circle` renderer: inline chrome slots along the disc's arc
114    /// bands (`CardConfig::arc_span` on top, `arc_bottom_span` on the
115    /// bottom - both default a quarter) as an "arc header" -
116    /// `CardImpl::circle_allocate` places them on the band's mid
117    /// radius, a `Title` pinned to the arc's center like a header
118    /// bar's. `Satellite` slots stay satellites; the edge grips have
119    /// no mount on a circle.
120    fn mount_arc_chrome(&self, config: &CardConfig) {
121        let imp = self.imp();
122        let span = config
123            .arc_span
124            .unwrap_or(Degrees::QUARTER_TURN)
125            .as_f64()
126            .clamp(Degrees::ZERO.as_f64(), Degrees::FULL_TURN.as_f64());
127        imp.arc_span.set(Degrees::new(span).to_radians());
128        let bottom_span = config
129            .arc_bottom_span
130            .unwrap_or(Degrees::new(span))
131            .as_f64()
132            .clamp(Degrees::ZERO.as_f64(), Degrees::FULL_TURN.as_f64());
133        imp.arc_bottom_span.set(Degrees::new(bottom_span).to_radians());
134        let title = imp.title_text.borrow().clone();
135        let slots = config.effective_chrome_slots();
136        let style = config.chrome_style.resolved();
137        let mut items = Vec::new();
138        for slot in &slots {
139            if slot.placement == ChromePlacement::Satellite {
140                skip_satellite(slot.element);
141                continue;
142            }
143            let widget: Widget = match slot.element {
144                ChromeElement::DragHandle => {
145                    let handle = drag_handle_label();
146                    let _ = imp.handle_label.set(handle.clone());
147                    handle.upcast()
148                }
149                ChromeElement::Title => {
150                    let label = title_label(&title);
151                    // The arc's centered item reads like a header bar's
152                    // title - it must not claim inline space.
153                    label.set_hexpand(false);
154                    label.set_halign(Align::Center);
155                    let _ = imp.title_label.set(label.clone());
156                    label.upcast()
157                }
158                ChromeElement::Minimize => {
159                    let button = self.build_minimize_button(ICON_MINIMIZE_INLINE);
160                    let _ = imp.fold_button.set(button.clone());
161                    button.upcast()
162                }
163                ChromeElement::Close => {
164                    let button = self.build_close_button();
165                    let _ = imp.close_button.set(button.clone());
166                    button.upcast()
167                }
168                ChromeElement::RotateClockwise => self.build_rotate_button(RotationDirection::Clockwise).upcast(),
169                ChromeElement::RotateCounterClockwise => self.build_rotate_button(RotationDirection::CounterClockwise).upcast(),
170                ChromeElement::NextSeat => self.build_pass_button(TurnDirection::Clockwise).upcast(),
171                ChromeElement::PreviousSeat => self.build_pass_button(TurnDirection::CounterClockwise).upcast(),
172                ChromeElement::ResizeGrip | ChromeElement::RotateGrip => {
173                    warn!(element = ?slot.element, "edge grip has no mount on a circle card - skipped");
174                    continue;
175                }
176            };
177            widget.add_css_class(CLASS_ARC_CHROME);
178            if style == ChromeStyle::Adwaita {
179                // The themed look chrome satellites get: a raised round
180                // button instead of the builtin flat icon.
181                widget.remove_css_class("flat");
182                widget.add_css_class("circular");
183            }
184            self.append(&widget);
185            items.push((widget, *slot));
186        }
187        // The painted band sits beneath the items (earlier children
188        // paint first) and is created only when items exist - an
189        // empty band would be a stray ring sector.
190        if !items.is_empty() {
191            let band = arc_band(self);
192            band.insert_before(self, items.first().map(|(first, _)| first));
193            let _ = imp.arc_band.set(band);
194        }
195        *imp.arc_items.borrow_mut() = items;
196    }

198    /// `Builtin` renderer: header strip as a `gtk4::Box` of inline
199    /// chrome slots, grips on the content overlay. Reproduces the
200    /// historical card look one-to-one (P1.3 regression baseline).
201    fn mount_builtin_chrome(&self, title: &str, slots: &[ChromeSlot]) {
202        let imp = self.imp();
203        let header = gtk4::Box::new(Orientation::Horizontal, 6);
204        header.add_css_class(CLASS_HEADER);
205        header.set_hexpand(true);
206        // Header children are grouped by placement: `InlineStart`
207        // children land before the neutral `Inline` run (which keeps
208        // slot order), `InlineEnd` children after it - so ends of the
209        // strip are controllable independent of slot order.
210        let mut start_children: Vec<Widget> = Vec::new();
211        let mut middle_children: Vec<Widget> = Vec::new();
212        let mut end_children: Vec<Widget> = Vec::new();
213        for slot in slots {
214            if slot.placement == ChromePlacement::Satellite {
215                skip_satellite(slot.element);
216                continue;
217            }
218            let target = match slot.placement {
219                ChromePlacement::InlineStart => &mut start_children,
220                ChromePlacement::InlineEnd => &mut end_children,
221                _ => &mut middle_children,
222            };
223            match slot.element {
224                ChromeElement::DragHandle => {
225                    let handle = drag_handle_label();
226                    target.push(handle.clone().upcast());
227                    let _ = imp.handle_label.set(handle);
228                }
229                ChromeElement::Title => {
230                    let label = title_label(title);
231                    target.push(label.clone().upcast());
232                    let _ = imp.title_label.set(label);
233                }
234                ChromeElement::Minimize => {
235                    let button = self.build_minimize_button(ICON_MINIMIZE_INLINE);
236                    target.push(button.clone().upcast());
237                    let _ = imp.fold_button.set(button);
238                }
239                ChromeElement::Close => {
240                    let button = self.build_close_button();
241                    target.push(button.clone().upcast());
242                    let _ = imp.close_button.set(button);
243                }
244                ChromeElement::RotateClockwise => target.push(self.build_rotate_button(RotationDirection::Clockwise).upcast()),
245                ChromeElement::RotateCounterClockwise => target.push(self.build_rotate_button(RotationDirection::CounterClockwise).upcast()),
246                ChromeElement::NextSeat => target.push(self.build_pass_button(TurnDirection::Clockwise).upcast()),
247                ChromeElement::PreviousSeat => target.push(self.build_pass_button(TurnDirection::CounterClockwise).upcast()),
248                ChromeElement::ResizeGrip => self.mount_grip(ChromeElement::ResizeGrip),
249                ChromeElement::RotateGrip => self.mount_grip(ChromeElement::RotateGrip),
250            }
251        }
252        let header_children = start_children.len() + middle_children.len() + end_children.len();
253        for child in start_children.iter().chain(&middle_children).chain(&end_children) {
254            header.append(child);
255        }
256        // A header without children is skipped entirely: no dead strip
257        // is drawn and `drag_source(DragHandle::Header)` falls back to
258        // the card widget, keeping a chrome-free card draggable.
259        if header_children > 0 {
260            imp.header_shown.set(true);
261            self.prepend(&header);
262            let _ = imp.header.set(header.upcast());
263        }
264    }

266    /// Mounts the grip widget for `element` on the card's content
267    /// overlay and records it in the matching `CardImpl` cell.
268    /// Non-grip elements are ignored.
269    fn mount_grip(&self, element: ChromeElement) {
270        let imp = self.imp();
271        let Some(overlay) = imp.content_overlay.get() else {
272            return;
273        };
274        let (grip, cell) = match element {
275            ChromeElement::ResizeGrip => (resize_grip(), &imp.resize_grip),
276            ChromeElement::RotateGrip => (rotate_grip(), &imp.rotate_grip),
277            _ => return,
278        };
279        overlay.add_overlay(&grip);
280        let _ = cell.set(grip);
281    }

362/// A `Satellite` slot never mounts inline - satellite-capable elements
363/// are picked up by the board's satellite mounting (`CardConfig::
364/// effective_satellites`); elements that can never be satellites warn.
365fn skip_satellite(element: ChromeElement) {
366    if element.supports_satellite() {
367        debug!("Chrome element {element:?} configured as satellite - mounted by the board, not inline");
368    } else {
369        warn!("Chrome element {element:?} does not support satellite placement - slot ignored");
370    }
371}

373/// The painted top-quarter annulus band behind a circle card's arc
374/// chrome (`ARC_CHROME_BAND` thick, ±45°): a `DrawingArea` stroking a
375/// round-capped arc on the band's mid radius in the card's header
376/// color - read through the CSS `color` property so palettes tint it.
377/// Input-transparent: the band is decoration, the ring stays a drag
378/// surface.
379fn arc_band(card: &Card) -> DrawingArea {
380    let band = DrawingArea::new();
381    band.add_css_class(CLASS_ARC_BAND);
382    band.set_can_target(false);
383    let card_weak = card.downgrade();
384    band.set_draw_func(move |area, cr, width, height| {
385        let radius = f64::from(width.min(height)) / 2.0 - ARC_CHROME_BAND / 2.0;
386        let Some(card) = card_weak.upgrade() else { return };
387        if radius <= 0.0 {
388            return;
389        }
390        let top_half_span = card.imp().arc_span.get() / 2.0;
391        let bottom_half_span = card.imp().arc_bottom_span.get() / 2.0;
392        let items = card.imp().arc_items.borrow();
393        let has_top = items.iter().any(|(_, slot)| slot.placement.is_top_arc());
394        let has_bottom = items.iter().any(|(_, slot)| slot.placement.is_bottom_arc());
395        drop(items);
396        let color = area.color();
397        cr.set_source_rgba(f64::from(color.red()), f64::from(color.green()), f64::from(color.blue()), f64::from(color.alpha()));
398        cr.set_line_width(ARC_CHROME_BAND);
399        cr.set_line_cap(LineCap::Round);
400        // Cairo measures angles from +x: the bands center on -90°
401        // (top) and +90° (bottom), each `arc_span`/`arc_bottom_span` wide.
402        for (center, present, half_span) in [(-FRAC_PI_2, has_top, top_half_span), (FRAC_PI_2, has_bottom, bottom_half_span)] {
403            if present {
404                cr.arc(
405                    f64::from(width) / 2.0,
406                    f64::from(height) / 2.0,
407                    radius,
408                    center - half_span.as_f64(),
409                    center + half_span.as_f64(),
410                );
411                let _ = cr.stroke();
412            }
413        }
414    });
415    band
416}

438/// Edge length of the resize grip's diagonal strokes (px).
439const RESIZE_GRIP_SIZE: i32 = 18;
440
441/// The bottom-right resize affordance: three diagonal strokes in the
442/// card's foreground color, `nwse-resize` cursor on hover.
443fn resize_grip() -> DrawingArea {
444    let grip = DrawingArea::new();
445    grip.add_css_class(CLASS_RESIZE_GRIP);
446    // Overlay children default to `Align::Fill` - without End
447    // alignment the grip would cover the whole content area.
448    grip.set_halign(Align::End);
449    grip.set_valign(Align::End);
450    grip.set_content_width(RESIZE_GRIP_SIZE);
451    grip.set_content_height(RESIZE_GRIP_SIZE);
452    grip.set_cursor_from_name(Some("nwse-resize"));
453    grip.set_tooltip_text(Some(&l10n().message("card-resize")));
454    grip.set_draw_func(|grip, cr, _, _| {
455        let color = grip.color();
456        let width = f64::from(grip.width());
457        let height = f64::from(grip.height());
458        cr.set_source_rgba(f64::from(color.red()), f64::from(color.green()), f64::from(color.blue()), f64::from(color.alpha()));
459        cr.set_line_width(1.5);
460        cr.set_line_cap(LineCap::Round);
461        for inset in [4.0, 8.0, 12.0] {
462            cr.move_to(width - inset, height - 2.0);
463            cr.line_to(width - 2.0, height - inset);
464            let _ = cr.stroke();
465        }
466    });
467    grip
468}

470/// Edge length of the rotate grip's circular arrow (px).
471const ROTATE_GRIP_SIZE: i32 = 18;
472
473/// The bottom-left rotate affordance: a circular arrow in the card's
474/// foreground color, `grab` cursor on hover.
475fn rotate_grip() -> DrawingArea {
476    let grip = DrawingArea::new();
477    grip.add_css_class(CLASS_ROTATE_GRIP);
478    // Overlay children default to `Align::Fill` - without Start/End
479    // alignment the grip would cover the whole content area.
480    grip.set_halign(Align::Start);
481    grip.set_valign(Align::End);
482    grip.set_content_width(ROTATE_GRIP_SIZE);
483    grip.set_content_height(ROTATE_GRIP_SIZE);
484    grip.set_cursor_from_name(Some("grab"));
485    grip.set_tooltip_text(Some(&l10n().message("card-rotate")));
486    grip.set_draw_func(|grip, cr, _, _| {
487        let color = grip.color();
488        let width = f64::from(grip.width());
489        let height = f64::from(grip.height());
490        cr.set_source_rgba(f64::from(color.red()), f64::from(color.green()), f64::from(color.blue()), f64::from(color.alpha()));
491        cr.set_line_width(1.5);
492        cr.set_line_cap(LineCap::Round);
493        // A three-quarter clockwise arc with an arrowhead on its tip:
494        // the circular arrow suggests turning in either direction.
495        let radius = width.min(height) / 2.0 - 3.0;
496        let center_x = width / 2.0;
497        let center_y = height / 2.0;
498        let start = -FRAC_PI_2 + 0.7;
499        let end = start + 3.0 * FRAC_PI_2;
500        cr.arc(center_x, center_y, radius, start, end);
501        let _ = cr.stroke();
502        let tip_x = center_x + radius * end.cos();
503        let tip_y = center_y + radius * end.sin();
504        let tangent = end + FRAC_PI_2;
505        for spread in [-0.7, 0.7] {
506            cr.move_to(tip_x, tip_y);
507            cr.line_to(tip_x - 4.0 * (tangent + spread).cos(), tip_y - 4.0 * (tangent + spread).sin());
508        }
509        let _ = cr.stroke();
510    });
511    grip
512}

514/// `Adwaita` renderer (feature `adwaita`): libadwaita chrome per P1.4 -
515/// `adw::HeaderBar` as the header strip (with `default-decoration`, the
516/// slim window-titlebar variant), `adw::WindowTitle` for the `Title`
517/// element. `Minimize`/`Close` render as real wired buttons (the same
518/// builders the builtin renderer uses) inside `windowcontrols`-named
519/// containers placed by `gtk-decoration-layout`, so the libadwaita
520/// stylesheet and GTK themes (e.g. Rewaita) treat them exactly like
521/// window controls; `DragHandle` and the grips stay custom widgets.

598    impl Card {
599        /// Builds the `adw::HeaderBar` chrome for the card.
600        pub(super) fn mount_adwaita_chrome(&self, config: &CardConfig, title: &str, slots: &[ChromeSlot]) {
601            let imp = self.imp();
602            if !adw::is_initialized() {
603                debug!("ChromeStyle::Adwaita configured - call `adw::init()` (or use `adw::Application`) for themed styling");
604            }
605            let header = adw::HeaderBar::new();
606            header.set_show_start_title_buttons(false);
607            header.set_show_end_title_buttons(false);
608            // The headerbar paints the window chrome background; keep the
609            // card's header CSS class so the palette/theme keeps styling
610            // the frame around it.
611            header.add_css_class(CLASS_HEADER);
612            // Real Adwaita *window* titlebars carry `default-decoration`:
613            // the stylesheet shrinks them to 37px with 22px controls. A
614            // plain `adw::HeaderBar` would render at the 46px app-header
615            // height - noticeably taller than both our builtin header and
616            // a real window titlebar.
617            header.add_css_class("default-decoration");
618            // The HeaderBar registers internal gestures that drive the
619            // *toplevel window* (move-drag, double-click maximize). On a
620            // card they race the card's own drag - a fast drag moved the
621            // whole window instead of the card. Watch for them from the
622            // start, then strip the subtree after our children exist.
623            let watcher = header.observe_controllers();
624            watch_foreign_gestures(&header, &watcher);
625            // The watcher only covers the header's *own* controller
626            // list; libadwaita can also register gestures lazily on
627            // inner widgets (windowhandle) at map time. Strip the whole
628            // subtree once more when the header is actually shown.
629            header.connect_map(strip_foreign_gestures);
630            // Wired control buttons, collected per slot and placed by
631            // `pack_window_controls` afterwards: side and order follow
632            // the `gtk-decoration-layout` setting (themes like Rewaita
633            // move window buttons to the left).
634            // Control buttons route by placement: `InlineStart`
635            // collects into the start `windowcontrols` container,
636            // `InlineEnd` into the end container, neutral `Inline`
637            // keeps the historical behavior - `Minimize`/`Close`
638            // follow `gtk-decoration-layout`, other actions append to
639            // the end controls.
640            let mut minimize_button: Option<gtk4::Button> = None;
641            let mut close_button: Option<gtk4::Button> = None;
642            let mut start_extras: Vec<gtk4::Button> = Vec::new();
643            let mut end_extras: Vec<gtk4::Button> = Vec::new();
644            let mut header_children = 0usize;
645            for slot in slots {
646                if slot.placement == ChromePlacement::Satellite {
647                    skip_satellite(slot.element);
648                    continue;
649                }
650                match slot.element {
651                    ChromeElement::DragHandle => {
652                        let handle = drag_handle_label();
653                        match slot.placement {
654                            ChromePlacement::InlineEnd => header.pack_end(&handle),
655                            _ => header.pack_start(&handle),
656                        }
657                        let _ = imp.handle_label.set(handle);
658                        header_children += 1;
659                    }
660                    ChromeElement::Title => {
661                        let title_widget = adw::WindowTitle::new(title, "");
662                        title_widget.set_visible(!config.headerless);
663                        match slot.placement {
664                            ChromePlacement::InlineStart => header.pack_start(&title_widget),
665                            ChromePlacement::InlineEnd => header.pack_end(&title_widget),
666                            _ => header.set_title_widget(Some(&title_widget)),
667                        }
668                        header_children += 1;
669                    }
670                    _ => {
671                        let (button, class) = match slot.element {
672                            ChromeElement::Minimize => (self.build_minimize_button(ICON_MINIMIZE_SYMBOLIC), "minimize"),
673                            ChromeElement::Close => (self.build_close_button(), "close"),
674                            ChromeElement::RotateClockwise => (self.build_rotate_button(RotationDirection::Clockwise), "rotate-clockwise"),
675                            ChromeElement::RotateCounterClockwise => {
676                                (self.build_rotate_button(RotationDirection::CounterClockwise), "rotate-counter-clockwise")
677                            }
678                            ChromeElement::NextSeat => (self.build_pass_button(TurnDirection::Clockwise), "next-seat"),
679                            ChromeElement::PreviousSeat => (self.build_pass_button(TurnDirection::CounterClockwise), "previous-seat"),
680                            ChromeElement::ResizeGrip => {
681                                self.mount_grip(ChromeElement::ResizeGrip);
682                                continue;
683                            }
684                            ChromeElement::RotateGrip => {
685                                self.mount_grip(ChromeElement::RotateGrip);
686                                continue;
687                            }
688                            ChromeElement::DragHandle | ChromeElement::Title => continue,
689                        };
690                        // A real wired button: `WindowControls` buttons
691                        // are decorative (they target a real toplevel).
692                        // Let the `windowcontrols` node rules own the
693                        // look (the `flat` class would fight their
694                        // padding/hover styling), and carry the role
695                        // class themes key colors off.
696                        button.remove_css_class("flat");
697                        button.add_css_class(class);
698                        match slot.placement {
699                            ChromePlacement::InlineStart => start_extras.push(button.clone()),
700                            ChromePlacement::InlineEnd => end_extras.push(button.clone()),
701                            _ => match slot.element {
702                                ChromeElement::Minimize => minimize_button = Some(button.clone()),
703                                ChromeElement::Close => close_button = Some(button.clone()),
704                                _ => end_extras.push(button.clone()),
705                            },
706                        }
707                        if slot.element == ChromeElement::Minimize {
708                            let _ = imp.fold_button.set(button.clone());
709                        }
710                        if slot.element == ChromeElement::Close {
711                            let _ = imp.close_button.set(button);
712                        }
713                        header_children += 1;
714                    }
715                }
716            }
717            pack_window_controls(&header, minimize_button, close_button, start_extras, end_extras);
718            strip_foreign_gestures(&header);
719            // Same rule as the builtin renderer: no children, no header -
720            // `drag_source(DragHandle::Header)` falls back to the card.
721            if header_children > 0 {
722                imp.header_shown.set(true);
723                self.prepend(&header);
724                let _ = imp.header.set(header.upcast());
725            }
726        }
727    }

729    /// Fallback when `gtk-decoration-layout` is unset: all controls on
730    /// the right in GNOME order.
731    const DEFAULT_DECORATION_LAYOUT: &str = ":minimize,maximize,close";
732
733    /// Splits `gtk-decoration-layout` into the button names left and
734    /// right of the `:` separator (e.g. Rewaita's `"close,minimize:"`
735    /// moves all controls to the left side).
736    fn decoration_layout_names() -> (Vec<String>, Vec<String>) {
737        let layout = gtk4::Settings::default()
738            .and_then(|settings| settings.gtk_decoration_layout())
739            .map(|layout| layout.to_string())
740            .unwrap_or_else(|| DEFAULT_DECORATION_LAYOUT.to_string());
741        let names = |side: &str| side.split(',').map(str::trim).filter(|name| !name.is_empty()).map(str::to_string).collect();
742        let mut sides = layout.splitn(2, ':');
743        let start = names(sides.next().unwrap_or_default());
744        let end = names(sides.next().unwrap_or_default());
745        (start, end)
746    }

748    /// Packs the card's control buttons into `windowcontrols` containers,
749    /// following the decoration layout: a button goes to its named side in
750    /// layout order; slotted buttons the layout does not mention append to
751    /// the end side (a configured slot is never dropped, only reordered).
752    /// `extras` are card actions outside the window-button vocabulary
753    /// (rotate/pass steps): `start_extras` lead the start controls
754    /// (outermost first, like GNOME's app-buttons-left convention),
755    /// `end_extras` trail the end controls.
756    fn pack_window_controls(
757        header: &adw::HeaderBar,
758        minimize: Option<gtk4::Button>,
759        close: Option<gtk4::Button>,
760        start_extras: Vec<gtk4::Button>,
761        end_extras: Vec<gtk4::Button>,
762    ) {
763        let (start_names, end_names) = decoration_layout_names();
764        let start_controls = ChromeControls::new();
765        let end_controls = ChromeControls::new();
766        let mut minimize = minimize;
767        let mut close = close;
768        let mut take = |name: &str| match name {
769            "minimize" => minimize.take(),
770            "close" => close.take(),
771            _ => None,
772        };
773        for extra in &start_extras {
774            start_controls.append(extra);
775        }
776        for name in &start_names {
777            if let Some(button) = take(name) {
778                start_controls.append(&button);
779            }
780        }
781        for name in &end_names {
782            if let Some(button) = take(name) {
783                end_controls.append(&button);
784            }
785        }
786        drop(take);
787        if let Some(button) = minimize.take() {
788            end_controls.append(&button);
789        }
790        if let Some(button) = close.take() {
791            end_controls.append(&button);
792        }
793        for extra in &end_extras {
794            end_controls.append(extra);
795        }
796        if start_controls.first_child().is_some() {
797            // `start`/`end` mirror what GtkWindowControls' own boxes
798            // carry; idempotent where the headerbar already adds them.
799            start_controls.add_css_class("start");
800            header.pack_start(&start_controls);
801        }
802        if end_controls.first_child().is_some() {
803            end_controls.add_css_class("end");
804            header.pack_end(&end_controls);
805        }
806    }

808    /// A gesture controller not tagged `CARD_GESTURE_NAME` - i.e. one of
809    /// the HeaderBar's internal window gestures.
810    fn is_foreign_gesture(controller: &gtk4::EventController) -> bool {
811        controller.is::<gtk4::Gesture>() && controller.name().as_deref() != Some(CARD_GESTURE_NAME)
812    }
813
814    /// Removes every foreign gesture on `widget` and its descendants,
815    /// recursing into children but skipping `gtk4::Button` subtrees
816    /// *before* touching controllers - a button's internal `GestureClick`
817    /// is a foreign gesture by the tag rule, yet it is exactly what
818    /// makes the button work.
819    fn strip_foreign_gestures(widget: &impl IsA<gtk4::Widget>) {
820        let widget = widget.upcast_ref::<gtk4::Widget>();
821        if widget.is::<gtk4::Button>() {
822            return;
823        }
824        let controllers = widget.observe_controllers();
825        for index in (0..controllers.n_items()).rev() {
826            let Some(item) = controllers.item(index) else { continue };
827            let Ok(controller) = item.downcast::<gtk4::EventController>() else {
828                continue;
829            };
830            if is_foreign_gesture(&controller) {
831                widget.remove_controller(&controller);
832            }
833        }
834        let mut child = widget.first_child();
835        while let Some(sub) = child {
836            strip_foreign_gestures(&sub);
837            child = sub.next_sibling();
838        }
839    }

841    /// Watches the header's controller list for foreign gestures
842    /// registered *after* construction (e.g. on realize) and removes
843    /// them. Removal is deferred out of the `items_changed` emission -
844    /// mutating a list model inside its own change handler is fragile.
845    fn watch_foreign_gestures(header: &adw::HeaderBar, controllers: &gtk4::gio::ListModel) {
846        let header_weak = header.downgrade();
847        controllers.connect_items_changed(move |model, position, _removed, added| {
848            for index in position..position + added {
849                let Some(item) = model.item(index) else { continue };
850                let Ok(controller) = item.downcast::<gtk4::EventController>() else {
851                    continue;
852                };
853                if !is_foreign_gesture(&controller) {
854                    continue;
855                }
856                let Some(header) = header_weak.upgrade() else { continue };
857                gtk4::glib::idle_add_local_once(move || header.remove_controller(&controller));
858            }
859        });
860    }
