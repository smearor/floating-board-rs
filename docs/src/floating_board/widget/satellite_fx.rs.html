1//! Optional `smearor-commons-fx` integration (feature `fx`, Phase 5.3):
2//! snapshot a live widget into a texture, mount an [`FxPlayer`] at its
3//! layer position, play an [`EffectSpec`], then swap the live widget
4//! back in on `animation-ended`.
5
6use crate::domain::BoardError;
7use crate::domain::CardId;
8use crate::domain::SatelliteId;
9use crate::widget::board::FloatingBoard;
10use glib::clone::Downgrade;
11use glib::object::Cast;
12use glib::subclass::types::ObjectSubclassIsExt;

17use smearor_commons_fx::gtk::FxPlayOptions;
18use smearor_commons_fx::gtk::FxPlayer;
19use smearor_commons_fx::gtk::snapshot;
20use smearor_commons_geometry::Pixels;
21use smearor_commons_geometry::Vec2;
22use std::time::Duration;

24impl FloatingBoard {
25    /// Plays an [`EffectSpec`] over one satellite of a card: the
26    /// satellite's transform stack is snapshotted to a texture, replaced
27    /// by an [`FxPlayer`] at the same layer position, and swapped back
28    /// in when `animation-ended` fires. The texture is static - geometry
29    /// changes during playback do not track; the live widget returns at
30    /// its current placement on the next sync.
31    ///
32    /// `FxPlayOptions::duration == Duration::ZERO` defaults to the
33    /// card's `animation_speed_ms` budget.
34    pub fn play_satellite_fx(&self, id: &CardId, satellite: &SatelliteId, spec: EffectSpec, options: FxPlayOptions) -> Result<(), BoardError> {
35        let cards = self.imp().cards.borrow();
36        let entry = cards.get(id).ok_or_else(|| BoardError::UnknownCard(id.clone()))?;
37        let satellites = entry.satellites.borrow();
38        let Some(satellite_entry) = satellites.iter().find(|entry| &entry.id == satellite) else {
39            return Err(BoardError::UnknownSatellite {
40                card: id.clone(),
41                satellite: satellite.clone(),
42            });
43        };
44        let target = satellite_entry.scaling.clone().upcast::<Widget>();
45        self.play_widget_fx(&target, spec, options, entry.config.animation_speed_ms)
46    }
47
48    /// Same FX swap as [`Self::play_satellite_fx`] for a whole card:
49    /// the card's transform stack is snapshotted, replaced by an
50    /// [`FxPlayer`], and swapped back in on `animation-ended`.
51    /// Satellites keep rendering normally while the effect plays.
52    pub fn play_card_fx(&self, id: &CardId, spec: EffectSpec, options: FxPlayOptions) -> Result<(), BoardError> {
53        let cards = self.imp().cards.borrow();
54        let entry = cards.get(id).ok_or_else(|| BoardError::UnknownCard(id.clone()))?;
55        let target = entry.scaling.clone().upcast::<Widget>();
56        self.play_widget_fx(&target, spec, options, entry.config.animation_speed_ms)
57    }
58
59    /// The snapshot -> [`FxPlayer`] -> swap-back pipeline shared by the
60    /// satellite and card FX entry points.
61    fn play_widget_fx(&self, target: &Widget, spec: EffectSpec, mut options: FxPlayOptions, speed_ms: u64) -> Result<(), BoardError> {
62        let Some(layer) = self.card_layer() else {
63            return Err(BoardError::Fx("board has no card layer".to_string()));
64        };
65        let buffer = snapshot(target).map_err(|error| BoardError::Fx(error.to_string()))?;
66        let player = FxPlayer::new();
67        player.set_effect(spec, buffer);
68        if options.duration.is_zero() {
69            options.duration = Duration::from_millis(speed_ms.max(1));
70        }
71        let position = Vec2::<Pixels>::from(layer.child_position(target));
72        let (width, height) = (target.width(), target.height());
73        if width > 0 && height > 0 {
74            player.set_size_request(width, height);
75        }
76        layer.put(&player, position.x, position.y);
77        let was_visible = target.is_visible();
78        target.set_visible(false);
79        let target_weak = target.downgrade();
80        let layer_weak = layer.downgrade();
81        player.connect_animation_ended(move |player| {
82            if let Some(layer) = layer_weak.upgrade() {
83                layer.remove(player);
84            }
85            if let Some(target) = target_weak.upgrade() {
86                target.set_visible(was_visible);
87            }
88        });
89        player.play(options);
90        Ok(())
91    }
92}
