34/// Placement and interaction configuration of a card.
35///
36/// Passed to `FloatingBoard::add_card`; all fields have defaults so
37/// `CardConfig::default()` yields a fully interactive, draggable card anchored
38/// at the board center.
39#[derive(Debug, Clone, PartialEq)]
40#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
41pub struct CardConfig {
42    /// Title shown in the header strip; falls back to a localized default when
43    /// the header is visible and no title is set.
44    pub title: Option<String>,
45    /// Initial placement anchor, resolved on first allocation.
46    pub position: CardAnchor,
47    /// Whether the card can be moved by dragging.
48    pub draggable: bool,
49    /// Whether the card can be rotated via the two-finger rotate gesture.
50    pub rotatable: bool,
51    /// Whether the card can be scaled via the pinch gesture.
52    pub scalable: bool,
53    /// Whether the card shows a bottom-right resize grip that scales the
54    /// card around its center when dragged (mouse and touch).
55    pub resizable: bool,
56    /// Whether the card shows a bottom-left rotate grip that steps the
57    /// rotation to the next quarter turn in the swipe direction - one
58    /// quarter turn per drag, no matter how far the swipe goes (mouse
59    /// and touch). Works independently of `rotatable`, which gates the
60    /// free two-finger rotate.
61    #[cfg_attr(feature = "serde", serde(default))]
62    pub rotate_grip: bool,
63    /// Whether the card can be folded down to its header.
64    pub foldable: bool,
65    /// Whether the header shows a close button.
66    pub closable: bool,
67    /// What the close button does - remove the card (default) or hide it
68    /// so a menu can re-open it later.
69    pub close_action: CardCloseAction,
70    /// Collapses the header to a slim grip when no title/buttons are needed.
71    pub headerless: bool,
72    /// Which part of the card acts as the drag affordance.
73    pub drag_handle: DragHandle,
74    /// The card's outline shape - `Rectangle` (default) or `Circle`.
75    ///
76    /// A circle card has no header strip and no corner grips: inline
77    /// chrome mounts along the disc's arc bands (`arc_span` on top,
78    /// `arc_bottom_span` on the bottom), `Satellite`
79    /// placements stay satellites, the edge grips are dropped with a
80    /// warning at mount, and taps in the bounding-box corners pass
81    /// through to whatever is underneath.
82    #[cfg_attr(feature = "serde", serde(default))]
83    pub shape: CardShape,
84    /// How a shaped card (`Circle`/`Semicircle`/`Hexagon`/`Octagon`)
85    /// allocates its content inside the bounding box: `Inscribed`
86    /// fits the shape's inscribed square, `Fill` the rim-shrunk
87    /// silhouette (the content draws its own shape), `Clipped` like
88    /// `Fill` plus a silhouette clip so picks in the cut regions fall
89    /// back onto the card. Ignored for `Rectangle`.
90    #[cfg_attr(feature = "serde", serde(default, alias = "circle_content"))]
91    pub content: CardContent,
92    /// Angular span of the arc-chrome band on `CardShape::Circle` and
93    /// `CardShape::Annulus` - the top arc the inline chrome items ride
94    /// on, centered on the card's top. `None` defaults to
95    /// `Degrees::QUARTER_TURN` (90°); values above a full turn are
96    /// clamped. Ignored for `Rectangle` and for cards without inline
97    /// chrome.
98    #[cfg_attr(feature = "serde", serde(default))]
99    pub arc_span: Option<Degrees>,
100    /// Angular span of the bottom arc-chrome band on
101    /// `CardShape::Circle` and `CardShape::Annulus`. `None` mirrors
102    /// `arc_span`; set it when the bottom band should wrap a different
103    /// share of the rim than the top band. Ignored for `Rectangle` and
104    /// for cards without `Bar*` chrome.
105    #[cfg_attr(feature = "serde", serde(default))]
106    pub arc_bottom_span: Option<Degrees>,
107    /// Fraction of the outer radius making the see-through,
108    /// click-through hole on `CardShape::Annulus`. `None` defaults to
109    /// `DEFAULT_HOLE_RATIO` (0.5); clamped to `0.05..=0.95`. Ignored
110    /// for every other shape.
111    #[cfg_attr(feature = "serde", serde(default))]
112    pub hole_ratio: Option<Fraction>,
113    /// Margin in pixels an edge or corner anchor keeps between the card's
114    /// footprint and the board edge; default `ANCHOR_MARGIN` (24 px).
115    /// Ignored for `Center` and `Explicit` placements.
116    #[cfg_attr(feature = "serde", serde(default = "default_anchor_margin"))]
117    pub anchor_margin: Pixels,
118    /// Inclusive scale limits for the pinch gesture; default `0.5 - 2.0`.
119    pub scale_limits: ScaleBounds,
120    /// Minimum overlap in pixels a card must keep inside the board bounds.
121    pub min_visible: MinVisible,
122    /// How strictly the card's footprint is kept inside the board bounds;
123    /// default `Loose` (only `min_visible` is enforced).
124    #[cfg_attr(feature = "serde", serde(default))]
125    pub bounds_mode: BoundsMode,
126    /// Per-card color scheme: a [`CardPalette`] mapped onto scoped CSS
127    /// at build time. `None` (or a palette with unfilled slots) keeps
128    /// the themed/default-card colors.
129    #[cfg_attr(feature = "serde", serde(default))]
130    pub palette: Option<CardPalette>,
131    /// Space in pixels between the card border and the content widget;
132    /// default `DEFAULT_CONTENT_PADDING` (10 px, the former fixed CSS
133    /// value). `0` lets the content sit flush with the card edge.
134    /// Applies to `CardContent::Inscribed` only - `Fill`/`Clipped`
135    /// carry the ring band as their designed inset already, and
136    /// `Cover` is edge-to-edge.
137    #[cfg_attr(feature = "serde", serde(default = "default_content_padding"))]
138    pub content_padding: Pixels,
139    /// Optional initial z-ordering; `None` appends the card on top.
140    pub z_index: Option<i32>,
141    /// Hard upper bound on the card's measured width; `None` lets
142    /// the card grow with its content. Caps growth caused by changing content
143    /// (e.g. a number gaining a digit): the layout allocates at most this
144    /// size and overflowing content is clipped by the default CSS.
145    #[cfg_attr(feature = "serde", serde(default))]
146    pub max_width: Option<Pixels>,
147    /// Hard upper bound on the card's measured height; see
148    /// `max_width`.
149    #[cfg_attr(feature = "serde", serde(default))]
150    pub max_height: Option<Pixels>,
151    /// Duration of rotation animations (gesture snapping and
152    /// `set_rotation_with_animation`) in milliseconds; default 500.
153    #[cfg_attr(feature = "serde", serde(default = "default_animation_speed_ms"))]
154    pub animation_speed_ms: u64,
155    /// Optional seat assignment on the tabletop edge the card belongs
156    /// to - decoupled from the card's visual rotation: rotating the
157    /// board or freely spinning the card does not change which user
158    /// owns it. `None` leaves the card unassigned; seat strategies and
159    /// `rotate_to_next_user` fall back to the nearest side of its
160    /// rotation.
161    #[cfg_attr(feature = "serde", serde(default))]
162    pub seat: Option<Seat>,
163    /// Custom widget shown while the card is folded. When set, folding
164    /// collapses the whole card - header included - to this widget and
165    /// adds the `floating-card-collapsed` class, so a round button or
166    /// a compact mini view can replace the full card. `None` keeps the
167    /// default fold that leaves the header strip visible.
168    ///
169    /// The folded widget should offer an unfold affordance wired to
170    /// `Card::set_folded(false)` or `Card::fold` - the card's own fold
171    /// button hides with the header. `unfold_on_click` provides a
172    /// ready-made affordance for cards without one.
173    #[cfg_attr(feature = "serde", serde(skip))]
174    pub folded_content: Option<Widget>,
175    /// A click on the folded card unfolds it (revealing the satellite
176    /// set again). The click gesture only releases on sequences the
177    /// card drag did not claim, so moving a folded card never unfolds
178    /// it. While enabled the circle card's `Minimize` fold exception is
179    /// unnecessary - the card is one tap away from unfolding - so the
180    /// satellite closes with the rest of the set.
181    #[cfg_attr(feature = "serde", serde(default))]
182    pub unfold_on_click: bool,
183    /// How the card transitions between unfolded and folded
184    /// presentation (`CardFoldAnimation`): an eased program over the
185    /// axes width, height, rotation, zoom and content opacity,
186    /// budgeted by `animation_speed_ms`. `Instant` (default) swaps
187    /// atomically; reduced motion lands instantly too.
188    #[cfg_attr(feature = "serde", serde(default))]
189    pub fold_animation: CardFoldAnimation,
190    /// The morph program for the unfold direction - `None` replays
191    /// `fold_animation` mirrored, so a single setting animates both
192    /// directions.
193    #[cfg_attr(feature = "serde", serde(default))]
194    pub unfold_animation: Option<CardFoldAnimation>,
195    /// When an animated fold plays relative to the satellite close
196    /// choreography: `Sequential` (default) morphs after the last
197    /// satellite sank into the card, `Overlap(fraction)` starts the
198    /// morph mid-close (`0.0` = parallel). Unfolds always morph while
199    /// the set reveals.
200    #[cfg_attr(feature = "serde", serde(default))]
201    pub fold_timing: CardFoldTiming,
202    /// The unfold-direction choreography timing relative to the
203    /// satellite reveal - `None` resolves to `Overlap(0.0)`, the
204    /// morph and the reveal run in lockstep.
205    #[cfg_attr(feature = "serde", serde(default))]
206    pub unfold_timing: Option<CardFoldTiming>,
207    /// Pins the card to the screen instead of the workspace: a sticky
208    /// card ignores the `View` camera entirely and keeps its on-screen
209    /// position, rotation and zoom while the view pans, zooms or
210    /// rotates. Its stored geometry is then expressed in widget
211    /// pixels; toggling converts between the two spaces, so pinning
212    /// never moves the card visually. Sticky cards are skipped by
213    /// `apply_layout`, and passing one to a seat releases the pin.
214    #[cfg_attr(feature = "serde", serde(default))]
215    pub sticky: bool,
216    /// Explicit chrome composition as an ordered slot list
217    /// (`CardConfig::chrome`).
218    ///
219    /// - `None` (default): derive slots from the legacy flags
220    ///   (`title`/`title_label` -> `Title`, `foldable` -> `Minimize`,
221    ///   `closable` -> `Close`, `resizable` -> `ResizeGrip`,
222    ///   `rotate_grip` -> `RotateGrip`,
223    ///   `drag_handle == Handle` -> `DragHandle`; all `Inline`).
224    /// - `Some(slots)`: overrides the legacy-derived chrome entirely -
225    ///   the list is the complete chrome description. `Satellite`
226    ///   placements are validated at build time; until satellite
227    ///   mounting lands they are skipped (no inline fallback).
228    #[cfg_attr(feature = "serde", serde(default))]
229    pub chrome: Option<Vec<ChromeSlot>>,
230    /// Which renderer draws the chrome (`CardConfig::chrome_style`):
231    /// `Builtin` (default) or `Adwaita` (requires the `adwaita`
232    /// feature; falls back to `Builtin` without it).
233    #[cfg_attr(feature = "serde", serde(default))]
234    pub chrome_style: ChromeStyle,
235    /// Satellite configuration: layout, rotation and collision
236    /// semantics plus the explicit satellite list (`CardSatellites`).
237    /// `None` (default) means no custom satellites; chrome elements
238    /// configured `ChromeSlot::satellite(..)` still mount with the
239    /// `CardSatellites` defaults (sidebar below the card).
240    #[cfg_attr(feature = "serde", serde(default))]
241    pub satellites: Option<CardSatellites>,
242}

289impl CardConfig {
290    /// The fold morph program for a transition direction:
291    /// `unfold_animation` falls back to `fold_animation`, so a single
292    /// setting animates both directions.
293    pub fn effective_fold_animation(&self, to_folded: bool) -> CardFoldAnimation {
294        match to_folded {
295            true => self.fold_animation,
296            false => self.unfold_animation.unwrap_or(self.fold_animation),
297        }
298    }
299
300    /// The choreography timing for a transition direction: the unfold
301    /// side defaults to `Overlap(0.0)` - morph and satellite reveal
302    /// run in lockstep.
303    pub fn effective_fold_timing(&self, to_folded: bool) -> CardFoldTiming {
304        match to_folded {
305            true => self.fold_timing,
306            false => self.unfold_timing.unwrap_or(CardFoldTiming::Overlap(Fraction::ZERO)),
307        }
308    }
309
310    /// The effective ordered chrome slots - either the explicit
311    /// `CardConfig::chrome` list or the list derived from the legacy
312    /// flags (default: `DragHandle`, `Title`, `Minimize`, `Close`,
313    /// `ResizeGrip`, `RotateGrip`, all `Inline`; elements whose flag
314    /// is off are absent).
315    pub fn effective_chrome_slots(&self) -> Vec<ChromeSlot> {
316        if let Some(chrome) = &self.chrome {
317            return chrome.clone();
318        }
319        self.legacy_chrome_slots()
320    }
321
322    /// Slots derived from the legacy boolean flags, in the historical
323    /// header order (leading: handle + title, trailing: minimize +
324    /// close; grips mount onto the content overlay).
325    fn legacy_chrome_slots(&self) -> Vec<ChromeSlot> {
326        let mut slots = Vec::with_capacity(6);
327        if self.drag_handle == DragHandle::Handle {
328            slots.push(ChromeSlot::inline(ChromeElement::DragHandle));
329        }
330        slots.push(ChromeSlot::inline(ChromeElement::Title));
331        if self.foldable {
332            slots.push(ChromeSlot::inline(ChromeElement::Minimize));
333        }
334        if self.closable {
335            slots.push(ChromeSlot::inline(ChromeElement::Close));
336        }
337        if self.resizable {
338            slots.push(ChromeSlot::inline(ChromeElement::ResizeGrip));
339        }
340        if self.rotate_grip {
341            slots.push(ChromeSlot::inline(ChromeElement::RotateGrip));
342        }
343        slots
344    }
345
346    /// The effective satellite set: `ChromeSlot::satellite` entries in
347    /// slot order (elements that cannot be satellites are warned about
348    /// once at mount), then the explicit `CardSatellites::satellites`
349    /// list. Layout, rotation and collision semantics come from
350    /// `satellites`, defaulting to a `CoRotate` sidebar below the card.
351    ///
352    /// Non-satellite placements are never satellites: on `Rectangle`
353    /// they mount in the header strip, on `Circle` in the top-arc
354    /// chrome band (`Card::mount_chrome` owns both mounts).
355    pub fn effective_satellites(&self) -> (CardSatellites, Vec<SatelliteConfig>) {
356        let settings = self.satellites.clone().unwrap_or_default();
357        let mut configs: Vec<SatelliteConfig> = self
358            .effective_chrome_slots()
359            .iter()
360            .filter(|slot| slot.placement == ChromePlacement::Satellite && slot.element.supports_satellite())
361            .map(|slot| SatelliteConfig::chrome(format!("chrome-{}", slot.element.satellite_key().unwrap_or("element")), slot.element))
362            .collect();
363        configs.extend(settings.satellites.iter().cloned());
364        (settings, configs)
365    }
366}
