70    /// Builds a card owned by `board`. Called from `FloatingBoard::add_card`.
71    pub(crate) fn build(id: &CardId, content: &Widget, board: &FloatingBoard, config: &CardConfig) -> Self {
72        let card: Self = glib::Object::builder().build();
73        let imp = card.imp();
74        imp.id.replace(id.clone());
75        imp.board.replace(board.downgrade());
76        imp.max_width.set(config.max_width);
77        imp.max_height.set(config.max_height);
78
79        imp.shape.set(config.shape);
80        imp.content.set(config.content);
81        imp.hole_ratio.set(Fraction::new(
82            config
83                .hole_ratio
84                .unwrap_or(crate::widget::card_config::DEFAULT_HOLE_RATIO)
85                .as_f64()
86                .clamp(0.05, 0.95),
87        ));
88        if config.shape == CardShape::Circle {
89            card.add_css_class(CLASS_CARD_CIRCLE);
90        }
91        if config.shape == CardShape::Semicircle {
92            card.add_css_class(CLASS_CARD_SEMICIRCLE);
93            // A dome's folded pane belongs on the flat edge, not the
94            // center like a circle's.
95            if let Some(holder) = imp.folded_holder.get() {
96                holder.set_valign(gtk4::Align::End);
97            }
98        }
99        if matches!(config.shape, CardShape::Hexagon | CardShape::Octagon | CardShape::Annulus) {
100            card.add_css_class(match config.shape {
101                CardShape::Hexagon => CLASS_CARD_HEXAGON,
102                CardShape::Octagon => CLASS_CARD_OCTAGON,
103                _ => CLASS_CARD_ANNULUS,
104            });
105            // The face paints the card background inside the clip
106            // silhouette (the widget's own CSS background renders
107            // outside it and stays transparent) - inserted beneath the
108            // stage so content and chrome draw on top. On `Annulus`
109            // the face is the ring: the hole stays see-through.
110            let face = DrawingArea::new();
111            face.add_css_class(CLASS_CARD_FACE);
112            face.set_can_target(false);
113            if let Some(stage) = card.first_child() {
114                face.insert_before(&card, Some(&stage));
115            }
116            let _ = imp.face.set(face);
117        }
118        if config.headerless {
119            card.add_css_class(CLASS_HEADERLESS);
120        }
121        if let Some(palette) = &config.palette {
122            imp.palette.replace(install_palette(&card, palette));
123        }
124        // Chrome is composed from `config.effective_chrome_slots()`:
125        // inline elements mount into a header strip prepended to the
126        // card, grips onto the content overlay, satellite placements
127        // are validated and deferred to Phase 2/3.
128        card.mount_chrome(config);
129        if let Some(label) = imp.title_label.get() {
130            label.set_visible(!config.headerless);
131        }
132        // `Clipped` on every non-rectangular card: a silhouette clip
133        // between overlay and holder cuts render and pick at the
134        // inner shape - picks in the cut regions fall back to the
135        // card, keeping the ring a drag affordance. `Cover` needs no
136        // clip here: the card's own silhouette clip is the bound.
137        if config.content == CardContent::Clipped
138            && config.shape != CardShape::Rectangle
139            && let (Some(overlay), Some(holder)) = (imp.content_overlay.get(), imp.content_holder.get())
140        {
141            let clip = ShapeClip::new(config.shape, imp.hole_ratio.get());
142            overlay.set_child(None::<&Widget>);
143            clip.set_child(holder);
144            overlay.set_child(Some(&clip));
145        }
146        if let Some(holder) = imp.content_holder.get() {
147            // Configured as margin on our own holder rather than padding on
148            // the consumer widget: the gap shows the card's background and
149            // does not clobber margins the consumer may set on `content`.
150            // The margin applies to `Inscribed` only - `Fill`/`Clipped`
151            // carry the ring band as their designed inset already, and
152            // `Cover` is edge-to-edge.
153            let padding = if config.content == CardContent::Inscribed {
154                config.content_padding.as_f64() as i32
155            } else {
156                0
157            };
158            // `Fill`/`Clipped`/`Cover` extend the content allocation
159            // over the card's cut corners. GTK descends into children
160            // unconditionally (`contains` is only the leaf check), so
161            // the content would claim corner clicks that belong to
162            // cards underneath - and GTK offers no non-rectangular
163            // pick masks. Marking the holder non-targetable skips the
164            // whole content subtree during picking; picks inside the
165            // silhouette still reach the card through the
166            // silhouette-bound containers. `Inscribed` and `Rectangle`
167            // content stays interactive: its bounds never cover cut
168            // corners. An `Annulus` card's hole passes picks through in
169            // every mode - even `Inscribed` allocations span it - so
170            // its content is display-only regardless.
171            let interactive = config.shape != CardShape::Annulus && (config.content == CardContent::Inscribed || config.shape == CardShape::Rectangle);
172            imp.content_interactive.set(interactive);
173            holder.set_can_target(interactive);
174            holder.set_margin_top(padding);
175            holder.set_margin_bottom(padding);
176            holder.set_margin_start(padding);
177            holder.set_margin_end(padding);
178            holder.append(content);
179        }
180        if let (Some(folded), Some(holder)) = (&config.folded_content, imp.folded_holder.get()) {
181            holder.append(folded);
182        }
183        card
184    }

345    /// The transform stack's stored position on the card layer - the
346    /// `(x, y)` the layer was last asked to place it at (`put`). The
347    /// widget chain is card -> rotation -> scaling -> layer, so the
348    /// scaling widget's parent is the layer and `child_position`
349    /// reports the stored spot. `None` while the widget tree is not
350    /// attached yet. Comparing `placed` with the rendered bounds shows
351    /// whether the layer actually moved the stack where
352    /// `apply_position` put it; comparing `placed + size/2` with
353    /// `position` shows whether the board's model and its placement
354    /// agree.
355    pub fn placed_position(&self) -> Option<Vec2<Pixels>> {
356        let scaling = self.parent()?.parent()?;
357        let layer = scaling.parent()?;
358        let layer = layer.downcast::<CardLayer>().ok()?;
359        let (x, y) = layer.child_position(&scaling);
360        Some(Vec2::new(x, y))
361    }

363    /// Every ancestor of the card up to the root with its widget
364    /// type, its `(0,0)` origin mapped to board coordinates and its
365    /// GTK allocation - the layer-by-layer view that pinpoints which
366    /// level of the card -> rotation -> scaling -> layer chain
367    /// introduces a render-vs-placement offset. Empty while the card
368    /// is not on a board.
369    pub fn ancestor_chain(&self) -> Vec<String> {
370        let Some(board) = self.board() else {
371            return Vec::new();
372        };
373        let mut chain = Vec::new();
374        let mut widget: Option<Widget> = Some(self.clone().upcast());
375        while let Some(current) = widget {
376            let origin = current.compute_point(&board, &gtk4::graphene::Point::new(0.0, 0.0));
377            let bounds = current.compute_bounds(&current);
378            chain.push(format!("{}: origin={:?} alloc={:?}", current.type_().name(), origin, bounds));
379            widget = current.parent();
380        }
381        chain
382    }

447    pub fn set_folded(&self, folded: bool) {
448        let imp = self.imp();
449        if imp.folded.get() == folded {
450            return;
451        }
452        imp.folded.set(folded);
453        self.apply_folded_presentation(folded);
454        // Folding changes the measured size; the center anchor must be kept.
455        if let Some(board) = self.board() {
456            board.recenter_card(&self.id());
457        }
458        debug!(card = %self.id(), folded, "card folded state changed");
459        self.emit_by_name::<()>(CardSignal::Folded.name(), &[]);
460    }

462    /// Applies the folded state's presentation: pane visibility, size
463    /// pins, grip availability and CSS classes. `set_folded` runs this
464    /// on each state change; the fold transition
465    /// (`CardConfig::fold_animation`) replays it at the transition end,
466    /// so the last animated frame hands over to the exact target state.
467    pub(crate) fn apply_folded_presentation(&self, folded: bool) {
468        let imp = self.imp();
469        // A fold morph's measure override ends with the target
470        // presentation - the card returns to child-driven measure
471        // (or the folded pin below).
472        imp.morph_measure.set(None);
473        let collapsed = imp.folded_holder.get().is_some_and(|holder| holder.first_child().is_some());
474        if let Some(holder) = imp.content_holder.get() {
475            if folded && !collapsed {
476                // Pin the card width while folded to the header: without
477                // the content the natural width shrinks and the fold
478                // button would end up outside the shrunken header. A
479                // collapsed card is meant to shrink instead.
480                let (_, natural) = self.measure_size();
481                // A folded card collapses onto its shape's silhouette:
482                // the circle keeps its ring, the dome its 2r x r
483                // proportion - `collapsed_extent` decides per shape.
484                let extent = imp.shape.get().collapsed_extent(natural);
485                self.set_size_request(extent.x, extent.y);
486            }
487            holder.set_visible(!folded);
488        }
489        // Arc and bar chrome belong to the curved card's full
490        // presentation - a folded dome shows its folded holder alone.
491        if let Some(band) = imp.arc_band.get() {
492            band.set_visible(!folded);
493        }
494        if let Some(band) = imp.bar_band.get() {
495            band.set_visible(!folded);
496        }
497        if let Some(band) = imp.top_band.get() {
498            band.set_visible(!folded);
499        }
500        for (widget, _) in imp.arc_items.borrow().iter() {
501            widget.set_visible(!folded);
502        }
503        for (widget, _) in imp.bar_items.borrow().iter() {
504            widget.set_visible(!folded);
505        }
506        for (widget, _) in imp.top_items.borrow().iter() {
507            widget.set_visible(!folded);
508        }
509        // A custom folded widget replaces the header too; restore it on
510        // unfold (the header is always visible while unfolded, also in
511        // headerless mode where CSS slims it).
512        if let Some(header) = imp.header.get()
513            && collapsed
514        {
515            header.set_visible(!folded && imp.header_shown.get());
516        }
517        if let Some(holder) = imp.folded_holder.get() {
518            holder.set_visible(folded && collapsed);
519        }
520        // The folded pane is an overlay child of the stage and never
521        // claims layout space - a collapsed card's size must be pinned
522        // to the folded presentation's natural measure explicitly.
523        if folded && collapsed {
524            // `folded_natural` decides per shape: the curved silhouettes
525            // measure through their shape measure, `Rectangle` through
526            // the folded pane itself.
527            if let Some(natural) = imp.folded_natural() {
528                self.set_size_request(natural.x, natural.y);
529            }
530        } else if !folded {
531            self.set_size_request(-1, -1);
532        }
533        // The grips sit on the content overlay - hide them while folded.
534        if let Some(grip) = imp.resize_grip.get() {
535            grip.set_visible(!folded && imp.resizable.get());
536        }
537        if let Some(grip) = imp.rotate_grip.get() {
538            grip.set_visible(!folded && imp.rotate_grip_enabled.get());
539        }
540        if let Some(button) = imp.fold_button.get() {
541            // Only the builtin `go-*` icon pair toggles - adwaita and
542            // satellite minimize buttons keep their symbolic icon.
543            if button.icon_name().is_some_and(|icon| icon.starts_with("go-")) {
544                button.set_icon_name(if folded { "go-up-symbolic" } else { "go-down-symbolic" });
545            }
546            button.set_tooltip_text(Some(&l10n().message(if folded { "card-unfold" } else { "card-fold" })));
547        }
548        if folded {
549            self.add_css_class(CLASS_FOLDED);
550            if collapsed {
551                self.add_css_class(CLASS_COLLAPSED);
552            }
553        } else {
554            self.remove_css_class(CLASS_FOLDED);
555            self.remove_css_class(CLASS_COLLAPSED);
556        }
557        // Fold-transition handoff: restore steady-state opacity and
558        // targeting on both panes (a running transition animates them).
559        if let Some(pane) = imp.content_overlay.get() {
560            pane.set_opacity(1.0);
561            pane.set_can_target(true);
562        }
563        if let Some(holder) = imp.folded_holder.get() {
564            holder.set_opacity(1.0);
565            holder.set_can_target(true);
566        }
567        if let Some(holder) = imp.content_holder.get() {
568            // Restore the build-time decision - immersive content on
569            // shaped cards (and every `Annulus` content, whose hole
570            // must stay click-through) stays non-targetable.
571            holder.set_can_target(imp.content_interactive.get());
572        }
573    }

575    /// Whether the card is folded to its header.
576    pub fn folded(&self) -> bool {

580    /// The card's natural size in its folded presentation, measured by
581    /// applying the folded state, measuring, and restoring - all
582    /// synchronously within one frame, so nothing renders the
583    /// intermediate state. The fold choreography needs the folded
584    /// footprint *before* the real swap when an `Overlap` morph starts
585    /// mid-close.
586    pub(crate) fn folded_natural_size(&self) -> Vec2<Pixels> {
587        self.apply_folded_presentation(true);
588        let (_, folded_natural) = self.measure_size();
589        self.apply_folded_presentation(false);
590        folded_natural.to_f64().with_unit::<Pixels>()
591    }

644    /// Widget acting as the drag affordance for the given handle mode
645    /// and the card's current folded state. `WholeCardFolded` and
646    /// `WholeCardUnfolded` drag the whole card in their named state and
647    /// fall back to the header strip in the other - an unfolded
648    /// `WholeCardFolded` card keeps its interactive content usable.
649    pub(crate) fn drag_source(&self, drag_handle: DragHandle) -> Widget {
650        let folded = self.imp().folded.get();
651        let whole_card = match drag_handle {
652            DragHandle::WholeCard => true,
653            DragHandle::WholeCardFolded => folded,
654            DragHandle::WholeCardUnfolded => !folded,
655            DragHandle::Header | DragHandle::Handle => false,
656        };
657        if whole_card {
658            return self.clone().upcast();
659        }
660        match drag_handle {
661            DragHandle::Handle => self
662                .imp()
663                .handle_label
664                .get()
665                .cloned()
666                .map(|h| h.upcast())
667                .unwrap_or_else(|| self.clone().upcast()),
668            _ => self.imp().header.get().cloned().unwrap_or_else(|| self.clone().upcast()),
669        }
670    }
