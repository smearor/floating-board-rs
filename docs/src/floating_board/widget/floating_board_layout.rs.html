1//! `FloatingBoardLayout` - the board's layout manager, which also
2//! drives the per-allocation board reconciliation.
3
4use crate::widget::imp::floating_board_layout::FloatingBoardLayoutImpl;
5use gtk4::glib;
6
7glib::wrapper! {
8    /// Layout manager of [`FloatingBoard`](crate::widget::FloatingBoard):
9    /// gives every child the full board area (the board is a surface,
10    /// not a stacking container - it manages exactly one overlay child)
11    /// and runs the board's position reconciliation each allocation
12    /// pass, where a plain `size_allocate` vfunc would never be called
13    /// (GTK4 delegates straight to the layout manager for widgets that
14    /// have one, like `GtkBox`).
15    pub struct FloatingBoardLayout(ObjectSubclass<FloatingBoardLayoutImpl>)
16        @extends gtk4::LayoutManager,
17        @implements gtk4::Orientable;
18}
