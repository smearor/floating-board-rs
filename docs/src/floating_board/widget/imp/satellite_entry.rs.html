10/// Board-owned runtime state of one mounted satellite: the same mini
11/// transform stack a card gets (`ScalingWidget` on the `CardLayer` ->
12/// `RotationWidget` -> satellite widget), so view zoom, card rotation
13/// and card scale apply through the existing machinery.
14pub(crate) struct SatelliteEntry {
15    /// Stable id - keys the resolver's `SatellitePlacement` results.
16    pub id: SatelliteId,
17    /// The satellite's kind - distinguishes chrome elements from
18    /// consumer widgets (needed for the circle card's fold exception:
19    /// its `Minimize` satellite is the only unfold control).
20    pub kind: SatelliteKind,
21    /// Fixed anchor angle for `SatelliteDistribution::Anchored`
22    /// (radial matter, Phase 3).
23    pub anchor: Option<Rotation>,
24    /// The satellite's own widget (chrome button or consumer widget).
25    pub widget: Widget,
26    /// Outer transform: scale wrapper placed on the `gtk4::Fixed` layer.
27    pub scaling: ScalingWidget,
28    /// Inner transform: rotation container holding `widget`.
29    pub rotation_widget: RotationWidget,
30    /// The last applied workspace placement - `None` until the first
31    /// sync. The clamp union footprints visible satellites from it.
32    pub last_geometry: Cell<Option<CardGeometry>>,
33    /// Runtime open flag (`CardSatellites::open` at mount,
34    /// `FloatingBoard::set_satellites_open` later): a closed satellite
35    /// hides like a folded card hides its set. The *animation* of
36    /// opening/closing is Phase-5 matter.
37    pub open: Cell<bool>,
38}
