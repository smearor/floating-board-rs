13/// Board-owned runtime state of one mounted satellite: the same mini
14/// transform stack a card gets (`ScalingWidget` on the `CardLayer` ->
15/// `RotationWidget` -> satellite widget), so view zoom, card rotation
16/// and card scale apply through the existing machinery.
17pub(crate) struct SatelliteEntry {
18    /// Stable id - keys the resolver's `SatellitePlacement` results.
19    pub id: SatelliteId,
20    /// The satellite's kind - distinguishes chrome elements from
21    /// consumer widgets (needed for the circle card's fold exception:
22    /// its `Minimize` satellite is the only unfold control).
23    pub kind: SatelliteKind,
24    /// Fixed anchor angle for `SatelliteDistribution::Anchored`
25    /// (radial matter, Phase 3).
26    pub anchor: Option<Rotation>,
27    /// The satellite's own widget (chrome button or consumer widget).
28    pub widget: Widget,
29    /// Outer transform: scale wrapper placed on the `gtk4::Fixed` layer.
30    pub scaling: ScalingWidget,
31    /// Inner transform: rotation container holding `widget`.
32    pub rotation_widget: RotationWidget,
33    /// The last applied workspace placement - `None` until the first
34    /// sync. The clamp union footprints visible satellites from it.
35    pub last_geometry: Cell<Option<CardGeometry>>,
36    /// Runtime open flag (`CardSatellites::open` at mount,
37    /// `FloatingBoard::set_satellites_open` later): a closed satellite
38    /// hides like a folded card hides its set. The *animation* of
39    /// opening/closing is Phase-5 matter.
40    pub open: Cell<bool>,
41    /// In-flight open/close/move transition (Phase 5). While set, the
42    /// applied placement interpolates `from` (or the collapsed seed)
43    /// toward the live resolved placement (or back into the seed).
44    pub transition: RefCell<Option<SatelliteTransition>>,
45    /// The resolved placement expressed in card-local space at the last
46    /// sync, used to distinguish "card moved" (rigid follow - no tween)
47    /// from "placement retargeted" (footprint/layout change - tween).
48    pub last_local_geometry: Cell<Option<CardGeometry>>,
49}

51/// An in-flight satellite placement transition (open, close or
52/// retarget). Both endpoints resolve live each frame where possible so
53/// card gestures keep carrying an animating satellite.
54pub(crate) struct SatelliteTransition {
55    /// Eased progress driver; staggered starts come from
56    /// `TimedAnimation::start_at` offsets.
57    pub animation: TimedAnimation<Span<f64>>,
58    /// Placement frozen at transition start (mid-flight reversal or a
59    /// retarget). `None` means emerge from the collapsed seed.
60    pub from: Option<CardGeometry>,
61    /// `true`: the end point is the live resolved placement (open /
62    /// retarget). `false`: the end point is the collapsed seed and the
63    /// satellite hides when the animation finishes (close).
64    pub opening: bool,
65}
