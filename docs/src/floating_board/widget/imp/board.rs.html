37/// Widget implementation for [`FloatingBoard`].
38pub struct FloatingBoardImpl {
39    /// Internal overlay stacking the card layer over the content.
40    pub overlay: OnceCell<gtk4::Overlay>,
41    /// Transparent card layer holding the card transform stacks.
42    /// Empty areas have no pickable, so gestures fall through to the content.
43    pub card_layer: OnceCell<CardLayer>,
44    /// All cards keyed by id (sorted, for deterministic snapshots).
45    pub(crate) cards: RefCell<BTreeMap<CardId, CardEntry>>,
46    /// Counter for auto-generated `card-N` ids and insertion order.
47    pub sequence: Cell<SequenceNumber>,
48    /// When `true`, an external consumer manages the surface input region
49    /// and `size_allocate` stops resetting it.
50    pub input_region_managed: Cell<bool>,
51    /// Extent mode of the virtual workspace the cards live on.
52    pub workspace_size: Cell<WorkspaceSize<Pixels>>,
53    /// The configured camera over the workspace; `None` resolves to the
54    /// default view covering the workspace extent (identity for the
55    /// `FollowWidget` default). An explicit `Some` is pinned and
56    /// survives resizes; `reset_view` clears back to `None`.
57    pub view: RefCell<Option<View<Pixels>>>,
58    /// Generation of the running `goto_card` camera flight; a new
59    /// flight or any programmatic/gesture view change bumps it so
60    /// stale ticks stop themselves.
61    pub view_flight: Cell<GenerationCounter>,
62    /// Empty-space gestures drive the view camera (pan, pinch zoom,
63    /// two-finger rotate) while set - default `false`, see
64    /// `FloatingBoard::set_view_gestures`.
65    pub view_gestures: Cell<bool>,
66    /// The corner view-control button box is visible while set -
67    /// default `false`, see `FloatingBoard::set_view_buttons`.
68    pub view_buttons: Cell<bool>,
69    /// Button box overlaid on the bottom-right corner (zoom in, zoom
70    /// out, reset). Created in `constructed`; visibility follows
71    /// `view_buttons`.
72    pub view_controls: OnceCell<gtk4::Box>,
73    /// Last cumulative offset reported by the view pan drag (layer
74    /// pixels since drag begin) - the workspace delta is computed
75    /// between consecutive offsets so the workspace point stays under
76    /// the finger while the view pans.
77    pub view_pan_offset: Cell<Vec2<Pixels>>,
78    /// Last cumulative scale reported by the view zoom gesture.
79    pub view_zoom_scale: Cell<ScaleFactor>,
80    /// Last cumulative angle delta reported by the view rotate gesture.
81    pub view_rotate_delta: Cell<RotationRad>,
82    /// Where pass-to-seat operations anchor seat edges - default
83    /// `SeatAnchor::View` (the screen edge where users sit).
84    pub seat_anchor: Cell<SeatAnchor>,
85}

109#[glib::object_subclass]
110impl ObjectSubclass for FloatingBoardImpl {
111    const NAME: &'static str = "FloatingBoard";
112    type Type = FloatingBoard;
113    type ParentType = gtk4::Box;
114
115    fn class_init(klass: &mut Self::Class) {
116        // `GtkBox` installs a `GtkBoxLayout`, and for widgets with a
117        // layout manager GTK4 calls `gtk_layout_manager_allocate`
118        // directly - a `size_allocate` vfunc would never run. The
119        // board therefore carries its own layout manager, whose
120        // `allocate` drives the per-pass card reconciliation.
121        klass.set_layout_manager_type::<FloatingBoardLayout>();
122    }
123}

125impl ObjectImpl for FloatingBoardImpl {
126    fn constructed(&self) {
127        self.parent_constructed();
128        let board = self.obj();
129        board.set_orientation(Orientation::Vertical);
130        board.add_css_class(CLASS_BOARD);
131
132        let overlay = gtk4::Overlay::new();
133        overlay.set_hexpand(true);
134        overlay.set_vexpand(true);
135
136        let card_layer = CardLayer::new();
137        card_layer.add_css_class(CLASS_CARD_LAYER);
138        // The layer must fill the board: a fixed-style container
139        // otherwise measures at its children bounding box, shrinking
140        // and shifting with card positions (picking and any geometry
141        // derived from its allocation would follow the card cluster).
142        card_layer.set_hexpand(true);
143        card_layer.set_vexpand(true);
144        overlay.add_overlay(&card_layer);
145
146        board.append(&overlay);
147        let _ = self.card_layer.set(card_layer);
148        let _ = self.overlay.set(overlay);
149
150        board.wire_view_gestures();
151        board.wire_view_controls();
152    }
153
154    fn signals() -> &'static [Signal] {
155        static SIGNALS: LazyLock<Vec<Signal>> = LazyLock::new(|| BOARD_SIGNALS.iter().map(|signal| signal.signal()).collect());
156        &SIGNALS
157    }
158}
