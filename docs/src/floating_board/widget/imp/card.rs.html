42/// Widget implementation for [`Card`].
43///
44/// The card is a view: it owns no geometry state. Position/rotation/scale live
45/// exclusively on the `FloatingBoard`; the backlink here only delegates calls.
46pub struct CardImpl {
47    /// Semantic identifier of this card.
48    pub id: RefCell<CardId>,
49    /// Weak backlink to the owning board.
50    pub board: RefCell<WeakRef<FloatingBoard>>,
51    /// The installed palette's generated CSS class + provider, so
52    /// `Card::set_palette` can uninstall a replaced palette.
53    pub palette: RefCell<Option<(String, gtk4::CssProvider)>>,
54    /// Header strip (drag affordance) - built by `chrome::mount_chrome`
55    /// during `Card::build`; `gtk4::Box` for the builtin style,
56    /// `adw::HeaderBar` for adwaita.
57    pub header: OnceCell<gtk4::Widget>,
58    /// Whether the header has chrome children (and may be shown);
59    /// `set_folded` restores to this instead of unconditionally
60    /// revealing an empty strip.
61    pub header_shown: Cell<bool>,
62    /// Overlay stacking the resize/rotate grips over the content.
63    pub content_overlay: OnceCell<Overlay>,
64    /// Container for the consumer's content widget.
65    pub content_holder: OnceCell<gtk4::Box>,
66    /// Resolved title - source of truth for `Card::title` regardless
67    /// of which chrome renderer displays it.
68    pub title_text: RefCell<String>,
69    /// Title label inside the header (builtin style only).
70    pub title_label: OnceCell<Label>,
71    /// Grip icon used in `DragHandle::Handle` mode.
72    pub handle_label: OnceCell<Label>,
73    /// Fold toggle button (visible when `CardConfig::foldable`).
74    pub fold_button: OnceCell<Button>,
75    /// Close button (visible when `CardConfig::closable`).
76    pub close_button: OnceCell<Button>,
77    /// Bottom-right resize grip (`CardConfig::resizable`) - hidden by
78    /// default, shown from `Card::build` and while unfolded.
79    pub resize_grip: OnceCell<DrawingArea>,
80    /// Whether the card offers the resize grip - kept so `set_folded`
81    /// can gate the grip's visibility on the fold state.
82    pub resizable: Cell<bool>,
83    /// Bottom-left rotate grip (`CardConfig::rotate_grip`) - hidden by
84    /// default, shown from `Card::build` and while unfolded.
85    pub rotate_grip: OnceCell<DrawingArea>,
86    /// Whether the card offers the rotate grip - kept so `set_folded`
87    /// can gate the grip's visibility on the fold state.
88    pub rotate_grip_enabled: Cell<bool>,
89    /// Whether the card is currently folded to its header.
90    pub folded: Cell<bool>,
91    /// Container for `CardConfig::folded_content` - visible only while
92    /// folded with a custom folded widget installed.
93    pub folded_holder: OnceCell<gtk4::Box>,
94    /// Upper bound on the reported width (`CardConfig::max_width`).
95    pub max_width: Cell<Option<Pixels>>,
96    /// Upper bound on the reported height (`CardConfig::max_height`).
97    pub max_height: Cell<Option<Pixels>>,
98    /// Fold-morph measure override (`CardFoldAnimation`): while a
99    /// morph runs, `CardShapeLayout` reports this pinned size instead
100    /// of measuring children - a `size_request` cannot shrink a card
101    /// below its content's own minimum, so the footprint tween needs
102    /// the layout-level override. `apply_folded_presentation` clears it.
103    pub morph_measure: Cell<Option<Vec2<Pixels, i32>>>,
104    /// The card's outline (`CardConfig::shape`) - drives the circle
105    /// branches in `measure`/`size_allocate`/`contains`.
106    pub shape: Cell<CardShape>,
107    /// Content allocation mode for `CardShape::Circle`
108    /// (`CardConfig::circle_content`).
109    pub circle_content: Cell<CircleContent>,
110}

143impl ObjectSubclass for CardImpl {
144    const NAME: &'static str = "FloatingBoardCard";
145    type Type = Card;
146    type ParentType = gtk4::Box;

148    fn class_init(klass: &mut Self::Class) {
149        // Shape-aware layout: GTK4 consults the layout manager for
150        // `measure`/`allocate` before the widget vfuncs, and `GtkBox`
151        // always carries one - the circle sizing only works from here.
152        klass.set_layout_manager_type::<CardShapeLayout>();
153    }
154}

156impl ObjectImpl for CardImpl {
157    fn constructed(&self) {
158        self.parent_constructed();
159        let card = self.obj();
160        card.set_orientation(Orientation::Vertical);
161        // No `set_spacing(0)`: it delegates to `gtk_box_layout_*` and
162        // hard-asserts on our custom `CardShapeLayout`; the delegate
163        // inside defaults to spacing 0 anyway.
164        // Clips content that exceeds a `max_width`/`max_height` cap and keeps
165        // it inside the rounded corners (GTK CSS has no `overflow` property).
166        card.set_overflow(Overflow::Hidden);
167        card.add_css_class(CLASS_CARD);
168
169        // The header strip is NOT built here: `chrome::mount_chrome`
170        // composes it during `Card::build` from the config's chrome
171        // slot list (and prepends it before the content overlay).
172        let content_holder = gtk4::Box::new(Orientation::Vertical, 0);
173        content_holder.add_css_class(CLASS_CONTENT);
174        content_holder.set_vexpand(true);
175
176        // The overlay stacks the resize/rotate grips over the
177        // content's corners without claiming layout space.
178        let content_overlay = Overlay::new();
179        content_overlay.set_child(Some(&content_holder));
180
181        let folded_holder = gtk4::Box::new(Orientation::Horizontal, 0);
182        folded_holder.add_css_class(crate::widget::css::CLASS_FOLDED_CONTENT);
183        folded_holder.set_visible(false);
184        folded_holder.set_halign(gtk4::Align::Center);
185        folded_holder.set_valign(gtk4::Align::Center);
186
187        // The stage stacks the folded pane over the content pane:
188        // while folded - or mid-crossfade in a `CardFoldAnimation`
189        // transition - both panes share the same area. The folded pane
190        // is an overlay child, so it never claims layout space; the
191        // folded state's size is pinned explicitly in `set_folded`.
192        let stage = Overlay::new();
193        stage.set_child(Some(&content_overlay));
194        stage.add_overlay(&folded_holder);
195
196        card.append(&stage);
197
198        let _ = self.content_overlay.set(content_overlay);
199        let _ = self.content_holder.set(content_holder);
200        let _ = self.folded_holder.set(folded_holder);
201    }
202
203    fn signals() -> &'static [Signal] {
204        static SIGNALS: LazyLock<Vec<Signal>> = LazyLock::new(|| CARD_SIGNALS.iter().map(|signal| signal.signal()).collect());
205        &SIGNALS
206    }
207}

209/// Width of the ring band around a circle card's inner circle (px).
210/// The band is card chrome: it carries the card's background and stays
211/// a drag affordance; the content never reaches into it.
212const CIRCLE_RING_BAND: f64 = 8.0;
213
214impl WidgetImpl for CardImpl {
215    // `Circle`: only the disk inside the bounding square is pickable -
216    // taps in the corners pass through to whatever is underneath
217    // (through the board layers' `contains` chain, same as the rotated
218    // corners of a rectangle card).
219    fn contains(&self, x: f64, y: f64) -> bool {
220        if self.shape.get() == CardShape::Circle {
221            let (width, height) = (f64::from(self.obj().width()), f64::from(self.obj().height()));
222            return Circle::<()>::new(Vec2::new(width / 2.0, height / 2.0), width.min(height) / 2.0).contains(&Vec2::new(x, y));
223        }
224        self.parent_contains(x, y)
225    }
226}

228impl CardImpl {
229    /// `Circle` measure: the card reports its bounding square - a
230    /// diameter derived from the visible content child's natural size
231    /// via `CircleContent::diameter_for` (the `folded_holder` while it
232    /// carries a collapsed widget, else the `content_overlay`). Called
233    /// from `CardShapeLayout`.
234    pub(crate) fn circle_measure(&self) -> (i32, i32, i32, i32) {
235        let content = self
236            .folded_holder
237            .get()
238            .filter(|holder| holder.is_visible() && holder.first_child().is_some())
239            .map(|holder| holder.clone().upcast::<Widget>())
240            .or_else(|| self.content_overlay.get().map(|overlay| overlay.clone().upcast()));
241        let Some(content) = content else {
242            return (0, 0, -1, -1);
243        };
244        let (_, natural_width, _, _) = content.measure(Orientation::Horizontal, -1);
245        let (_, natural_height, _, _) = content.measure(Orientation::Vertical, -1);
246        // A disc does not flex: minimum == natural keeps the bounding
247        // square stable (the parent layer allocates at the minimum
248        // width - a smaller value would render a shrunken circle whose
249        // satellites still orbit the natural diameter).
250        let diameter = self
251            .circle_content
252            .get()
253            .diameter_for(f64::from(natural_width.max(natural_height)), CIRCLE_RING_BAND);
254        (diameter.ceil() as i32, diameter.ceil() as i32, -1, -1)
255    }
256
257    /// `Circle` allocate: every visible child gets the centered content
258    /// square (`CircleContent::content_square_for` of the allocated
259    /// diameter); the margin around it renders as the ring. Called from
260    /// `CardShapeLayout`.
261    pub(crate) fn circle_allocate(&self, width: i32, height: i32, baseline: i32) {
262        let diameter = f64::from(width.min(height));
263        let side = self.circle_content.get().content_square_for(diameter, CIRCLE_RING_BAND);
264        let transform = Transform::new().translate(&Point::new(((f64::from(width) - side) / 2.0) as f32, ((f64::from(height) - side) / 2.0) as f32));
265        let side = side.ceil() as i32;
266        let mut child = self.obj().first_child();
267        while let Some(widget) = child {
268            if widget.is_visible() {
269                widget.allocate(side, side, baseline, Some(transform.clone()));
270            }
271            child = widget.next_sibling();
272        }
273    }
274}
