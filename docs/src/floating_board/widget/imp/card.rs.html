244impl ObjectImpl for CardImpl {
245    fn constructed(&self) {
246        self.parent_constructed();
247        let card = self.obj();
248        card.set_orientation(Orientation::Vertical);
249        // No `set_spacing(0)`: it delegates to `gtk_box_layout_*` and
250        // hard-asserts on our custom `CardShapeLayout`; the delegate
251        // inside defaults to spacing 0 anyway.
252        // Clips content that exceeds a `max_width`/`max_height` cap and keeps
253        // it inside the rounded corners (GTK CSS has no `overflow` property).
254        card.set_overflow(Overflow::Hidden);
255        card.add_css_class(CLASS_CARD);
256
257        // The header strip is NOT built here: `chrome::mount_chrome`
258        // composes it during `Card::build` from the config's chrome
259        // slot list (and prepends it before the content overlay).
260        let content_holder = SilhouetteBox::new(Orientation::Vertical, 0);
261        content_holder.add_css_class(CLASS_CONTENT);
262        content_holder.set_vexpand(true);
263
264        // The overlay stacks the resize/rotate grips over the
265        // content's corners without claiming layout space - the
266        // silhouette-bound `SilhouetteStage` lets picks in the cut
267        // regions outside the shape fall through to cards below.
268        let content_overlay = crate::widget::silhouette_stage::SilhouetteStage::new();
269        content_overlay.set_child(Some(&content_holder));
270
271        let folded_holder = SilhouetteBox::new(Orientation::Horizontal, 0);
272        folded_holder.add_css_class(crate::widget::css::CLASS_FOLDED_CONTENT);
273        folded_holder.set_visible(false);
274        folded_holder.set_halign(Align::Center);
275        folded_holder.set_valign(Align::Center);
276
277        // The stage stacks the folded pane over the content pane:
278        // while folded - or mid-crossfade in a `CardFoldAnimation`
279        // transition - both panes share the same area. The folded pane
280        // is an overlay child, so it never claims layout space; the
281        // folded state's size is pinned explicitly in `set_folded`.
282        let stage = SilhouetteStage::new();
283        stage.set_child(Some(&content_overlay));
284        stage.add_overlay(&folded_holder);
285
286        card.append(&stage);
287
288        let _ = self.content_overlay.set(content_overlay);
289        let _ = self.content_holder.set(content_holder.upcast());
290        let _ = self.folded_holder.set(folded_holder.upcast());
291    }

299/// Width of the ring band around a circle card's inner circle (px).
300/// The band is card chrome: it carries the card's background and stays
301/// a drag affordance; the content never reaches into it.
302const CIRCLE_RING_BAND: f64 = 8.0;
303
304/// `tan(22.5°) = sqrt(2) - 1` - a flat-top octagon's flat edge is
305/// `2 * a * tan(22.5°)` wide for inradius `a`, and its diagonal
306/// edges widen a cross-section at a 1:1 gain per unit height.
307const TAN_22_5: f64 = 0.414_213_562_373_095_03;
308/// Thickness of the arc-chrome band on `CardShape::Circle` (px): the
309/// chrome items' centers ride the band's mid radius.
310pub(crate) const ARC_CHROME_BAND: f64 = 36.0;
311impl WidgetImpl for CardImpl {
312    // `Circle`: only the disk inside the bounding square is pickable -
313    // taps in the corners pass through to whatever is underneath
314    // (through the board layers' `contains` chain, same as the rotated
315    // corners of a rectangle card).
316    fn contains(&self, x: f64, y: f64) -> bool {
317        let bounds = Rect::<()>::from_size(self.obj().allocated_size().to_f64());
318        // `Rectangle` keeps the rectangular parent test - every other
319        // shape answers through its inscribed `CardOutline`.
320        self.shape
321            .get()
322            .contains(&bounds, self.hole_ratio.get(), &Vec2::new(x, y))
323            .unwrap_or_else(|| self.parent_contains(x, y))
324    }
325
326    /// Silhouette clip: the card's snapshot is clipped to the
327    /// shape's path before children render - the single bound for
328    /// `Cover` content on every shape (for `Circle`/`Semicircle`
329    /// the path coincides with the CSS rounded background).
330    fn snapshot(&self, snapshot: &Snapshot) {
331        if let Some(path) = self.silhouette_path() {
332            snapshot.push_fill(&path, FillRule::Winding);
333            self.parent_snapshot(snapshot);
334            snapshot.pop();
335        } else {
336            self.parent_snapshot(snapshot);
337        }
338    }
339}

341/// Traces a closed polygon path through `vertices` on `builder` -
342/// shared by the card's silhouette clip for `Hexagon`/`Octagon`.
343fn polygon_path(builder: &PathBuilder, vertices: &[Vec2<()>]) {
344    builder.move_to(vertices[0].x as f32, vertices[0].y as f32);
345    for vertex in &vertices[1..] {
346        builder.line_to(vertex.x as f32, vertex.y as f32);
347    }
348    builder.close();
349}

351impl CardImpl {
352    /// The silhouette path of the card, relative to its bounding
353    /// box: the inscribed circle for `Circle`, the dome (a circle
354    /// centered on the flat bottom edge - inside the widget bounds
355    /// only the dome remains) for `Semicircle`, the flat-top
356    /// polygon for `Hexagon`/`Octagon`. `None` for `Rectangle`.
357    fn silhouette_path(&self) -> Option<Path> {
358        let bounds = Rect::<()>::from_size(self.obj().allocated_size().to_f64());
359        let builder = PathBuilder::new();
360        match self.shape.get().outline(&bounds, self.hole_ratio.get()) {
361            // The dome adds its virtual circle - the bounds box cuts
362            // it back to the dome's upper half.
363            Some(CardOutline::Circle(circle)) => {
364                builder.add_circle(&Point::new(circle.center.x as f32, circle.center.y as f32), circle.radius as f32);
365            }
366            Some(CardOutline::Semicircle(dome)) => {
367                builder.add_circle(&Point::new(dome.center.x as f32, dome.center.y as f32), dome.radius as f32);
368            }
369            Some(CardOutline::Hexagon(hexagon)) => {
370                polygon_path(&builder, &hexagon.vertices());
371            }
372            Some(CardOutline::Octagon(octagon)) => {
373                polygon_path(&builder, &octagon.vertices());
374            }
375            // The ring: the outer circle plus the hole circle added
376            // reversed - under the winding fill rule the hole area
377            // stays outside the clip, so it renders transparent.
378            Some(CardOutline::Annulus(annulus)) => {
379                let center = Point::new(annulus.center.x as f32, annulus.center.y as f32);
380                builder.add_circle(&center, annulus.outer as f32);
381                let hole = PathBuilder::new();
382                hole.add_circle(&center, annulus.inner as f32);
383                builder.add_reverse_path(&hole.to_path());
384            }
385            None => return None,
386        }
387        Some(builder.to_path())
388    }

390    /// `Circle` measure: the card reports its bounding square - a
391    /// diameter derived from the visible content child's natural size
392    /// via `CircleContent::diameter_for` (the `folded_holder` while it
393    /// carries a collapsed widget, else the `content_overlay`). Called
394    /// from `CardShapeLayout`.
395    /// The widget whose natural size drives the shape measure: the
396    /// `folded_holder` while it carries a collapsed widget, else the
397    /// `content_overlay`.
398    fn visible_content_widget(&self) -> Option<Widget> {
399        self.folded_holder
400            .get()
401            .filter(|holder| holder.is_visible() && holder.first_child().is_some())
402            .map(|holder| holder.clone().upcast::<Widget>())
403            .or_else(|| self.content_overlay.get().map(|overlay| overlay.clone().upcast()))
404    }

406    /// The natural size the folded card pins to when a custom folded
407    /// widget is installed: the shape measure for the curved shapes
408    /// (their silhouette decides the pin), the `folded_holder`'s own
409    /// measure for `Rectangle`. `None` without a folded widget.
410    pub(crate) fn folded_natural(&self) -> Option<Size<i32>> {
411        match self.shape.get() {
412            CardShape::Circle | CardShape::Annulus => {
413                let (_, natural_width, _, natural_height) = self.circle_measure();
414                Some(Size::new(natural_width, natural_height))
415            }
416            CardShape::Semicircle => {
417                let (_, natural_width, ..) = self.semicircle_measure(Orientation::Horizontal);
418                let (_, natural_height, ..) = self.semicircle_measure(Orientation::Vertical);
419                Some(Size::new(natural_width, natural_height))
420            }
421            CardShape::Hexagon => {
422                let (_, natural_width, ..) = self.hexagon_measure(Orientation::Horizontal);
423                let (_, natural_height, ..) = self.hexagon_measure(Orientation::Vertical);
424                Some(Size::new(natural_width, natural_height))
425            }
426            CardShape::Octagon => {
427                let (_, natural_width, ..) = self.octagon_measure(Orientation::Horizontal);
428                let (_, natural_height, ..) = self.octagon_measure(Orientation::Vertical);
429                Some(Size::new(natural_width, natural_height))
430            }
431            CardShape::Rectangle => self.folded_holder.get().map(|holder| {
432                let (_, natural_width, _, _) = holder.measure(Orientation::Horizontal, -1);
433                let (_, natural_height, _, _) = holder.measure(Orientation::Vertical, -1);
434                Size::new(natural_width, natural_height)
435            }),
436        }
437    }

439    pub(crate) fn circle_measure(&self) -> (i32, i32, i32, i32) {
440        let Some(content) = self.visible_content_widget() else {
441            return (0, 0, -1, -1);
442        };
443        let (_, natural_width, _, _) = content.measure(Orientation::Horizontal, -1);
444        let (_, natural_height, _, _) = content.measure(Orientation::Vertical, -1);
445        // A disc does not flex: minimum == natural keeps the bounding
446        // square stable (the parent layer allocates at the minimum
447        // width - a smaller value would render a shrunken circle whose
448        // satellites still orbit the natural diameter).
449        let diameter = self
450            .content
451            .get()
452            .diameter_for(Distance::new(f64::from(natural_width.max(natural_height))), Distance::new(CIRCLE_RING_BAND));
453        (diameter.as_f64().ceil() as i32, diameter.as_f64().ceil() as i32, -1, -1)
454    }

456    /// `Circle` allocate: every visible child gets the centered content
457    /// square (`CircleContent::content_square_for` of the allocated
458    /// diameter); the margin around it renders as the ring. Arc-chrome
459    /// items instead keep their natural size and sit upright on the
460    /// top-quarter band's mid radius (`arc_angle`). Called from
461    /// `CardShapeLayout`.
462    pub(crate) fn circle_allocate(&self, size: Size<i32>, baseline: i32) {
463        let diameter = f64::from(size.x.min(size.y));
464        let side = self
465            .content
466            .get()
467            .content_square_for(Distance::new(diameter), Distance::new(CIRCLE_RING_BAND))
468            .as_f64();
469        let transform = Transform::new().translate(&Point::new(((f64::from(size.x) - side) / 2.0) as f32, ((f64::from(size.y) - side) / 2.0) as f32));
470        let side = side.ceil() as i32;
471        let arc_items = self.arc_items.borrow();
472        let arc_band = self.arc_band.get();
473        let angles = ChromeSlots::from(arc_items.as_slice()).arc_angles(self.arc_span.get() / 2.0, self.arc_bottom_span.get() / 2.0);
474        // The chrome line: just inside the rim on a disc, on the
475        // band's mid radius on a ring.
476        let bounds = Rect::<()>::from_size(Vec2::new(f64::from(size.x), f64::from(size.y)));
477        let arc_radius = self.shape.get().arc_chrome_radius(&bounds, self.hole_ratio.get(), ARC_CHROME_BAND);
478        let mut child = self.obj().first_child();
479        while let Some(widget) = child {
480            if widget.is_visible() {
481                if arc_band.is_some_and(|band| band.upcast_ref::<Widget>() == &widget)
482                    || self.face.get().is_some_and(|face| face.upcast_ref::<Widget>() == &widget)
483                {
484                    // The painted band covers the whole disc square -
485                    // its draw func clips itself to the arc sectors.
486                    // The face covers the whole bounding box - the
487                    // silhouette clip shapes it (the ring band on an
488                    // `Annulus` card).
489                    widget.allocate(size.x, size.y, baseline, None);
490                } else if let Some(index) = arc_items.iter().position(|(item, _)| *item == widget) {
491                    let (_, natural) = widget.measure_size();
492                    let up = Vec2::<Pixels>::new(0.0, -1.0);
493                    let center = Vec2::<Pixels>::new(f64::from(size.x) / 2.0, f64::from(size.y) / 2.0) + up.rotated(angles[index]) * arc_radius;
494                    let origin = Rect::<Pixels>::centered(center, Vec2::new(f64::from(natural.x), f64::from(natural.y))).min;
495                    let arc_transform = Transform::new().translate(&Point::new(origin.x as f32, origin.y as f32));
496                    widget.allocate(natural.x, natural.y, baseline, Some(arc_transform));
497                } else {
498                    widget.allocate(side, side, baseline, Some(transform.clone()));
499                }
500            }
501            child = widget.next_sibling();
502        }
503    }

505    /// `Semicircle` measure: the card reports its bounding dome - a
506    /// `2r × r` box where `r` derives from the visible content
507    /// child's natural size via `CircleContent::dome_radius_for` (the
508    /// `folded_holder` while it carries a collapsed widget, else the
509    /// `content_overlay`). The reported extent is `2r` horizontally
510    /// and `r` vertically. Called from `CardShapeLayout`.
511    pub(crate) fn semicircle_measure(&self, orientation: Orientation) -> (i32, i32, i32, i32) {
512        let Some(content) = self.visible_content_widget() else {
513            return (0, 0, -1, -1);
514        };
515        let (_, natural_width, _, _) = content.measure(Orientation::Horizontal, -1);
516        let (_, natural_height, _, _) = content.measure(Orientation::Vertical, -1);
517        // A dome does not flex: minimum == natural keeps the bounding
518        // box stable (the parent layer allocates at the minimum
519        // width).
520        // The measure must reserve the same chrome insets the
521        // allocation applies - otherwise the content extent shrinks
522        // below the child's natural size and GTK clips it.
523        let insets = self.dome_insets();
524        let radius = self
525            .content
526            .get()
527            .dome_radius_for(Vec2::new(f64::from(natural_width), f64::from(natural_height)), insets);
528        let extent = match orientation {
529            Orientation::Horizontal => radius * 2.0,
530            _ => radius,
531        };
532        (extent.as_f64().ceil() as i32, extent.as_f64().ceil() as i32, -1, -1)
533    }

535    /// The dome's chrome insets `(ring, bottom_inset)`: `ring`
536    /// reserves the arc band's width inside the rim, `bottom_inset`
537    /// lifts the content onto the straight bar's top. While folded
538    /// both bands are hidden and reserve nothing.
539    fn dome_insets(&self) -> DomeInsets {
540        if self.folded.get() {
541            return DomeInsets::new(Distance::new(CIRCLE_RING_BAND), Distance::ZERO);
542        }
543        let ring = Distance::new(CIRCLE_RING_BAND + if self.arc_items.borrow().is_empty() { 0.0 } else { ARC_CHROME_BAND });
544        let bottom_inset = Distance::new(if self.bar_items.borrow().is_empty() { 0.0 } else { ARC_CHROME_BAND });
545        DomeInsets::new(ring, bottom_inset)
546    }

548    /// `Semicircle` allocate: the dome's virtual circle centers at
549    /// the bounding box's bottom middle with radius `min(w/2, h)`.
550    /// Every visible child gets the bottom-aligned content extent
551    /// (`CircleContent::dome_content_for`); arc items sit upright on
552    /// the top band's mid radius and bar items upright on the bottom
553    /// bar's mid line (`bar_offsets`). Called from `CardShapeLayout`.
554    pub(crate) fn semicircle_allocate(&self, size: Size<i32>, baseline: i32) {
555        let (width, height) = (f64::from(size.x), f64::from(size.y));
556        let radius = Semicircle::<()>::inscribed_in(&Rect::<()>::from_size(Vec2::new(width, height))).radius;
557        let arc_items = self.arc_items.borrow();
558        let bar_items = self.bar_items.borrow();
559        // Each painted band pushes the content inward: the arc band
560        // grows the ring (the band rides the dome's rim), the straight
561        // bar lifts the content's bottom edge onto the bar's top.
562        // `dome_insets` mirrors `semicircle_measure` - folded cards
563        // keep their full dome for the folded pane.
564        let insets = self.dome_insets();
565        let content = self.content.get().dome_content_for(Distance::new(radius), insets);
566        let (content_width, content_height) = (content.x, content.y);
567        let transform = Transform::new().translate(&Point::new(
568            (width / 2.0 - content_width / 2.0) as f32,
569            (height - content_height - insets.bottom_inset.as_f64()) as f32,
570        ));
571        let (content_width, content_height) = (content_width.ceil() as i32, content_height.ceil() as i32);
572        // A dome has no bottom arc - `arc_items` holds top-zone slots
573        // only; the bottom span is unused.
574        let half_span = self.arc_span.get() / 2.0;
575        let angles = ChromeSlots::from(arc_items.as_slice()).arc_angles(half_span, half_span);
576        // The bar's safe half-extent at its mid line - item centers
577        // are additionally clamped so an item's edge stays inside the
578        // dome's chord.
579        let bar_half_width = (radius.powi(2) - (ARC_CHROME_BAND / 2.0).powi(2)).max(0.0).sqrt();
580        let bar_offsets = ChromeSlots::from(bar_items.as_slice()).bar_offsets(bar_half_width);
581        let arc_radius = radius - ARC_CHROME_BAND / 2.0;
582        let mut child = self.obj().first_child();
583        while let Some(widget) = child {
584            if widget.is_visible() {
585                if self.arc_band.get().is_some_and(|band| band.upcast_ref::<Widget>() == &widget)
586                    || self.bar_band.get().is_some_and(|band| band.upcast_ref::<Widget>() == &widget)
587                {
588                    // The painted bands cover the whole dome box -
589                    // their draw funcs clip themselves to the shape.
590                    widget.allocate(size.x, size.y, baseline, None);
591                } else if let Some(index) = arc_items.iter().position(|(item, _)| *item == widget) {
592                    let (_, natural) = widget.measure_size();
593                    let up = Vec2::<Pixels>::new(0.0, -1.0);
594                    let center = Vec2::<Pixels>::new(width / 2.0, height) + up.rotated(angles[index]) * arc_radius;
595                    let origin = Rect::<Pixels>::centered(center, Vec2::new(f64::from(natural.x), f64::from(natural.y))).min;
596                    let arc_transform = Transform::new().translate(&Point::new(origin.x as f32, origin.y as f32));
597                    widget.allocate(natural.x, natural.y, baseline, Some(arc_transform));
598                } else if let Some(index) = bar_items.iter().position(|(item, _)| *item == widget) {
599                    let (_, natural) = widget.measure_size();
600                    let max_offset = (bar_half_width - f64::from(natural.x) / 2.0).max(0.0);
601                    let center = Vec2::<Pixels>::new(width / 2.0 + bar_offsets[index].clamp(-max_offset, max_offset), height - ARC_CHROME_BAND / 2.0);
602                    let origin = Rect::<Pixels>::centered(center, Vec2::new(f64::from(natural.x), f64::from(natural.y))).min;
603                    let bar_transform = Transform::new().translate(&Point::new(origin.x as f32, origin.y as f32));
604                    widget.allocate(natural.x, natural.y, baseline, Some(bar_transform));
605                } else {
606                    widget.allocate(content_width, content_height, baseline, Some(transform.clone()));
607                }
608            }
609            child = widget.next_sibling();
610        }
611    }

636    /// `Hexagon` measure: the card reports its bounding box `2R x
637    /// sqrt(3)*R` where `R` derives from the visible content child's
638    /// natural size via `hexagon_radius_for` (the `folded_holder`
639    /// while it carries a collapsed widget, else the
640    /// `content_overlay`). Called from `CardShapeLayout`.
641    pub(crate) fn hexagon_measure(&self, orientation: Orientation) -> (i32, i32, i32, i32) {
642        let Some(content) = self.visible_content_widget() else {
643            return (0, 0, -1, -1);
644        };
645        let (_, natural_width, _, _) = content.measure(Orientation::Horizontal, -1);
646        let (_, natural_height, _, _) = content.measure(Orientation::Vertical, -1);
647        // Like the disc the hexagon does not flex: minimum == natural.
648        // The bottom inset mirrors `hexagon_allocate` so the reported
649        // extent never undersizes the content (same invariant
650        // `dome_insets` maintains for the dome).
651        let bar_inset = self.hexagon_bottom_inset() + self.hexagon_top_inset();
652        let radius = match self.content.get() {
653            CardContent::Inscribed => hexagon_radius_for(f64::from(natural_width.max(natural_height)), CIRCLE_RING_BAND, bar_inset),
654            // `Fill`/`Clipped`: the rim-shrunk hexagon's bounding box
655            // holds the content - the bars overlay it, so the bar
656            // inset does not apply. `Cover` drops the ring too.
657            CardContent::Fill | CardContent::Clipped => hexagon_fill_radius_for(f64::from(natural_width), f64::from(natural_height), CIRCLE_RING_BAND),
658            CardContent::Cover => hexagon_fill_radius_for(f64::from(natural_width), f64::from(natural_height), 0.0),
659        };
660        let bounding = Hexagon::<()>::bounding_size_for_circumradius(radius, PolygonOrientation::FlatTop);
661        let extent = match orientation {
662            Orientation::Horizontal => bounding.x,
663            _ => bounding.y,
664        };
665        (extent.ceil() as i32, extent.ceil() as i32, -1, -1)
666    }

668    /// `Hexagon` allocate: every visible child gets the centered
669    /// inscribed square (`hexagon_content_for` of the allocated
670    /// radius); bar items keep their natural size and sit upright on
671    /// the straight bar's mid line along the flat bottom edge.
672    /// Called from `CardShapeLayout`.
673    pub(crate) fn hexagon_allocate(&self, size: Size<i32>, baseline: i32) {
674        let (width, height) = (f64::from(size.x), f64::from(size.y));
675        let radius = Hexagon::<()>::circumradius_bounded_by(Vec2::new(width, height), PolygonOrientation::FlatTop);
676        let bar_items = self.bar_items.borrow();
677        let top_items = self.top_items.borrow();
678        let bar_inset = self.hexagon_bottom_inset() + self.hexagon_top_inset();
679        // `Inscribed` gets the centered inscribed square,
680        // `Fill`/`Clipped` the rim-shrunk hexagon's bounding box -
681        // the content draws its own hexagon silhouette.
682        let (content_width, content_height) = match self.content.get() {
683            CardContent::Inscribed => {
684                let side = hexagon_content_for(radius, CIRCLE_RING_BAND, bar_inset);
685                (side, side)
686            }
687            CardContent::Fill | CardContent::Clipped => hexagon_fill_extent(radius, CIRCLE_RING_BAND),
688            CardContent::Cover => hexagon_fill_extent(radius, 0.0),
689        };
690        let transform = Transform::new().translate(&Point::new(((width - content_width) / 2.0) as f32, ((height - content_height) / 2.0) as f32));
691        let (content_width, content_height) = (content_width.ceil() as i32, content_height.ceil() as i32);
692        // The bars' safe half-extent at their mid line: the hexagon's
693        // cross-section there - `R/2` (the flat edge's half width)
694        // grown by the slant gain `BAND * w / (4h)` over half the band.
695        let bar_half_width = radius / 2.0 + ARC_CHROME_BAND * width / (4.0 * height);
696        let bar_offsets = ChromeSlots::from(bar_items.as_slice()).bar_offsets(bar_half_width);
697        let top_offsets = ChromeSlots::from(top_items.as_slice()).top_offsets(bar_half_width);
698        let mut child = self.obj().first_child();
699        while let Some(widget) = child {
700            if widget.is_visible() {
701                if self.bar_band.get().is_some_and(|band| band.upcast_ref::<Widget>() == &widget)
702                    || self.top_band.get().is_some_and(|band| band.upcast_ref::<Widget>() == &widget)
703                    || self.face.get().is_some_and(|face| face.upcast_ref::<Widget>() == &widget)
704                {
705                    // The face covers the whole bounding box - the
706                    // silhouette clip shapes it.
707                    widget.allocate(size.x, size.y, baseline, None);
708                } else if let Some(index) = bar_items.iter().position(|(item, _)| *item == widget) {
709                    let (_, natural) = widget.measure_size();
710                    let max_offset = (bar_half_width - f64::from(natural.x) / 2.0).max(0.0);
711                    let center = Vec2::<Pixels>::new(width / 2.0 + bar_offsets[index].clamp(-max_offset, max_offset), height - ARC_CHROME_BAND / 2.0);
712                    let origin = Rect::<Pixels>::centered(center, Vec2::new(f64::from(natural.x), f64::from(natural.y))).min;
713                    let bar_transform = Transform::new().translate(&Point::new(origin.x as f32, origin.y as f32));
714                    widget.allocate(natural.x, natural.y, baseline, Some(bar_transform));
715                } else if let Some(index) = top_items.iter().position(|(item, _)| *item == widget) {
716                    let (_, natural) = widget.measure_size();
717                    let max_offset = (bar_half_width - f64::from(natural.x) / 2.0).max(0.0);
718                    let center = Vec2::<Pixels>::new(width / 2.0 + top_offsets[index].clamp(-max_offset, max_offset), ARC_CHROME_BAND / 2.0);
719                    let origin = Rect::<Pixels>::centered(center, Vec2::new(f64::from(natural.x), f64::from(natural.y))).min;
720                    let top_transform = Transform::new().translate(&Point::new(origin.x as f32, origin.y as f32));
721                    widget.allocate(natural.x, natural.y, baseline, Some(top_transform));
722                } else {
723                    widget.allocate(content_width, content_height, baseline, Some(transform.clone()));
724                }
725            }
726            child = widget.next_sibling();
727        }
728    }

762        let (_, natural_width, _, _) = content.measure(Orientation::Horizontal, -1);
763        let (_, natural_height, _, _) = content.measure(Orientation::Vertical, -1);

766        // extent never undersizes the content.

769            CardContent::Inscribed => octagon_inradius_for(f64::from(natural_width.max(natural_height)), CIRCLE_RING_BAND, bar_inset),

773            CardContent::Fill | CardContent::Clipped => octagon_fill_inradius_for(f64::from(natural_width.max(natural_height)), CIRCLE_RING_BAND),

782    /// `Octagon` allocate: every visible child gets the centered
783    /// inscribed square (`octagon_content_for` of the allocated
784    /// inradius); bar items keep their natural size and sit upright
785    /// on the straight bar's mid line along a flat edge. Called from
786    /// `CardShapeLayout`.
787    pub(crate) fn octagon_allocate(&self, size: Size<i32>, baseline: i32) {
788        let (width, height) = (f64::from(size.x), f64::from(size.y));
789        let inradius = Octagon::inscribed_in(&Rect::<()>::from_size(Vec2::new(width, height)), PolygonOrientation::FlatTop).inradius();
790        let bar_items = self.bar_items.borrow();
791        let top_items = self.top_items.borrow();
792        let bar_inset = self.octagon_bottom_inset() + self.octagon_top_inset();
793        // `Inscribed` gets the centered inscribed square,
794        // `Fill`/`Clipped` the rim-shrunk octagon's bounding box -
795        // the content draws its own octagon silhouette.
796        let side = match self.content.get() {
797            CardContent::Inscribed => octagon_content_for(inradius, CIRCLE_RING_BAND, bar_inset),
798            CardContent::Fill | CardContent::Clipped => octagon_fill_extent(inradius, CIRCLE_RING_BAND),
799            CardContent::Cover => octagon_fill_extent(inradius, 0.0),
800        };
801        let transform = Transform::new().translate(&Point::new(((width - side) / 2.0) as f32, ((height - side) / 2.0) as f32));
802        let side = side.ceil() as i32;
803        // The bars' safe half-extent at their mid line: the octagon's
804        // cross-section there - the flat edge's half width
805        // `a * tan(22.5°)` grown by the diagonal's 1:1 gain `BAND/2`
806        // over half the band.
807        let bar_half_width = width / 2.0 * TAN_22_5 + ARC_CHROME_BAND / 2.0;
808        let bar_offsets = ChromeSlots::from(bar_items.as_slice()).bar_offsets(bar_half_width);
809        let top_offsets = ChromeSlots::from(top_items.as_slice()).top_offsets(bar_half_width);
810        let mut child = self.obj().first_child();
811        while let Some(widget) = child {
812            if widget.is_visible() {
813                if self.bar_band.get().is_some_and(|band| band.upcast_ref::<Widget>() == &widget)
814                    || self.top_band.get().is_some_and(|band| band.upcast_ref::<Widget>() == &widget)
815                    || self.face.get().is_some_and(|face| face.upcast_ref::<Widget>() == &widget)
816                {
817                    // The face covers the whole bounding box - the
818                    // silhouette clip shapes it.
819                    widget.allocate(size.x, size.y, baseline, None);
820                } else if let Some(index) = bar_items.iter().position(|(item, _)| *item == widget) {
821                    let (_, natural) = widget.measure_size();
822                    let max_offset = (bar_half_width - f64::from(natural.x) / 2.0).max(0.0);
823                    let center = Vec2::<Pixels>::new(width / 2.0 + bar_offsets[index].clamp(-max_offset, max_offset), height - ARC_CHROME_BAND / 2.0);
824                    let origin = Rect::<Pixels>::centered(center, Vec2::new(f64::from(natural.x), f64::from(natural.y))).min;
825                    let bar_transform = Transform::new().translate(&Point::new(origin.x as f32, origin.y as f32));
826                    widget.allocate(natural.x, natural.y, baseline, Some(bar_transform));
827                } else if let Some(index) = top_items.iter().position(|(item, _)| *item == widget) {
828                    let (_, natural) = widget.measure_size();
829                    let max_offset = (bar_half_width - f64::from(natural.x) / 2.0).max(0.0);
830                    let center = Vec2::<Pixels>::new(width / 2.0 + top_offsets[index].clamp(-max_offset, max_offset), ARC_CHROME_BAND / 2.0);
831                    let origin = Rect::<Pixels>::centered(center, Vec2::new(f64::from(natural.x), f64::from(natural.y))).min;
832                    let top_transform = Transform::new().translate(&Point::new(origin.x as f32, origin.y as f32));
833                    widget.allocate(natural.x, natural.y, baseline, Some(top_transform));
834                } else {
835                    widget.allocate(side, side, baseline, Some(transform.clone()));
836                }
837            }
838            child = widget.next_sibling();
839        }
840    }
