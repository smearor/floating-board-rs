42/// Widget implementation for [`Card`].
43///
44/// The card is a view: it owns no geometry state. Position/rotation/scale live
45/// exclusively on the `FloatingBoard`; the backlink here only delegates calls.
46pub struct CardImpl {
47    /// Semantic identifier of this card.
48    pub id: RefCell<CardId>,
49    /// Weak backlink to the owning board.
50    pub board: RefCell<WeakRef<FloatingBoard>>,
51    /// Header strip (drag affordance), built in `constructed`.
52    pub header: OnceCell<gtk4::Box>,
53    /// Container for the consumer's content widget.
54    pub content_holder: OnceCell<gtk4::Box>,
55    /// Title label inside the header.
56    pub title_label: OnceCell<Label>,
57    /// Grip icon used in `DragHandle::Handle` mode.
58    pub handle_label: OnceCell<Label>,
59    /// Fold toggle button (visible when `CardConfig::foldable`).
60    pub fold_button: OnceCell<Button>,
61    /// Close button (visible when `CardConfig::closable`).
62    pub close_button: OnceCell<Button>,
63    /// Bottom-right resize grip (`CardConfig::resizable`) - hidden by
64    /// default, shown from `Card::build` and while unfolded.
65    pub resize_grip: OnceCell<DrawingArea>,
66    /// Whether the card offers the resize grip - kept so `set_folded`
67    /// can gate the grip's visibility on the fold state.
68    pub resizable: Cell<bool>,
69    /// Bottom-left rotate grip (`CardConfig::rotate_grip`) - hidden by
70    /// default, shown from `Card::build` and while unfolded.
71    pub rotate_grip: OnceCell<DrawingArea>,
72    /// Whether the card offers the rotate grip - kept so `set_folded`
73    /// can gate the grip's visibility on the fold state.
74    pub rotate_grip_enabled: Cell<bool>,
75    /// Whether the card is currently folded to its header.
76    pub folded: Cell<bool>,
77    /// Container for `CardConfig::folded_content` - visible only while
78    /// folded with a custom folded widget installed.
79    pub folded_holder: OnceCell<gtk4::Box>,
80    /// Upper bound on the reported width (`CardConfig::max_width`).
81    pub max_width: Cell<Option<Pixels>>,
82    /// Upper bound on the reported height (`CardConfig::max_height`).
83    pub max_height: Cell<Option<Pixels>>,
84}

116impl ObjectImpl for CardImpl {
117    fn constructed(&self) {
118        self.parent_constructed();
119        let card = self.obj();
120        card.set_orientation(Orientation::Vertical);
121        card.set_spacing(0);
122        // Clips content that exceeds a `max_width`/`max_height` cap and keeps
123        // it inside the rounded corners (GTK CSS has no `overflow` property).
124        card.set_overflow(Overflow::Hidden);
125        card.add_css_class(CLASS_CARD);
126
127        let header = gtk4::Box::new(Orientation::Horizontal, 6);
128        header.add_css_class(CLASS_HEADER);
129        header.set_hexpand(true);
130
131        let handle_label = Label::new(Some("\u{2261}"));
132        handle_label.add_css_class(CLASS_HANDLE);
133        handle_label.set_visible(false);
134        header.append(&handle_label);
135
136        let title_label = Label::new(None);
137        title_label.set_hexpand(true);
138        title_label.set_halign(Align::Start);
139        title_label.set_ellipsize(EllipsizeMode::End);
140        header.append(&title_label);
141
142        let fold_button = Button::from_icon_name("go-down-symbolic");
143        fold_button.add_css_class("flat");
144        fold_button.set_visible(false);
145        header.append(&fold_button);
146
147        let close_button = Button::from_icon_name("window-close-symbolic");
148        close_button.add_css_class("flat");
149        close_button.set_visible(false);
150        header.append(&close_button);
151
152        let content_holder = gtk4::Box::new(Orientation::Vertical, 0);
153        content_holder.add_css_class(CLASS_CONTENT);
154        content_holder.set_vexpand(true);
155
156        // The overlay stacks the resize grip over the content's
157        // bottom-right corner without claiming layout space.
158        let content_overlay = Overlay::new();
159        content_overlay.set_child(Some(&content_holder));
160        let resize_grip = resize_grip();
161        content_overlay.add_overlay(&resize_grip);
162        let rotate_grip = rotate_grip();
163        content_overlay.add_overlay(&rotate_grip);
164
165        let folded_holder = gtk4::Box::new(Orientation::Horizontal, 0);
166        folded_holder.add_css_class(crate::widget::css::CLASS_FOLDED_CONTENT);
167        folded_holder.set_visible(false);
168
169        card.append(&header);
170        card.append(&content_overlay);
171        card.append(&folded_holder);
172
173        let _ = self.header.set(header);
174        let _ = self.content_holder.set(content_holder);
175        let _ = self.title_label.set(title_label);
176        let _ = self.handle_label.set(handle_label);
177        let _ = self.fold_button.set(fold_button);
178        let _ = self.close_button.set(close_button);
179        let _ = self.resize_grip.set(resize_grip);
180        let _ = self.rotate_grip.set(rotate_grip);
181        let _ = self.folded_holder.set(folded_holder);
182    }

190/// Edge length of the resize grip's diagonal strokes (px).
191const RESIZE_GRIP_SIZE: i32 = 18;
192
193/// Builds the bottom-right resize affordance: three diagonal strokes in
194/// the card's foreground color, `nwse-resize` cursor on hover.
195fn resize_grip() -> DrawingArea {
196    let grip = DrawingArea::new();
197    grip.add_css_class(CLASS_RESIZE_GRIP);
198    // Overlay children default to `Align::Fill` - without End
199    // alignment the grip would cover the whole content area.
200    grip.set_halign(Align::End);
201    grip.set_valign(Align::End);
202    grip.set_content_width(RESIZE_GRIP_SIZE);
203    grip.set_content_height(RESIZE_GRIP_SIZE);
204    grip.set_cursor_from_name(Some("nwse-resize"));
205    grip.set_visible(false);
206    grip.set_draw_func(|grip, cr, _, _| {
207        let color = grip.color();
208        let width = f64::from(grip.width());
209        let height = f64::from(grip.height());
210        cr.set_source_rgba(f64::from(color.red()), f64::from(color.green()), f64::from(color.blue()), f64::from(color.alpha()));
211        cr.set_line_width(1.5);
212        cr.set_line_cap(LineCap::Round);
213        for inset in [4.0, 8.0, 12.0] {
214            cr.move_to(width - inset, height - 2.0);
215            cr.line_to(width - 2.0, height - inset);
216            let _ = cr.stroke();
217        }
218    });
219    grip
220}

222/// Edge length of the rotate grip's circular arrow (px).
223const ROTATE_GRIP_SIZE: i32 = 18;
224
225/// Builds the bottom-left rotate affordance: a circular arrow in the
226/// card's foreground color, `grab` cursor on hover.
227fn rotate_grip() -> DrawingArea {
228    let grip = DrawingArea::new();
229    grip.add_css_class(CLASS_ROTATE_GRIP);
230    // Overlay children default to `Align::Fill` - without Start/End
231    // alignment the grip would cover the whole content area.
232    grip.set_halign(Align::Start);
233    grip.set_valign(Align::End);
234    grip.set_content_width(ROTATE_GRIP_SIZE);
235    grip.set_content_height(ROTATE_GRIP_SIZE);
236    grip.set_cursor_from_name(Some("grab"));
237    grip.set_visible(false);
238    grip.set_draw_func(|grip, cr, _, _| {
239        let color = grip.color();
240        let width = f64::from(grip.width());
241        let height = f64::from(grip.height());
242        cr.set_source_rgba(f64::from(color.red()), f64::from(color.green()), f64::from(color.blue()), f64::from(color.alpha()));
243        cr.set_line_width(1.5);
244        cr.set_line_cap(LineCap::Round);
245        // A three-quarter clockwise arc with an arrowhead on its tip:
246        // the circular arrow suggests turning in either direction.
247        let radius = width.min(height) / 2.0 - 3.0;
248        let center_x = width / 2.0;
249        let center_y = height / 2.0;
250        let start = -FRAC_PI_2 + 0.7;
251        let end = start + 3.0 * FRAC_PI_2;
252        cr.arc(center_x, center_y, radius, start, end);
253        let _ = cr.stroke();
254        let tip_x = center_x + radius * end.cos();
255        let tip_y = center_y + radius * end.sin();
256        let tangent = end + FRAC_PI_2;
257        for spread in [-0.7, 0.7] {
258            cr.move_to(tip_x, tip_y);
259            cr.line_to(tip_x - 4.0 * (tangent + spread).cos(), tip_y - 4.0 * (tangent + spread).sin());
260        }
261        let _ = cr.stroke();
262    });
263    grip
264}

266impl WidgetImpl for CardImpl {
267    // Caps the reported size at `CardConfig::max_width`/`max_height` so the
268    // card stops growing when its content grows; the parent layer allocates
269    // at most the capped size and the default CSS clips the overflow.
270    fn measure(&self, orientation: Orientation, for_size: i32) -> (i32, i32, i32, i32) {
271        let (minimum, natural, minimum_baseline, natural_baseline) = self.parent_measure(orientation, for_size);
272        let limit = match orientation {
273            Orientation::Horizontal => self.max_width.get(),
274            Orientation::Vertical => self.max_height.get(),
275            _ => None,
276        };
277        match limit {
278            // Fractional caps floor so the reported size never exceeds the bound.
279            Some(cap) => {
280                let cap = cap.as_f64() as i32;
281                (minimum.min(cap), natural.min(cap), -1, -1)
282            }
283            None => (minimum, natural, minimum_baseline, natural_baseline),
284        }
285    }
286}
