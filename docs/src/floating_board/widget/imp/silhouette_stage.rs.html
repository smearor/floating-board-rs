19/// Implementation of [`SilhouetteStage`].
20#[derive(Default)]
21pub struct SilhouetteStageImpl {
22    /// The base child (allocated the full extent); overlay children
23    /// follow it in the child list and claim no layout space.
24    pub child: RefCell<Option<gtk4::Widget>>,
25}
26
27#[glib::object_subclass]
28impl ObjectSubclass for SilhouetteStageImpl {
29    const NAME: &'static str = "FloatingBoardSilhouetteStage";
30    type Type = SilhouetteStage;
31    type ParentType = gtk4::Widget;
32
33    fn class_init(klass: &mut Self::Class) {
34        klass.set_layout_manager_type::<StageLayout>();
35    }
36}
37
38impl ObjectImpl for SilhouetteStageImpl {
39    fn dispose(&self) {
40        while let Some(child) = self.obj().first_child() {
41            child.unparent();
42        }
43    }
44}
45
46impl WidgetImpl for SilhouetteStageImpl {
47    // `gtk_widget_do_pick` consults `contains` only as a leaf check:
48    // a plain overlay claims picks in the cut regions its allocation
49    // covers outside the card's silhouette, swallowing them instead
50    // of letting them fall through to the widgets below. Delegating
51    // to the ancestor card's silhouette keeps the stage's claimable
52    // region identical to the card's own.
53    fn contains(&self, x: f64, y: f64) -> bool {
54        silhouette_contains::within_card(&*self.obj(), x, y)
55    }
56}

58/// Implementation of [`StageLayout`].
59#[derive(Default)]
60pub struct StageLayoutImpl;
61
62#[glib::object_subclass]
63impl ObjectSubclass for StageLayoutImpl {
64    const NAME: &'static str = "FloatingBoardStageLayout";
65    type Type = StageLayout;
66    type ParentType = gtk4::LayoutManager;
67}
68
69impl ObjectImpl for StageLayoutImpl {}
70
71impl StageLayoutImpl {
72    /// `overlay children share the base's extent.
73    fn aligned_extent(align: gtk4::Align, natural: i32, total: i32) -> (i32, i32) {
74        match align {
75            gtk4::Align::Fill => (total, 0),
76            gtk4::Align::Start => (natural, 0),
77            gtk4::Align::End => (natural, total - natural),
78            _ => (natural, (total - natural) / 2),
79        }
80    }
81}
82
83impl LayoutManagerImpl for StageLayoutImpl {
84    // `gtk4::Overlay` semantics: only the base child measures - the
85    // overlay children claim no layout space.
86    fn measure(&self, widget: &gtk4::Widget, orientation: Orientation, for_size: i32) -> (i32, i32, i32, i32) {
87        widget.first_child().map(|child| child.measure(orientation, for_size)).unwrap_or_default()
88    }
89
90    fn allocate(&self, widget: &gtk4::Widget, width: i32, height: i32, baseline: i32) {
91        let mut child = widget.first_child();
92        let mut base = true;
93        while let Some(current) = child {
94            if base {
95                current.allocate(width, height, baseline, None);
96                base = false;
97            } else {
98                let (_, natural_width, _, _) = current.measure(Orientation::Horizontal, -1);
99                let (_, natural_height, _, _) = current.measure(Orientation::Vertical, natural_width);
100                let (child_width, x) = Self::aligned_extent(current.halign(), natural_width, width);
101                let (child_height, y) = Self::aligned_extent(current.valign(), natural_height, height);
102                let transform = Transform::new().translate(&Point::new(x as f32, y as f32));
103                current.allocate(child_width, child_height, baseline, Some(transform));
104            }
105            child = current.next_sibling();
106        }
107    }
108}
