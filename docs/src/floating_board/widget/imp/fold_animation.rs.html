1//! Runtime state of a playing fold transition on a card entry - the
2//! board's tick evaluates `animation`'s eased progress through
3//! `CardFoldAnimation::pose` and applies the pose per frame.
4
5use crate::domain::CardFoldAnimation;
6use smearor_commons_animation::Span;
7use smearor_commons_animation::TimedAnimation;
8use smearor_commons_geometry::Fraction;
9use smearor_commons_geometry::Pixels;
10use smearor_commons_geometry::RotationRad;
11use smearor_commons_geometry::ScaleFactor;
12use smearor_commons_geometry::Vec2;

14/// One in-flight fold transition: eased progress drives `kind`'s pose
15/// between `start_size` and `end_size`; rotation/zoom excursions
16/// compose on the captured rendered bases so user gestures before
17/// the fold are preserved.
18pub struct FoldAnimation {
19    /// Eased progress driver (`0.0` = start state, `1.0` = end state).
20    pub animation: TimedAnimation<Span<f64>>,
21    /// `true` = animating into the folded state.
22    pub to_folded: bool,
23    /// The direction-resolved morph program
24    /// (`CardConfig::effective_fold_animation`).
25    pub kind: CardFoldAnimation,
26    /// Eased-progress fraction at which a gated satellite reveal
27    /// fires (unfold direction only): `unfold_timing`'s `Overlap`
28    /// fraction, `Fraction::ONE` for `Sequential`.
29    pub reveal_at: Fraction,
30    /// Card footprint in px at the transition start.
31    pub start_size: Vec2<Pixels>,
32    /// Card footprint in px at the transition end.
33    pub end_size: Vec2<Pixels>,
34    /// Rendered base rotation the excursion composes on.
35    pub base_rotation: RotationRad,
36    /// Rendered base scale the zoom excursion composes on.
37    pub base_scale: ScaleFactor,
38}
