80/// Minimum live points before the quad transform applies - a single touch
81/// stays a plain touch and never moves the card.

84/// Frame-to-frame angle jump (60°) treated as a degenerate finger crossing
85/// (~180° flips when points cross the centroid) rather than real rotation.

88/// How long a tracked touch may stay silent before it counts as dead - its
89/// `TouchEnd` was routed to a claiming gesture or lost on the wire. Real

129    /// Adds a card with an auto-generated `card-N` id and returns its id.
130    pub fn add_card(&self, content: &impl IsA<Widget>, config: CardConfig) -> CardId {
131        loop {
132            let sequence = self.imp().sequence.get().incremented();
133            let id = CardId::new(format!("card-{sequence}"));
134            match self.add_card_with_id(id.clone(), content, config.clone()) {

136                Err(BoardError::DuplicateCard(_)) => continue,
137                Err(_) => return id,
138            }

142    /// Adds a card with a caller-chosen semantic id (e.g. `"controls"`).
143    ///
144    /// The id stays stable in persisted layouts, so consumers can restore
145    /// geometry per semantic key.
146    pub fn add_card_with_id(&self, id: impl Into<CardId>, content: &impl IsA<Widget>, config: CardConfig) -> Result<CardId, BoardError> {
147        let id = id.into();
148        if self.imp().cards.borrow().contains_key(&id) {
149            return Err(BoardError::DuplicateCard(id));
150        }
151        let Some(layer) = self.card_layer() else {
152            return Err(BoardError::UnknownCard(id));
153        };
154
155        let card = Card::build(&id, content.upcast_ref(), self, &config);
156        let rotation_widget = RotationWidget::new(SmearorRotation::DEG_0)
157            // Rotation comes from the board's own quad transform gesture;
158            // smearor's built-in rotate gesture stays off.
159            .with_gesture_rotation_enabled(false)
160            // The card is nested inside the board's `Fixed` layer - the
161            // surface-global input-region clamp would kill input for the
162            // rest of the window (`smearor-wrot-rotation` >= 0.2.1).
163            .with_input_region_enabled(false)
164            // Allocate the card its natural size at any angle - the legacy
165            // inverse-solve shrinks non-square children near 45°.
166            .with_natural_child_size_enabled(true);
167        rotation_widget.set_animation_speed(config.animation_speed_ms);
168        rotation_widget.set_child(Some(&card));
169        let scaling = ScalingWidget::new(&rotation_widget);
170        layer.put(&scaling, 0.0, 0.0);
171
172        let sequence = self.imp().sequence.replace(self.imp().sequence.get().incremented());
173        let (initial_position, pending_anchor) = match config.position {
174            CardAnchor::Explicit(position) => (position, None),
175            anchor => (CardPosition::default(), Some(anchor)),
176        };
177        let entry = CardEntry {
178            card: card.clone(),
179            scaling: scaling.clone(),
180            rotation_widget: rotation_widget.clone(),
181            config: config.clone(),
182            position: Cell::new(initial_position),
183            scale: Cell::new(ScaleFactor::default()),
184            pending_anchor: RefCell::new(pending_anchor),
185            drag_start: Cell::new(CardPosition::default()),
186            drag_grab: Cell::new(None),
187            quad_touches: RefCell::new(Vec::new()),
188            quad_baseline: RefCell::new(None),
189            claimed: RefCell::new(HashMap::new()),
190            watched: RefCell::new(HashSet::new()),
191            sequence,
192        };
193        self.imp().cards.borrow_mut().insert(id.clone(), entry);
194
195        // Place the stack immediately for explicit anchors; named anchors
196        // resolve at the first board allocation.
197        if pending_anchor.is_none() {
198            let cards = self.imp().cards.borrow();
199            if let Some(entry) = cards.get(&id) {
200                self.apply_position(entry);
201            }
202        }
203
204        self.wire_gestures(&id, &card, &scaling, &rotation_widget, &config);
205        if config.z_index.is_some() {
206            self.restack();
207        }
208
209        debug!(card = %id, "card added");
210        self.emit_by_name::<()>(BoardSignal::CardAdded.name(), &[&card]);
211        Ok(id)
212    }

251    /// Restores a previously snapshotted layout.
252    ///
253    /// Unknown card ids are skipped (logged, not an error); geometry is
254    /// clamped to the current board bounds so nothing restores off-screen.
255    pub fn restore_layout(&self, layout: &BoardLayout) {
256        for placement in &layout.cards {
257            let (card, rotation_widget, scaling) = {
258                let cards = self.imp().cards.borrow();
259                match cards.get(&placement.id) {
260                    Some(entry) => {
261                        entry.position.set(placement.geometry.position);
262                        entry.scale.set(placement.geometry.scale.clamped(&entry.config.scale_limits));
263                        entry.pending_anchor.borrow_mut().take();
264                        (entry.card.clone(), entry.rotation_widget.clone(), entry.scaling.clone())
265                    }
266                    None => {
267                        warn!(card = %placement.id, "skipping unknown card in layout restore");
268                        continue;
269                    }
270                }
271            };
272            // Notify handlers re-borrow the card map, so widget calls happen
273            // outside of it.
274            rotation_widget.set_rotation(SmearorRotation(placement.geometry.rotation));
275            scaling.set_scale(placement.geometry.scale.value());
276            card.set_folded(placement.folded);
277            let cards = self.imp().cards.borrow();
278            if let Some(entry) = cards.get(&placement.id) {
279                self.apply_position(entry);
280                self.clamp_entry(entry);
281            }
282        }
283        self.emit_geometry_changed();
284    }

291    /// Applies a full geometry update, clamped to the board bounds.
292    pub(crate) fn set_card_geometry(&self, id: &CardId, geometry: &CardGeometry) {
293        let (rotation_widget, scaling, scale) = {
294            let cards = self.imp().cards.borrow();
295            match cards.get(id) {
296                Some(entry) => {
297                    entry.position.set(geometry.position);
298                    let scale = geometry.scale.clamped(&entry.config.scale_limits);
299                    entry.scale.set(scale);
300                    entry.pending_anchor.borrow_mut().take();
301                    (entry.rotation_widget.clone(), entry.scaling.clone(), scale)
302                }
303                None => return,
304            }
305        };
306        rotation_widget.set_rotation(SmearorRotation(geometry.rotation));
307        scaling.set_scale(scale.value());
308        let cards = self.imp().cards.borrow();
309        if let Some(entry) = cards.get(id) {
310            self.apply_position(entry);
311            self.clamp_entry(entry);
312        }
313        drop(cards);
314        self.emit_geometry_changed();
315    }

353    /// Resets a card to its configured initial geometry: the `position`
354    /// anchor re-resolved against the current bounds, `Deg0` rotation and
355    /// scale `1.0`.
356    ///
357    /// Rotation goes through the widget first (the `notify::rotation` handler
358    /// applies the interim state); the anchor then resolves against the
359    /// reset footprint.
360    pub fn reset_card(&self, id: &CardId) -> Result<(), BoardError> {
361        let (rotation_widget, scaling) = {
362            let cards = self.imp().cards.borrow();
363            let entry = cards.get(id).ok_or_else(|| BoardError::UnknownCard(id.clone()))?;
364            entry.scale.set(ScaleFactor::default());
365            (entry.rotation_widget.clone(), entry.scaling.clone())
366        };
367        rotation_widget.set_rotation(SmearorRotation::DEG_0);
368        scaling.set_scale(1.0);
369        {
370            let cards = self.imp().cards.borrow();
371            let Some(entry) = cards.get(id) else {
372                return Ok(());
373            };
374            match entry.config.position {
375                CardAnchor::Explicit(position) => {
376                    entry.pending_anchor.borrow_mut().take();
377                    entry.position.set(position);
378                }
379                anchor => {
380                    let bounds = self.board_bounds();
381                    if bounds.is_degenerate() {
382                        entry.pending_anchor.replace(Some(anchor));
383                    } else {
384                        entry.pending_anchor.borrow_mut().take();
385                        let footprint = self.predicted_size(entry);
386                        entry.position.set(anchor.resolve(&bounds, footprint, entry.config.anchor_margin));
387                    }
388                }
389            }
390            self.apply_position(entry);
391            self.clamp_entry(entry);
392        }
393        self.emit_geometry_changed();
394        Ok(())
395    }

433    /// Union of every card's rotated footprint in card-layer (board)
434    /// coordinates. Consumers restricting the surface input region to
435    /// interactive chrome - e.g. touch-passthrough overlays - union this
436    /// with their own regions after translating to surface coordinates.
437    pub fn cards_region(&self) -> Region {
438        let region = Region::create();
439        for entry in self.imp().cards.borrow().values() {
440            // Hidden cards keep their geometry but contribute no input
441            // footprint - a stale rect would dead-zone the canvas.
442            if !entry.scaling.is_visible() {
443                continue;
444            }
445            if let Err(error) = region.union(&self.entry_footprint(entry)) {
446                warn!(%error, "failed to union card footprint");
447            }
448        }
449        region
450    }

510    /// Called from `WidgetImpl::size_allocate`: resolves pending anchors and
511    /// re-clamps all cards against the new bounds.
512    pub(crate) fn on_board_allocated(&self) {
513        let bounds = self.board_bounds();
514        if bounds.is_degenerate() {
515            return;
516        }
517        let mut resolved = false;
518        {
519            let cards = self.imp().cards.borrow();
520            for entry in cards.values() {
521                if let Some(anchor) = entry.pending_anchor.borrow_mut().take() {
522                    let footprint = self.predicted_size(entry);
523                    entry.position.set(anchor.resolve(&bounds, footprint, entry.config.anchor_margin));
524                    resolved = true;
525                }
526                // Reconcile render state with position state on every
527                // allocation: `gtk4::Fixed` positions are only ever moved via
528                // apply_position, so cards added before the first allocation
529                // would otherwise stay at (0,0).
530                self.apply_position(entry);
531            }
532        }
533        self.clamp_all();
534        if resolved {
535            self.emit_geometry_changed();
536        }
537    }

539    /// `notify::rotation` on the card's `RotatedLayout`: re-centers the card
540    /// (the rotated bounding box changes size), clamps and emits signals.
541    fn on_card_rotated(&self, id: &CardId) {
542        let card = {
543            let cards = self.imp().cards.borrow();
544            let Some(entry) = cards.get(id) else {
545                return;
546            };
547            // smearor-wrot-rotation queues allocation rather than resize, so
548            // the RotatedLayout measure would stay stale; `layout_changed`
549            // invalidates the measure chain and a 90 degree turn correctly
550            // swaps the child's width/height.
551            if let Some(manager) = entry.rotation_widget.layout_manager() {
552                manager.layout_changed();
553            }
554            entry.scaling.queue_resize();
555            self.apply_position(entry);
556            self.clamp_entry(entry);
557            entry.card.clone()
558        };
559        card.emit_card_rotated();
560        self.emit_geometry_changed();
561    }

563    /// Slot activation at sequence begin: the quad drags are `exclusive`,
564    /// so GTK's own `GtkGestureSingle` distribution hands each slot a
565    /// different touch sequence and `begin` only fires for the sequence this
566    /// slot actually owns. The slot records the finger's board-space
567    /// position at touch-down, not only once GTK recognizes a drag.
568    /// Routes a raw touch event from the quad tracker: `TouchBegin`
569    /// registers the sequence as a live point, `TouchUpdate` moves it and
570    /// feeds the similarity transform, `TouchEnd`/`TouchCancel` removes it.
571    fn on_quad_touch_event(
572        &self,
573        id: &CardId,
574        controller: &gtk4::EventControllerLegacy,
575        event: &gtk4::gdk::Event,
576        card_drag: Option<&gtk4::GestureDrag>,
577        click: &gtk4::GestureClick,
578    ) {
579        let event_type = event.event_type();
580        if !matches!(event_type, EventType::TouchBegin | EventType::TouchUpdate | EventType::TouchEnd | EventType::TouchCancel) {
581            return;
582        }
583        // `event_sequence` is a plain (non-Option) binding - calling it on
584        // sequence-less events asserts on a NULL pointer, so it may only run
585        // after the touch-type check above.
586        self.sweep_quad_touches(id);
587        let sequence = event.event_sequence();
588        match event_type {
589            EventType::TouchBegin => {
590                // Diagnostic: two TouchBegins for one physical contact
591                // (slot + emulated/pointer dupe) would look identical
592                // here - sequence address + device distinguishes them.
593                info!(card = %id, sequence = ?sequence, "quad touch begin");
594                if let Some(point) = self.event_point_in_layer(controller, event) {
595                    let cards = self.imp().cards.borrow();
596                    if let Some(entry) = cards.get(id) {
597                        self.activate_quad_touch(entry, sequence, point, card_drag, click);
598                    }
599                }
600                self.emit_geometry_changed();
601            }
602            EventType::TouchUpdate => {
603                let Some(point) = self.event_point_in_layer(controller, event) else {
604                    return;
605                };
606                {
607                    let cards = self.imp().cards.borrow();
608                    let Some(entry) = cards.get(id) else {
609                        return;
610                    };
611                    let tracked = {
612                        let mut touches = entry.quad_touches.borrow_mut();
613                        let Some(tracked) = touches.iter_mut().find(|touch| touch.sequence == sequence) else {
614                            return;
615                        };
616                        tracked.point = point;
617                        tracked.last_event = Instant::now();
618                        touches.len()
619                    };
620                    debug!(card = %id, tracked, point = ?point, "quad touch update");
621                    self.apply_quad_transform(entry);
622                }
623                self.emit_geometry_changed();
624            }
625            EventType::TouchEnd | EventType::TouchCancel => {
626                self.release_quad_touch(id, &sequence);
627            }
628            _ => {}
629        }
630    }

632    /// Attaches a `sequence-state-changed` watcher to every `Gesture`
633    /// controller in the card subtree that does not have one yet. The
634    /// watcher marks claimed sequences on the entry - the mark
635    /// survives the gesture's own reset, which a `sequence_state`
636    /// query would not. Walked lazily on each sweep so gestures
637    /// children add at runtime are covered too.
638    fn watch_claims(&self, entry: &CardEntry) {
639        let root = entry.scaling.clone().upcast::<gtk4::Widget>();
640        let card_id = entry.card.id();
641        let mut stack = vec![root];
642        while let Some(widget) = stack.pop() {
643            for controller in widget.observe_controllers().iter::<glib::Object>() {
644                let Ok(controller) = controller else {
645                    continue;
646                };
647                let Some(gesture) = controller.downcast_ref::<gtk4::Gesture>() else {
648                    continue;
649                };
650                // The wrapper reference is a temporary - the stable
651                // identity is the underlying GObject pointer.
652                let address = glib::translate::ToGlibPtr::<*mut gtk4::ffi::GtkGesture>::to_glib_none(&gesture).0 as usize;
653                if !entry.watched.borrow_mut().insert(address) {
654                    continue;
655                }
656                let board_weak = self.downgrade();
657                let card_id = card_id.clone();
658                gesture.connect_sequence_state_changed(move |_, sequence, state| {
659                    if state != EventSequenceState::Claimed {
660                        return;
661                    }
662                    if let (Some(board), Some(sequence)) = (board_weak.upgrade(), sequence) {
663                        let cards = board.imp().cards.borrow();
664                        if let Some(entry) = cards.get(&card_id) {
665                            entry.claimed.borrow_mut().insert(sequence.clone(), Instant::now());
666                        }
667                    }
668                });
669            }
670            let mut child = widget.first_child();
671            while let Some(ref sibling) = child {
672                stack.push(sibling.clone());
673                child = sibling.next_sibling();
674            }
675        }
676    }

678    /// Drops tracked touches that are no longer live. Two cases:
679    /// a touch *claimed* by another gesture (its remaining events are routed
680    /// to the claimer only) and a touch that went *silent* - its `TouchEnd`
681    /// already went to a claimer or was lost on the wire, so only the event
682    /// timestamp reveals it is dead. Without the sweep a dead touch lingers
683    /// as a phantom point, keeps the transform "active" for a single new
684    /// finger and distorts radius/angle. Claims are checked against every
685    /// gesture in the card subtree - the board's own click/drag as well as
686    /// child widgets (buttons, toggles), even ones added at runtime.
687    fn sweep_quad_touches(&self, id: &CardId) {
688        let pruned = {
689            let cards = self.imp().cards.borrow();
690            let Some(entry) = cards.get(id) else {
691                return;
692            };
693            self.watch_claims(entry);
694            // Slot ids recycle: a mark older than the stale window can
695            // only belong to a press that already ended - expire it so
696            // the next touch on the same slot is not shadowed.
697            entry.claimed.borrow_mut().retain(|_, ts| ts.elapsed() < Duration::from_millis(QUAD_STALE_MS));
698            let pruned = {
699                let mut touches = entry.quad_touches.borrow_mut();
700                let claimed = entry.claimed.borrow();
701                let before = touches.len();
702                touches.retain(|touch| touch.last_event.elapsed() < Duration::from_millis(QUAD_STALE_MS) && !claimed.contains_key(&touch.sequence));
703                before - touches.len()
704            };
705            if pruned > 0 {
706                info!(card = %id, pruned, "quad touches pruned (claimed or stale)");
707                if entry.quad_touches.borrow().len() >= QUAD_MIN_POINTS {
708                    self.refresh_quad_baseline(entry);
709                } else {
710                    entry.quad_baseline.borrow_mut().take();
711                }
712            }
713            pruned
714        };
715        if pruned > 0 {
716            self.emit_geometry_changed();
717        }
718    }

720    /// Drops a quad touch: re-baselines for the remaining fingers or
721    /// finishes the transform and emits the collected signals.
722    fn release_quad_touch(&self, id: &CardId, sequence: &gtk4::gdk::EventSequence) {
723        let (card, moved, rotated, scaled) = {
724            let cards = self.imp().cards.borrow();
725            let Some(entry) = cards.get(id) else {
726                return;
727            };
728            let active = {
729                let mut touches = entry.quad_touches.borrow_mut();
730                let Some(index) = touches.iter().position(|touch| touch.sequence == *sequence) else {
731                    return;
732                };
733                touches.remove(index);
734                entry.claimed.borrow_mut().remove(sequence);
735                touches.len()
736            };
737            info!(card = %id, active, "quad touch released");
738            if active >= QUAD_MIN_POINTS {
739                // A finger left while the transform is still running: snap a
740                // fresh baseline so the remaining fingers continue smoothly.
741                self.refresh_quad_baseline(entry);
742                (entry.card.clone(), false, false, false)
743            } else {
744                let baseline = entry.quad_baseline.borrow_mut().take();
745                self.clamp_entry(entry);
746                let flags = baseline.map(|b| (b.moved, b.rotated, b.scaled)).unwrap_or_default();
747                (entry.card.clone(), flags.0, flags.1, flags.2)
748            }
749        };
750        if moved {
751            card.emit_card_moved();
752        }
753        if rotated {
754            card.emit_card_rotated();
755        }
756        if scaled {
757            card.emit_card_scaled();
758        }
759        self.emit_geometry_changed();
760    }

762    /// Activates a quad touch: records the finger's board-space position,
763    /// denies the card's own drag once the transform threshold is reached
764    /// (no double-move while the quad transform runs) and (re-)takes the
765    /// baseline snapshot.
766    fn activate_quad_touch(
767        &self,
768        entry: &CardEntry,
769        sequence: gtk4::gdk::EventSequence,
770        point: Vec2<Pixels>,
771        card_drag: Option<&gtk4::GestureDrag>,
772        click: &gtk4::GestureClick,
773    ) {
774        if entry.claimed.borrow().contains_key(&sequence) {
775            debug!(point = ?point, "quad touch skipped - sequence claimed");
776            return;
777        }
778        let active = {
779            let mut touches = entry.quad_touches.borrow_mut();
780            touches.push(TrackedTouch {
781                sequence,
782                point,
783                last_event: Instant::now(),
784            });
785            touches.len()
786        };
787        info!(active, point = ?point, "quad touch activated");
788        if active == QUAD_MIN_POINTS {
789            info!("quad transform active - denying card drag and click");
790            if let Some(card_drag) = card_drag {
791                deny_gesture(card_drag);
792            }
793            // GestureClick claims every sequence on press; the claim
794            // steals the first touch's remaining events and makes the
795            // sweep prune it, so the tracker could never
796            // reach two live points. Denying releases the claims - both
797            // sequences stream to the tracker again (raise-on-release
798            // is suppressed mid-gesture).
799            deny_gesture(click);
800        }
801        if active >= QUAD_MIN_POINTS {
802            self.refresh_quad_baseline(entry);
803        }
804    }

806    /// Touch position in board coordinates: the event reports its position
807    /// in surface space; it is mapped into the controller widget's
808    /// coordinate space through the native's `compute_point` and then down
809    /// the transform stack into layer (board) space.
810    fn event_point_in_layer(&self, controller: &EventControllerLegacy, event: &gtk4::gdk::Event) -> Option<Vec2<Pixels>> {
811        let position = Vec2::<Pixels>::from(event.position()?);
812        let widget = controller.widget()?;
813        let native = widget.native()?;
814        let surface_point = Point::from(position.to_f32());
815        let widget_point = native.compute_point(&widget, &surface_point)?;
816        let layer = self.card_layer()?;
817        let point = widget.compute_point(&layer, &widget_point)?;
818        Some(Vec2::<Pixels, f32>::from(point).to_f64())
819    }

821    /// Takes a fresh baseline: current geometry plus the live quad points.
822    /// Called on activation and whenever the point count changes, so adding
823    /// or lifting fingers never produces a jump.
824    fn refresh_quad_baseline(&self, entry: &CardEntry) {
825        let points: Vec<Vec2<Pixels>> = entry.quad_touches.borrow().iter().map(|touch| touch.point).collect();
826        let previous = entry.quad_baseline.borrow_mut().take();
827        if points.len() < QUAD_MIN_POINTS {
828            return;
829        }
830        let (moved, rotated, scaled) = previous.map(|b| (b.moved, b.rotated, b.scaled)).unwrap_or_default();
831        info!(points = ?points, position = ?entry.position.get(), scale = entry.scale.get().value(), rotation_deg = entry.rotation_widget.rotation().to_degrees(), "quad baseline snapshot");
832        entry.quad_baseline.replace(Some(QuadBaseline {
833            position: entry.position.get(),
834            scale: entry.scale.get(),
835            rotation: entry.rotation_widget.rotation(),
836            points,
837            moved,
838            rotated,
839            scaled,
840            applied_angle: RotationRad::ZERO,
841        }));
842    }

844    /// Applies the similarity transform fitted between the baseline point
845    /// cloud and the current points: the card center follows the same
846    /// rotation/scale/translation the fingers describe.
847    fn apply_quad_transform(&self, entry: &CardEntry) {
848        let current: Vec<Vec2<Pixels>> = entry.quad_touches.borrow().iter().map(|touch| touch.point).collect();
849        let mut baseline_ref = entry.quad_baseline.borrow_mut();
850        let Some(baseline) = baseline_ref.as_mut() else {
851            return;
852        };
853        let Some(fit) = SimilarityTransform::fit(&baseline.points, &current) else {
854            return;
855        };
856        // Finger crossings through the centroid flip the point-cloud angle
857        // by ~180° in a single update - far faster than a real hand can
858        // rotate. Re-baseline on such jumps instead of applying them.
859        if baseline.applied_angle.signed_delta_to(fit.rotation).abs() > QUAD_ANGLE_JUMP {
860            drop(baseline_ref);
861            self.refresh_quad_baseline(entry);
862            return;
863        }
864        baseline.applied_angle = fit.rotation;
865        let angle = if entry.config.rotatable { fit.rotation } else { RotationRad::ZERO };
866        let scale = if entry.config.scalable {
867            ScaleFactor::new(baseline.scale.value() * fit.scale.as_f64()).clamped(&entry.config.scale_limits)
868        } else {
869            baseline.scale
870        };
871        debug!(
872            fitted_angle_deg = fit.rotation.to_degrees(),
873            fitted_scale = fit.scale.as_f64(),
874            applied_angle_deg = angle.to_degrees(),
875            applied_scale = scale.value(),
876            centroid_now = ?fit.current_centroid,
877            "quad transform computed"
878        );
879        let applied_ratio = scale / baseline.scale;
880        if entry.config.draggable {
881            // The center follows the gated/clamped transform, not the raw
882            // fit, so `apply` is recomputed with the effective angle/ratio.
883            let rel = baseline.position.0 - fit.baseline_centroid;
884            let position = CardPosition::from(fit.current_centroid + rel.rotated(angle) * f64::from(applied_ratio));
885            if position != entry.position.get() {
886                baseline.moved = true;
887                entry.position.set(position);
888            }
889        }
890        if entry.config.rotatable {
891            let rotation = baseline.rotation + angle;
892            if rotation.signed_delta_to(entry.rotation_widget.rotation()).abs() > RotationRad::from_degrees(0.01) {
893                baseline.rotated = true;
894                entry.rotation_widget.set_rotation(SmearorRotation::from(rotation));
895            }
896        }
897        if entry.config.scalable && scale != entry.scale.get() {
898            baseline.scaled = true;
899            entry.scale.set(scale);
900            entry.scaling.set_scale(scale.value());
901        }
902        drop(baseline_ref);
903        self.apply_position(entry);
904        if entry.config.bounds_mode == BoundsMode::Contained {
905            self.clamp_entry(entry);
906        }
907    }

909    /// Records the card position and the pointer's board coordinates at drag
910    /// begin; the update delta is measured between two board-space pointer
911    /// samples, which stays stable while the card follows the pointer.
912    fn on_card_drag_begin(&self, id: &CardId, gesture: &GestureDrag) {
913        if let Some(entry) = self.imp().cards.borrow().get(id) {
914            entry.drag_start.set(entry.position.get());
915            entry.drag_grab.set(self.pointer_in_layer(gesture));
916        }
917    }

919    /// Moves the card so it follows the pointer.
920    ///
921    /// `GestureDrag` reports offsets in the coordinate space of the drag
922    /// source - which lives *inside* the rotating/scaling transform stack and
923    /// *moves* with the card. Mapping that delta with `compute_point` under
924    /// the current transform creates a feedback loop (the reference frame
925    /// shifts as the card tracks the pointer) that shows up as jitter and
926    /// roughly half pointer speed. Instead, the pointer's board position is
927    /// taken straight from the event's surface coordinates, which are
928    /// independent of the widget's own transform.
929    fn on_card_drag_update(&self, id: &CardId, gesture: &GestureDrag, offset: Vec2<Pixels>) {
930        {
931            let cards = self.imp().cards.borrow();
932            let Some(entry) = cards.get(id) else {
933                return;
934            };
935            let delta = self.drag_delta(gesture, offset, entry);
936            let position = entry.drag_start.get().translated(delta);
937            debug!(card = %id, delta = ?delta, position = ?(position.x().as_f64(), position.y().as_f64()), "drag update");
938            entry.position.set(position);
939            self.apply_position(entry);
940            if entry.config.bounds_mode == BoundsMode::Contained {
941                self.clamp_entry(entry);
942            }
943        }
944        self.emit_geometry_changed();
945    }
946
947    fn on_card_drag_end(&self, id: &CardId, gesture: &GestureDrag, offset: Vec2<Pixels>) {
948        self.on_card_drag_update(id, gesture, offset);
949        let card = {
950            let cards = self.imp().cards.borrow();
951            let Some(entry) = cards.get(id) else {
952                return;
953            };
954            self.clamp_entry(entry);
955            entry.card.clone()
956        };
957        card.emit_card_moved();
958        self.emit_geometry_changed();
959    }

961    /// Pointer position in card-layer (board) coordinates, taken from the
962    /// gesture's own point tracking (`start_point + offset` in its widget
963    /// space, mapped through `compute_point`). The round trip through the
964    /// transform stack yields the true board-space position: GTK delivers the
965    /// widget-space point via the inverse transform, so applying the forward
966    /// transform returns the surface point - independent of the card's own
967    /// motion, and always fresh at signal emission time (unlike
968    /// `current_event`, which can lag one touch event behind).
969    fn pointer_in_layer(&self, gesture: &GestureDrag) -> Option<Vec2<Pixels>> {
970        let start = Vec2::<Pixels>::from(gesture.start_point()?);
971        let offset = Vec2::<Pixels>::from(gesture.offset()?);
972        let widget = gesture.widget()?;
973        let layer = self.card_layer()?;
974        let point = widget.compute_point(&layer, &Point::from((start + offset).to_f32()))?;
975        Some(Vec2::<Pixels, f32>::from(point).to_f64())
976    }

978    /// Drag delta in board coordinates: current pointer position minus the
979    /// pointer position captured at drag begin. Falls back to undoing the
980    /// inverse `rotate . scale` on the widget-space offset before the first
981    /// allocation, when no surface mapping is available yet.
982    fn drag_delta(&self, gesture: &GestureDrag, offset: Vec2<Pixels>, entry: &CardEntry) -> Vec2<Pixels> {
983        if let (Some(now), Some(grab)) = (self.pointer_in_layer(gesture), entry.drag_grab.get()) {
984            return now - grab;
985        }
986        let rotation = entry.rotation_widget.rotation();
987        let scale = entry.scale.get().value();
988        offset.rotated(rotation) * scale
989    }

991    /// Reorders the card layer by `z_index` (stable by insertion sequence).
992    fn restack(&self) {
993        let Some(layer) = self.card_layer() else {
994            return;
995        };
996        let cards = self.imp().cards.borrow();
997        let mut ordered: Vec<&CardEntry> = cards.values().collect();
998        ordered.sort_by_key(|entry| (entry.config.z_index.unwrap_or(i32::MAX), entry.sequence));
999        for entry in ordered {
1000            let top_left = self.displayed_footprint(entry).min;
1001            layer.remove(&entry.scaling);
1002            layer.put(&entry.scaling, top_left.x, top_left.y);
1003        }
1004    }

1006    fn wire_gestures(&self, id: &CardId, card: &Card, scaling: &ScalingWidget, rotation_widget: &RotationWidget, config: &CardConfig) {
1007        // Fold changes the card footprint - layout persistence and
1008        // input-region consumers track it through `geometry-changed`.
1009        let board_weak = self.downgrade();
1010        card.connect_local("card-folded", false, move |_| {
1011            if let Some(board) = board_weak.upgrade() {
1012                board.emit_geometry_changed();
1013            }
1014            None
1015        });
1016
1017        // Rotation tracking: RotatedLayout exposes `rotation` as a notify-able
1018        // property, covering both gesture and programmatic changes.
1019        if let Some(manager) = rotation_widget.layout_manager() {
1020            let board_weak = self.downgrade();
1021            let card_id = id.clone();
1022            manager.connect_notify_local(Some("rotation"), move |_, _| {
1023                if let Some(board) = board_weak.upgrade() {
1024                    board.on_card_rotated(&card_id);
1025                }
1026            });
1027        }
1028
1029        // Raise on release: a click raises the card, and a finished drag ends
1030        // raised too. Raising on press would cancel the in-flight sequence.
1031        let click = GestureClick::new();
1032        click.set_button(0);
1033        let board_weak = self.downgrade();
1034        let card_id = id.clone();
1035        click.connect_released(move |_, _, _, _| {
1036            if let Some(board) = board_weak.upgrade() {
1037                board.raise_card(&card_id);
1038            }
1039        });
1040        scaling.add_controller(click.clone());
1041
1042        let mut card_drag: Option<GestureDrag> = None;
1043        if config.draggable {
1044            let drag = GestureDrag::new();
1045            drag.set_button(0);
1046            let board_weak = self.downgrade();
1047            let card_id = id.clone();
1048            drag.connect_drag_begin(move |gesture, _, _| {
1049                if let Some(board) = board_weak.upgrade() {
1050                    board.on_card_drag_begin(&card_id, gesture);
1051                }
1052            });
1053            let board_weak = self.downgrade();
1054            let card_id = id.clone();
1055            drag.connect_drag_update(move |gesture, offset_x, offset_y| {
1056                if let Some(board) = board_weak.upgrade() {
1057                    board.on_card_drag_update(&card_id, gesture, Vec2::new(offset_x, offset_y));
1058                }
1059            });
1060            let board_weak = self.downgrade();
1061            let card_id = id.clone();
1062            drag.connect_drag_end(move |gesture, offset_x, offset_y| {
1063                if let Some(board) = board_weak.upgrade() {
1064                    board.on_card_drag_end(&card_id, gesture, Vec2::new(offset_x, offset_y));
1065                }
1066            });
1067            card.drag_source(config.drag_handle).add_controller(drag.clone());
1068            card_drag = Some(drag);
1069        }
1070
1071        // Quad transform tracker: a raw `EventControllerLegacy` observing
1072        // every touch sequence on the card. Unlike `GestureSingle` siblings
1073        // it sees all sequences without claiming any, so each finger maps to
1074        // a tracked point and interactive content keeps working. With
1075        // `QUAD_MIN_POINTS` or more live points a similarity transform
1076        // applies - centroid delta moves, radius ratio scales, point-cloud
1077        // rotation rotates - so move, rotate and zoom are one mechanism
1078        // instead of arbitrated gestures. `FBD_NO_QUAD` disables it for
1079        // A/B testing.
1080        let quad_enabled = std::env::var_os("FBD_NO_QUAD").is_none() && (config.draggable || config.rotatable || config.scalable);
1081        if quad_enabled {
1082            let tracker = EventControllerLegacy::new();
1083            let board_weak = self.downgrade();
1084            let card_id = id.clone();
1085            let click_for_tracker = click.clone();
1086            tracker.connect_event(move |controller, event| {
1087                if let Some(board) = board_weak.upgrade() {
1088                    board.on_quad_touch_event(&card_id, controller, event, card_drag.as_ref(), &click_for_tracker);
1089                }
1090                Propagation::Proceed
1091            });
1092            scaling.add_controller(tracker);
1093        }
1094    }

1117    /// Positions the transform stack so its displayed center matches
1118    /// `entry.position` (`gtk4::Fixed` positions by top-left). No-ops when the
1119    /// stack already sits at the target spot, avoiding needless relayouts.
1120    fn apply_position(&self, entry: &CardEntry) {
1121        let Some(layer) = self.card_layer() else {
1122            return;
1123        };
1124        let target = self.displayed_footprint(entry).min;
1125        let current = Vec2::<Pixels>::from(layer.child_position(&entry.scaling));
1126        let delta = current - target;
1127        if delta.x.abs() > 0.5 || delta.y.abs() > 0.5 {
1128            layer.move_(&entry.scaling, target.x, target.y);
1129        }
1130    }

1132    /// Pushes the card back inside the board bounds. `BoundsMode::Loose`
1133    /// cards keep only `min_visible` pixels inside; `Contained` cards get
1134    /// their whole footprint pulled back.
1135    fn clamp_entry(&self, entry: &CardEntry) {
1136        let bounds = self.board_bounds();
1137        if bounds.is_degenerate() {
1138            return;
1139        }
1140        let footprint = self.displayed_footprint(entry);
1141        let delta = match entry.config.bounds_mode {
1142            BoundsMode::Loose => footprint.clamp_offset(&bounds, entry.config.min_visible.value()),
1143            BoundsMode::Contained => footprint.contain_offset(&bounds),
1144        };
1145        if delta != Vec2::ZERO {
1146            entry.position.set(entry.position.get().translated(delta));
1147            self.apply_position(entry);
1148        }
1149    }

1152        let cards = self.imp().cards.borrow();
1153        for entry in cards.values() {
1154            self.clamp_entry(entry);
1155        }
1156    }

1158    /// The size the transform stack occupies on the layer: the predicted size
1159    /// (what `gtk4::Fixed` will allocate next, reflecting pending rotation and
1160    /// scale changes), falling back to the allocated size before measuring is
1161    /// possible.
1162    fn displayed_size(&self, entry: &CardEntry) -> Vec2<Pixels> {
1163        let size = self.predicted_size(entry);
1164        if size.x > 0.0 && size.y > 0.0 {
1165            size
1166        } else {
1167            entry.scaling.allocated_size().to_f64().with_unit::<Pixels>()
1168        }
1169    }
1170
1171    /// Predicted displayed size: rotated bounding box of the card's natural
1172    /// size, times the scale factor.
1173    fn predicted_size(&self, entry: &CardEntry) -> Vec2<Pixels> {
1174        let (_, natural) = entry.card.measure_size();
1175        (natural.to_f64().with_unit::<Pixels>() * entry.scale.get().value()).rotated_bounds(entry.rotation_widget.rotation())
1176    }

1179        CardGeometry {
1180            position: entry.position.get(),
1181            rotation: Rotation::quantized(entry.rotation_widget.rotation()),
1182            scale: entry.scale.get(),
1183        }
1184    }
