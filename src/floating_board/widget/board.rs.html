181    /// Adds a card with a caller-chosen semantic id (e.g. `"controls"`).
182    ///
183    /// The id stays stable in persisted layouts, so consumers can restore
184    /// geometry per semantic key.
185    pub fn add_card_with_id(&self, id: impl Into<CardId>, content: &impl IsA<Widget>, config: CardConfig) -> Result<CardId, BoardError> {
186        let id = id.into();
187        if self.imp().cards.borrow().contains_key(&id) {
188            return Err(BoardError::DuplicateCard(id));
189        }
190        let Some(layer) = self.card_layer() else {
191            return Err(BoardError::UnknownCard(id));
192        };
193
194        let card = Card::build(&id, content.upcast_ref(), self, &config);
195        let rotation_widget = RotationWidget::new(SmearorRotation::DEG_0)
196            // Rotation comes from the board's own quad transform gesture;
197            // smearor's built-in rotate gesture stays off.
198            .with_gesture_rotation_enabled(false)
199            // The card is nested inside the board's `Fixed` layer - the
200            // surface-global input-region clamp would kill input for the
201            // rest of the window (`smearor-wrot-rotation` >= 0.2.1).
202            .with_input_region_enabled(false)
203            // Allocate the card its natural size at any angle - the legacy
204            // inverse-solve shrinks non-square children near 45°.
205            .with_natural_child_size_enabled(true);
206        rotation_widget.set_animation_speed(config.animation_speed_ms);
207        rotation_widget.set_child(Some(&card));
208        let scaling = ScalingWidget::new(&rotation_widget);
209        layer.put(&scaling, 0.0, 0.0);
210
211        let sequence = self.imp().sequence.replace(self.imp().sequence.get().incremented());
212        let (initial_position, pending_anchor) = match config.position {
213            CardAnchor::Explicit(position) => (position, None),
214            anchor => (CardPosition::default(), Some(anchor)),
215        };
216        let entry = CardEntry {
217            card: card.clone(),
218            scaling: scaling.clone(),
219            rotation_widget: rotation_widget.clone(),
220            config: config.clone(),
221            position: Cell::new(initial_position),
222            rotation: Cell::new(RotationRad::ZERO),
223            scale: Cell::new(ScaleFactor::default()),
224            pending_anchor: RefCell::new(pending_anchor),
225            drag_start: Cell::new(CardPosition::default()),
226            drag_grab: Cell::new(None),
227            resize_grab: Cell::new(None),
228            resize_scale: Cell::new(ScaleFactor::default()),
229            rotate_grab: Cell::new(None),
230            rotate_stepped: Cell::new(false),
231            quad_touches: RefCell::new(Vec::new()),
232            quad_baseline: RefCell::new(None),
233            claimed: RefCell::new(HashMap::new()),
234            watched: RefCell::new(HashSet::new()),
235            sequence,
236            seat: Cell::new(config.seat),
237            layout_generation: Cell::new(GenerationCounter::default()),
238            glide_target: Cell::new(None),
239        };
240        self.imp().cards.borrow_mut().insert(id.clone(), entry);
241
242        // Place the stack immediately for explicit anchors; named anchors
243        // resolve at the first board allocation.
244        if pending_anchor.is_none() {
245            let cards = self.imp().cards.borrow();
246            if let Some(entry) = cards.get(&id) {
247                self.apply_position(entry);
248            }
249        }
250
251        self.wire_gestures(&id, &card, &scaling, &rotation_widget, &config);
252        if config.z_index.is_some() {
253            self.restack();
254        }
255
256        debug!(card = %id, "card added");
257        self.emit_by_name::<()>(BoardSignal::CardAdded.name(), &[&card]);
258        Ok(id)
259    }

341    /// Restores a previously snapshotted layout.
342    ///
343    /// Unknown card ids are skipped (logged, not an error); geometry is
344    /// clamped to the current workspace bounds so nothing restores
345    /// off-screen. The stored workspace size and view are applied first.
346    pub fn restore_layout(&self, layout: &BoardLayout) {
347        self.cancel_view_flight();
348        self.imp().workspace_size.set(layout.workspace);
349        self.imp().view.replace(layout.view);
350        self.imp().seat_anchor.set(layout.seat_anchor);
351        self.on_board_allocated();
352        for placement in &layout.cards {
353            let (card, rotation_widget, scaling) = {
354                let cards = self.imp().cards.borrow();
355                match cards.get(&placement.id) {
356                    Some(entry) => {
357                        entry.position.set(placement.geometry.position);
358                        entry.scale.set(placement.geometry.scale.clamped(&entry.config.scale_limits));
359                        entry.pending_anchor.borrow_mut().take();
360                        // A restore supersedes in-flight glides: bumping
361                        // the generation stops their ticks and retires
362                        // the recorded target for seat fan-out.
363                        entry.layout_generation.set(entry.layout_generation.get().incremented());
364                        entry.glide_target.take();
365                        (entry.card.clone(), entry.rotation_widget.clone(), entry.scaling.clone())
366                    }
367                    None => {
368                        warn!(card = %placement.id, "skipping unknown card in layout restore");
369                        continue;
370                    }
371                }
372            };
373            // Notify handlers re-borrow the card map, so widget calls happen
374            // outside of it.
375            rotation_widget.set_rotation(SmearorRotation::from(placement.geometry.rotation.to_rotation_rad() + self.view().rotation));
376            scaling.set_scale(placement.geometry.scale.value() * self.view_zoom());
377            card.set_folded(placement.folded);
378            let _ = self.set_card_seat(&placement.id, placement.seat);
379            let cards = self.imp().cards.borrow();
380            if let Some(entry) = cards.get(&placement.id) {
381                self.apply_position(entry);
382                self.clamp_entry(entry);
383            }
384        }
385        self.emit_geometry_changed();
386    }

388    /// Arranges cards with an `smearor-commons-layout` strategy and
389    /// applies the result in one shot (unanimated).
390    ///
391    /// `ids` selects which cards participate: `Some(&[CardId])`
392    /// arranges just those (every id must exist), `None` arranges all
393    /// currently visible cards in insertion order. Each card becomes a
394    /// [`LayoutItem`] carrying its displayed footprint, its live
395    /// rotation and the seat from `CardConfig::seat`; the returned
396    /// placements are applied through the regular geometry path, so
397    /// bounds clamping, scale limits and `geometry-changed` behave
398    /// exactly like a manual manipulation.
399    ///
400    /// `Placement::size` is the target footprint - cards keep their
401    /// configured scale, only position and rotation are applied.
402    ///
403    /// `animated` glides each card on the frame clock towards its
404    /// placement (`EaseInOut`, duration from
405    /// `CardConfig::animation_speed_ms`); `false` applies immediately.
406    /// A new call supersedes animations still running on the same
407    /// card.
408    ///
409    /// Returns [`BoardError::Layout`] when the strategy reports that
410    /// the cards do not fit; the board is left untouched then.
411    pub fn apply_layout(&self, strategy: &dyn LayoutStrategy<Pixels>, ids: Option<&[CardId]>, animated: bool) -> Result<(), BoardError> {
412        let mut selected: Vec<(CardId, ScaleFactor, LayoutItem<Pixels>)> = Vec::new();
413        {
414            let cards = self.imp().cards.borrow();
415            match ids {
416                Some(ids) => {
417                    for id in ids {
418                        let Some(entry) = cards.get(id) else {
419                            return Err(BoardError::UnknownCard(id.clone()));
420                        };
421                        selected.push((id.clone(), entry.scale.get(), self.layout_item(entry)));
422                    }
423                }
424                None => {
425                    let mut visible: Vec<(&CardId, &CardEntry)> = cards.iter().filter(|(_, entry)| entry.scaling.is_visible()).collect();
426                    visible.sort_by_key(|(_, entry)| entry.sequence);
427                    for (id, entry) in visible {
428                        selected.push((id.clone(), entry.scale.get(), self.layout_item(entry)));
429                    }
430                }
431            }
432        }
433
434        let items: Vec<LayoutItem<Pixels>> = selected.iter().map(|(_, _, item)| *item).collect();
435        let placements = strategy.arrange(self.workspace_bounds(), &items)?;
436
437        for ((id, scale, _), placement) in selected.iter().zip(placements.iter()) {
438            // The placement expresses the target footprint; the card's
439            // configured scale is preserved.
440            let target = CardGeometry {
441                position: CardPosition::from(placement.center),
442                rotation: placement.rotation,
443                scale: *scale,
444            };
445            if animated {
446                self.glide_card_geometry(id, target)?;
447            } else {
448                self.set_card_geometry(id, &target);
449            }
450        }
451        Ok(())
452    }

454    /// Passes the card to the next user's seat around the table.
455    ///
456    /// The starting seat is the card's explicit `seat` assignment -
457    /// falling back to `Seat::nearest(rotation)` for unassigned cards -
458    /// stepped once in `direction` (the `offset` along the edge is
459    /// preserved). The card glides there animated: it rotates to the
460    /// target side's reading orientation and, with `move_to_edge`
461    /// (the common case), slides to `anchor_margin` in front of that
462    /// edge. With `move_to_edge = false` it only rotates in place.
463    ///
464    /// The step is on table *sides* - use
465    /// [`Self::rotate_to_next_occupied_seat`] to hop between occupied
466    /// seats instead. The passed card slips into the nearest free spot
467    /// at the seat's edge lane (or the lane behind it when the edge
468    /// queue reaches farther than a second row would), never on top of
469    /// another card.
470    ///
471    /// An explicitly assigned seat is advanced and persisted
472    /// (`card-seat-changed`); unassigned cards stay unassigned - the
473    /// next pass derives their seat from the rotation again.
474    ///
475    /// Returns [`BoardError::UnknownCard`] for an id not on the board.
476    pub fn rotate_to_next_user(&self, id: &CardId, direction: TurnDirection, move_to_edge: bool) -> Result<(), BoardError> {
477        let seat = self.card_seat_or_nearest(id)?;
478        let target = match direction {
479            TurnDirection::Clockwise => seat.clockwise(),
480            TurnDirection::CounterClockwise => seat.counter_clockwise(),
481        };
482        self.glide_card_to_seat(id, target, move_to_edge, false)
483    }

485    /// Passes the card to the next *occupied* seat of `seats`.
486    ///
487    /// Like [`Self::rotate_to_next_user`], but the destination is the
488    /// roster seat with the smallest clockwise (or counter-clockwise)
489    /// distance along the table perimeter: empty sides are skipped and
490    /// several seats on one side are visited in order. This is the
491    /// semantic a consumer's user list carries - "next player", not
492    /// "next edge".
493    ///
494    /// Unlike [`Self::rotate_to_next_user`], a reached roster seat is
495    /// always assigned to the card (`card-seat-changed`), including
496    /// previously unassigned cards: being passed to a user makes the
497    /// card belong to their seat, and the persisted position is what
498    /// lets repeated passes keep hopping the roster instead of
499    /// re-deriving a mid-edge starting point.
500    ///
501    /// The passed card slips into the nearest free spot at the seat's
502    /// edge lane (or the lane behind it when the edge queue reaches
503    /// farther than a second row would), never on top of another card.
504    ///
505    /// Returns [`BoardError::UnknownCard`] for an id not on the board
506    /// and [`BoardError::NoNextSeat`] when the roster holds no seat
507    /// other than the card's current one (empty rosters included).
508    pub fn rotate_to_next_occupied_seat(&self, id: &CardId, seats: &[Seat], direction: TurnDirection, move_to_edge: bool) -> Result<(), BoardError> {
509        let seat = self.card_seat_or_nearest(id)?;
510        let target = match direction {
511            TurnDirection::Clockwise => seat.next_in(seats),
512            TurnDirection::CounterClockwise => seat.previous_in(seats),
513        };
514        let Some(target) = target else {
515            return Err(BoardError::NoNextSeat(id.clone()));
516        };
517        self.glide_card_to_seat(id, target, move_to_edge, true)
518    }

525    /// Applies a full geometry update, clamped to the board bounds.
526    pub(crate) fn set_card_geometry(&self, id: &CardId, geometry: &CardGeometry) {
527        let (rotation_widget, scaling, scale) = {
528            let cards = self.imp().cards.borrow();
529            match cards.get(id) {
530                Some(entry) => {
531                    entry.position.set(geometry.position);
532                    let scale = geometry.scale.clamped(&entry.config.scale_limits);
533                    entry.scale.set(scale);
534                    entry.pending_anchor.borrow_mut().take();
535                    (entry.rotation_widget.clone(), entry.scaling.clone(), scale)
536                }
537                None => return,
538            }
539        };
540        rotation_widget.set_rotation(SmearorRotation::from(geometry.rotation.to_rotation_rad() + self.view().rotation));
541        scaling.set_scale(scale.value() * self.view_zoom());
542        let cards = self.imp().cards.borrow();
543        if let Some(entry) = cards.get(id) {
544            self.apply_position(entry);
545            self.clamp_entry(entry);
546        }
547        drop(cards);
548        self.emit_geometry_changed();
549    }

626            self.clamp_entry(entry);
627        }

772    /// Reconciles model and render state after each allocation pass -
773    /// resolves pending anchors, syncs card transforms to the view and
774    /// re-clamps. Called by the board's own layout manager
775    /// (`FloatingBoardLayout::allocate`): GTK4 delegates allocation to
776    /// the layout manager for widgets that have one (like `GtkBox`), so
777    /// a `size_allocate` vfunc would never run.
778    pub(crate) fn on_board_allocated(&self) {
779        // smearor-wrot-rotation is designed for layer-shell surfaces and
780        // narrows the surface input region to each card's footprint on every
781        // allocation. Inside the board all card allocations run through this
782        // pass, so restoring the full input region here wins per layout pass.
783        // Skipped while a consumer manages the region itself
784        // (`set_input_region_managed`) - e.g. touch-passthrough overlays that
785        // restrict input to the cards.
786        if !self.imp().input_region_managed.get()
787            && let Some(surface) = self.native().and_then(|native| native.surface())
788        {
789            surface.set_input_region(None);
790        }
791        let bounds = self.workspace_bounds();
792        if bounds.is_degenerate() {
793            return;
794        }
795        let mut resolved = false;
796        {
797            let cards = self.imp().cards.borrow();
798            for entry in cards.values() {
799                if let Some(anchor) = entry.pending_anchor.borrow_mut().take() {
800                    let footprint = self.workspace_card_size(entry);
801                    entry.position.set(anchor.resolve(&bounds, footprint, entry.config.anchor_margin));
802                    resolved = true;
803                }
804                // Reconcile render state with position state on every
805                // allocation: `gtk4::Fixed` positions are only ever moved via
806                // apply_position, so cards added before the first allocation
807                // would otherwise stay at (0,0). The view zoom can also
808                // change with the allocation (a `Fixed` workspace fits to
809                // the widget), so the rendered transform is re-synced.
810                self.sync_rendered_transform(entry);
811            }
812        }
813        self.clamp_all();
814        if resolved {
815            self.emit_geometry_changed();
816        }
817    }

847    /// Slot activation at sequence begin: the quad drags are `exclusive`,
848    /// so GTK's own `GtkGestureSingle` distribution hands each slot a
849    /// different touch sequence and `begin` only fires for the sequence this
850    /// slot actually owns. The slot records the finger's workspace
851    /// position at touch-down, not only once GTK recognizes a drag.
852    /// Routes a raw touch event from the quad tracker: `TouchBegin`
853    /// registers the sequence as a live point, `TouchUpdate` moves it and
854    /// feeds the similarity transform, `TouchEnd`/`TouchCancel` removes it.
855    fn on_quad_touch_event(
856        &self,
857        id: &CardId,
858        controller: &gtk4::EventControllerLegacy,
859        event: &gtk4::gdk::Event,
860        card_drag: Option<&gtk4::GestureDrag>,
861        click: &gtk4::GestureClick,
862    ) {
863        let event_type = event.event_type();
864        if !matches!(event_type, EventType::TouchBegin | EventType::TouchUpdate | EventType::TouchEnd | EventType::TouchCancel) {
865            return;
866        }
867        // `event_sequence` is a plain (non-Option) binding - calling it on
868        // sequence-less events asserts on a NULL pointer, so it may only run
869        // after the touch-type check above.
870        self.sweep_quad_touches(id);
871        let sequence = event.event_sequence();
872        match event_type {
873            EventType::TouchBegin => {
874                // Diagnostic: two TouchBegins for one physical contact
875                // (slot + emulated/pointer dupe) would look identical
876                // here - sequence address + device distinguishes them.
877                info!(card = %id, sequence = ?sequence, "quad touch begin");
878                if let Some(point) = self.event_point_in_workspace(controller, event) {
879                    let cards = self.imp().cards.borrow();
880                    if let Some(entry) = cards.get(id) {
881                        self.activate_quad_touch(entry, sequence, point, card_drag, click);
882                    }
883                }
884                self.emit_geometry_changed();
885            }
886            EventType::TouchUpdate => {
887                let Some(point) = self.event_point_in_workspace(controller, event) else {
888                    return;
889                };
890                {
891                    let cards = self.imp().cards.borrow();
892                    let Some(entry) = cards.get(id) else {
893                        return;
894                    };
895                    let tracked = {
896                        let mut touches = entry.quad_touches.borrow_mut();
897                        let Some(tracked) = touches.iter_mut().find(|touch| touch.sequence == sequence) else {
898                            return;
899                        };
900                        tracked.point = point;
901                        tracked.last_event = Instant::now();
902                        touches.len()
903                    };
904                    debug!(card = %id, tracked, point = ?point, "quad touch update");
905                    self.apply_quad_transform(entry);
906                }
907                self.emit_geometry_changed();
908            }
909            EventType::TouchEnd | EventType::TouchCancel => {
910                self.release_quad_touch(id, &sequence);
911            }
912            _ => {}
913        }
914    }

916    /// Attaches a `sequence-state-changed` watcher to every `Gesture`
917    /// controller in the card subtree that does not have one yet. The
918    /// watcher marks claimed sequences on the entry - the mark
919    /// survives the gesture's own reset, which a `sequence_state`
920    /// query would not. Walked lazily on each sweep so gestures
921    /// children add at runtime are covered too.
922    fn watch_claims(&self, entry: &CardEntry) {
923        let root = entry.scaling.clone().upcast::<gtk4::Widget>();
924        let card_id = entry.card.id();
925        let mut stack = vec![root];
926        while let Some(widget) = stack.pop() {
927            for controller in widget.observe_controllers().iter::<glib::Object>() {
928                let Ok(controller) = controller else {
929                    continue;
930                };
931                let Some(gesture) = controller.downcast_ref::<gtk4::Gesture>() else {
932                    continue;
933                };
934                // The wrapper reference is a temporary - the stable
935                // identity is the underlying GObject pointer.
936                let address = glib::translate::ToGlibPtr::<*mut gtk4::ffi::GtkGesture>::to_glib_none(&gesture).0 as usize;
937                if !entry.watched.borrow_mut().insert(address) {
938                    continue;
939                }
940                let board_weak = self.downgrade();
941                let card_id = card_id.clone();
942                gesture.connect_sequence_state_changed(move |_, sequence, state| {
943                    if state != EventSequenceState::Claimed {
944                        return;
945                    }
946                    if let (Some(board), Some(sequence)) = (board_weak.upgrade(), sequence) {
947                        let cards = board.imp().cards.borrow();
948                        if let Some(entry) = cards.get(&card_id) {
949                            entry.claimed.borrow_mut().insert(sequence.clone(), Instant::now());
950                        }
951                    }
952                });
953            }
954            let mut child = widget.first_child();
955            while let Some(ref sibling) = child {
956                stack.push(sibling.clone());
957                child = sibling.next_sibling();
958            }
959        }
960    }

962    /// Drops tracked touches that are no longer live. Two cases:
963    /// a touch *claimed* by another gesture (its remaining events are routed
964    /// to the claimer only) and a touch that went *silent* - its `TouchEnd`
965    /// already went to a claimer or was lost on the wire, so only the event
966    /// timestamp reveals it is dead. Without the sweep a dead touch lingers
967    /// as a phantom point, keeps the transform "active" for a single new
968    /// finger and distorts radius/angle. Claims are checked against every
969    /// gesture in the card subtree - the board's own click/drag as well as
970    /// child widgets (buttons, toggles), even ones added at runtime.
971    fn sweep_quad_touches(&self, id: &CardId) {
972        let pruned = {
973            let cards = self.imp().cards.borrow();
974            let Some(entry) = cards.get(id) else {
975                return;
976            };
977            self.watch_claims(entry);
978            // Slot ids recycle: a mark older than the stale window can
979            // only belong to a press that already ended - expire it so
980            // the next touch on the same slot is not shadowed.
981            entry.claimed.borrow_mut().retain(|_, ts| ts.elapsed() < Duration::from_millis(QUAD_STALE_MS));
982            let pruned = {
983                let mut touches = entry.quad_touches.borrow_mut();
984                let claimed = entry.claimed.borrow();
985                let before = touches.len();
986                touches.retain(|touch| touch.last_event.elapsed() < Duration::from_millis(QUAD_STALE_MS) && !claimed.contains_key(&touch.sequence));
987                before - touches.len()
988            };
989            if pruned > 0 {
990                info!(card = %id, pruned, "quad touches pruned (claimed or stale)");
991                if entry.quad_touches.borrow().len() >= QUAD_MIN_POINTS {
992                    self.refresh_quad_baseline(entry);
993                } else {
994                    entry.quad_baseline.borrow_mut().take();
995                }
996            }
997            pruned
998        };
999        if pruned > 0 {
1000            self.emit_geometry_changed();
1001        }
1002    }

1004    /// Drops a quad touch: re-baselines for the remaining fingers or
1005    /// finishes the transform and emits the collected signals.
1006    fn release_quad_touch(&self, id: &CardId, sequence: &gtk4::gdk::EventSequence) {
1007        let (card, moved, rotated, scaled) = {
1008            let cards = self.imp().cards.borrow();
1009            let Some(entry) = cards.get(id) else {
1010                return;
1011            };
1012            let active = {
1013                let mut touches = entry.quad_touches.borrow_mut();
1014                let Some(index) = touches.iter().position(|touch| touch.sequence == *sequence) else {
1015                    return;
1016                };
1017                touches.remove(index);
1018                entry.claimed.borrow_mut().remove(sequence);
1019                touches.len()
1020            };
1021            info!(card = %id, active, "quad touch released");
1022            if active >= QUAD_MIN_POINTS {
1023                // A finger left while the transform is still running: snap a
1024                // fresh baseline so the remaining fingers continue smoothly.
1025                self.refresh_quad_baseline(entry);
1026                (entry.card.clone(), false, false, false)
1027            } else {
1028                let baseline = entry.quad_baseline.borrow_mut().take();
1029                self.clamp_entry(entry);
1030                let flags = baseline.map(|b| (b.moved, b.rotated, b.scaled)).unwrap_or_default();
1031                (entry.card.clone(), flags.0, flags.1, flags.2)
1032            }
1033        };
1034        if moved {
1035            card.emit_card_moved();
1036        }
1037        if rotated {
1038            card.emit_card_rotated();
1039        }
1040        if scaled {
1041            card.emit_card_scaled();
1042        }
1043        self.emit_geometry_changed();
1044    }

1046    /// Activates a quad touch: records the finger's workspace position,
1047    /// denies the card's own drag once the transform threshold is reached
1048    /// (no double-move while the quad transform runs) and (re-)takes the
1049    /// baseline snapshot.
1050    fn activate_quad_touch(
1051        &self,
1052        entry: &CardEntry,
1053        sequence: gtk4::gdk::EventSequence,
1054        point: Vec2<Pixels>,
1055        card_drag: Option<&gtk4::GestureDrag>,
1056        click: &gtk4::GestureClick,
1057    ) {
1058        if entry.claimed.borrow().contains_key(&sequence) {
1059            debug!(point = ?point, "quad touch skipped - sequence claimed");
1060            return;
1061        }
1062        let active = {
1063            let mut touches = entry.quad_touches.borrow_mut();
1064            touches.push(TrackedTouch {
1065                sequence,
1066                point,
1067                last_event: Instant::now(),
1068            });
1069            touches.len()
1070        };
1071        info!(active, point = ?point, "quad touch activated");
1072        if active == QUAD_MIN_POINTS {
1073            info!("quad transform active - denying card drag and click");
1074            if let Some(card_drag) = card_drag {
1075                deny_gesture(card_drag);
1076            }
1077            // GestureClick claims every sequence on press; the claim
1078            // steals the first touch's remaining events and makes the
1079            // sweep prune it, so the tracker could never
1080            // reach two live points. Denying releases the claims - both
1081            // sequences stream to the tracker again (raise-on-release
1082            // is suppressed mid-gesture).
1083            deny_gesture(click);
1084        }
1085        if active >= QUAD_MIN_POINTS {
1086            self.refresh_quad_baseline(entry);
1087        }
1088    }

1131    /// Applies the similarity transform fitted between the baseline point
1132    /// cloud and the current points: the card center follows the same
1133    /// rotation/scale/translation the fingers describe.
1134    fn apply_quad_transform(&self, entry: &CardEntry) {
1135        let current: Vec<Vec2<Pixels>> = entry.quad_touches.borrow().iter().map(|touch| touch.point).collect();
1136        let mut baseline_ref = entry.quad_baseline.borrow_mut();
1137        let Some(baseline) = baseline_ref.as_mut() else {
1138            return;
1139        };
1140        let Some(fit) = SimilarityTransform::fit(&baseline.points, &current) else {
1141            return;
1142        };
1143        // Finger crossings through the centroid flip the point-cloud angle
1144        // by ~180° in a single update - far faster than a real hand can
1145        // rotate. Re-baseline on such jumps instead of applying them.
1146        if baseline.applied_angle.signed_delta_to(fit.rotation).abs() > QUAD_ANGLE_JUMP {
1147            drop(baseline_ref);
1148            self.refresh_quad_baseline(entry);
1149            return;
1150        }
1151        baseline.applied_angle = fit.rotation;
1152        let angle = if entry.config.rotatable { fit.rotation } else { RotationRad::ZERO };
1153        let scale = if entry.config.scalable {
1154            ScaleFactor::new(baseline.scale.value() * fit.scale.as_f64()).clamped(&entry.config.scale_limits)
1155        } else {
1156            baseline.scale
1157        };
1158        debug!(
1159            fitted_angle_deg = fit.rotation.to_degrees(),
1160            fitted_scale = fit.scale.as_f64(),
1161            applied_angle_deg = angle.to_degrees(),
1162            applied_scale = scale.value(),
1163            centroid_now = ?fit.current_centroid,
1164            "quad transform computed"
1165        );
1166        let applied_ratio = scale / baseline.scale;
1167        if entry.config.draggable {
1168            // The center follows the gated/clamped transform, not the raw
1169            // fit, so `apply` is recomputed with the effective angle/ratio.
1170            let rel = baseline.position.0 - fit.baseline_centroid;
1171            let position = CardPosition::from(fit.current_centroid + rel.rotated(angle) * f64::from(applied_ratio));
1172            if position != entry.position.get() {
1173                baseline.moved = true;
1174                entry.position.set(position);
1175            }
1176        }
1177        if entry.config.rotatable {
1178            let rendered = baseline.rotation + angle + self.view().rotation;
1179            if rendered.signed_delta_to(entry.rotation_widget.rotation()).abs() > RotationRad::from_degrees(0.01) {
1180                baseline.rotated = true;
1181                entry.rotation_widget.set_rotation(SmearorRotation::from(rendered));
1182            }
1183        }
1184        if entry.config.scalable && scale != entry.scale.get() {
1185            baseline.scaled = true;
1186            entry.scale.set(scale);
1187            entry.scaling.set_scale(scale.value() * self.view_zoom());
1188        }
1189        drop(baseline_ref);
1190        self.apply_position(entry);
1191        if entry.config.bounds_mode == BoundsMode::Contained {
1192            self.clamp_entry(entry);
1193        }
1194    }

1206    /// Moves the card so it follows the pointer.
1207    ///
1208    /// `GestureDrag` reports offsets in the coordinate space of the drag
1209    /// source - which lives *inside* the rotating/scaling transform stack and
1210    /// *moves* with the card. Mapping that delta with `compute_point` under
1211    /// the current transform creates a feedback loop (the reference frame
1212    /// shifts as the card tracks the pointer) that shows up as jitter and
1213    /// roughly half pointer speed. Instead, the pointer's board position is
1214    /// taken straight from the event's surface coordinates, which are
1215    /// independent of the widget's own transform.
1216    fn on_card_drag_update(&self, id: &CardId, gesture: &GestureDrag, offset: Vec2<Pixels>) {
1217        {
1218            let cards = self.imp().cards.borrow();
1219            let Some(entry) = cards.get(id) else {
1220                return;
1221            };
1222            let delta = self.drag_delta(gesture, offset, entry);
1223            let position = entry.drag_start.get().translated(delta);
1224            debug!(card = %id, delta = ?delta, position = ?(position.x().as_f64(), position.y().as_f64()), "drag update");
1225            entry.position.set(position);
1226            self.apply_position(entry);
1227            if entry.config.bounds_mode == BoundsMode::Contained {
1228                self.clamp_entry(entry);
1229            }
1230        }
1231        self.emit_geometry_changed();
1232    }

1360    /// Drag delta in workspace coordinates: current pointer position minus
1361    /// the pointer position captured at drag begin. Falls back to undoing
1362    /// the card's own `rotate . scale` on the widget-space offset before
1363    /// the first allocation (the view factors cancel: the offset lives
1364    /// inside the card's rendered transform).
1365    fn drag_delta(&self, gesture: &GestureDrag, offset: Vec2<Pixels>, entry: &CardEntry) -> Vec2<Pixels> {
1366        if let (Some(now), Some(grab)) = (self.pointer_in_workspace(gesture), entry.drag_grab.get()) {
1367            return now - grab;
1368        }
1369        let rotation = entry.rotation.get();
1370        let scale = entry.scale.get().value();
1371        offset.rotated(rotation) * scale
1372    }

1374    /// Reorders the card layer by `z_index` (stable by insertion sequence).
1375    fn restack(&self) {
1376        let Some(layer) = self.card_layer() else {
1377            return;
1378        };
1379        let cards = self.imp().cards.borrow();
1380        let mut ordered: Vec<&CardEntry> = cards.values().collect();
1381        ordered.sort_by_key(|entry| (entry.config.z_index.unwrap_or(i32::MAX), entry.sequence));
1382        for entry in ordered {
1383            let top_left = self.rendered_footprint(entry).min;
1384            layer.remove(&entry.scaling);
1385            layer.put(&entry.scaling, top_left.x, top_left.y);
1386        }
1387    }

1389    fn wire_gestures(&self, id: &CardId, card: &Card, scaling: &ScalingWidget, rotation_widget: &RotationWidget, config: &CardConfig) {
1390        // Fold changes the card footprint - layout persistence and
1391        // input-region consumers track it through `geometry-changed`.
1392        let board_weak = self.downgrade();
1393        card.connect_local("card-folded", false, move |_| {
1394            if let Some(board) = board_weak.upgrade() {
1395                board.emit_geometry_changed();
1396            }
1397            None
1398        });
1399
1400        // Rotation tracking: RotatedLayout exposes `rotation` as a notify-able
1401        // property, covering both gesture and programmatic changes.
1402        if let Some(manager) = rotation_widget.layout_manager() {
1403            let board_weak = self.downgrade();
1404            let card_id = id.clone();
1405            manager.connect_notify_local(Some("rotation"), move |_, _| {
1406                if let Some(board) = board_weak.upgrade() {
1407                    board.on_card_rotated(&card_id);
1408                }
1409            });
1410        }
1411
1412        // Raise on release: a click raises the card, and a finished drag ends
1413        // raised too. Raising on press would cancel the in-flight sequence.
1414        let click = GestureClick::new();
1415        click.set_button(0);
1416        let board_weak = self.downgrade();
1417        let card_id = id.clone();
1418        click.connect_released(move |_, _, _, _| {
1419            if let Some(board) = board_weak.upgrade() {
1420                board.raise_card(&card_id);
1421            }
1422        });
1423        scaling.add_controller(click.clone());
1424
1425        let mut card_drag: Option<GestureDrag> = None;
1426        if config.draggable {
1427            let drag = GestureDrag::new();
1428            drag.set_button(0);
1429            let board_weak = self.downgrade();
1430            let card_id = id.clone();
1431            drag.connect_drag_begin(move |gesture, _, _| {
1432                if let Some(board) = board_weak.upgrade() {
1433                    board.on_card_drag_begin(&card_id, gesture);
1434                }
1435            });
1436            let board_weak = self.downgrade();
1437            let card_id = id.clone();
1438            drag.connect_drag_update(move |gesture, offset_x, offset_y| {
1439                if let Some(board) = board_weak.upgrade() {
1440                    board.on_card_drag_update(&card_id, gesture, Vec2::new(offset_x, offset_y));
1441                }
1442            });
1443            let board_weak = self.downgrade();
1444            let card_id = id.clone();
1445            drag.connect_drag_end(move |gesture, offset_x, offset_y| {
1446                if let Some(board) = board_weak.upgrade() {
1447                    board.on_card_drag_end(&card_id, gesture, Vec2::new(offset_x, offset_y));
1448                }
1449            });
1450            card.drag_source(config.drag_handle).add_controller(drag.clone());
1451            card_drag = Some(drag);
1452        }
1453
1454        // Resize grip: a dedicated drag on the card's bottom-right
1455        // affordance scales the card around its center. The grip sits deeper
1456        // in the bubble chain than the card drag and the `scaling` click, so
1457        // its press wins the sequence claim before either can react.
1458        if config.resizable
1459            && let Some(grip) = card.resize_grip()
1460        {
1461            let resize = GestureDrag::new();
1462            resize.set_button(0);
1463            // Mark claimed sequences immediately: the grip drag claims a
1464            // press before the quad tracker sees it, and without the mark
1465            // the touch would register as a (phantom) transform point.
1466            let board_weak = self.downgrade();
1467            let card_id = id.clone();
1468            resize.connect_sequence_state_changed(move |_, sequence, state| {
1469                if state != EventSequenceState::Claimed {
1470                    return;
1471                }
1472                if let (Some(board), Some(sequence)) = (board_weak.upgrade(), sequence) {
1473                    let cards = board.imp().cards.borrow();
1474                    if let Some(entry) = cards.get(&card_id) {
1475                        entry.claimed.borrow_mut().insert(sequence.clone(), Instant::now());
1476                    }
1477                }
1478            });
1479            let board_weak = self.downgrade();
1480            let card_id = id.clone();
1481            resize.connect_drag_begin(move |gesture, _, _| {
1482                if let Some(board) = board_weak.upgrade() {
1483                    board.on_card_resize_begin(&card_id, gesture);
1484                }
1485            });
1486            let board_weak = self.downgrade();
1487            let card_id = id.clone();
1488            resize.connect_drag_update(move |gesture, _, _| {
1489                if let Some(board) = board_weak.upgrade() {
1490                    board.on_card_resize_update(&card_id, gesture);
1491                }
1492            });
1493            let board_weak = self.downgrade();
1494            let card_id = id.clone();
1495            resize.connect_drag_end(move |gesture, _, _| {
1496                if let Some(board) = board_weak.upgrade() {
1497                    board.on_card_resize_end(&card_id, gesture);
1498                }
1499            });
1500            grip.add_controller(resize);
1501        }
1502
1503        // Rotate grip: a dedicated drag on the card's bottom-left
1504        // affordance steps the rotation to the next quarter turn in the
1505        // swipe direction. Like the resize grip it sits deeper in the
1506        // bubble chain than the card drag and the `scaling` click, so
1507        // its press wins the sequence claim before either can react.
1508        if config.rotate_grip
1509            && let Some(grip) = card.rotate_grip()
1510        {
1511            let rotate = GestureDrag::new();
1512            rotate.set_button(0);
1513            // Same phantom-touch guard as the resize grip: mark claimed
1514            // sequences so the quad tracker never counts them.
1515            let board_weak = self.downgrade();
1516            let card_id = id.clone();
1517            rotate.connect_sequence_state_changed(move |_, sequence, state| {
1518                if state != EventSequenceState::Claimed {
1519                    return;
1520                }
1521                if let (Some(board), Some(sequence)) = (board_weak.upgrade(), sequence) {
1522                    let cards = board.imp().cards.borrow();
1523                    if let Some(entry) = cards.get(&card_id) {
1524                        entry.claimed.borrow_mut().insert(sequence.clone(), Instant::now());
1525                    }
1526                }
1527            });
1528            let board_weak = self.downgrade();
1529            let card_id = id.clone();
1530            rotate.connect_drag_begin(move |gesture, _, _| {
1531                if let Some(board) = board_weak.upgrade() {
1532                    board.on_card_rotate_begin(&card_id, gesture);
1533                }
1534            });
1535            let board_weak = self.downgrade();
1536            let card_id = id.clone();
1537            rotate.connect_drag_update(move |gesture, _, _| {
1538                if let Some(board) = board_weak.upgrade() {
1539                    board.on_card_rotate_update(&card_id, gesture);
1540                }
1541            });
1542            let board_weak = self.downgrade();
1543            let card_id = id.clone();
1544            rotate.connect_drag_end(move |gesture, _, _| {
1545                if let Some(board) = board_weak.upgrade() {
1546                    board.on_card_rotate_end(&card_id, gesture);
1547                }
1548            });
1549            grip.add_controller(rotate);
1550        }
1551
1552        // Quad transform tracker: a raw `EventControllerLegacy` observing
1553        // every touch sequence on the card. Unlike `GestureSingle` siblings
1554        // it sees all sequences without claiming any, so each finger maps to
1555        // a tracked point and interactive content keeps working. With
1556        // `QUAD_MIN_POINTS` or more live points a similarity transform
1557        // applies - centroid delta moves, radius ratio scales, point-cloud
1558        // rotation rotates - so move, rotate and zoom are one mechanism
1559        // instead of arbitrated gestures. `FBD_NO_QUAD` disables it for
1560        // A/B testing.
1561        let quad_enabled = std::env::var_os("FBD_NO_QUAD").is_none() && (config.draggable || config.rotatable || config.scalable);
1562        if quad_enabled {
1563            let tracker = EventControllerLegacy::new();
1564            let board_weak = self.downgrade();
1565            let card_id = id.clone();
1566            let click_for_tracker = click.clone();
1567            tracker.connect_event(move |controller, event| {
1568                if let Some(board) = board_weak.upgrade() {
1569                    board.on_quad_touch_event(&card_id, controller, event, card_drag.as_ref(), &click_for_tracker);
1570                }
1571                Propagation::Proceed
1572            });
1573            scaling.add_controller(tracker);
1574        }
1575    }

1691    /// Flies the camera onto a card: the target view centers the
1692    /// card's workspace footprint grown by `GOTO_CARD_PADDING`, and
1693    /// `rotation` decides whether the camera keeps its rotation or
1694    /// turns the card upright for the viewer. `transition` picks the
1695    /// flight - `Snap` applies the target instantly, `Glide` flies
1696    /// straight there and `Transit` pulls back to an overview of both
1697    /// regions first (Prezi-style). The target card's
1698    /// `animation_speed_ms` times the flight.
1699    ///
1700    /// A running flight is replaced by the next `goto_card`, and any
1701    /// programmatic or gesture-driven view change ends it. The
1702    /// destination is a pinned view like [`Self::set_view`] -
1703    /// [`Self::reset_view`] releases it.
1704    ///
1705    /// Returns [`BoardError::UnknownCard`] for an id not on the board.
1706    pub fn goto_card(&self, id: &CardId, transition: GotoTransition, rotation: GotoRotation) -> Result<(), BoardError> {
1707        let (footprint, card_rotation, duration) = {
1708            let cards = self.imp().cards.borrow();
1709            let Some(entry) = cards.get(id) else {
1710                return Err(BoardError::UnknownCard(id.clone()));
1711            };
1712            // The camera targets the in-flight glide destination, not
1713            // the mid-flight spot.
1714            let geometry = self.effective_geometry(entry);
1715            (
1716                self.workspace_footprint_at(entry, &geometry),
1717                geometry.rotation,
1718                Duration::from_millis(entry.config.animation_speed_ms),
1719            )
1720        };
1721        let from = self.view();
1722        let target = View {
1723            region: Rect::centered(footprint.center(), footprint.size * ViewScaleFactor::new(GOTO_CARD_PADDING)),
1724            rotation: match rotation {
1725                GotoRotation::Keep => from.rotation,
1726                // Rendered rotation is `card + view`, so a view of
1727                // `-card` draws the card upright on screen.
1728                GotoRotation::Upright => (-card_rotation.to_rotation_rad()).normalized(),
1729            },
1730        };
1731        self.fly_view_to(from, target, transition, duration);
1732        Ok(())
1733    }

2320    /// Shared tail of the pass-to-seat operations: computes the target
2321    /// placement for `target_seat`, persists the seat
2322    /// (`card-seat-changed`) when the card already had an explicit
2323    /// assignment or `assign_seat` asks for it, and glides the card
2324    /// there.
2325    fn glide_card_to_seat(&self, id: &CardId, target_seat: Seat, move_to_edge: bool, assign_seat: bool) -> Result<(), BoardError> {
2326        let anchor = self.imp().seat_anchor.get();
2327        let view = self.view();
2328        let widget = self.widget_bounds();
2329        let zoom = self.view_zoom();
2330        let (geometry, footprint, had_seat, margin) = {
2331            let cards = self.imp().cards.borrow();
2332            let Some(entry) = cards.get(id) else {
2333                return Err(BoardError::UnknownCard(id.clone()));
2334            };
2335            let geometry = self.entry_geometry(entry);
2336            // The footprint is expressed at the target orientation, so
2337            // the placement centers the *rotated* bounding box. `View`
2338            // works in screen space: the rendered size at the rendered
2339            // current rotation, the screen-pixel margin.
2340            let turn = target_seat.side.to_quarter_turn();
2341            let footprint = match anchor {
2342                SeatAnchor::View => footprint_for(self.rendered_size(entry), Rotation::quantized(entry.rotation.get() + view.rotation), turn),
2343                SeatAnchor::Workspace => footprint_for(self.workspace_card_size(entry), geometry.rotation, turn),
2344            };
2345            (
2346                geometry,
2347                footprint,
2348                entry.seat.get().is_some(),
2349                match anchor {
2350                    SeatAnchor::View => entry.config.anchor_margin.as_f64() * zoom,
2351                    SeatAnchor::Workspace => entry.config.anchor_margin.as_f64(),
2352                },
2353            )
2354        };
2355
2356        if had_seat || assign_seat {
2357            self.set_card_seat(id, Some(target_seat))?;
2358        }
2359
2360        let bounds = self.seat_bounds();
2361        let placement = if move_to_edge && !bounds.is_degenerate() {
2362            // Every other card physically occupies space near the seat:
2363            // the fan-out solver slips the passed card into the nearest
2364            // free lane slot beside them instead of stacking on top.
2365            // `View` runs the whole lane in screen space.
2366            let taken: Vec<(Vec2<Pixels>, Vec2<Pixels>)> = self
2367                .imp()
2368                .cards
2369                .borrow()
2370                .iter()
2371                .filter(|(other_id, _)| *other_id != id)
2372                .map(|(_, entry)| {
2373                    // Cards gliding to a seat count at their
2374                    // destination: a pass issued while a sibling is
2375                    // still moving fans out around its landing spot
2376                    // instead of stacking on it.
2377                    let geometry = self.effective_geometry(entry);
2378                    let footprint = match anchor {
2379                        SeatAnchor::View => self.rendered_footprint_at(entry, &geometry),
2380                        SeatAnchor::Workspace => self.workspace_footprint_at(entry, &geometry),
2381                    };
2382                    (footprint.center(), footprint.size)
2383                })
2384                .collect();
2385            target_seat.fanned_placement(bounds, footprint, margin, &taken, margin)
2386        } else {
2387            target_seat.placement(bounds, footprint, margin)
2388        };
2389        let target = CardGeometry {
2390            // Degenerate bounds (unrealized board) fall back to
2391            // rotation-in-place, like `clamp_entry` early-returns.
2392            position: if move_to_edge && !bounds.is_degenerate() {
2393                CardPosition::from(match anchor {
2394                    SeatAnchor::View => view.to_workspace(placement.center, widget),
2395                    SeatAnchor::Workspace => placement.center,
2396                })
2397            } else {
2398                geometry.position
2399            },
2400            // Under `View` the placement's quarter turn is a *screen*
2401            // orientation: subtracting the camera rotation makes the
2402            // rendered card land on it. Under `Workspace` it is the
2403            // table-space rotation already.
2404            rotation: match anchor {
2405                SeatAnchor::View => Rotation::quantized(placement.rotation.to_rotation_rad() - view.rotation),
2406                SeatAnchor::Workspace => placement.rotation,
2407            },
2408            scale: geometry.scale,
2409        };
2410
2411        self.glide_card_geometry(id, target)
2412    }

2414    /// Glides a card towards `target` on the frame clock. Each tick
2415    /// applies the eased `CardGeometry` through the regular
2416    /// `set_card_geometry` path, so re-centering, clamping and signal
2417    /// emission stay identical to an instant update.
2418    ///
2419    /// Starting a new glide bumps the card's `layout_generation`;
2420    /// ticks of a superseded animation notice the stale generation and
2421    /// stop themselves.
2422    ///
2423    /// Returns [`BoardError::UnknownCard`] for an id not on the board.
2424    pub fn glide_card_geometry(&self, id: &CardId, target: CardGeometry) -> Result<(), BoardError> {
2425        let (scaling, from, duration, generation) = {
2426            let cards = self.imp().cards.borrow();
2427            let Some(entry) = cards.get(id) else {
2428                return Err(BoardError::UnknownCard(id.clone()));
2429            };
2430            let generation = entry.layout_generation.get().incremented();
2431            entry.layout_generation.set(generation);
2432            // Seat fan-out reads the destination as the card's
2433            // effective geometry, so passes during the glide fan out
2434            // around the landing spot instead of the mid-flight one.
2435            entry.glide_target.set(Some((generation, target)));
2436            (
2437                entry.scaling.clone(),
2438                self.entry_geometry(entry),
2439                Duration::from_millis(entry.config.animation_speed_ms),
2440                generation,
2441            )
2442        };
2443        let mut animation = TimedAnimation::new(Span::new(from, target), duration, EasingFunction::EaseInOut);
2444        animation.start();
2445        let board = self.clone();
2446        let id = id.clone();
2447        scaling.add_tick_callback(move |_, _frame_clock| {
2448            let Some(geometry) = animation.value() else {
2449                return ControlFlow::Break;
2450            };
2451            {
2452                let cards = board.imp().cards.borrow();
2453                match cards.get(&id) {
2454                    Some(entry) if entry.layout_generation.get() == generation => {}
2455                    _ => return ControlFlow::Break,
2456                }
2457            }
2458            board.set_card_geometry(&id, &geometry);
2459            if animation.is_complete() {
2460                let cards = board.imp().cards.borrow();
2461                if let Some(entry) = cards.get(&id) {
2462                    entry.glide_target.take();
2463                }
2464                return ControlFlow::Break;
2465            }
2466            ControlFlow::Continue
2467        });
2468        Ok(())
2469    }
