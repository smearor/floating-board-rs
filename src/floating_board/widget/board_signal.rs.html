5/// The signals `FloatingBoard` emits, as a type-safe alternative to
6/// stringly-typed signal names. Used by `signals()` in the widget
7/// implementation and by every `emit_by_name` call site so each name lives
8/// in exactly one place.
9#[derive(Debug, Clone, Copy, PartialEq, Eq)]
10pub(crate) enum BoardSignal {
11    /// Emitted with the `Card` after a card was added to the board.
12    CardAdded,
13    /// Emitted with the `Card` after a card was removed from the board.
14    CardRemoved,
15    /// Emitted with the `Card` when `set_card_visible` toggles visibility.
16    CardVisibilityChanged,
17    /// Emitted when any card's position, rotation, scale or z-order changed.
18    GeometryChanged,
19}
20
21/// Every signal of the board, for `ObjectImpl::signals()`.
22pub(crate) const BOARD_SIGNALS: &[BoardSignal] = &[
23    BoardSignal::CardAdded,
24    BoardSignal::CardRemoved,
25    BoardSignal::CardVisibilityChanged,
26    BoardSignal::GeometryChanged,
27];
28
29impl BoardSignal {
30    /// The static GTK signal name (`"card-added"`).
31    pub(crate) fn name(self) -> &'static str {
32        match self {
33            Self::CardAdded => "card-added",
34            Self::CardRemoved => "card-removed",
35            Self::CardVisibilityChanged => "card-visibility-changed",
36            Self::GeometryChanged => "geometry-changed",
37        }
38    }
39
40    /// Builds the GTK signal declaration with its parameter types - the
41    /// card lifecycle signals carry the affected `Card`.
42    pub(crate) fn signal(self) -> Signal {
43        match self {
44            Self::GeometryChanged => Signal::builder(self.name()).build(),
45            Self::CardAdded | Self::CardRemoved | Self::CardVisibilityChanged => Signal::builder(self.name()).param_types([Card::static_type()]).build(),
46        }
47    }
48}
