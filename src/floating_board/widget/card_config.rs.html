6/// Placement and interaction configuration of a card.
7///
8/// Passed to `FloatingBoard::add_card`; all fields have defaults so
9/// `CardConfig::default()` yields a fully interactive, draggable card anchored
10/// at the board center.
11#[derive(Debug, Clone, PartialEq)]
12#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
13pub struct CardConfig {
14    /// Title shown in the header strip; falls back to a localized default when
15    /// the header is visible and no title is set.
16    pub title: Option<String>,
17    /// Initial placement anchor, resolved on first allocation.
18    pub position: CardAnchor,
19    /// Whether the card can be moved by dragging.
20    pub draggable: bool,
21    /// Whether the card can be rotated via the two-finger rotate gesture.
22    pub rotatable: bool,
23    /// Whether the card can be scaled via the pinch gesture.
24    pub scalable: bool,
25    /// Whether the card can be folded down to its header.
26    pub foldable: bool,
27    /// Whether the header shows a close button that removes the card.
28    pub closable: bool,
29    /// Collapses the header to a slim grip when no title/buttons are needed.
30    pub headerless: bool,
31    /// Which part of the card acts as the drag affordance.
32    pub drag_handle: DragHandle,
33    /// Inclusive scale limits for the pinch gesture; default `0.5 - 2.0`.
34    pub scale_limits: ScaleBounds,
35    /// Minimum overlap in pixels a card must keep inside the board bounds.
36    pub min_visible: MinVisible,
37    /// Optional initial z-ordering; `None` appends the card on top.
38    pub z_index: Option<i32>,
39}
40
41impl Default for CardConfig {
42    fn default() -> Self {
43        Self {
44            title: None,
45            position: CardAnchor::Center,
46            draggable: true,
47            rotatable: true,
48            scalable: true,
49            foldable: false,
50            closable: false,
51            headerless: false,
52            drag_handle: DragHandle::Header,
53            scale_limits: ScaleBounds::default(),
54            min_visible: MinVisible::default(),
55            z_index: None,
56        }
57    }
58}
