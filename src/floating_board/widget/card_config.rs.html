17/// Placement and interaction configuration of a card.
18///
19/// Passed to `FloatingBoard::add_card`; all fields have defaults so
20/// `CardConfig::default()` yields a fully interactive, draggable card anchored
21/// at the board center.
22#[derive(Debug, Clone, PartialEq)]
23#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
24pub struct CardConfig {
25    /// Title shown in the header strip; falls back to a localized default when
26    /// the header is visible and no title is set.
27    pub title: Option<String>,
28    /// Initial placement anchor, resolved on first allocation.
29    pub position: CardAnchor,
30    /// Whether the card can be moved by dragging.
31    pub draggable: bool,
32    /// Whether the card can be rotated via the two-finger rotate gesture.
33    pub rotatable: bool,
34    /// Whether the card can be scaled via the pinch gesture.
35    pub scalable: bool,
36    /// Whether the card shows a bottom-right resize grip that scales the
37    /// card around its center when dragged (mouse and touch).
38    pub resizable: bool,
39    /// Whether the card shows a bottom-left rotate grip that steps the
40    /// rotation to the next quarter turn in the swipe direction - one
41    /// quarter turn per drag, no matter how far the swipe goes (mouse
42    /// and touch). Works independently of `rotatable`, which gates the
43    /// free two-finger rotate.
44    #[cfg_attr(feature = "serde", serde(default))]
45    pub rotate_grip: bool,
46    /// Whether the card can be folded down to its header.
47    pub foldable: bool,
48    /// Whether the header shows a close button.
49    pub closable: bool,
50    /// What the close button does - remove the card (default) or hide it
51    /// so a menu can re-open it later.
52    pub close_action: CardCloseAction,
53    /// Collapses the header to a slim grip when no title/buttons are needed.
54    pub headerless: bool,
55    /// Which part of the card acts as the drag affordance.
56    pub drag_handle: DragHandle,
57    /// Margin in pixels an edge or corner anchor keeps between the card's
58    /// footprint and the board edge; default `ANCHOR_MARGIN` (24 px).
59    /// Ignored for `Center` and `Explicit` placements.
60    #[cfg_attr(feature = "serde", serde(default = "default_anchor_margin"))]
61    pub anchor_margin: Pixels,
62    /// Inclusive scale limits for the pinch gesture; default `0.5 - 2.0`.
63    pub scale_limits: ScaleBounds,
64    /// Minimum overlap in pixels a card must keep inside the board bounds.
65    pub min_visible: MinVisible,
66    /// How strictly the card's footprint is kept inside the board bounds;
67    /// default `Loose` (only `min_visible` is enforced).
68    #[cfg_attr(feature = "serde", serde(default))]
69    pub bounds_mode: BoundsMode,
70    /// Per-card color scheme: a [`CardPalette`] mapped onto scoped CSS
71    /// at build time. `None` (or a palette with unfilled slots) keeps
72    /// the themed/default-card colors.
73    #[cfg_attr(feature = "serde", serde(default))]
74    pub palette: Option<CardPalette>,
75    /// Space in pixels between the card border and the content widget;
76    /// default `DEFAULT_CONTENT_PADDING` (10 px, the former fixed CSS
77    /// value). `0` lets the content sit flush with the card edge.
78    #[cfg_attr(feature = "serde", serde(default = "default_content_padding"))]
79    pub content_padding: Pixels,
80    /// Optional initial z-ordering; `None` appends the card on top.
81    pub z_index: Option<i32>,
82    /// Hard upper bound on the card's measured width; `None` lets
83    /// the card grow with its content. Caps growth caused by changing content
84    /// (e.g. a number gaining a digit): the layout allocates at most this
85    /// size and overflowing content is clipped by the default CSS.
86    #[cfg_attr(feature = "serde", serde(default))]
87    pub max_width: Option<Pixels>,
88    /// Hard upper bound on the card's measured height; see
89    /// `max_width`.
90    #[cfg_attr(feature = "serde", serde(default))]
91    pub max_height: Option<Pixels>,
92    /// Duration of rotation animations (gesture snapping and
93    /// `set_rotation_with_animation`) in milliseconds; default 500.
94    #[cfg_attr(feature = "serde", serde(default = "default_animation_speed_ms"))]
95    pub animation_speed_ms: u64,
96    /// Optional seat assignment on the tabletop edge the card belongs
97    /// to - decoupled from the card's visual rotation: rotating the
98    /// board or freely spinning the card does not change which user
99    /// owns it. `None` leaves the card unassigned; seat strategies and
100    /// `rotate_to_next_user` fall back to the nearest side of its
101    /// rotation.
102    #[cfg_attr(feature = "serde", serde(default))]
103    pub seat: Option<Seat>,
104}

106impl Default for CardConfig {
107    fn default() -> Self {
108        Self {
109            title: None,
110            position: CardAnchor::Center,
111            draggable: true,
112            rotatable: true,
113            scalable: true,
114            resizable: false,
115            rotate_grip: false,
116            foldable: false,
117            closable: false,
118            close_action: CardCloseAction::Remove,
119            headerless: false,
120            drag_handle: DragHandle::Header,
121            anchor_margin: ANCHOR_MARGIN,
122            scale_limits: ScaleBounds::default(),
123            min_visible: MinVisible::default(),
124            bounds_mode: BoundsMode::default(),
125            palette: None,
126            content_padding: DEFAULT_CONTENT_PADDING,
127            z_index: None,
128            max_width: None,
129            max_height: None,
130            animation_speed_ms: 500,
131            seat: None,
132        }
133    }
134}
