21/// Board-owned state of one card: the transform stack widgets plus the
22/// geometry cells. `Card` objects carry only a weak backlink - this entry is
23/// the single owner of position/scale state (rotation lives inside the
24/// `RotationWidget`'s layout and is read on demand).
25pub(crate) struct CardEntry {
26    /// The card view widget.
27    pub card: Card,
28    /// Outer transform: scale wrapper placed on the `gtk4::Fixed` layer.
29    pub scaling: ScalingWidget,
30    /// Inner transform: rotation container holding the card.
31    pub rotation_widget: RotationWidget,
32    /// Immutable configuration passed at `add_card`.
33    pub config: CardConfig,
34    /// Current center position in board coordinates.
35    pub position: Cell<CardPosition>,
36    /// Current scale factor.
37    pub scale: Cell<ScaleFactor>,
38    /// Anchor awaiting resolution at the next board allocation.
39    pub pending_anchor: RefCell<Option<CardAnchor>>,
40    /// Card center at the start of the active drag gesture.
41    pub drag_start: Cell<CardPosition>,
42    /// Pointer position in board coordinates at drag begin.
43    pub drag_grab: Cell<Option<Vec2<Pixels>>>,
44    /// Pointer position in board coordinates at resize-drag begin;
45    /// `Some` while a resize gesture on the card's grip is active.
46    pub resize_grab: Cell<Option<Vec2<Pixels>>>,
47    /// Scale factor captured at resize-drag begin; updates apply the
48    /// ratio between the current and the grab radial distance.
49    pub resize_scale: Cell<ScaleFactor>,
50    /// Live touch points for the quad transform, one entry per finger
51    /// currently on the card. The tracker's `EventControllerLegacy` keeps
52    /// this in sync - a plain event observer that sees every sequence
53    /// without claiming it, so interactive content keeps working (unlike
54    /// `GestureSingle`, which locks onto the first sequence it sees and
55    /// cannot distribute touches across siblings).
56    pub quad_touches: RefCell<Vec<TrackedTouch>>,
57    /// Baseline for the active quad transform.
58    pub quad_baseline: RefCell<Option<QuadBaseline>>,
59    /// Sequences any gesture in the card subtree has claimed, each
60    /// with the claim timestamp. Claims reset when a tap ends, so the
61    /// state is not queryable retroactively - the mark keeps it: a
62    /// claimed sequence's `TouchEnd` went to the claimer, never to
63    /// the tracker, so its quad touch is a phantom. `EventSequence`
64    /// is only a touch-slot id and is reused for the next press on
65    /// the same slot, so marks expire after the stale window -
66    /// afterwards the phantom would die on the stale check anyway.
67    pub claimed: RefCell<HashMap<EventSequence, Instant>>,
68    /// Gesture object addresses already carrying a
69    /// `sequence-state-changed` watcher, so the lazy walk connects
70    /// each controller only once.
71    pub watched: RefCell<HashSet<usize>>,
72    /// Insertion sequence number for stable z-ordering.
73    pub sequence: SequenceNumber,
74}

76/// Smallest radial grab distance the resize drag accepts as its
77/// reference - a press (nearly) on the card center would make tiny
78/// pointer moves blow up the ratio.
79const RESIZE_MIN_RADIUS: Pixels = Pixels::new(4.0);
80
81impl CardEntry {
82    /// Scale target for a resize-drag sample at pointer position `now`
83    /// (board coordinates): the grab-time scale multiplied by the
84    /// radial-distance ratio to the card center. `None` while no
85    /// resize grab is active.
86    pub(crate) fn resize_target(&self, now: Vec2<Pixels>) -> Option<ScaleFactor> {
87        let grab = self.resize_grab.get()?;
88        Some(resize_scale(grab, now, self.center(), self.resize_scale.get(), &self.config.scale_limits))
89    }
90
91    /// Current center position as a board-coordinate vector.
92    fn center(&self) -> Vec2<Pixels> {
93        Vec2::<Pixels>::from(self.position.get())
94    }
95}
96
97/// Scale target for one resize-drag sample: `start` multiplied by the
98/// ratio between the pointer's current and grab-time radial distance to
99/// the card `center`, clamped to `limits`. Radial distance is
100/// rotation-invariant, so a rotated card resizes identically to an
101/// unrotated one. `start` is returned while the grab sits too close to
102/// `center` to yield a stable ratio.
103fn resize_scale(grab: Vec2<Pixels>, now: Vec2<Pixels>, center: Vec2<Pixels>, start: ScaleFactor, limits: &ScaleBounds) -> ScaleFactor {
104    let grab_radius = grab.distance(&center);
105    if grab_radius < RESIZE_MIN_RADIUS {
106        return start;
107    }
108    // `Pixels / Pixels` yields the dimensionless same-unit ratio.
109    let ratio = now.distance(&center) / grab_radius;
110    ScaleFactor::new(start.value() * ratio).clamped(limits)
111}

113#[cfg(test)]
114mod tests {
115    use super::*;
116
117    fn resize(grab: (f64, f64), now: (f64, f64), start: f64) -> f64 {
118        resize_scale(
119            Vec2::<Pixels>::from(grab),
120            Vec2::<Pixels>::from(now),
121            Vec2::<Pixels>::ZERO,
122            ScaleFactor::new(start),
123            &ScaleBounds::default(),
124        )
125        .value()
126    }
127
128    #[test]
129    fn doubling_the_radial_distance_doubles_the_scale() {
130        assert!((resize((100.0, 0.0), (200.0, 0.0), 1.0) - 2.0).abs() < f64::EPSILON);
131    }
132
133    #[test]
134    fn halving_the_radial_distance_halves_the_scale() {
135        assert!((resize((100.0, 0.0), (50.0, 0.0), 1.0) - 0.5).abs() < f64::EPSILON);
136    }
137
138    #[test]
139    fn resize_ratio_is_rotation_invariant() {
140        // Same radius, different angle: a rotated card scales identically.
141        let radial = resize((100.0, 0.0), (200.0, 0.0), 1.0);
142        let diagonal = resize((70.71067811865476, 70.71067811865476), (141.4213562373095, 141.4213562373095), 1.0);
143        assert!((radial - diagonal).abs() < 1e-9);
144    }
145
146    #[test]
147    fn resize_clamps_to_scale_bounds() {
148        assert!((resize((100.0, 0.0), (1_000_000.0, 0.0), 1.0) - 2.0).abs() < f64::EPSILON);
149        assert!((resize((100.0, 0.0), (0.000001, 0.0), 1.0) - 0.5).abs() < f64::EPSILON);
150    }
151
152    #[test]
153    fn grab_on_the_center_keeps_the_start_scale() {
154        assert!((resize((1.0, 0.0), (500.0, 0.0), 1.5) - 1.5).abs() < f64::EPSILON);
155    }
