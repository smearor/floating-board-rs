20/// Board-owned state of one card: the transform stack widgets plus the
21/// geometry cells. `Card` objects carry only a weak backlink - this entry is
22/// the single owner of position/scale state (rotation lives inside the
23/// `RotationWidget`'s layout and is read on demand).
24pub(crate) struct CardEntry {
25    /// The card view widget.
26    pub card: Card,
27    /// Outer transform: scale wrapper placed on the `gtk4::Fixed` layer.
28    pub scaling: ScalingWidget,
29    /// Inner transform: rotation container holding the card.
30    pub rotation_widget: RotationWidget,
31    /// Immutable configuration passed at `add_card`.
32    pub config: CardConfig,
33    /// Current center position in board coordinates.
34    pub position: Cell<CardPosition>,
35    /// Current scale factor.
36    pub scale: Cell<ScaleFactor>,
37    /// Anchor awaiting resolution at the next board allocation.
38    pub pending_anchor: RefCell<Option<CardAnchor>>,
39    /// Card center at the start of the active drag gesture.
40    pub drag_start: Cell<CardPosition>,
41    /// Pointer position in board coordinates at drag begin.
42    pub drag_grab: Cell<Option<Vec2<Pixels>>>,
43    /// Live touch points for the quad transform, one entry per finger
44    /// currently on the card. The tracker's `EventControllerLegacy` keeps
45    /// this in sync - a plain event observer that sees every sequence
46    /// without claiming it, so interactive content keeps working (unlike
47    /// `GestureSingle`, which locks onto the first sequence it sees and
48    /// cannot distribute touches across siblings).
49    pub quad_touches: RefCell<Vec<TrackedTouch>>,
50    /// Baseline for the active quad transform.
51    pub quad_baseline: RefCell<Option<QuadBaseline>>,
52    /// Sequences any gesture in the card subtree has claimed, each
53    /// with the claim timestamp. Claims reset when a tap ends, so the
54    /// state is not queryable retroactively - the mark keeps it: a
55    /// claimed sequence's `TouchEnd` went to the claimer, never to
56    /// the tracker, so its quad touch is a phantom. `EventSequence`
57    /// is only a touch-slot id and is reused for the next press on
58    /// the same slot, so marks expire after the stale window -
59    /// afterwards the phantom would die on the stale check anyway.
60    pub claimed: RefCell<HashMap<EventSequence, Instant>>,
61    /// Gesture object addresses already carrying a
62    /// `sequence-state-changed` watcher, so the lazy walk connects
63    /// each controller only once.
64    pub watched: RefCell<HashSet<usize>>,
65    /// Insertion sequence number for stable z-ordering.
66    pub sequence: SequenceNumber,
67}
