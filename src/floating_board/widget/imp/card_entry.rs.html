23/// Board-owned state of one card: the transform stack widgets plus the
24/// geometry cells. `Card` objects carry only a weak backlink - this entry is
25/// the single owner of position/scale state (rotation lives inside the
26/// `RotationWidget`'s layout and is read on demand).
27pub(crate) struct CardEntry {
28    /// The card view widget.
29    pub card: Card,
30    /// Outer transform: scale wrapper placed on the `gtk4::Fixed` layer.
31    pub scaling: ScalingWidget,
32    /// Inner transform: rotation container holding the card.
33    pub rotation_widget: RotationWidget,
34    /// Immutable configuration passed at `add_card`.
35    pub config: CardConfig,
36    /// Current center position in board coordinates.
37    pub position: Cell<CardPosition>,
38    /// Current scale factor.
39    pub scale: Cell<ScaleFactor>,
40    /// Anchor awaiting resolution at the next board allocation.
41    pub pending_anchor: RefCell<Option<CardAnchor>>,
42    /// Card center at the start of the active drag gesture.
43    pub drag_start: Cell<CardPosition>,
44    /// Pointer position in board coordinates at drag begin.
45    pub drag_grab: Cell<Option<Vec2<Pixels>>>,
46    /// Pointer position in board coordinates at resize-drag begin;
47    /// `Some` while a resize gesture on the card's grip is active.
48    pub resize_grab: Cell<Option<Vec2<Pixels>>>,
49    /// Scale factor captured at resize-drag begin; updates apply the
50    /// ratio between the current and the grab radial distance.
51    pub resize_scale: Cell<ScaleFactor>,
52    /// Pointer position in board coordinates at rotate-drag begin;
53    /// `Some` while a rotate gesture on the card's grip is active.
54    pub rotate_grab: Cell<Option<Vec2<Pixels>>>,
55    /// Whether the active rotate drag already consumed its quarter-turn
56    /// step - the grip performs at most one step per drag, no matter
57    /// how far the swipe continues past the trigger.
58    pub rotate_stepped: Cell<bool>,
59    /// Live touch points for the quad transform, one entry per finger
60    /// currently on the card. The tracker's `EventControllerLegacy` keeps
61    /// this in sync - a plain event observer that sees every sequence
62    /// without claiming it, so interactive content keeps working (unlike
63    /// `GestureSingle`, which locks onto the first sequence it sees and
64    /// cannot distribute touches across siblings).
65    pub quad_touches: RefCell<Vec<TrackedTouch>>,
66    /// Baseline for the active quad transform.
67    pub quad_baseline: RefCell<Option<QuadBaseline>>,
68    /// Sequences any gesture in the card subtree has claimed, each
69    /// with the claim timestamp. Claims reset when a tap ends, so the
70    /// state is not queryable retroactively - the mark keeps it: a
71    /// claimed sequence's `TouchEnd` went to the claimer, never to
72    /// the tracker, so its quad touch is a phantom. `EventSequence`
73    /// is only a touch-slot id and is reused for the next press on
74    /// the same slot, so marks expire after the stale window -
75    /// afterwards the phantom would die on the stale check anyway.
76    pub claimed: RefCell<HashMap<EventSequence, Instant>>,
77    /// Gesture object addresses already carrying a
78    /// `sequence-state-changed` watcher, so the lazy walk connects
79    /// each controller only once.
80    pub watched: RefCell<HashSet<usize>>,
81    /// Insertion sequence number for stable z-ordering.
82    pub sequence: SequenceNumber,
83    /// Seat the card is assigned to on the tabletop edge -
84    /// `Some` mirrors `config.seat` until a seat-changing operation
85    /// (pass to next user) updates it.
86    pub seat: Cell<Option<Seat>>,
87}

89/// Smallest radial grab distance the resize drag accepts as its
90/// reference - a press (nearly) on the card center would make tiny
91/// pointer moves blow up the ratio.
92const RESIZE_MIN_RADIUS: Pixels = Pixels::new(4.0);
93
94/// Angular sweep around the card center that triggers the rotate
95/// grip's quarter-turn step (5 degrees) - above pointer jitter, well
96/// below the quarter turn itself.
97const ROTATE_TRIGGER: RotationRad = RotationRad::new(5.0 * std::f64::consts::PI / 180.0);
98
99/// Smallest pointer-to-center distance for a meaningful swipe
100/// direction - a sample (nearly) on the card center has no stable
101/// angle.
102const ROTATE_MIN_RADIUS: Pixels = Pixels::new(8.0);
103
104impl CardEntry {
105    /// Scale target for a resize-drag sample at pointer position `now`
106    /// (board coordinates): the grab-time scale multiplied by the
107    /// radial-distance ratio to the card center. `None` while no
108    /// resize grab is active.
109    pub(crate) fn resize_target(&self, now: Vec2<Pixels>) -> Option<ScaleFactor> {
110        let grab = self.resize_grab.get()?;
111        Some(resize_scale(grab, now, self.center(), self.resize_scale.get(), &self.config.scale_limits))
112    }
113
114    /// Quarter-turn target for a rotate-drag sample at pointer position
115    /// `now` (board coordinates): once the pointer's angular sweep
116    /// around the card center passes `ROTATE_TRIGGER`, the next 90
117    /// degree boundary in the sweep direction off `current`. `None`
118    /// while no rotate grab is active, while the swipe is still below
119    /// the trigger, or once the drag's single step was taken.
120    pub(crate) fn rotate_target(&self, now: Vec2<Pixels>, current: RotationRad) -> Option<RotationRad> {
121        if self.rotate_stepped.get() {
122            return None;
123        }
124        let grab = self.rotate_grab.get()?;
125        let center = self.center();
126        if grab.distance(&center) < ROTATE_MIN_RADIUS || now.distance(&center) < ROTATE_MIN_RADIUS {
127            return None;
128        }
129        let sweep = (grab - center).direction().signed_delta_to((now - center).direction());
130        if sweep.abs() < ROTATE_TRIGGER {
131            return None;
132        }
133        Some(next_quarter_turn(current, sweep > RotationRad::ZERO))
134    }
135
136    /// Current center position as a board-coordinate vector.
137    fn center(&self) -> Vec2<Pixels> {
138        Vec2::<Pixels>::from(self.position.get())
139    }
140}
141
142/// Scale target for one resize-drag sample: `start` multiplied by the
143/// ratio between the pointer's current and grab-time radial distance to
144/// the card `center`, clamped to `limits`. Radial distance is
145/// rotation-invariant, so a rotated card resizes identically to an
146/// unrotated one. `start` is returned while the grab sits too close to
147/// `center` to yield a stable ratio.
148fn resize_scale(grab: Vec2<Pixels>, now: Vec2<Pixels>, center: Vec2<Pixels>, start: ScaleFactor, limits: &ScaleBounds) -> ScaleFactor {
149    let grab_radius = grab.distance(&center);
150    if grab_radius < RESIZE_MIN_RADIUS {
151        return start;
152    }
153    // `Pixels / Pixels` yields the dimensionless same-unit ratio.
154    let ratio = now.distance(&center) / grab_radius;
155    ScaleFactor::new(start.value() * ratio).clamped(limits)
156}
157
158/// The quarter-turn boundary next in `direction` from `current`: the
159/// smallest 90 degree multiple past `current` when `clockwise`, the
160/// largest below it when not. A `current` already on a boundary steps
161/// a full quarter turn; the epsilon keeps float noise on snapped
162/// angles from staying put.
163fn next_quarter_turn(current: RotationRad, clockwise: bool) -> RotationRad {
164    const QUARTER: f64 = 90.0;
165    const EPSILON: f64 = 1e-6;
166    let turns = current.to_degrees() / QUARTER;
167    let next = if clockwise {
168        (turns + EPSILON).floor() + 1.0
169    } else {
170        (turns - EPSILON).ceil() - 1.0
171    };
172    RotationRad::from_degrees(next * QUARTER)
173}

175#[cfg(test)]
176mod tests {
177    use super::*;
178
179    fn resize(grab: (f64, f64), now: (f64, f64), start: f64) -> f64 {
180        resize_scale(
181            Vec2::<Pixels>::from(grab),
182            Vec2::<Pixels>::from(now),
183            Vec2::<Pixels>::ZERO,
184            ScaleFactor::new(start),
185            &ScaleBounds::default(),
186        )
187        .value()
188    }
189
190    #[test]
191    fn doubling_the_radial_distance_doubles_the_scale() {
192        assert!((resize((100.0, 0.0), (200.0, 0.0), 1.0) - 2.0).abs() < f64::EPSILON);
193    }
194
195    #[test]
196    fn halving_the_radial_distance_halves_the_scale() {
197        assert!((resize((100.0, 0.0), (50.0, 0.0), 1.0) - 0.5).abs() < f64::EPSILON);
198    }
199
200    #[test]
201    fn resize_ratio_is_rotation_invariant() {
202        // Same radius, different angle: a rotated card scales identically.
203        let radial = resize((100.0, 0.0), (200.0, 0.0), 1.0);
204        let diagonal = resize((70.71067811865476, 70.71067811865476), (141.4213562373095, 141.4213562373095), 1.0);
205        assert!((radial - diagonal).abs() < 1e-9);
206    }
207
208    #[test]
209    fn resize_clamps_to_scale_bounds() {
210        assert!((resize((100.0, 0.0), (1_000_000.0, 0.0), 1.0) - 2.0).abs() < f64::EPSILON);
211        assert!((resize((100.0, 0.0), (0.000001, 0.0), 1.0) - 0.5).abs() < f64::EPSILON);
212    }
213
214    #[test]
215    fn grab_on_the_center_keeps_the_start_scale() {
216        assert!((resize((1.0, 0.0), (500.0, 0.0), 1.5) - 1.5).abs() < f64::EPSILON);
217    }
218
219    fn next(current: f64, clockwise: bool) -> f64 {
220        next_quarter_turn(RotationRad::from_degrees(current), clockwise).to_degrees()
221    }
222
223    #[test]
224    fn snapped_rotation_steps_a_full_quarter_turn() {
225        assert_eq!(next(0.0, true), 90.0);
226        assert_eq!(next(90.0, true), 180.0);
227        assert_eq!(next(90.0, false), 0.0);
228        assert_eq!(next(0.0, false), -90.0);
229    }
230
231    #[test]
232    fn free_rotation_lands_on_the_next_boundary() {
233        assert_eq!(next(30.0, true), 90.0);
234        assert_eq!(next(30.0, false), 0.0);
235        assert_eq!(next(-10.0, false), -90.0);
236        assert_eq!(next(269.0, true), 270.0);
237    }
238
239    #[test]
240    fn float_noise_on_a_boundary_still_steps() {

242        assert_eq!(next(90.0 - 1e-9, false), 0.0);
243    }
