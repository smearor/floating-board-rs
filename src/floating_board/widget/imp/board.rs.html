35/// Board-owned state of one card: the transform stack widgets plus the
36/// geometry cells. `Card` objects carry only a weak backlink - this entry is
37/// the single owner of position/scale state (rotation lives inside the
38/// `RotationWidget`'s layout and is read on demand).
39pub(crate) struct CardEntry {
40    /// The card view widget.
41    pub card: Card,
42    /// Outer transform: scale wrapper placed on the `gtk4::Fixed` layer.
43    pub scaling: ScalingWidget,
44    /// Inner transform: rotation container holding the card.
45    pub rotation_widget: RotationWidget,
46    /// Immutable configuration passed at `add_card`.
47    pub config: CardConfig,
48    /// Current center position in board coordinates.
49    pub position: Cell<CardPosition>,
50    /// Current scale factor.
51    pub scale: Cell<ScaleFactor>,
52    /// Anchor awaiting resolution at the next board allocation.
53    pub pending_anchor: RefCell<Option<CardAnchor>>,
54    /// Card center at the start of the active drag gesture.
55    pub drag_start: Cell<CardPosition>,
56    /// Pointer position in board coordinates at drag begin.
57    pub drag_grab: Cell<Option<(f64, f64)>>,
58    /// Card scale at the start of the active pinch gesture.
59    pub pinch_start: Cell<f64>,
60    /// Insertion sequence number for stable z-ordering.
61    pub sequence: u64,
62}

64/// Widget implementation for [`FloatingBoard`].
65pub struct FloatingBoardImpl {
66    /// Internal overlay stacking the card layer over the content.
67    pub overlay: OnceCell<gtk4::Overlay>,
68    /// Transparent card layer holding the card transform stacks.
69    /// Empty areas have no pickable, so gestures fall through to the content.
70    pub card_layer: OnceCell<CardLayer>,
71    /// All cards keyed by id (sorted, for deterministic snapshots).
72    pub(crate) cards: RefCell<BTreeMap<CardId, CardEntry>>,
73    /// Counter for auto-generated `card-N` ids and insertion order.
74    pub sequence: Cell<u64>,
75    /// When `true`, an external consumer manages the surface input region
76    /// and `size_allocate` stops resetting it.
77    pub input_region_managed: Cell<bool>,
78}

99impl ObjectImpl for FloatingBoardImpl {
100    fn constructed(&self) {
101        self.parent_constructed();
102        let board = self.obj();
103        board.set_orientation(Orientation::Vertical);
104        board.add_css_class(CLASS_BOARD);
105
106        let overlay = gtk4::Overlay::new();
107        overlay.set_hexpand(true);
108        overlay.set_vexpand(true);
109
110        let card_layer = CardLayer::new();
111        card_layer.add_css_class(CLASS_CARD_LAYER);
112        overlay.add_overlay(&card_layer);
113
114        board.append(&overlay);
115        let _ = self.card_layer.set(card_layer);
116        let _ = self.overlay.set(overlay);
117    }
118
119    fn signals() -> &'static [Signal] {
120        static SIGNALS: LazyLock<Vec<Signal>> = LazyLock::new(|| {
121            vec![
122                Signal::builder("card-added").param_types([Card::static_type()]).build(),
123                Signal::builder("card-removed").param_types([Card::static_type()]).build(),
124                Signal::builder("geometry-changed").build(),
125            ]
126        });
127        &SIGNALS
128    }
129}

131impl WidgetImpl for FloatingBoardImpl {
132    fn size_allocate(&self, width: i32, height: i32, baseline: i32) {
133        self.parent_size_allocate(width, height, baseline);
134        // smearor-wrot-rotation is designed for layer-shell surfaces and
135        // narrows the surface input region to each card's footprint on every
136        // allocation. Inside the board all card allocations run through this
137        // vfunc, so restoring the full input region here wins per layout pass.
138        // Skipped while a consumer manages the region itself
139        // (`set_input_region_managed`) - e.g. touch-passthrough overlays that
140        // restrict input to the cards.
141        if !self.input_region_managed.get()
142            && let Some(surface) = self.obj().native().and_then(|native| native.surface())
143        {
144            surface.set_input_region(None);
145        }
146        self.obj().on_board_allocated();
147    }
148}
