30/// Widget implementation for [`FloatingBoard`].
31pub struct FloatingBoardImpl {
32    /// Internal overlay stacking the card layer over the content.
33    pub overlay: OnceCell<gtk4::Overlay>,
34    /// Transparent card layer holding the card transform stacks.
35    /// Empty areas have no pickable, so gestures fall through to the content.
36    pub card_layer: OnceCell<CardLayer>,
37    /// All cards keyed by id (sorted, for deterministic snapshots).
38    pub(crate) cards: RefCell<BTreeMap<CardId, CardEntry>>,
39    /// Counter for auto-generated `card-N` ids and insertion order.
40    pub sequence: Cell<SequenceNumber>,
41    /// When `true`, an external consumer manages the surface input region
42    /// and `size_allocate` stops resetting it.
43    pub input_region_managed: Cell<bool>,
44}
45
46impl Default for FloatingBoardImpl {
47    fn default() -> Self {
48        Self {
49            overlay: OnceCell::new(),
50            card_layer: OnceCell::new(),
51            cards: RefCell::new(BTreeMap::new()),
52            sequence: Cell::new(SequenceNumber::default()),
53            input_region_managed: Cell::new(false),
54        }
55    }
56}

65impl ObjectImpl for FloatingBoardImpl {
66    fn constructed(&self) {
67        self.parent_constructed();
68        let board = self.obj();
69        board.set_orientation(Orientation::Vertical);
70        board.add_css_class(CLASS_BOARD);
71
72        let overlay = gtk4::Overlay::new();
73        overlay.set_hexpand(true);
74        overlay.set_vexpand(true);
75
76        let card_layer = CardLayer::new();
77        card_layer.add_css_class(CLASS_CARD_LAYER);
78        overlay.add_overlay(&card_layer);
79
80        board.append(&overlay);
81        let _ = self.card_layer.set(card_layer);
82        let _ = self.overlay.set(overlay);
83    }
84
85    fn signals() -> &'static [Signal] {
86        static SIGNALS: LazyLock<Vec<Signal>> = LazyLock::new(|| BOARD_SIGNALS.iter().map(|signal| signal.signal()).collect());
87        &SIGNALS
88    }
89}

91impl WidgetImpl for FloatingBoardImpl {
92    fn size_allocate(&self, width: i32, height: i32, baseline: i32) {
93        self.parent_size_allocate(width, height, baseline);
94        // smearor-wrot-rotation is designed for layer-shell surfaces and
95        // narrows the surface input region to each card's footprint on every
96        // allocation. Inside the board all card allocations run through this
97        // vfunc, so restoring the full input region here wins per layout pass.
98        // Skipped while a consumer manages the region itself
99        // (`set_input_region_managed`) - e.g. touch-passthrough overlays that
100        // restrict input to the cards.
101        if !self.input_region_managed.get()
102            && let Some(surface) = self.obj().native().and_then(|native| native.surface())
103        {
104            surface.set_input_region(None);
105        }
106        self.obj().on_board_allocated();
107    }
108}
