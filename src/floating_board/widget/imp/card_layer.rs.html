1//! `CardLayerImpl` - the `gtk4::Fixed` card layer that only lets its
2//! children pick events.
3
4use glib::object_subclass;
5use gtk4::subclass::prelude::FixedImpl;
6use gtk4::subclass::prelude::ObjectImpl;
7use gtk4::subclass::prelude::ObjectSubclass;
8use gtk4::subclass::prelude::WidgetImpl;
9
10use crate::widget::CardLayer;
11
12/// Implementation of [`CardLayer`].
13#[derive(Default)]
14pub struct CardLayerImpl;
15
16#[object_subclass]
17impl ObjectSubclass for CardLayerImpl {
18    const NAME: &'static str = "FloatingBoardCardLayer";
19    type Type = CardLayer;
20    type ParentType = gtk4::Fixed;
21}
22
23impl ObjectImpl for CardLayerImpl {}
24
25impl WidgetImpl for CardLayerImpl {
26    /// The layer itself is never an event target - only the cards
27    /// placed on it are. GTK pick checks children *before* consulting
28    /// `contains` for the parent itself, so reporting `false` here
29    /// passes clicks on empty areas through to the content below
30    /// while the cards stay interactive.
31    fn contains(&self, _x: f64, _y: f64) -> bool {
32        false
33    }
34}
35
36impl FixedImpl for CardLayerImpl {}
