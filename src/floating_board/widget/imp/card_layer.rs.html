1//! `CardLayerImpl` - the `gtk4::Fixed` card layer that only lets its
2//! children pick events.
3
4use crate::widget::CardLayer;
5use crate::widget::FloatingBoard;
6use glib::WeakRef;
7use glib::object_subclass;
8use gtk4::subclass::prelude::FixedImpl;
9use gtk4::subclass::prelude::ObjectImpl;
10use gtk4::subclass::prelude::ObjectSubclass;
11use gtk4::subclass::prelude::WidgetImpl;

15/// Implementation of [`CardLayer`].
16#[derive(Default)]
17pub struct CardLayerImpl {
18    /// Owning board - reconciles card positions while children
19    /// re-allocate (see `size_allocate`).
20    pub board: RefCell<Option<WeakRef<FloatingBoard>>>,
21}
22
23#[object_subclass]
24impl ObjectSubclass for CardLayerImpl {
25    const NAME: &'static str = "FloatingBoardCardLayer";
26    type Type = CardLayer;
27    type ParentType = gtk4::Fixed;
28}
29
30impl ObjectImpl for CardLayerImpl {}
31
32impl WidgetImpl for CardLayerImpl {
33    /// The layer itself is never an event target - only the cards
34    /// placed on it are. GTK pick checks children *before* consulting
35    /// `contains` for the parent itself, so reporting `false` here
36    /// passes clicks on empty areas through to the content below
37    /// while the cards stay interactive.
38    fn contains(&self, _x: f64, _y: f64) -> bool {
39        false
40    }
41
42    /// A `gtk4::Fixed` re-allocates whenever a child's measured size
43    /// changes - the board's own allocation is unaffected, so its
44    /// position reconciliation would miss the pass. Re-applying the
45    /// stored `put` positions here keeps grown or rotated cards centered
46    /// on their model position in the same layout pass that allocates
47    /// their new size (otherwise the card renders off its footprint and
48    /// off the input region until the board reallocates).
49    fn size_allocate(&self, width: i32, height: i32, baseline: i32) {
50        if let Some(board) = self.board.borrow().as_ref().and_then(WeakRef::upgrade) {
51            board.reconcile_card_positions();
52        }
53        self.parent_size_allocate(width, height, baseline);
54    }
55}
56
57impl FixedImpl for CardLayerImpl {}
