27/// Installs the crate's default stylesheet on the default display.
28///
29/// The provider is registered with `STYLE_PROVIDER_PRIORITY_FALLBACK`, so
30/// consumer themes and additional providers always win over these defaults.
31/// Called lazily from `FloatingBoard` construction - before a display exists
32/// the registration is skipped and retried on the next board.
33pub(crate) fn ensure_default_css() {
34    if CSS_LOADED.load(Ordering::Acquire) {
35        return;
36    }
37    let Some(display) = Display::default() else {
38        return;
39    };
40    let provider = CssProvider::new();
41    provider.load_from_string(DEFAULT_CSS);
42    gtk4::style_context_add_provider_for_display(&display, &provider, gtk4::STYLE_PROVIDER_PRIORITY_FALLBACK);
43    CSS_LOADED.store(true, Ordering::Release);
44}
