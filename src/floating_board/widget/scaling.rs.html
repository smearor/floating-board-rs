18glib::wrapper! {
19    /// Container that renders its child scaled around the center.
20    ///
21    /// Outer element of the card transform stack:
22    /// `ScalingWidget { RotationWidget { Card } }`.
23    pub struct ScalingWidget(ObjectSubclass<ScalingWidgetImpl>)
24        @extends Widget,
25        @implements Accessible, Buildable, ConstraintTarget;
26}
27
28impl ScalingWidget {
29    /// Creates a scaling container for `child` at scale `1.0`.
30    pub fn new(child: &impl IsA<Widget>) -> Self {
31        let widget: Self = glib::Object::builder().build();
32        widget.set_child(Some(child));
33        widget
34    }
35
36    /// Returns the scaling layout of this widget.
37    pub fn scaling_layout(&self) -> Option<ScalingLayout> {
38        self.layout_manager().and_then(|manager| manager.downcast::<ScalingLayout>().ok())
39    }
40
41    /// Current uniform scale factor (`1.0` = natural size).
42    pub fn scale(&self) -> f64 {
43        self.scaling_layout().map(|layout| layout.imp().scale.get()).unwrap_or(1.0)
44    }
45
46    /// Sets the scale factor and triggers a re-measure so the parent
47    /// (`gtk4::Fixed`) re-allocates the resized footprint.
48    pub fn set_scale(&self, scale: f64) {
49        if let Some(layout) = self.scaling_layout() {
50            layout.imp().scale.set(scale);
51            layout.layout_changed();
52        }
53        self.queue_resize();
54    }
55
56    /// Returns the wrapped child widget.
57    pub fn child(&self) -> Option<Widget> {
58        self.imp().child.borrow().clone()
59    }
60
61    /// Replaces the wrapped child widget.
62    pub fn set_child(&self, child: Option<&impl IsA<Widget>>) {
63        let mut stored = self.imp().child.borrow_mut();
64        if let Some(old_child) = stored.take() {
65            old_child.unparent();
66        }
67        if let Some(new_child) = child {
68            new_child.set_parent(self);
69            *stored = Some(new_child.clone().upcast());
70        }
71        drop(stored);
72        self.queue_resize();
73    }
74}
