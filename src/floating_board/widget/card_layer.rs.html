1//! `CardLayer` - the transparent `gtk4::Fixed` subclass holding the
2//! card transform stacks.
3
4use gtk4::glib;
5
6use crate::widget::imp::card_layer::CardLayerImpl;
7
8glib::wrapper! {
9    /// Card layer: a `gtk4::Fixed` that is never itself an event
10    /// target - picks pass through to its card children or to the
11    /// board's content below.
12    pub struct CardLayer(ObjectSubclass<CardLayerImpl>)
13        @extends gtk4::Fixed, gtk4::Widget,
14        @implements gtk4::Accessible, gtk4::Buildable, gtk4::ConstraintTarget;
15}
16
17impl CardLayer {
18    /// Creates the empty card layer.
19    pub fn new() -> Self {
20        glib::Object::builder().build()
21    }
22}
23
24impl Default for CardLayer {
25    fn default() -> Self {
26        Self::new()
27    }
