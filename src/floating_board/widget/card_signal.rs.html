1use glib::subclass::Signal;
2
3/// The signals `Card` emits, as a type-safe alternative to stringly-typed
4/// signal names. Used by `signals()` in the widget implementation and by
5/// every `emit_by_name` call site so each name lives in exactly one place.
6#[derive(Debug, Clone, Copy, PartialEq, Eq)]
7pub(crate) enum CardSignal {
8    /// Emitted after a drag or quad transform changed the position.
9    Moved,
10    /// Emitted on every rotation change (gesture or programmatic).
11    Rotated,
12    /// Emitted after a transform or `set_scale` changed the scale.
13    Scaled,
14    /// Emitted when the card is raised to the top.
15    Raised,
16    /// Emitted when the fold state toggles.
17    Folded,
18}
19
20/// Every signal of the card, for `ObjectImpl::signals()`.
21pub(crate) const CARD_SIGNALS: &[CardSignal] = &[
22    CardSignal::Moved,
23    CardSignal::Rotated,
24    CardSignal::Scaled,
25    CardSignal::Raised,
26    CardSignal::Folded,
27];
28
29impl CardSignal {
30    /// The static GTK signal name (`"card-moved"`).
31    pub(crate) fn name(self) -> &'static str {
32        match self {
33            Self::Moved => "card-moved",
34            Self::Rotated => "card-rotated",
35            Self::Scaled => "card-scaled",
36            Self::Raised => "card-raised",
37            Self::Folded => "card-folded",
38        }
39    }
40
41    /// Builds the parameterless GTK signal declaration.
42    pub(crate) fn signal(self) -> Signal {
43        Signal::builder(self.name()).build()
44    }
45}
