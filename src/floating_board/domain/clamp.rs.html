4/// Computes the minimal translation `(dx, dy)` needed so that at least
5/// `min_visible` pixels of `footprint` overlap `bounds` on each axis.
6///
7/// Applying the returned offset to the card's position moves the footprint
8/// back inside the board. The overlap requirement is capped at half the
9/// bounds size so it stays satisfiable on very small boards.
10///
11/// Pure function - unit-testable without GTK.
12pub fn clamp_offset(footprint: &Rect, bounds: &Rect, min_visible: MinVisible) -> (f64, f64) {
13    (
14        clamp_axis(footprint.x, footprint.width, bounds.x, bounds.width, min_visible),
15        clamp_axis(footprint.y, footprint.height, bounds.y, bounds.height, min_visible),
16    )
17}
18
19fn clamp_axis(footprint_min: f64, footprint_size: f64, bounds_min: f64, bounds_size: f64, min_visible: MinVisible) -> f64 {
20    let overlap = min_visible.value().min(bounds_size / 2.0);
21    // footprint_min must keep `overlap` pixels inside the bounds on this axis.
22    let allowed_min = bounds_min + overlap - footprint_size;
23    let allowed_max = bounds_min + bounds_size - overlap;
24    footprint_min.clamp(allowed_min, allowed_max) - footprint_min
25}

27#[cfg(test)]
28mod tests {
29    use super::*;
30
31    const BOUNDS: Rect = Rect {
32        x: 0.0,
33        y: 0.0,
34        width: 1000.0,
35        height: 800.0,
36    };
37    const MIN_VISIBLE: MinVisible = MinVisible(40.0);
38
39    #[test]
40    fn fully_inside_footprint_is_untouched() {
41        let footprint = Rect::new(100.0, 100.0, 200.0, 100.0);
42        assert_eq!(clamp_offset(&footprint, &BOUNDS, MIN_VISIBLE), (0.0, 0.0));
43    }
44
45    #[test]
46    fn footprint_pushed_back_from_right_edge() {
47        // Only 10 px of the card remain inside on the right edge.
48        let footprint = Rect::new(990.0, 100.0, 200.0, 100.0);
49        let (dx, dy) = clamp_offset(&footprint, &BOUNDS, MIN_VISIBLE);
50        assert_eq!(dy, 0.0);
51        assert_eq!(dx, -30.0); // 40 px of the card stay inside the bounds
52    }
53
54    #[test]
55    fn footprint_pushed_back_from_left_edge() {
56        let footprint = Rect::new(-190.0, 100.0, 200.0, 100.0);
57        let (dx, _) = clamp_offset(&footprint, &BOUNDS, MIN_VISIBLE);
58        assert_eq!(dx, 30.0); // 40 px of the card stay inside the bounds
59    }
60
61    #[test]
62    fn footprint_pushed_back_from_bottom_edge() {
63        let footprint = Rect::new(100.0, 785.0, 200.0, 100.0);
64        let (_, dy) = clamp_offset(&footprint, &BOUNDS, MIN_VISIBLE);
65        assert_eq!(dy, -25.0); // 40 px of the card stay inside the bounds
66    }
67
68    #[test]
69    fn oversized_footprint_is_still_clamped() {
70        // Card wider than the board: only 30 px overlap on the right edge.
71        let footprint = Rect::new(970.0, 100.0, 2000.0, 100.0);
72        let (dx, _) = clamp_offset(&footprint, &BOUNDS, MIN_VISIBLE);
73        assert_eq!(dx, -10.0); // right edge pulled back to 1000 - 40
74    }
75
76    #[test]
77    fn oversized_centered_footprint_is_untouched() {
78        let footprint = Rect::new(-500.0, 100.0, 2000.0, 100.0);
79        assert_eq!(clamp_offset(&footprint, &BOUNDS, MIN_VISIBLE).0, 0.0);
80    }
