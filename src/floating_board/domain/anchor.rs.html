5/// Margin in pixels between a card's footprint and the board edge when an
6/// edge/corner anchor is used.
7pub const ANCHOR_MARGIN: f64 = 24.0;
8
9/// Resolves a [`CardAnchor`] into a concrete center position.
10///
11/// `bounds` is the board rectangle, `footprint` the card's displayed size
12/// (already accounting for rotation and scale). Edge anchors place the
13/// footprint `ANCHOR_MARGIN` inside the bounds; [`CardAnchor::Explicit`]
14/// returns its position unchanged.
15pub fn resolve_anchor(anchor: &CardAnchor, bounds: &Rect, footprint_width: f64, footprint_height: f64) -> CardPosition {
16    if let CardAnchor::Explicit(position) = anchor {
17        return *position;
18    }
19    let horizontal = anchor.horizontal_weight().unwrap_or(0.5);
20    let vertical = anchor.vertical_weight().unwrap_or(0.5);
21    let inset_x = ANCHOR_MARGIN + footprint_width / 2.0;
22    let inset_y = ANCHOR_MARGIN + footprint_height / 2.0;
23    let x = bounds.x + inset_x + (bounds.width - 2.0 * inset_x).max(0.0) * horizontal;
24    let y = bounds.y + inset_y + (bounds.height - 2.0 * inset_y).max(0.0) * vertical;
25    CardPosition::new(x, y)
26}

28#[cfg(test)]
29mod tests {
30    use super::*;
31
32    const BOUNDS: Rect = Rect {
33        x: 0.0,
34        y: 0.0,
35        width: 1000.0,
36        height: 800.0,
37    };
38
39    #[test]
40    fn center_anchor_centers_card() {
41        let position = resolve_anchor(&CardAnchor::Center, &BOUNDS, 200.0, 100.0);
42        assert_eq!(position, CardPosition::new(500.0, 400.0));
43    }
44
45    #[test]
46    fn bottom_center_anchor_insets_from_bottom() {
47        let position = resolve_anchor(&CardAnchor::BottomCenter, &BOUNDS, 200.0, 100.0);
48        assert_eq!(position, CardPosition::new(500.0, 800.0 - ANCHOR_MARGIN - 50.0));
49    }
50
51    #[test]
52    fn top_left_anchor_insets_both_axes() {
53        let position = resolve_anchor(&CardAnchor::TopLeft, &BOUNDS, 200.0, 100.0);
54        assert_eq!(position, CardPosition::new(ANCHOR_MARGIN + 100.0, ANCHOR_MARGIN + 50.0));
55    }
56
57    #[test]
58    fn explicit_anchor_passthrough() {
59        let explicit = CardPosition::new(123.0, 456.0);
60        assert_eq!(resolve_anchor(&CardAnchor::Explicit(explicit), &BOUNDS, 200.0, 100.0), explicit);
61    }
62
63    #[test]
64    fn tiny_board_stays_inside() {
65        let bounds = Rect::new(0.0, 0.0, 100.0, 80.0);
66        let position = resolve_anchor(&CardAnchor::BottomRight, &bounds, 400.0, 300.0);
67        assert_eq!(position, CardPosition::new(ANCHOR_MARGIN + 200.0, ANCHOR_MARGIN + 150.0));
68    }
