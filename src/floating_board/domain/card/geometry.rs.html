7/// Complete geometric state of a card: center position, rotation and scale.
8///
9/// This is the unit of persistence: a `CardGeometry` per card ends up in
10/// [`crate::domain::BoardLayout`] snapshots.
11#[derive(Debug, Clone, Copy)]
12#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
13pub struct CardGeometry {
14    /// Center point of the card in board coordinates.
15    pub position: CardPosition,
16    /// Rotation of the card; serialized as degrees (`rotation = 180.0`).
17    #[cfg_attr(feature = "serde", serde(with = "rotation_degrees"))]
18    pub rotation: Rotation,
19    /// Scale factor of the card (`1.0` = natural size).
20    pub scale: ScaleFactor,
21}
22
23impl Default for CardGeometry {
24    fn default() -> Self {
25        Self {
26            position: CardPosition::default(),
27            rotation: Rotation::ZERO,
28            scale: ScaleFactor::default(),
29        }
30    }
31}
32
33/// Component-wise interpolation on one clock: the position slides
34/// along `Vec2<Pixels>`, the rotation follows the shortest arc via
35/// `RotationRad` (re-quantized so snapped targets stay `QuarterTurn`)
36/// and the scale blends linearly - the animation unit of layout
37/// transitions.
38impl Interpolate for CardGeometry {
39    fn interpolate(self, target: Self, t: Fraction) -> Self {
40        Self {
41            position: CardPosition::from(self.position.0.interpolate(target.position.0, t)),
42            rotation: Rotation::quantized(self.rotation.to_rotation_rad().interpolate(target.rotation.to_rotation_rad(), t)),
43            scale: self.scale.interpolate(target.scale, t),
44        }
45    }
46}

48#[cfg(feature = "serde")]
49mod rotation_degrees {
50    use serde::Deserialize;
51    use serde::Deserializer;
52    use serde::Serialize;
53    use serde::Serializer;
54    use smearor_commons_geometry::Rotation;
55
56    /// Serializes the rotation as a plain degree value (`180.0`), matching the
57    /// persisted layout format - independent of `Rotation`'s own serde
58    /// representation.
59    pub fn serialize<S: Serializer>(rotation: &Rotation, serializer: S) -> Result<S::Ok, S::Error> {
60        rotation.degrees().serialize(serializer)
61    }
62
63    /// Deserializes a degree value into [`Rotation`].
64    pub fn deserialize<'de, D: Deserializer<'de>>(deserializer: D) -> Result<Rotation, D::Error> {
65        Ok(Rotation::from_degrees(f64::deserialize(deserializer)?))
66    }
67}

75    /// Interpolation endpoints map `from`/`to` exactly.
76    #[test]
77    fn interpolate_endpoints_are_exact() {
78        let from = CardGeometry {
79            position: CardPosition::from(Vec2::new(10.0, 20.0)),
80            rotation: Rotation::from_degrees(30.0),
81            scale: ScaleFactor::new(0.5),
82        };
83        let to = CardGeometry {
84            position: CardPosition::from(Vec2::new(110.0, 220.0)),
85            rotation: Rotation::from_degrees(120.0),
86            scale: ScaleFactor::new(2.0),
87        };
88        let start = from.interpolate(to, Fraction::new(0.0));
89        assert_eq!(start.position.0, from.position.0);
90        assert_eq!(start.rotation.degrees(), from.rotation.degrees());
91        assert_eq!(start.scale.value(), from.scale.value());
92        let end = from.interpolate(to, Fraction::new(1.0));
93        assert_eq!(end.position.0, to.position.0);
94        assert_eq!(end.rotation.degrees(), to.rotation.degrees());
95        assert_eq!(end.scale.value(), to.scale.value());
96    }

98    /// The midpoint blends all three components.
99    #[test]
100    fn interpolate_midpoint_blends_all_components() {
101        let from = CardGeometry {
102            position: CardPosition::from(Vec2::new(0.0, 0.0)),
103            rotation: Rotation::from_degrees(0.0),
104            scale: ScaleFactor::new(1.0),
105        };
106        let to = CardGeometry {
107            position: CardPosition::from(Vec2::new(100.0, 200.0)),
108            rotation: Rotation::from_degrees(90.0),
109            scale: ScaleFactor::new(3.0),
110        };
111        let mid = from.interpolate(to, Fraction::new(0.5));
112        assert_eq!(mid.position.0, Vec2::new(50.0, 100.0));
113        assert_eq!(mid.rotation.degrees(), 45.0);
114        assert_eq!(mid.scale.value(), 2.0);
115    }
