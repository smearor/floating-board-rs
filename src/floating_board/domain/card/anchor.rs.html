1use crate::domain::CardPosition;
2use smearor_commons_geometry::Pixels;
3use smearor_commons_geometry::Rect;
4use smearor_commons_geometry::Vec2;

6/// Default margin in pixels between a card's footprint and the board edge
7/// when an edge/corner anchor is used. Cards override it via
8/// `CardConfig::anchor_margin`.
9pub const ANCHOR_MARGIN: Pixels = Pixels::new(24.0);
10
11/// Declarative initial placement of a card on the board.
12///
13/// Anchors are resolved against the board bounds once the card's natural size
14/// is known (first allocation). [`CardAnchor::Explicit`] bypasses resolution
15/// and uses the given center position directly.
16#[derive(Debug, Clone, Copy, PartialEq, Default)]
17#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize), serde(rename_all = "kebab-case"))]
18pub enum CardAnchor {
19    /// Top edge, horizontally flush left.
20    TopLeft,
21    /// Top edge, horizontally centered.
22    TopCenter,
23    /// Top edge, horizontally flush right.
24    TopRight,
25    /// Left edge, vertically centered.
26    CenterLeft,
27    /// Board center.
28    #[default]
29    Center,
30    /// Right edge, vertically centered.
31    CenterRight,
32    /// Bottom edge, horizontally flush left.
33    BottomLeft,
34    /// Bottom edge, horizontally centered.
35    BottomCenter,
36    /// Bottom edge, horizontally flush right.
37    BottomRight,
38    /// Explicit center position in board coordinates.
39    Explicit(CardPosition),
40}
41
42impl CardAnchor {
43    /// Horizontal anchor weight: `0.0` = left, `0.5` = center, `1.0` = right.
44    pub(crate) fn horizontal_weight(&self) -> Option<f64> {
45        match self {
46            Self::TopLeft | Self::CenterLeft | Self::BottomLeft => Some(0.0),
47            Self::TopCenter | Self::Center | Self::BottomCenter => Some(0.5),
48            Self::TopRight | Self::CenterRight | Self::BottomRight => Some(1.0),
49            Self::Explicit(_) => None,
50        }
51    }
52
53    /// Vertical anchor weight: `0.0` = top, `0.5` = center, `1.0` = bottom.
54    pub(crate) fn vertical_weight(&self) -> Option<f64> {
55        match self {
56            Self::TopLeft | Self::TopCenter | Self::TopRight => Some(0.0),
57            Self::CenterLeft | Self::Center | Self::CenterRight => Some(0.5),
58            Self::BottomLeft | Self::BottomCenter | Self::BottomRight => Some(1.0),
59            Self::Explicit(_) => None,
60        }
61    }
62
63    /// Resolves the anchor into a concrete center position.
64    ///
65    /// `bounds` is the board rectangle, `footprint` the card's displayed
66    /// size (already accounting for rotation and scale). Edge anchors place
67    /// the footprint `margin` pixels inside the bounds;
68    /// [`CardAnchor::Explicit`] returns its position unchanged.
69    pub fn resolve(&self, bounds: &Rect<Pixels>, footprint: Vec2<Pixels>, margin: Pixels) -> CardPosition {
70        if let Self::Explicit(position) = self {
71            return *position;
72        }
73        let weights = Vec2::new(self.horizontal_weight().unwrap_or(0.5), self.vertical_weight().unwrap_or(0.5));
74        let inset = footprint * 0.5 + Vec2::splat(margin);
75        let span = (bounds.size - inset * 2.0).map(|c| c.max(0.0));
76        CardPosition::from(bounds.min + inset + span.component_mul(&weights))
77    }
78}

80#[cfg(test)]
81mod tests {
82    use super::*;
83
84    #[test]
85    fn anchor_weights() {
86        assert_eq!(CardAnchor::BottomRight.horizontal_weight(), Some(1.0));
87        assert_eq!(CardAnchor::BottomRight.vertical_weight(), Some(1.0));
88        assert_eq!(CardAnchor::Center.horizontal_weight(), Some(0.5));
89        assert_eq!(CardAnchor::Explicit(CardPosition::from(Vec2::new(1.0, 2.0))).horizontal_weight(), None);
90    }
91
92    const BOUNDS: Rect<Pixels> = Rect::from_xywh(0.0, 0.0, 1000.0, 800.0);
93
94    #[test]
95    fn center_anchor_centers_card() {
96        let position = CardAnchor::Center.resolve(&BOUNDS, Vec2::new(200.0, 100.0), ANCHOR_MARGIN);
97        assert_eq!(position, CardPosition::from(Vec2::new(500.0, 400.0)));
98    }
99
100    #[test]
101    fn bottom_center_anchor_insets_from_bottom() {
102        let position = CardAnchor::BottomCenter.resolve(&BOUNDS, Vec2::new(200.0, 100.0), ANCHOR_MARGIN);
103        assert_eq!(position, CardPosition::from(Vec2::new(500.0, 800.0 - ANCHOR_MARGIN.as_f64() - 50.0)));
104    }
105
106    #[test]
107    fn top_left_anchor_insets_both_axes() {
108        let position = CardAnchor::TopLeft.resolve(&BOUNDS, Vec2::new(200.0, 100.0), ANCHOR_MARGIN);
109        assert_eq!(position, CardPosition::from(Vec2::new(ANCHOR_MARGIN.as_f64() + 100.0, ANCHOR_MARGIN.as_f64() + 50.0)));
110    }
111
112    #[test]
113    fn explicit_anchor_passthrough() {
114        let explicit = CardPosition::from(Vec2::new(123.0, 456.0));
115        assert_eq!(CardAnchor::Explicit(explicit).resolve(&BOUNDS, Vec2::new(200.0, 100.0), ANCHOR_MARGIN), explicit);
116    }
117
118    #[test]
119    fn tiny_board_stays_inside() {
120        let bounds = Rect::<Pixels>::from_xywh(0.0, 0.0, 100.0, 80.0);
121        let position = CardAnchor::BottomRight.resolve(&bounds, Vec2::new(400.0, 300.0), ANCHOR_MARGIN);
122        assert_eq!(position, CardPosition::from(Vec2::new(ANCHOR_MARGIN.as_f64() + 200.0, ANCHOR_MARGIN.as_f64() + 150.0)));
123    }
